3//! Device-node creation and deletion through configurable hooks.
4//!
5//! The device model delegates `/dev` node operations to [`KernelHooks`].
6//! Implement this trait in the code that manages the device filesystem,
7//! then call [`install_hooks`] once it is ready to create and delete nodes.
8//! Node creation requests are queued until [`install_hooks`] installs the hooks and replays them.
9//! Each [`DevNodeRequest`] carries the device number, path relative to `/dev`, and permissions.
10
11use alloc::vec::Vec;
12
13use ostd::sync::Mutex;
14use spin::Once;
15
16use crate::common::{DevNum, Error, Result, SysStr};
17
18/// Installs the kernel hooks and replays every request queued before.
19///
20/// The hooks must be ready to handle requests when installed.
21/// Node creation requests are queued before installation;
22/// removing a device cancels its queued request.
23/// Calling this a second time has no effect.
24pub fn install_hooks(hooks: &'static dyn KernelHooks) {
25    HOOKS.install(hooks);
26}
27
28/// Callbacks for creating and deleting `/dev` nodes.
29///
30/// Installed through [`install_hooks`] once the callbacks are ready to handle requests.
31/// Until then, node creation requests are queued.
32pub trait KernelHooks: Send + Sync + 'static {
33    /// Creates a `/dev` node.
34    ///
35    /// A path with `/` in it, such as `input/event0`,
36    /// means the intermediate directories are created too.
37    fn create_devnode(&self, request: &DevNodeRequest) -> core::result::Result<(), HookError>;
38
39    /// Deletes a `/dev` node created earlier.
40    /// The request is the one that created the node.
41    fn delete_devnode(&self, request: &DevNodeRequest) -> core::result::Result<(), HookError>;
42}
43
44/// A request to create or delete a `/dev` node.
45#[derive(Clone, Debug, PartialEq, Eq)]
46pub struct DevNodeRequest {
47    /// The device number the node refers to.
48    pub devnum: DevNum,
49    /// The path of the node relative to `/dev`, e.g. `null` or `input/event0`.
50    pub path: SysStr,
51    /// The permission bits of the node.
52    pub mode: u16,
53}
54
55/// An error from a kernel hook.
56#[derive(Clone, Copy, Debug)]
57pub struct HookError;
58
59pub(crate) fn create_devnode(request: DevNodeRequest) -> Result<()> {
60    HOOKS.create_devnode(request)
61}
62
63pub(crate) fn delete_devnode(request: &DevNodeRequest) -> Result<()> {
64    HOOKS.delete_devnode(request)
65}
66
67/// The hooks, once installed, and the requests waiting for them.
68pub(crate) struct HookSlot {
69    hooks: Once<&'static dyn KernelHooks>,
70    pending: Mutex<Vec<DevNodeRequest>>,
71}
72
73impl HookSlot {
74    pub(crate) const fn new() -> Self {
75        Self {
76            hooks: Once::new(),
77            pending: Mutex::new(Vec::new()),
78        }
79    }
80
81    /// Installs the hooks and replays every request queued before.
82    ///
83    /// Installing, draining and replaying all happen under the queue lock,
84    /// so a request made concurrently is either replayed here or delivered directly,
85    /// and a removal cannot overtake the create it cancels.
86    /// Calling this a second time has no effect.
87    pub(crate) fn install(&self, hooks: &'static dyn KernelHooks) {
88        let mut pending = self.pending.lock();
89        let installed = self.hooks.call_once(|| hooks);
90        for request in core::mem::take(&mut *pending) {
91            // A failure here cannot be reported to the caller that queued
92            // the request long ago; the node is simply absent.
93            let _ = installed.create_devnode(&request);
94        }
95    }
96
97    pub(crate) fn create_devnode(&self, request: DevNodeRequest) -> Result<()> {
98        let mut queue = self.pending.lock();
99        match self.hooks.get() {
100            Some(hooks) => {
101                drop(queue);
102                hooks.create_devnode(&request).map_err(|_| Error::Hook)
103            }
104            None => {
105                queue.push(request);
106                Ok(())
107            }
108        }
109    }
110
111    pub(crate) fn delete_devnode(&self, request: &DevNodeRequest) -> Result<()> {
112        let mut queue = self.pending.lock();
113        match self.hooks.get() {
114            Some(hooks) => {
115                drop(queue);
116                hooks.delete_devnode(request).map_err(|_| Error::Hook)
117            }
118            None => {
119                // The node was never created; forget the queued request.
120                queue.retain(|pending| pending != request);
121                Ok(())
122            }
123        }
124    }
125}
126
127/// The shared hook slot.
128static HOOKS: HookSlot = HookSlot::new();
