3//! EFI-side management of TDX unaccepted memory.
4//!
5//! The EFI stub records unaccepted memory ranges from the final UEFI memory map
6//! in a bitmap and passes the table to the kernel through the boot parameters.
7//! It reserves the bitmap storage before exiting boot services, then registers
8//! the ranges after the final memory map is available.
9//!
10//! Asterinas uses a fixed kernel load address below 4 GiB, so the EFI stub does
11//! not need to accept memory for an arbitrary kernel destination. When ranges
12//! are registered, small regions and unaligned edges are accepted immediately;
13//! the remaining aligned regions are left for the kernel to accept later.
14
15use core::{ops::Range, ptr::NonNull};
16
17use linux_boot_params::{BootParams, EfiInfo};
18use tdx_guest::unaccepted_memory::{EfiUnacceptedMemory, EfiUnacceptedMemoryBuilder};
19use uefi::{
20    boot::AllocateType,
21    mem::memory_map::{MemoryMap, MemoryType},
22};
23
24use crate::x86::amd64_efi::{alloc, efi};

26/// Creates a builder for the unaccepted memory bitmap.
27///
28/// This method is called before exiting the EFI boot services to make memory allocations. Afterwards,
29/// the builder can build the final bitmap without allocations. See [`finish_unaccepted_memory`].
30pub(super) fn prepare_unaccepted_memory(
31    boot_params: &mut BootParams,
32) -> Option<EfiUnacceptedMemoryBuilder> {
33    if !tdx_guest::is_tdx_guest_early() {
34        return None;
35    }
36
37    boot_params.efi_info.efi_loader_signature = EfiInfo::TDX_LOADER_SIGNATURE;
38    boot_params.efi_info.efi_systab = 0;
39    boot_params.efi_info.efi_systab_hi = 0;
40
41    // Compute a single range that covers all unaccepted memory.
42    let coverage = {
43        let pre_exit_memory_map = uefi::boot::memory_map(MemoryType::LOADER_DATA)
44            .expect("failed to fetch pre-exit memory map for unaccepted bitmap setup");
45        let mut ranges = iter_unaccepted_memory(&pre_exit_memory_map);
46
47        let first_range = ranges.next()?;
48        ranges.fold(first_range, |coverage, range| {
49            coverage.start.min(range.start)..coverage.end.max(range.end)
50        })
51    };
52
53    // Allocate pages for the unaccepted memory bitmap.
54    let builder = {
55        let required_size = EfiUnacceptedMemory::required_size_for_range(coverage.clone())
56            .expect("invalid unaccepted memory bounds");
57
58        let allocation = alloc::alloc_pages(AllocateType::MaxAddress(u32::MAX as u64), required_size);
59        let allocation_ptr = NonNull::new(allocation.as_mut_ptr().cast()).unwrap();
60        // SAFETY: `allocation` is writable, page-aligned, and sized for `coverage`.
61        // After this call, the allocation is accessed only through the builder and
62        // the resulting table, and the EFI stub never frees it.
63        unsafe { EfiUnacceptedMemory::builder(allocation_ptr, allocation.len(), coverage) }
64            .expect("failed to initialize unaccepted memory builder")
65    };
66
67    uefi::println!("[EFI stub] Passing unaccepted memory bitmap to the kernel");
68
69    Some(builder)
70}

72/// Builds the unaccepted memory bitmap according to `memory_map` and publishes the table address to
73/// the kernel.
74///
75/// This method may accept memory to align the regions of unaccepted memory, allowing them to be
76/// recorded in the bitmap. `memory_map` will not be updated.
77///
78/// # Safety
79///
80/// `memory_map` must accurately describe the regions of unaccepted memory.
81pub(super) unsafe fn finish_unaccepted_memory<M: MemoryMap>(
82    boot_params: &mut BootParams,
83    builder: Option<EfiUnacceptedMemoryBuilder>,
84    memory_map: &M,
85) {
86    let Some(mut builder) = builder else {
87        return;
88    };
89
90    for range in iter_unaccepted_memory(memory_map) {
91        // SAFETY: The caller guarantees that `range` describes a region of unaccepted memory.
92        unsafe { builder.register_range(range.start, range.end) }
93            .expect("failed to register unaccepted memory range");
94    }
95
96    let table = builder.build();
97    let table_addr = core::ptr::from_ref(table.as_ref().get_ref()).addr();
98    let table_addr = table_addr
99        .try_into()
100        .expect("unaccepted memory table is above 4 GiB");
101    boot_params.efi_info.efi_systab = table_addr;
102}
103
104fn iter_unaccepted_memory<M: MemoryMap>(memory_map: &M) -> impl Iterator<Item = Range<u64>> + '_ {
105    memory_map.entries().filter_map(|entry| {
106        if entry.ty != MemoryType::UNACCEPTED {
107            return None;
108        }
109
110        let size = entry.page_count * efi::PAGE_SIZE;
111        let end = entry.phys_start + size;
112        Some(entry.phys_start..end)
113    })
114}
