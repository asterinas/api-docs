11impl DrmFile {
12    pub(super) fn dispatch_ioctl(&self, raw_ioctl: RawIoctl) -> Result<i32> {
13        use ioctl_defs::*;
14
15        dispatch_ioctl!(match raw_ioctl {
16            // General ioctl cmds.
17            cmd @ Version => {
18                self.check_ioctl_access(DrmIoctlAccess::RENDER_ALLOW)?;
19                self.drm_get_version(cmd)
20            }
21            cmd @ GetUnique => {
22                self.check_ioctl_access(DrmIoctlAccess::empty())?;
23                self.drm_get_unique(cmd)
24            }
25            cmd @ GetMagic => {
26                self.check_ioctl_access(DrmIoctlAccess::empty())?;
27                self.drm_get_magic(cmd)
28            }
29            cmd @ GetCap => {
30                self.check_ioctl_access(DrmIoctlAccess::RENDER_ALLOW)?;
31                self.drm_get_cap(cmd)
32            }
33            cmd @ AuthMagic => {
34                self.check_ioctl_access(DrmIoctlAccess::MASTER)?;
35                self.drm_auth_magic(cmd)
36            }
37            cmd @ SetMaster => {
38                self.check_ioctl_access(DrmIoctlAccess::empty())?;
39                self.drm_set_master(cmd)
40            }
41            cmd @ DropMaster => {
42                self.check_ioctl_access(DrmIoctlAccess::empty())?;
43                self.drm_drop_master(cmd)
44            }
45            // GEM ioctl cmds.
46            cmd @ GemClose => {
47                self.check_gem_ioctl_access(DrmIoctlAccess::RENDER_ALLOW)?;
48                self.drm_gem_close(cmd)
49            }
50            cmd @ ModeCreateDumb => {
51                self.check_gem_ioctl_access(DrmIoctlAccess::empty())?;
52                self.drm_mode_create_dumb(cmd)
53            }
54            cmd @ ModeMapDumb => {
55                self.check_gem_ioctl_access(DrmIoctlAccess::empty())?;
56                self.drm_mode_map_dumb(cmd)
57            }
58            cmd @ ModeDestroyDumb => {
59                self.check_gem_ioctl_access(DrmIoctlAccess::empty())?;
60                self.drm_mode_destroy_dumb(cmd)
61            }
62            // KMS ioctl cmds.
63            cmd @ SetClientCap => {
64                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
65                self.drm_set_client_cap(cmd)
66            }
67            cmd @ ModeGetResources => {
68                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
69                self.drm_mode_get_resources(cmd)
70            }
71            cmd @ ModeGetCrtc => {
72                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
73                self.drm_mode_get_crtc(cmd)
74            }
75            cmd @ ModeSetCrtc => {
76                self.check_kms_ioctl_access(DrmIoctlAccess::MASTER)?;
77                self.drm_mode_set_crtc(cmd)
78            }
79            cmd @ ModeGetEncoder => {
80                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
81                self.drm_mode_get_encoder(cmd)
82            }
83            cmd @ ModeGetConnector => {
84                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
85                self.drm_mode_get_connector(cmd)
86            }
87            cmd @ ModeGetProperty => {
88                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
89                self.drm_mode_get_property(cmd)
90            }
91            cmd @ ModeGetPropBlob => {
92                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
93                self.drm_mode_get_blob(cmd)
94            }
95            cmd @ ModeAddFb => {
96                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
97                self.drm_mode_add_fb(cmd)
98            }
99            cmd @ ModeRmFb => {
100                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
101                self.drm_mode_rm_fb(cmd)
102            }
103            cmd @ ModeDirtyFb => {
104                self.check_kms_ioctl_access(DrmIoctlAccess::MASTER)?;
105                self.drm_mode_dirty_fb(cmd)
106            }
107            cmd @ ModeGetPlaneResources => {
108                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
109                self.drm_mode_get_plane_resources(cmd)
110            }
111            cmd @ ModeGetPlane => {
112                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
113                self.drm_mode_get_plane(cmd)
114            }
115            cmd @ ModeObjectGetProps => {
116                self.check_kms_ioctl_access(DrmIoctlAccess::empty())?;
117                self.drm_mode_object_get_props(cmd)
118            }
119            _ => {
120                ostd::warn!(
121                    "unknown ioctl minor={:?} cmd={:#x}",
122                    self.minor_type(),
123                    raw_ioctl.cmd()
124                );
125                return_errno_with_message!(Errno::ENOTTY, "the DRM ioctl command is unknown")
126            }
127        })
128    }
129
130    fn check_gem_ioctl_access(&self, required_access: DrmIoctlAccess) -> Result<()> {
131        if self.device().as_gem_ops().is_none() {
132            return_errno_with_message!(Errno::EOPNOTSUPP, "the DRM device does not support GEM");
133        }
134
135        self.check_ioctl_access(required_access)
136    }
137
138    fn check_kms_ioctl_access(&self, required_access: DrmIoctlAccess) -> Result<()> {
139        if self.device().as_kms_ops().is_none() {
140            return_errno_with_message!(Errno::EOPNOTSUPP, "the DRM device does not support KMS");
141        }
142
143        self.check_ioctl_access(required_access)
144    }
145
146    fn check_ioctl_access(&self, required_access: DrmIoctlAccess) -> Result<()> {
147        match self.minor_type() {
148            DrmMinorType::Primary => {
149                if required_access.contains(DrmIoctlAccess::AUTH) && !self.is_authenticated() {
150                    return_errno_with_message!(
151                        Errno::EACCES,
152                        "the DRM ioctl requires an authenticated primary client"
153                    );
154                }
155            }
156            DrmMinorType::Render => {
157                if !required_access.contains(DrmIoctlAccess::RENDER_ALLOW) {
158                    return_errno_with_message!(
159                        Errno::EACCES,
160                        "the DRM ioctl is not allowed on a render node"
161                    );
162                }
163            }
164        }
165
166        if required_access.contains(DrmIoctlAccess::ROOT_ONLY) && !has_current_sys_admin() {
167            return_errno_with_message!(Errno::EACCES, "the DRM ioctl requires CAP_SYS_ADMIN");
168        }
169        if required_access.contains(DrmIoctlAccess::MASTER) && !self.is_master() {
170            return_errno_with_message!(Errno::EACCES, "the DRM client is not the current master");
171        }
172
173        Ok(())
174    }
175}

177bitflags::bitflags! {
178    struct DrmIoctlAccess: u32 {
179        /// Requires the client to be authenticated or the current master.
180        const AUTH = 1 << 0;
181        /// Requires the current file to be the DRM master.
182        const MASTER = 1 << 1;
183        /// Requires the caller to have `CAP_SYS_ADMIN`.
184        const ROOT_ONLY = 1 << 2;
185        /// Allows the ioctl on a render node.
186        const RENDER_ALLOW = 1 << 3;
187    }
188}
