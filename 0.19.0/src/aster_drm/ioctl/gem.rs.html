3use aster_core::prelude::*;
4use ostd::mm::PAGE_SIZE;
5
6use crate::{file::DrmFile, ioctl::ioctl_defs};
7
8impl DrmFile {
9    pub(super) fn drm_gem_close(&self, cmd: ioctl_defs::GemClose) -> Result<i32> {
10        let args = cmd.read()?;
11        self.remove_gem_object(args.handle)?;
12
13        Ok(0)
14    }
15
16    pub(super) fn drm_mode_create_dumb(&self, cmd: ioctl_defs::ModeCreateDumb) -> Result<i32> {
17        let mut args = cmd.read()?;
18
19        if args.width == 0 || args.height == 0 || args.bpp == 0 {
20            return_errno_with_message!(
21                Errno::EINVAL,
22                "the dumb-buffer dimensions and bits per pixel must be nonzero"
23            );
24        }
25
26        let bits_per_row = u64::from(args.width) * u64::from(args.bpp);
27        let Ok(pitch) = u32::try_from(bits_per_row.div_ceil(8)) else {
28            return_errno_with_message!(Errno::EINVAL, "the dumb-buffer pitch overflows u32");
29        };
30
31        // Linux rejects dumb buffers whose page-aligned size does not fit in `u32`,
32        // even though the UAPI exposes the resulting size as `u64`.
33        let Some(size) = pitch
34            .checked_mul(args.height)
35            .and_then(|size| size.checked_next_multiple_of(PAGE_SIZE as u32))
36        else {
37            return_errno_with_message!(
38                Errno::EINVAL,
39                "the page-aligned dumb-buffer size overflows u32"
40            );
41        };
42
43        let gem_ops = self.device().as_gem_ops().ok_or(Errno::EOPNOTSUPP)?;
44        let gem_object = gem_ops.create_dumb(size as usize)?;
45
46        args.pitch = pitch;
47        args.size = gem_object.size() as u64;
48
49        args.handle = self.add_gem_object(gem_object)?;
50        if let Err(err) = cmd.write(&args) {
51            let _ = self.remove_gem_object(args.handle);
52            return Err(err);
53        }
54
55        Ok(0)
56    }
57
58    pub(super) fn drm_mode_map_dumb(&self, cmd: ioctl_defs::ModeMapDumb) -> Result<i32> {
59        let mut args = cmd.read()?;
60        args.offset = self.map_gem_handle(args.handle)?;
61        cmd.write(&args)?;
62
63        Ok(0)
64    }
65
66    pub(super) fn drm_mode_destroy_dumb(&self, cmd: ioctl_defs::ModeDestroyDumb) -> Result<i32> {
67        let args = cmd.read()?;
68        self.remove_gem_object(args.handle)?;
69
70        Ok(0)
71    }
72}
