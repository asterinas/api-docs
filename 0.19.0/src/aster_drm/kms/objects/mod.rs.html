3//! KMS objects and their device-local object store.
4//!
5//! This module defines planes, CRTCs, encoders, connectors, framebuffers, properties, and property blobs.
6//!
7//! Static objects belonging to a DRM device are registered by type in [`DrmKmsObjectStore`].
8//!
9//! Static objects and per-file objects such as framebuffers share a device-local `KmsObjectId` namespace.
10//!
11//! [`builder::DrmKmsObjectStoreBuilder`] constructs and validates the static KMS topology
12//! before materializing the objects in the store.
13

40pub(crate) type KmsObjectId = u32;
41
42pub(crate) const MAX_OBJECTS_PER_TYPE: usize = u32::BITS as usize;
43
44/// A zero-based position in one KMS object type's registration order.
45///
46/// The index is local to an object type.
47///
48/// For example, encoder index 0 and CRTC index 0 refer to entries in different lists.
49pub type KmsObjectIndex = RangedUsize<0, { MAX_OBJECTS_PER_TYPE - 1 }>;
50
51type KmsObjectMask = BitArray<[u32; MAX_OBJECTS_PER_TYPE / u32::BITS as usize]>;
52
53/// Registers all KMS objects belonging to a DRM device.
54///
55/// Static objects are grouped by type and looked up by their device-local `KmsObjectId`.
56/// Each object stores its own per-type registration index.
57///
58/// The ID allocator is also used for per-file KMS objects,
59/// but those objects are owned and tracked outside this store.
60#[derive(Debug)]
61pub struct DrmKmsObjectStore {
62    id_allocator: SparseIdAlloc,
63    planes: HashMap<KmsObjectId, Arc<DrmPlane>>,
64    crtcs: HashMap<KmsObjectId, Arc<DrmCrtc>>,
65    encoders: HashMap<KmsObjectId, Arc<DrmEncoder>>,
66    connectors: HashMap<KmsObjectId, Arc<DrmConnector>>,
67
68    properties: HashMap<KmsObjectId, Arc<DrmProperty>>,
69    /// Provides a semantic index for standard properties;
70    /// driver-specific properties remain accessible through `properties` by object ID.
71    standard_property_ids: HashMap<DrmStandardProperty, KmsObjectId>,
72    // TODO: Support both device-lifetime blobs created by the kernel and
73    // per-file blobs created and owned by userspace clients.
74    property_blobs: HashMap<KmsObjectId, Arc<DrmPropertyBlob>>,
75}

91    pub(crate) fn alloc_object_id(&mut self) -> Result<KmsObjectId> {
92        let id = self
93            .id_allocator
94            .alloc()
95            .ok_or_else(|| Error::with_message(Errno::ENOSPC, "all KMS object IDs are in use"))?;
96
97        Ok(id)
98    }

120    pub(crate) fn collect_sorted_crtc_ids(&self) -> Vec<KmsObjectId> {
121        let mut indexed_ids = self
122            .crtcs()
123            .map(|crtc| (crtc.index(), crtc.id()))
124            .collect::<Vec<_>>();
125
126        indexed_ids.sort_unstable_by_key(|(index, _)| *index);
127        indexed_ids.into_iter().map(|(_, id)| id).collect()
128    }

197    /// Looks up a KMS mode object's property table without specifying its type.
198    ///
199    /// The object ID may refer to any KMS mode object,
200    /// such as a plane, CRTC, encoder, or connector.
201    /// This is an object ID, not a property ID.
202    ///
203    /// Returns `EINVAL` for an encoder, which has no property table,
204    /// and `None` if no supported object matches the ID.
205    pub(crate) fn lookup_any_object_properties(
206        &self,
207        id: KmsObjectId,
208    ) -> Result<Option<&DrmPropertyAttachments>> {
209        if let Some(plane) = self.lookup_plane_by_id(id) {
210            return Ok(Some(plane.properties()));
211        }
212
213        if let Some(crtc) = self.lookup_crtc_by_id(id) {
214            return Ok(Some(crtc.properties()));
215        }
216
217        if let Some(connector) = self.lookup_connector_by_id(id) {
218            return Ok(Some(connector.properties()));
219        }
220
221        if self.lookup_encoder_by_id(id).is_some() {
222            return_errno_with_message!(Errno::EINVAL, "the DRM encoder has no property table");
223        }
224
225        Ok(None)
226    }
227
228    fn get_or_create_standard_property(
229        &mut self,
230        standard: DrmStandardProperty,
231    ) -> Result<Arc<DrmProperty>> {
232        if let Some(property) = self
233            .standard_property_ids
234            .get(&standard)
235            .and_then(|id| self.properties.get(id))
236        {
237            return Ok(property.clone());
238        }
239
240        let id = self.alloc_object_id()?;
241        let property = Arc::new(standard.create_property(id));
242
243        let previous = self.properties.insert(id, property.clone());
244        debug_assert!(previous.is_none());
245        let previous = self.standard_property_ids.insert(standard, id);
246        debug_assert!(previous.is_none());
247        Ok(property)
248    }
249
250    fn create_property_blob(&mut self, data: Box<[u8]>) -> Result<Arc<DrmPropertyBlob>> {
251        let id = self.alloc_object_id()?;
252        let blob = Arc::new(DrmPropertyBlob::new(id, data));
253
254        let previous = self.property_blobs.insert(id, blob.clone());
255        debug_assert!(previous.is_none());
256        Ok(blob)
257    }
