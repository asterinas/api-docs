3use aster_core::prelude::*;
4
5/// Encodes four bytes as a Linux DRM FOURCC pixel-format identifier.
6///
7/// Each byte occupies eight bits of the resulting value,
8/// with the first byte placed in the least-significant bits.
9const fn fourcc_code(a: u8, b: u8, c: u8, d: u8) -> u32 {
10    (a as u32) | ((b as u32) << 8) | ((c as u32) << 16) | ((d as u32) << 24)
11}
12
13/// A DRM framebuffer pixel format supported by the kernel.
14///
15/// Each discriminant is a Linux DRM FOURCC identifier exposed through the userspace API.
16/// The numeric values must remain compatible with the corresponding `DRM_FORMAT_*` definitions.
17/// This enum contains only the currently supported subset of DRM pixel formats.
18///
19/// Reference: <https://elixir.bootlin.com/linux/v6.17/source/include/uapi/drm/drm_fourcc.h#L113>.
20#[repr(u32)]
21#[derive(Clone, Copy, Debug, Eq, PartialEq)]
22pub enum DrmPixelFormat {
23    XRGB8888 = fourcc_code(b'X', b'R', b'2', b'4'),
24    ARGB8888 = fourcc_code(b'A', b'R', b'2', b'4'),
25    XBGR8888 = fourcc_code(b'X', b'B', b'2', b'4'),
26    RGBX8888 = fourcc_code(b'R', b'X', b'2', b'4'),
27    BGRX8888 = fourcc_code(b'B', b'X', b'2', b'4'),
28    C8 = fourcc_code(b'C', b'8', b' ', b' '),
29    XRGB1555 = fourcc_code(b'X', b'R', b'1', b'5'),
30    RGB565 = fourcc_code(b'R', b'G', b'1', b'6'),
31    RGB888 = fourcc_code(b'R', b'G', b'2', b'4'),
32    BGR888 = fourcc_code(b'B', b'G', b'2', b'4'),
33    XRGB2101010 = fourcc_code(b'X', b'R', b'3', b'0'),
34}
35
36impl DrmPixelFormat {
37    pub fn bytes_per_pixel(&self) -> usize {
38        match self {
39            Self::C8 => 1,
40            Self::XRGB1555 | Self::RGB565 => 2,
41            Self::RGB888 | Self::BGR888 => 3,
42            Self::XRGB8888
43            | Self::ARGB8888
44            | Self::XBGR8888
45            | Self::RGBX8888
46            | Self::BGRX8888
47            | Self::XRGB2101010 => 4,
48        }
49    }
50}
51
52impl TryFrom<(u32, u32)> for DrmPixelFormat {
53    type Error = Error;
54
55    fn try_from(value: (u32, u32)) -> Result<Self, Self::Error> {
56        match value {
57            (8, 8) => Ok(DrmPixelFormat::C8),
58            (16, 15) => Ok(DrmPixelFormat::XRGB1555),
59            (16, 16) => Ok(DrmPixelFormat::RGB565),
60            (24, 24) => Ok(DrmPixelFormat::RGB888),
61            (32, 24) => Ok(DrmPixelFormat::XRGB8888),
62            (32, 30) => Ok(DrmPixelFormat::XRGB2101010),
63            (32, 32) => Ok(DrmPixelFormat::ARGB8888),
64            _ => return_errno_with_message!(
65                Errno::EINVAL,
66                "the legacy DRM framebuffer format is unsupported"
67            ),
68        }
69    }
70}
