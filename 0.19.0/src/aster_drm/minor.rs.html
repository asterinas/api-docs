25const DRM_MAJOR_ID: u16 = 226;
26const PRIMARY_MINOR_BASE: u32 = 0;
27const RENDER_MINOR_BASE: u32 = 128;
28
29/// Represents a DRM minor node exposed to userspace.
30///
31/// A `DrmMinor` corresponds to a single character device registered under
32/// `/dev/dri/` (such as `/dev/dri/cardX` or `/dev/dri/renderDX`). It does not
33/// own hardware state by itself; instead, it provides a userspace-facing
34/// access point with a specific permission and usage model.
35///
36/// Multiple `DrmMinor` instances may reference the same underlying
37/// `DrmDevice`, sharing the same driver instance and global device state.
38/// The semantic differences between minors (e.g. authentication requirements,
39/// ioctl visibility, access restrictions) are expressed via `type_` and
40/// enforced by the minor and ioctl layers.
41#[derive(Debug)]
42pub(super) struct DrmMinor {
43    type_: DrmMinorType,
44    registered_device: Arc<RegisteredDrmDevice>,
45    weak_self: Weak<Self>,
46}
47
48impl DrmMinor {
49    pub(super) fn new(
50        registered_device: Arc<RegisteredDrmDevice>,
51        type_: DrmMinorType,
52    ) -> Arc<Self> {
53        Arc::new_cyclic(move |weak_ref| Self {
54            type_,
55            registered_device,
56            weak_self: weak_ref.clone(),
57        })
58    }
59
60    pub(super) fn type_(&self) -> DrmMinorType {
61        self.type_
62    }
63
64    pub(super) fn device(&self) -> &Arc<dyn DrmDevice> {
65        self.registered_device.device()
66    }
67
68    /// Opens a client through this minor and returns its master context, if applicable.
69    pub(super) fn open_client(&self, client_id: u64) -> Option<Arc<DrmMaster>> {
70        match self.type_ {
71            DrmMinorType::Primary => Some(self.registered_device.open_primary_client(client_id)),
72            _ => None,
73        }
74    }
75
76    pub(super) fn is_client_master(&self, client_id: u64) -> bool {
77        match self.type_ {
78            DrmMinorType::Primary => self.registered_device.is_client_master(client_id),
79            _ => false,
80        }
81    }
82
83    pub(super) fn authenticate_magic(&self, client_id: u64, magic: u32) -> Result<()> {
84        self.check_primary()?;
85        self.registered_device.authenticate_magic(client_id, magic)
86    }
87
88    pub(super) fn set_master(
89        &self,
90        client_id: u64,
91        retained_master: Option<&Arc<DrmMaster>>,
92    ) -> Result<Arc<DrmMaster>> {
93        self.check_primary()?;
94        self.registered_device
95            .set_master(client_id, retained_master)
96    }
97
98    pub(super) fn drop_master(&self, client_id: u64) -> Result<()> {
99        self.check_primary()?;
100        self.registered_device.drop_master(client_id)
101    }
102
103    fn check_primary(&self) -> Result<()> {
104        if !matches!(self.type_, DrmMinorType::Primary) {
105            return_errno_with_message!(Errno::EACCES, "the DRM operation requires a primary node");
106        }
107
108        Ok(())
109    }
110
111    pub(super) fn gem_mmap_offset(&self, object: &Arc<DrmGemObject>) -> Result<u64> {
112        self.registered_device.gem_mmap_offset(object)
113    }
114
115    pub(super) fn create_gem_mapped_object(
116        &self,
117        client_id: u64,
118        offset: usize,
119        size: usize,
120    ) -> Result<Box<dyn MappedObject>> {
121        self.registered_device
122            .create_gem_mapped_object(client_id, offset, size)
123    }
124}
125
126impl Device for DrmMinor {
127    fn id(&self) -> DeviceId {
128        let index = self.registered_device.index();
129        let minor_id = match self.type_ {
130            DrmMinorType::Primary => PRIMARY_MINOR_BASE + index,
131            DrmMinorType::Render => RENDER_MINOR_BASE + index,
132        };
133        DeviceId::new(MajorId::new(DRM_MAJOR_ID), MinorId::new(minor_id))
134    }
135
136    fn type_(&self) -> DeviceType {
137        DeviceType::Char
138    }
139
140    fn devtmpfs_meta(&self) -> Option<DevtmpfsNodeMeta> {
141        let index = self.registered_device.index();
142        match self.type_ {
143            DrmMinorType::Primary => {
144                Some(DevtmpfsNodeMeta::new(format!("dri/card{}", index)).unwrap())
145            }
146            DrmMinorType::Render => Some(
147                DevtmpfsNodeMeta::new(format!("dri/renderD{}", index + RENDER_MINOR_BASE)).unwrap(),
148            ),
149        }
150    }
151
152    fn open(&self) -> Result<Box<dyn PerOpenFileOps>> {
153        let drm_minor = self.weak_self.upgrade().unwrap();
154        Ok(Box::new(DrmFile::new(drm_minor)))
155    }
156}

158/// The userspace-facing role of a DRM device node.
159///
160/// Primary nodes provide display-management and authenticated legacy
161/// operations, while render nodes provide unprivileged rendering access.
162#[derive(Clone, Copy, Debug)]
163pub(super) enum DrmMinorType {
164    Primary,
165    Render,
166}
