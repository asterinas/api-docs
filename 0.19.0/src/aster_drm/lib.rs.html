3//! The Direct Rendering Manager subsystem of Asterinas.
4//!
5//! This crate provides the kernel-side framework for exposing graphics devices
6//! through the Linux-compatible DRM userspace API. It sits between graphics
7//! devices and the character-device layer, providing the shared object model,
8//! lifecycle management, permission checks, and ioctl infrastructure needed by
9//! DRM devices.
10//!
11//! Graphics devices implement [`device::DrmDevice`] and the relevant operation
12//! traits to supply hardware-specific behavior. The DRM core owns the common
13//! userspace-facing semantics and coordinates access to the device, keeping
14//! policy and ABI handling independent of individual device implementations.
15

53pub fn register_device(device: Arc<dyn DrmDevice>) -> Result<()> {
54    let registered_device = Arc::new(RegisteredDrmDevice::new(device)?);
55    let render_minor = if registered_device.device().has_features(DrmFeatures::RENDER) {
56        let minor = DrmMinor::new(registered_device.clone(), DrmMinorType::Render);
57        char::register(minor.clone())?;
58        Some(minor)
59    } else {
60        None
61    };
62
63    let primary_minor = DrmMinor::new(registered_device, DrmMinorType::Primary);
64
65    if let Err(error) = char::register(primary_minor) {
66        if let Some(render_minor) = render_minor {
67            let _ = char::unregister(render_minor.id());
68        }
69        return Err(error);
70    }
71
72    Ok(())
73}

75fn has_current_sys_admin() -> bool {
76    let task = Task::current().unwrap();
77    let posix_thread = task.as_posix_thread().unwrap();
78
79    lsm_hook::on_capable(CapableContext::new(
80        UserNamespace::get_init_singleton().as_ref(),
81        posix_thread,
82        CapSet::SYS_ADMIN,
83    ))
84    .is_ok()
85}
