4//! Functions for finding and validating the unaccepted-memory table from EFI tables.
5
6use core::{
7    mem::{align_of, size_of},
8    ptr::NonNull,
9    sync::atomic::AtomicU64,
10};
11
12use super::{
13    EfiUnacceptedMemory, LINUX_EFI_UNACCEPTED_MEM_TABLE_GUID,
14    LINUX_EFI_UNACCEPTED_MEM_TABLE_VERSION,
15};
16
17impl EfiUnacceptedMemory {
18    /// Locates and validates the unaccepted-memory table from an EFI system table.
19    ///
20    /// # Safety
21    ///
22    /// The caller must ensure that:
23    /// - if `systab` is non-null, it points to a valid, readable EFI system table;
24    /// - if the system table's configuration table pointer is non-null, it points to a readable
25    ///   array containing `number_of_configuration_table_entries` valid entries;
26    /// - any non-null vendor table pointer in a matching configuration table entry points to a
27    ///   valid, readable [`EfiUnacceptedMemory`] header.
28    pub unsafe fn from_system_table(
29        systab: *const uefi_raw::table::system::SystemTable,
30    ) -> Option<NonNull<Self>> {
31        if systab.is_null() || !systab.is_aligned() {
32            log::warn!("EFI system table is null or misaligned");
33            return None;
34        }
35
36        // SAFETY: Caller guarantees `systab` points to a valid, accessible EFI System Table.
37        let systab = unsafe { &*systab };
38
39        let configuration_table = systab.configuration_table;
40        if configuration_table.is_null() || !configuration_table.is_aligned() {
41            log::warn!("EFI configuration table is null or misaligned");
42            return None;
43        }
44
45        let configuration_table_size = systab
46            .number_of_configuration_table_entries
47            .checked_mul(size_of::<uefi_raw::table::configuration::ConfigurationTable>());
48        if configuration_table_size.is_none_or(|size| size > isize::MAX as usize) {
49            log::warn!("EFI configuration table is too large");
50            return None;
51        }
52
53        // SAFETY: `configuration_table` is non-null, suitably aligned, and its total byte length fits in a slice.
54        let entries = unsafe {
55            core::slice::from_raw_parts(
56                configuration_table,
57                systab.number_of_configuration_table_entries,
58            )
59        };
60
61        // SAFETY: Caller guarantees accessible memory for table pointers.
62        unsafe { Self::from_configuration_tables(entries) }
63    }
64
65    /// Locates and validates the unaccepted-memory table from EFI configuration table entries.
66    ///
67    /// # Safety
68    ///
69    /// The vendor table pointers within matching configuration table entries must point to
70    /// valid and accessible memory if present.
71    pub unsafe fn from_configuration_tables(
72        entries: &[uefi_raw::table::configuration::ConfigurationTable],
73    ) -> Option<NonNull<Self>> {
74        let table_ptr = entries
75            .iter()
76            .find(|entry| entry.vendor_guid == LINUX_EFI_UNACCEPTED_MEM_TABLE_GUID)?
77            .vendor_table
78            .cast::<Self>();
79
80        let non_null_table = NonNull::new(table_ptr)?;
81        if !table_ptr.is_aligned() {
82            return None;
83        }
84
85        // SAFETY: The pointer is non-null and aligned. Caller ensures it points to accessible memory.
86        let table = unsafe { non_null_table.as_ref() };
87
88        if table.version() != LINUX_EFI_UNACCEPTED_MEM_TABLE_VERSION {
89            log::warn!(
90                "Unknown unaccepted memory table version: {}",
91                table.version()
92            );
93            return None;
94        }
95
96        if table.unit_size_bytes() == 0 || !table.unit_size_bytes().is_power_of_two() {
97            log::warn!(
98                "Invalid unaccepted memory table unit size: {}",
99                table.unit_size_bytes()
100            );
101            return None;
102        }
103
104        let bitmap_addr = table_ptr.addr().checked_add(size_of::<Self>())?;
105        if !bitmap_addr.is_multiple_of(align_of::<AtomicU64>())
106            || table.bitmap_size_bytes() == 0
107            || !table
108                .bitmap_size_bytes()
109                .is_multiple_of(size_of::<AtomicU64>() as u64)
110        {
111            log::warn!(
112                "Invalid unaccepted memory table bitmap size: {}",
113                table.bitmap_size_bytes()
114            );
115            return None;
116        }
117
118        let total_bits = table.bitmap_size_bytes().checked_mul(8)?;
119        let total_size = total_bits.checked_mul(u64::from(table.unit_size_bytes()))?;
120        if table.phys_base().checked_add(total_size).is_none() {
121            log::warn!("Unaccepted memory table coverage overflows");
122            return None;
123        }
124
125        Some(non_null_table)
126    }
127}
