4//! Acceptance-path operations for unaccepted-memory bitmap ranges.
5//!
6//! This file contains range acceptance flows, plus pending-run claim/restore helpers.
7
8use super::{bitmap::BitIndex, EfiUnacceptedMemory};
9use crate::{accept_memory, AcceptError};
10
11impl EfiUnacceptedMemory {
12    /// Accepts bitmap-marked units that overlap `start..end`, then clears accepted bits.
13    ///
14    /// # Safety
15    ///
16    /// The caller must ensure this table and bitmap describe pending private-memory units,
17    /// and the target GPA ranges are valid for TDX acceptance.
18    pub unsafe fn accept_range(&self, start: u64, end: u64) -> Result<(), AcceptError> {
19        let Some((first_bit, last_bit, unit_size)) = self.overlapping_bit_range(start, end) else {
20            return Ok(());
21        };
22
23        let phys_base = self.phys_base;
24        let bitmap = self.bitmap_ref();
25        let mut scan = first_bit;
26        while let Some(run_start) = bitmap.find_next_set(scan, last_bit) {
27            let run_end = bitmap
28                .find_next_zero(run_start, last_bit)
29                .unwrap_or(last_bit);
30
31            let run_gpa_start = Self::bit_to_gpa(phys_base, run_start, unit_size)?;
32            let run_gpa_end = Self::bit_to_gpa(phys_base, run_end, unit_size)?;
33
34            // SAFETY: Caller guarantees bitmap/GPA mapping validity for pending private pages.
35            unsafe { accept_memory(run_gpa_start, run_gpa_end)? };
36            bitmap.clear_range(run_start, run_end);
37
38            scan = run_end;
39        }
40
41        Ok(())
42    }
43
44    /// Finds the first contiguous run of set bits overlapping `[start, end)`,
45    /// clears those bits, and returns the corresponding GPA range.
46    ///
47    /// # Safety
48    ///
49    /// The caller must ensure:
50    /// - No concurrent operation touches the same bitmap bits.
51    pub unsafe fn claim_next_pending_run(
52        &self,
53        start: u64,
54        end: u64,
55    ) -> Result<Option<(u64, u64)>, AcceptError> {
56        let Some((first_bit, last_bit, unit_size)) = self.overlapping_bit_range(start, end) else {
57            return Ok(None);
58        };
59
60        // SAFETY: Public concurrent API contract guarantees valid writable bitmap
61        // payload and atomic-access discipline for overlapping ranges.
62        let bitmap = self.bitmap_ref();
63        let Some(run_start) = bitmap.find_next_set(first_bit, last_bit) else {
64            return Ok(None);
65        };
66        let run_end = bitmap
67            .find_next_zero(run_start, last_bit)
68            .unwrap_or(last_bit);
69
70        let gpa_start = Self::bit_to_gpa(self.phys_base, run_start, unit_size)?;
71        let gpa_end = Self::bit_to_gpa(self.phys_base, run_end, unit_size)?;
72
73        bitmap.clear_range(run_start, run_end);
74        Ok(Some((gpa_start, gpa_end)))
75    }
76
77    /// Re-sets bitmap bits for a GPA range whose TDX accept failed.
78    ///
79    /// # Safety
80    ///
81    /// The caller must ensure:
82    /// - No concurrent operation touches the same bitmap bits.
83    /// - `start..end` is exactly a unit-aligned range previously returned by
84    ///   [`Self::claim_next_pending_run`] and has not been accepted or restored.
85    pub unsafe fn restore_pending_range(&self, start: u64, end: u64) {
86        let Some((first_bit, last_bit, _unit_size)) = self.overlapping_bit_range(start, end) else {
87            return;
88        };
89
90        // SAFETY: Public concurrent API contract guarantees valid writable bitmap
91        // payload and atomic-access discipline for overlapping ranges.
92        let bitmap = self.bitmap_ref();
93        bitmap.set_range(first_bit, last_bit);
94    }
95
96    fn bit_to_gpa(phys_base: u64, bit: BitIndex, unit_size: u64) -> Result<u64, AcceptError> {
97        let offset = bit
98            .checked_mul(unit_size)
99            .ok_or(AcceptError::ArithmeticOverflow)?;
100        phys_base
101            .checked_add(offset)
102            .ok_or(AcceptError::ArithmeticOverflow)
103    }
104}
