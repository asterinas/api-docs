3//! This module defines struct `ProcessVm`
4//! to represent the layout of user space process virtual memory.
5//!
6//! The `ProcessVm` struct contains `Vmar`,
7//! which stores all existing memory mappings.
8//! The `Vm` also contains
9//! the basic info of process level vm segments,
10//! like init stack and heap.

34/*
35 * The user's virtual memory space layout looks like below.
36 *
37 *  (high address)
38 *  +---------------------+ <------+ The top of Vmar, which is the highest address usable
39 *  |                     |          Randomly padded pages
40 *  +---------------------+ <------+ The base of the initial user stack
41 *  | User stack          |
42 *  |                     |
43 *  +---------||----------+ <------+ The user stack limit, can be extended lower
44 *  |         \/          |
45 *  | ...                 |
46 *  |                     |
47 *  | MMAP Spaces         |
48 *  |                     |
49 *  | ...                 |
50 *  |         /\          |
51 *  +---------||----------+ <------+ The current program break
52 *  | User heap           |
53 *  |                     |
54 *  +---------------------+ <------+ The original program break
55 *  |                     |          Randomly padded pages
56 *  +---------------------+ <------+ The end of the program's last segment
57 *  |                     |
58 *  | Loaded segments     |
59 *  | .text, .data, .bss  |
60 *  | , etc.              |
61 *  |                     |
62 *  +---------------------+ <------+ The bottom of Vmar at 0x1_0000
63 *  |                     |          64 KiB unusable space
64 *  +---------------------+
65 *  (low address)
66 */
67
68/// The process user space virtual memory
69pub(crate) struct ProcessVm {
70    /// The initial portion of the main stack of a process.
71    init_stack: InitStack,
72    /// The user heap
73    heap: Heap,
74    /// The code range from the executable file.
75    code_range: SpinLock<Range<Vaddr>>,
76    /// The data range from the executable file.
77    data_range: SpinLock<Range<Vaddr>>,
78    /// The path of the executable file.
79    executable_path: Path,
80    /// The base address for vDSO segment
81    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
82    vdso_base: AtomicUsize,
83}
84
85impl ProcessVm {
86    /// Creates a new `ProcessVm` without mapping anything.
87    pub(crate) fn new(executable_path: Path) -> Self {
88        Self {
89            init_stack: InitStack::new(),
90            heap: Heap::new_uninitialized(),
91            code_range: SpinLock::new(0..0),
92            data_range: SpinLock::new(0..0),
93            executable_path,
94            #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
95            vdso_base: AtomicUsize::new(0),
96        }
97    }
98
99    /// Creates a new `ProcessVm` with identical contents of an existing one.
100    pub(crate) fn fork_from(process_vm: &Self, heap_guard: &LockedHeap) -> Self {
101        Self {
102            init_stack: process_vm.init_stack.clone(),
103            heap: Heap::fork_from(heap_guard),
104            code_range: SpinLock::new(process_vm.code_range.lock().clone()),
105            data_range: SpinLock::new(process_vm.data_range.lock().clone()),
106            executable_path: process_vm.executable_path.clone(),
107            #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
108            vdso_base: AtomicUsize::new(process_vm.vdso_base.load(Ordering::Relaxed)),
109        }
110    }
111
112    /// Returns the initial portion of the main stack of a process.
113    pub(crate) fn init_stack(&self) -> &InitStack {
114        &self.init_stack
115    }
116
117    /// Returns the user heap.
118    pub(crate) fn heap(&self) -> &Heap {
119        &self.heap
120    }
121
122    /// Returns the code range from the executable file.
123    pub(crate) fn code_range(&self) -> Range<Vaddr> {
124        self.code_range.lock().clone()
125    }
126
127    /// Returns the data range from the executable file.
128    pub(crate) fn data_range(&self) -> Range<Vaddr> {
129        self.data_range.lock().clone()
130    }
131
132    /// Returns a reference to the executable `Path`.
133    pub(crate) fn executable_path(&self) -> &Path {
134        &self.executable_path
135    }
136
137    /// Maps and writes the initial portion of the main stack of a process.
138    pub(super) fn map_and_write_init_stack(
139        &self,
140        vmar: &Vmar,
141        argv: Vec<CString>,
142        envp: Vec<CString>,
143        aux_vec: AuxVec,
144    ) -> Result<()> {
145        self.init_stack().map_and_write(vmar, argv, envp, aux_vec)
146    }
147
148    /// Maps and initializes the heap virtual memory.
149    pub(super) fn map_and_init_heap(
150        &self,
151        vmar: &Vmar,
152        data_segment_size: usize,
153        heap_base: Vaddr,
154    ) -> Result<()> {
155        self.heap()
156            .map_and_init_heap(vmar, data_segment_size, heap_base)
157    }
158
159    /// Updates the code range from the executable file.
160    pub(super) fn set_code_range(&self, range: Range<Vaddr>) {
161        *self.code_range.lock() = range;
162    }
163
164    /// Updates the data range from the executable file.
165    pub(super) fn set_data_range(&self, range: Range<Vaddr>) {
166        *self.data_range.lock() = range;
167    }
168
169    /// Returns the base address for vDSO segment.
170    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
171    pub(super) fn vdso_base(&self) -> Vaddr {
172        self.vdso_base.load(Ordering::Relaxed)
173    }
174
175    /// Sets the base address for vDSO segment.
176    #[cfg(any(target_arch = "riscv64", target_arch = "aarch64"))]
177    pub(super) fn set_vdso_base(&self, addr: Vaddr) {
178        self.vdso_base.store(addr, Ordering::Relaxed);
179    }
180}

182/// A guard to the [`Vmar`] used by a process.
183///
184/// It is bound to a [`Process`] and can only be obtained from
185/// the [`Process::lock_vmar`] method.
186///
187/// [`Process`]: super::process::Process
188/// [`Process::lock_vmar`]: super::process::Process::lock_vmar
189pub(crate) struct ProcessVmarGuard<'a> {
190    inner: MutexGuard<'a, Option<Arc<Vmar>>>,
191}
192
193/// A snapshot of the process VMAR identity.
194///
195/// This type is used only for identity comparison.
196#[derive(Clone, Debug)]
197pub(crate) struct VmarSnapshot(Weak<Vmar>);
198
199impl VmarSnapshot {
200    /// Returns the raw identity pointer of the captured `Vmar`.
201    pub(crate) fn as_ptr(&self) -> *const Vmar {
202        Weak::as_ptr(&self.0)
203    }
204
205    /// Returns whether two snapshots refer to the same `Vmar`.
206    pub(crate) fn ptr_eq(&self, other: &Self) -> bool {
207        Weak::ptr_eq(&self.0, &other.0)
208    }
209}
210
211impl From<Weak<Vmar>> for VmarSnapshot {
212    fn from(snapshot: Weak<Vmar>) -> Self {
213        Self(snapshot)
214    }
215}
216
217impl<'a> ProcessVmarGuard<'a> {
218    /// Creates a new VMAR guard from the mutex guard.
219    ///
220    /// This method should only used by [`Process::lock_vmar`].
221    ///
222    /// [`Process::lock_vmar`]: super::process::Process::lock_vmar
223    pub(super) fn new(inner: MutexGuard<'a, Option<Arc<Vmar>>>) -> Self {
224        Self { inner }
225    }
226
227    /// Unwraps and returns a reference to the process VMAR.
228    ///
229    /// # Panics
230    ///
231    /// This method will panic if the process has exited and its VMAR has been dropped.
232    pub(crate) fn unwrap(&self) -> &Vmar {
233        self.inner.as_ref().unwrap()
234    }
235
236    /// Returns a reference to the process VMAR if it exists.
237    ///
238    /// Returns `None` if the process has exited and its VMAR has been dropped.
239    pub(crate) fn as_ref(&self) -> Option<&Vmar> {
240        self.inner.as_ref().map(|v| &**v)
241    }
242
243    /// Takes a snapshot of the current VMAR identity.
244    pub(crate) fn snapshot(&self) -> VmarSnapshot {
245        VmarSnapshot(self.inner.as_ref().map(Arc::downgrade).unwrap_or_default())
246    }
247
248    /// Returns whether the current VMAR has the same identity as the `snapshot`.
249    pub(crate) fn is_same_as(&self, snapshot: &VmarSnapshot) -> bool {
250        self.inner
251            .as_ref()
252            .is_some_and(|vmar| core::ptr::eq(Arc::as_ptr(vmar), Weak::as_ptr(&snapshot.0)))
253    }
254
255    /// Sets a new VMAR for the binding process.
256    ///
257    /// This method will return the old VMAR.
258    ///
259    /// If the `new_vmar` is `None`, this method will remove the
260    /// current VMAR.
261    pub(super) fn set_vmar(&mut self, new_vmar: Option<Arc<Vmar>>) -> Option<Arc<Vmar>> {
262        core::mem::replace(&mut *self.inner, new_vmar)
263    }
264
265    /// Returns a reader for reading contents from
266    /// the initial portion of the main stack of a process.
267    ///
268    /// Returns `None` if the process has exited and its VMAR has been dropped.
269    pub(crate) fn init_stack_reader(&self) -> Option<InitStackReader<'_>> {
270        self.as_ref()
271            .map(|vmar| vmar.process_vm().init_stack.reader(vmar))
272    }
273}

275/// Activates the [`Vmar`] in the current process's context.
276///
277/// Returns a [`ProcessVmarGuard`] that keeps the process VMAR lock held and the old [`Vmar`].
278pub(super) fn activate_vmar<'a>(
279    ctx: &'a Context<'a>,
280    new_vmar: VmarHandle,
281) -> (ProcessVmarGuard<'a>, VmarHandle) {
282    let vmar_arc = new_vmar.clone_arc();
283
284    let mut vmar_guard = ctx.process.lock_vmar();
285
286    // Disable preemption because `thread_local::vmar()` will be borrowed during a context switch.
287    let old_vmar = {
288        let _preempt_guard = disable_preempt();
289        let old_vmar = ctx
290            .thread_local
291            .vmar()
292            .borrow_mut()
293            .replace(new_vmar)
294            .unwrap();
295        vmar_arc.vm_space().activate();
296        old_vmar
297    };
298    vmar_guard.set_vmar(Some(vmar_arc));
299
300    (vmar_guard, old_vmar)
301}
