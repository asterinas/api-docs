15/// The inner data of a kernel thread.
16pub(super) struct KernelThread {
17    vmar: Option<Arc<Vmar>>,
18}
19
20impl KernelThread {
21    /// Returns the VMAR associated with the kernel thread.
22    pub(super) fn vmar(&self) -> Option<&Arc<Vmar>> {
23        self.vmar.as_ref()
24    }
25}
26
27/// A trait to provide the `as_kernel_thread` method for tasks and threads.
28pub(super) trait AsKernelThread {
29    /// Returns the associated [`KernelThread`].
30    fn as_kernel_thread(&self) -> Option<&KernelThread>;
31}
32
33impl AsKernelThread for Thread {
34    fn as_kernel_thread(&self) -> Option<&KernelThread> {
35        self.data().downcast_ref::<KernelThread>()
36    }
37}
38
39impl AsKernelThread for Task {
40    fn as_kernel_thread(&self) -> Option<&KernelThread> {
41        self.as_thread()?.as_kernel_thread()
42    }
43}

45/// Options to create or spawn a new kernel thread.
46pub(crate) struct ThreadOptions {
47    func: Option<Box<dyn FnOnce() + Send>>,
48    cpu_affinity: CpuSet,
49    sched_policy: SchedPolicy,
50    vmar: Option<Arc<Vmar>>,
51}
52
53impl ThreadOptions {
54    /// Creates the thread options with the thread function.
55    pub(crate) fn new<F>(func: F) -> Self
56    where
57        F: FnOnce() + Send + 'static,
58    {
59        let cpu_affinity = CpuSet::new_full();
60        let sched_policy = SchedPolicy::Fair(Nice::default());
61        Self {
62            func: Some(Box::new(func)),
63            cpu_affinity,
64            sched_policy,
65            vmar: None,
66        }
67    }
68
69    /// Sets the CPU affinity of the new thread.
70    pub(crate) fn cpu_affinity(mut self, cpu_affinity: CpuSet) -> Self {
71        self.cpu_affinity = cpu_affinity;
72        self
73    }
74
75    /// Sets the scheduling policy.
76    pub(crate) fn sched_policy(mut self, sched_policy: SchedPolicy) -> Self {
77        self.sched_policy = sched_policy;
78        self
79    }
80
81    /// Associates a VMAR with the kernel thread.
82    ///
83    /// The VMAR is activated whenever the thread is scheduled, allowing fallible userspace memory
84    /// operations to access it directly and handle page faults against it. The association cannot
85    /// be changed after the thread is built.
86    ///
87    /// The `Arc` keeps the VMAR object alive, but does not prevent its mappings from being cleared
88    /// after the last [`VmarHandle`](crate::vm::vmar::VmarHandle) is dropped. A caller that needs
89    /// the mappings to outlive their userspace owner must hold a separate owner-memory lease.
90    #[cfg_attr(not(ktest), expect(dead_code))]
91    pub(crate) fn vmar(mut self, vmar: Arc<Vmar>) -> Self {
92        self.vmar = Some(vmar);
93        self
94    }
95}
96
97impl ThreadOptions {
98    /// Builds a new kernel thread without running it immediately.
99    pub(crate) fn build(mut self) -> Arc<Task> {
100        let task_fn = self.func.take().unwrap();
101        let thread_fn = move || {
102            let _ = oops::catch_panics_as_oops(task_fn);
103            // Ensure that the thread exits.
104            current_thread!().exit();
105        };
106
107        Arc::new_cyclic(|weak_task| {
108            let thread = {
109                let kernel_thread = KernelThread { vmar: self.vmar };
110                let cpu_affinity = self.cpu_affinity;
111                let sched_policy = self.sched_policy;
112                Arc::new(Thread::new(
113                    weak_task.clone(),
114                    kernel_thread,
115                    cpu_affinity,
116                    sched_policy,
117                ))
118            };
119
120            TaskOptions::new(thread_fn).data(thread).build().unwrap()
121        })
122    }
123
124    /// Builds a new kernel thread and runs it immediately.
125    #[track_caller]
126    pub(crate) fn spawn(self) -> Arc<Thread> {
127        let task = self.build();
128        let thread = task.as_thread().unwrap().clone();
129        thread.run();
130        thread
131    }
132}

155    #[ktest]
156    fn associated_vmar_is_reactivated_after_context_switch() {
157        super::super::init();
158
159        let vmar = VmarHandle::new(new_process_vm());
160        let map_addr = PAGE_SIZE * 16;
161        let map_size = PAGE_SIZE * 4;
162        let vmo = VmoOptions::new(map_size).alloc().unwrap();
163        assert_eq!(
164            vmar.new_map(map_size, VmPerms::READ | VmPerms::WRITE)
165                .offset(VmarMapOffset::FixedNoReplace(map_addr))
166                .vmo(vmo)
167                .build()
168                .unwrap(),
169            map_addr
170        );
171
172        let access_addr = map_addr + 17;
173        let access_len = PAGE_SIZE * 3 + 37;
174        let mut expected = vec![0; access_len];
175        for (index, byte) in expected.iter_mut().enumerate() {
176            *byte = (index % 251) as u8;
177        }
178
179        let source = expected.clone();
180        let result = Arc::new(Mutex::new(None));
181        let worker_result = result.clone();
182        let worker_vmar = vmar.clone_arc();
183        let associated_vmar = worker_vmar.clone();
184        let test_cpu = CpuId::current_racy();
185
186        let switcher_vmar_handle = VmarHandle::new(new_process_vm());
187        let switcher_vmar = switcher_vmar_handle.clone_arc();
188        let associated_switcher_vmar = switcher_vmar.clone();
189
190        let worker = ThreadOptions::new(move || {
191            let current = Thread::current().unwrap();
192            assert!(
193                current
194                    .as_kernel_thread()
195                    .and_then(|kernel_thread| kernel_thread.vmar())
196                    .is_some_and(|vmar| Arc::ptr_eq(vmar, &worker_vmar))
197            );
198
199            let mut user_writer = worker_vmar
200                .vm_space()
201                .writer(access_addr, access_len)
202                .unwrap();
203            let mut source_reader = VmReader::from(source.as_slice()).to_fallible();
204            assert_eq!(
205                user_writer.write_fallible(&mut source_reader).unwrap(),
206                access_len
207            );
208
209            let switcher = ThreadOptions::new(move || {
210                let current = Thread::current().unwrap();
211                assert!(
212                    current
213                        .as_kernel_thread()
214                        .and_then(|kernel_thread| kernel_thread.vmar())
215                        .is_some_and(|vmar| Arc::ptr_eq(vmar, &switcher_vmar))
216                );
217                assert!(switcher_vmar.vm_space().reader(map_addr, 0).is_ok());
218            })
219            .cpu_affinity(test_cpu.into())
220            .vmar(associated_switcher_vmar)
221            .spawn();
222
223            switcher.join();
224
225            let mut user_reader = worker_vmar
226                .vm_space()
227                .reader(access_addr, access_len)
228                .unwrap();
229            let mut output = vec![0; access_len];
230            let mut output_writer = VmWriter::from(output.as_mut_slice()).to_fallible();
231            assert_eq!(
232                user_reader.read_fallible(&mut output_writer).unwrap(),
233                access_len
234            );
235            *worker_result.lock() = Some(output);
236        })
237        .cpu_affinity(test_cpu.into())
238        .vmar(associated_vmar)
239        .spawn();
240
241        worker.join();
242        assert_eq!(result.lock().take().unwrap(), expected);
243    }

245    #[ktest]
246    fn unmapped_vmar_access_returns_page_fault() {
247        super::super::init();
248
249        const UNMAPPED_ADDR: usize = PAGE_SIZE * 16;
250        let vmar_handle = VmarHandle::new(new_process_vm());
251        let vmar = vmar_handle.clone_arc();
252        let associated_vmar = vmar.clone();
253
254        let worker = ThreadOptions::new(move || {
255            let mut output = [0u8; 1];
256            let mut output_writer = VmWriter::from(output.as_mut_slice()).to_fallible();
257            let mut user_reader = vmar.vm_space().reader(UNMAPPED_ADDR, 1).unwrap();
258            assert_eq!(
259                user_reader.read_fallible(&mut output_writer),
260                Err((Error::PageFault, 0))
261            );
262
263            let input = [1u8; 1];
264            let mut input_reader = VmReader::from(input.as_slice()).to_fallible();
265            let mut user_writer = vmar.vm_space().writer(UNMAPPED_ADDR, 1).unwrap();
266            assert_eq!(
267                user_writer.write_fallible(&mut input_reader),
268                Err((Error::PageFault, 0))
269            );
270        })
271        .vmar(associated_vmar)
272        .spawn();
273
274        worker.join();
275    }
