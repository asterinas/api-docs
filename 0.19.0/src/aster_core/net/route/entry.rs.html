8/// A route that describes how to handle packets for a destination network.
9#[derive(Clone)]
10pub(super) struct RouteEntry<A: RouteAddressFamily> {
11    dst: A::Cidr,
12    #[expect(dead_code)]
13    gateway: Option<A>,
14    output_iface: Arc<Iface>,
15    table_id: RouteTableId,
16    metric: RouteMetric,
17    type_: RouteType,
18}
19
20impl<A: RouteAddressFamily> RouteEntry<A> {
21    /// Creates a route entry.
22    pub(super) fn new(
23        dst: A::Cidr,
24        gateway: Option<A>,
25        output_iface: Arc<Iface>,
26        table_id: RouteTableId,
27        metric: RouteMetric,
28        type_: RouteType,
29    ) -> Result<Self> {
30        if dst != A::network(&dst) {
31            return_errno_with_message!(Errno::EINVAL, "the route destination is not canonical");
32        }
33
34        Ok(Self {
35            dst,
36            gateway,
37            output_iface,
38            table_id,
39            metric,
40            type_,
41        })
42    }
43
44    /// Returns the destination network.
45    pub(super) const fn dst(&self) -> &A::Cidr {
46        &self.dst
47    }
48
49    /// Returns the next-hop gateway.
50    #[expect(dead_code)]
51    const fn gateway(&self) -> Option<A> {
52        self.gateway
53    }
54
55    /// Returns the output interface.
56    pub(super) const fn output_iface(&self) -> &Arc<Iface> {
57        &self.output_iface
58    }
59
60    /// Returns the ID of the table containing the route.
61    pub(super) const fn table_id(&self) -> RouteTableId {
62        self.table_id
63    }
64
65    /// Returns the route metric.
66    pub(super) const fn metric(&self) -> RouteMetric {
67        self.metric
68    }
69
70    /// Returns the route type.
71    pub(super) const fn type_(&self) -> RouteType {
72        self.type_
73    }
74
75    pub(super) fn matches_lookup(&self, key: &RouteLookupKey<A>) -> bool {
76        A::contains(&self.dst, key.dst())
77    }
78}

80/// An IP address family supported by the routing manager.
81pub(super) trait RouteAddressFamily: Copy + Eq + Ord {
82    /// The CIDR type associated with this address family.
83    type Cidr: Copy + Eq + Ord;
84
85    /// Returns whether `cidr` contains `address`.
86    fn contains(cidr: &Self::Cidr, address: &Self) -> bool;
87
88    /// Returns the canonical network represented by `cidr`.
89    fn network(cidr: &Self::Cidr) -> Self::Cidr;
90
91    /// Returns the prefix length of `cidr`.
92    fn prefix_len(cidr: &Self::Cidr) -> u8;
93}
94
95impl RouteAddressFamily for Ipv4Address {
96    type Cidr = Ipv4Cidr;
97
98    fn contains(cidr: &Self::Cidr, address: &Self) -> bool {
99        cidr.contains_addr(address)
100    }
101
102    fn network(cidr: &Self::Cidr) -> Self::Cidr {
103        cidr.network()
104    }
105
106    fn prefix_len(cidr: &Self::Cidr) -> u8 {
107        cidr.prefix_len()
108    }
109}
110
111impl RouteAddressFamily for Ipv6Address {
112    type Cidr = Ipv6Cidr;
113
114    fn contains(cidr: &Self::Cidr, address: &Self) -> bool {
115        cidr.contains_addr(address)
116    }
117
118    fn network(cidr: &Self::Cidr) -> Self::Cidr {
119        Ipv6Cidr::new(
120            cidr.address().mask(cidr.prefix_len()).into(),
121            cidr.prefix_len(),
122        )
123    }
124
125    fn prefix_len(cidr: &Self::Cidr) -> u8 {
126        cidr.prefix_len()
127    }
128}

130#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
131pub(super) struct RouteTableId(u32);
132
133impl RouteTableId {
134    /// The table ID of the default table,
135    /// which is reserved for routes used after earlier rules do not match.
136    pub(super) const DEFAULT: Self = Self(253);
137    /// The table ID of the main table.
138    pub(super) const MAIN: Self = Self(254);
139    /// The table ID of the local table,
140    /// which manages routes for addresses assigned locally.
141    pub(super) const LOCAL: Self = Self(255);
142}

149/// The forwarding behavior of a route.

152#[derive(Clone, Copy, Debug, Eq, PartialEq)]
153pub(super) enum RouteType {
