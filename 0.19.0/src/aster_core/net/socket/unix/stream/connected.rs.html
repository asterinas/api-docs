27pub(super) struct Connected {
28    // `addr` should be dropped as soon as the socket file is closed,
29    // so it must not belong to `Inner`.
30    addr: Option<UnixSocketAddrBound>,
31    inner: Endpoint<Inner>,
32}

34impl Connected {
35    pub(super) fn new_pair(
36        addr: Option<UnixSocketAddrBound>,
37        peer_addr: Option<UnixSocketAddrBound>,
38        state: EndpointState,
39        peer_state: EndpointState,
40        cred: SocketCred,
41        peer_cred: SocketCred,
42    ) -> (Connected, Connected) {
43        let (this_writer, peer_reader) = RingBuffer::new(UNIX_STREAM_DEFAULT_BUF_SIZE).split();
44        let (peer_writer, this_reader) = RingBuffer::new(UNIX_STREAM_DEFAULT_BUF_SIZE).split();
45
46        let this_inner = Inner {
47            addr: Once::new(),
48            state,
49            reader: Mutex::new(this_reader),
50            writer: Mutex::new(this_writer),
51            all_aux: Mutex::new(VecDeque::new()),
52            has_aux: AtomicBool::new(false),
53            is_pass_cred: AtomicBool::new(false),
54            cred,
55        };
56        let peer_inner = Inner {
57            addr: Once::new(),
58            state: peer_state,
59            reader: Mutex::new(peer_reader),
60            writer: Mutex::new(peer_writer),
61            all_aux: Mutex::new(VecDeque::new()),
62            has_aux: AtomicBool::new(false),
63            is_pass_cred: AtomicBool::new(false),
64            cred: peer_cred,
65        };
66
67        if let Some(addr) = addr.as_ref() {
68            this_inner.addr.call_once(|| addr.clone().into());
69        }
70        if let Some(peer_addr) = peer_addr.as_ref() {
71            peer_inner.addr.call_once(|| peer_addr.clone().into());
72        }
73
74        let (this_inner, peer_inner) = Endpoint::new_pair(this_inner, peer_inner);
75
76        let this = Connected {
77            addr,
78            inner: this_inner,
79        };
80        let peer = Connected {
81            addr: peer_addr,
82            inner: peer_inner,
83        };
84
85        (this, peer)
86    }

92    pub(super) fn peer_addr(&self) -> UnixSocketAddr {
93        self.inner
94            .peer_end()
95            .addr
96            .get()
97            .cloned()
98            .unwrap_or(UnixSocketAddr::Unnamed)
99    }

101    pub(super) fn bind(&mut self, addr_to_bind: UnixSocketAddr) -> Result<()> {
102        if self.addr.is_some() {
103            return addr_to_bind.bind_unnamed();
104        }
105
106        let bound_addr = addr_to_bind.bind()?;
107        self.inner
108            .this_end()
109            .addr
110            .call_once(|| bound_addr.clone().into());
111        self.addr = Some(bound_addr);
112
113        Ok(())
114    }

116    pub(super) fn try_read(
117        &self,
118        writer: &mut dyn MultiWrite,
119        is_seqpacket: bool,
120        flags: RecvFlags,
121    ) -> Result<(RecvOutput, Vec<ControlMessage>)> {
122        let is_empty = writer.is_empty();
123        if is_empty && !is_seqpacket {
124            if self.inner.this_end().reader.lock().is_empty() {
125                return_errno_with_message!(Errno::EAGAIN, "the channel is empty");
126            }
127            return Ok((RecvOutput::new_for_stream(0), Vec::new()));
128        }
129
130        let this_end = self.inner.this_end();
131        let peer_end = self.inner.peer_end();
132
133        let mut reader = this_end.reader.lock();
134        // `reader.len()` is an `Acquire` operation. So it can guarantee that the `has_aux`
135        // check below sees the up-to-date value.
136        let no_aux_len = reader.len();
137
138        let is_pass_cred = this_end.is_pass_cred.load(Ordering::Relaxed);
139        let behavior = flags.receive_behavior();
140
141        // Fast path: There are no auxiliary data to receive.
142        if !peer_end.has_aux.load(Ordering::Relaxed) {
143            let read_len = self.inner.read_with(|| {
144                let head = reader.head();
145                let copy_range = head..head + Wrapping(no_aux_len);
146                reader.ring_buffer().pick_fallible(copy_range, writer)
147            })?;
148            if behavior.will_consume_data() {
149                reader.commit_read(read_len);
150                if read_len > 0 {
151                    self.inner.notify_read();
152                }
153            }
154
155            let ctrl_msgs = if is_pass_cred {
156                AuxiliaryData::default().generate_control(behavior, is_pass_cred)
157            } else {
158                Vec::new()
159            };
160            return Ok((RecvOutput::new_for_stream(read_len), ctrl_msgs));
161        }
162
163        let mut all_aux = peer_end.all_aux.lock();
164        let mut aux_pos = 0;
165
166        let read_base = reader.head();
167        let mut read_tot_len = 0;
168        let mut trunc_len = 0;
169
170        loop {
171            let read_start = read_base + Wrapping(read_tot_len);
172            let (aux_len, aux_front) = if let Some(front) = all_aux.get(aux_pos) {
173                if front.start == read_start {
174                    ((front.end - read_start).0, Some(front))
175                } else {
176                    ((front.start - read_start).0, None)
177                }
178            } else {
179                ((reader.tail() - read_start).0, None)
180            };
181
182            // Unless the auxiliary data we have already received is a subset of the current
183            // auxiliary data, we cannot receive additional bytes.
184            if read_tot_len > 0 {
185                let is_subset = match (aux_pos.checked_sub(1), aux_front.as_ref()) {
186                    (Some(prev_pos), Some(front)) => {
187                        let prev = all_aux.get(prev_pos).unwrap();
188                        prev.data.is_subset_of(&front.data, is_pass_cred)
189                    }
190                    (Some(prev_pos), None) => {
191                        let prev = all_aux.get(prev_pos).unwrap();
192                        prev.data
193                            .is_subset_of(&AuxiliaryData::default(), is_pass_cred)
194                    }
195                    (None, Some(front)) => {
196                        AuxiliaryData::default().is_subset_of(&front.data, is_pass_cred)
197                    }
198                    (None, None) => true,
199                };
200                if !is_subset {
201                    break;
202                }
203            }
204
205            // Read the payload bytes of the current auxiliary data.
206            let read_res = if !is_empty && (aux_len > 0 || aux_front.is_none()) {
207                self.inner.read_with(|| {
208                    let copy_range = read_start..read_start + Wrapping(aux_len);
209                    reader.ring_buffer().pick_fallible(copy_range, writer)
210                })
211            } else {
212                Ok(0)
213            };
214            let read_len = match read_res {
215                Ok(read_len) => read_len,
216                Err(_) if read_tot_len > 0 => break,
217                Err(err) => return Err(err),
218            };
219
220            read_tot_len += read_len;
221            if aux_front.is_some() {
222                aux_pos += 1;
223            }
224
225            // Record the current auxiliary data. Break if the read is incomplete or this is a
226            // `SOCK_SEQPACKET` socket.
227            if is_seqpacket {
228                if read_len < aux_len {
229                    warn!("setting MSG_TRUNC is not supported");
230                    trunc_len = aux_len - read_len;
231                }
232                break;
233            } else if read_len < aux_len {
234                break;
235            }
236        }
237
238        // Consume the payload bytes that we've read.
239        if behavior.will_consume_data() {
240            let consume_tot_len = read_tot_len + trunc_len;
241            reader.commit_read(consume_tot_len);
242            if consume_tot_len > 0 {
243                self.inner.notify_read();
244            }
245        }
246        drop(reader);
247
248        // Consume the auxiliary data that we've read.
249        let ctrl_msgs = if aux_pos >= 1 {
250            let aux_data = all_aux.get_mut(aux_pos - 1).unwrap();
251            debug_assert!((aux_data.start - read_base).0 <= read_tot_len);
252
253            let ctrl_msgs = aux_data.data.generate_control(behavior, is_pass_cred);
254            if behavior.will_consume_data() {
255                let remaining_aux_count = all_aux.len() - (aux_pos - 1);
256                all_aux.retain_back(remaining_aux_count);
257                let consume_len = read_tot_len + trunc_len;
258                if (all_aux.front().unwrap().end - read_base).0 <= consume_len {
259                    all_aux.pop_front();
260                } else {
261                    all_aux.front_mut().unwrap().start = read_base + Wrapping(consume_len);
262                }
263                peer_end
264                    .has_aux
265                    .store(!all_aux.is_empty(), Ordering::Relaxed);
266            }
267            ctrl_msgs
268        } else {
269            let mut default_aux_data = AuxiliaryData::default();
270            default_aux_data.generate_control(behavior, is_pass_cred)
271        };
272
273        debug_assert!(is_seqpacket || read_tot_len != 0);
274        let output = if is_seqpacket {
275            let message_len = read_tot_len + trunc_len;
276            RecvOutput::new_for_packet(flags, read_tot_len, message_len)
277        } else {
278            RecvOutput::new_for_stream(read_tot_len)
279        };
280
281        Ok((output, ctrl_msgs))
282    }

284    pub(super) fn try_write(
285        &self,
286        reader: &mut dyn MultiRead,
287        aux_data: &mut AuxiliaryData,
288        is_seqpacket: bool,
289    ) -> Result<usize> {
290        let is_empty = reader.is_empty();
291        if is_empty {
292            if self.inner.is_shutdown() {
293                return_errno_with_message!(Errno::EPIPE, "the channel is shut down");
294            }
295            if !is_seqpacket {
296                return Ok(0);
297            }
298        }
299
300        if is_seqpacket && reader.sum_lens() >= UNIX_STREAM_DEFAULT_BUF_SIZE {
301            return_errno_with_message!(Errno::EMSGSIZE, "the message is too large");
302        }
303
304        let this_end = self.inner.this_end();
305        let need_pass_cred = this_end.is_pass_cred.load(Ordering::Relaxed)
306            || self.inner.peer_end().is_pass_cred.load(Ordering::Relaxed);
307
308        // Fast path: There are no auxiliary data to transmit.
309        if aux_data.is_empty() && !is_seqpacket && !need_pass_cred {
310            let mut writer = this_end.writer.lock();
311            let write_len = self
312                .inner
313                .write_with(move || writer.write_fallible(reader))?;
314            self.inner.notify_write();
315            return Ok(write_len);
316        }
317
318        let mut all_aux = this_end.all_aux.lock();
319
320        // No matter we succeed later or not, set the flag first to ensure that the auxiliary
321        // data are always visible to `try_recv`.
322        this_end.has_aux.store(true, Ordering::Relaxed);
323
324        // Write the payload bytes.
325        let (write_start, write_res) = if !is_empty {
326            let mut writer = this_end.writer.lock();
327            let write_start = writer.tail();
328            let write_res = self.inner.write_with(move || {
329                if is_seqpacket && writer.free_len() < reader.sum_lens() {
330                    return Ok(0);
331                }
332                writer.write_fallible(reader)
333            });
334            (write_start, write_res)
335        } else {
336            (this_end.writer.lock().tail(), Ok(0))
337        };
338        let Ok(write_len) = write_res else {
339            this_end
340                .has_aux
341                .store(!all_aux.is_empty(), Ordering::Relaxed);
342            return write_res;
343        };
344
345        if need_pass_cred {
346            aux_data.fill_cred();
347        }
348
349        // Store the auxiliary data.
350        let aux_range = RangedAuxiliaryData {
351            data: core::mem::take(aux_data),
352            start: write_start,
353            end: write_start + Wrapping(write_len),
354        };
355        all_aux.push_back(aux_range);
356
357        self.inner.notify_write();
358
359        Ok(write_len)
360    }

362    pub(super) fn shutdown(&self, cmd: SockShutdownCmd) {
363        if cmd.shut_read() {
364            self.inner.peer_shutdown();
365        }
366
367        if cmd.shut_write() {
368            self.inner.shutdown();
369        }
370    }

391    pub(super) fn check_io_events(&self) -> IoEvents {
392        let this_end = self.inner.this_end();
393        let mut events = IoEvents::empty();
394
395        if !this_end.reader.lock().is_empty() {
396            events |= IoEvents::IN;
397        }
398
399        if !this_end.writer.lock().is_full() {
400            events |= IoEvents::OUT;
401        }
402
403        events
404    }

415impl Drop for Connected {
416    fn drop(&mut self) {
417        self.inner.shutdown();
418        self.inner.peer_shutdown();
419    }

422struct Inner {
423    addr: Once<UnixSocketAddr>,
424    state: EndpointState,
425    // Lock order: `reader` -> `all_aux` & `all_aux` -> `writer`
426    reader: Mutex<RbConsumer<u8>>,
427    writer: Mutex<RbProducer<u8>>,
428    all_aux: Mutex<VecDeque<RangedAuxiliaryData>>,
429    has_aux: AtomicBool,
430    is_pass_cred: AtomicBool,
431    cred: SocketCred,
432}
433
434impl AsRef<EndpointState> for Inner {
435    fn as_ref(&self) -> &EndpointState {
436        &self.state
437    }
438}

440struct RangedAuxiliaryData {
441    data: AuxiliaryData,
442    start: Wrapping<usize>, // inclusive
443    end: Wrapping<usize>,   // exclusive
444}
