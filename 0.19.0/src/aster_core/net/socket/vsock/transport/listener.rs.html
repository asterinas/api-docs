19/// A uniquely owned vsock listener handle; dropping it will close the listener.
20pub(in vsock) struct Listener {
21    inner: Arc<ListenerInner>,
22}
23
24impl Listener {
25    pub(super) fn new(inner: Arc<ListenerInner>) -> Self {
26        Self { inner }
27    }
28
29    /// Accepts one pending connection.
30    pub(in vsock) fn try_accept(&self) -> Result<Connection> {
31        self.inner.pop_incoming().map(Connection::new)
32    }
33
34    /// Updates the listen backlog.
35    pub(in vsock) fn set_backlog(&self, backlog: usize) {
36        self.inner.set_backlog(backlog);
37    }
38
39    /// Returns the local listening address.
40    pub(in vsock) fn local_addr(&self) -> VsockSocketAddr {
41        self.inner.bound_port.local_addr()
42    }
43
44    /// Returns the currently observable I/O readiness for the listener.
45    pub(in vsock) fn check_io_events(&self) -> IoEvents {
46        self.inner.check_io_events()
47    }
48}
49
50impl Drop for Listener {
51    fn drop(&mut self) {
52        let vsock_space = self.inner.bound_port.vsock_space();
53        vsock_space.remove_listener(&self.inner);
54    }
55}
56
57pub(super) struct ListenerInner {
58    bound_port: BoundPort,
59    pollee: Pollee,
60    backlog: AtomicUsize,
61    num_conns: AtomicUsize,
62    incoming_conns: SpinLock<VecDeque<Arc<ConnectionInner>>, BottomHalfDisabled>,
63}
64
65impl ListenerInner {
66    pub(super) fn new(bound_port: BoundPort, backlog: usize, pollee: Pollee) -> Arc<Self> {
67        pollee.invalidate();
68
69        Arc::new(Self {
70            bound_port,
71            pollee,
72            backlog: AtomicUsize::new(backlog.min(MAX_BACKLOG)),
73            num_conns: AtomicUsize::new(0),
74            incoming_conns: SpinLock::new(VecDeque::new()),
75        })
76    }
77
78    pub(super) fn is_full(&self) -> bool {
79        // Race conditions don't matter here. We use `>` instead of `>=` because Linux allows to
80        // have `backlog + 1` connections in the backlog queue.
81        self.num_conns.load(Ordering::Relaxed) > self.backlog.load(Ordering::Relaxed)
82    }
83
84    pub(super) fn push_incoming(&self, connection: Arc<ConnectionInner>) {
85        let mut incoming_conns = self.incoming_conns.lock();
86        incoming_conns.push_back(connection);
87        self.num_conns
88            .store(incoming_conns.len(), Ordering::Relaxed);
89
90        drop(incoming_conns);
91        self.pollee.notify(IoEvents::IN);
92    }
93
94    pub(self) fn pop_incoming(&self) -> Result<Arc<ConnectionInner>> {
95        let mut incoming_conns = self.incoming_conns.lock();
96        let Some(connection) = incoming_conns.pop_front() else {
97            return_errno_with_message!(Errno::EAGAIN, "no pending connection is available");
98        };
99        self.num_conns
100            .store(incoming_conns.len(), Ordering::Relaxed);
101
102        drop(incoming_conns);
103        self.pollee.invalidate();
104
105        Ok(connection)
106    }
107
108    pub(self) fn set_backlog(&self, backlog: usize) {
109        self.backlog
110            .store(backlog.min(MAX_BACKLOG), Ordering::Relaxed);
111    }
112
113    pub(self) fn check_io_events(&self) -> IoEvents {
114        let incoming_conns = self.incoming_conns.lock();
115
116        if incoming_conns.is_empty() {
117            IoEvents::empty()
118        } else {
119            IoEvents::IN
120        }
121    }
122
123    pub(super) fn take_incoming_on_removal(&self) -> VecDeque<Arc<ConnectionInner>> {
124        core::mem::take(&mut *self.incoming_conns.lock())
125    }
126
127    pub(super) fn bound_port(&self) -> &BoundPort {
128        &self.bound_port
129    }
130}
