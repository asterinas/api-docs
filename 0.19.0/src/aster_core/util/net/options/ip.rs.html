71pub(crate) fn new_ip_option(name: i32) -> Result<Box<dyn RawSocketOption>> {
72    let name = CIpOptionName::try_from(name).map_err(|_| Errno::ENOPROTOOPT)?;
73    match name {
74        CIpOptionName::TOS => Ok(Box::new(Tos::new())),
75        CIpOptionName::TTL => Ok(Box::new(Ttl::new())),
76        CIpOptionName::HDRINCL => Ok(Box::new(Hdrincl::new())),
77        CIpOptionName::RECVERR => Ok(Box::new(Recverr::new())),
78        _ => return_errno_with_message!(Errno::ENOPROTOOPT, "unsupported ip level option"),
79    }
80}

82trait ReadIpOption: Sized {
83    fn read_ip_option(addr: Vaddr, max_len: u32) -> Result<Self>;
84}
85
86macro_rules! impl_raw_ip_socket_option {
87    ($option:ty) => {
88        impl RawSocketOption for $option {
89            fn read_from_user(&mut self, addr: Vaddr, max_len: u32) -> Result<()> {
90                let input = ReadIpOption::read_ip_option(addr, max_len)?;
91                self.set(input);
92                Ok(())
93            }
94
95            fn write_to_user(&self, addr: Vaddr, max_len: &mut u32) -> Result<usize> {
96                use $crate::util::net::options::utils::WriteToUser;
97
98                let output = self.get().unwrap();
99                output.write_to_user(addr, *max_len)
100            }
101
102            fn as_sock_option_mut(&mut self) -> &mut dyn super::SocketOption {
103                self
104            }
105
106            fn as_sock_option(&self) -> &dyn super::SocketOption {
107                self
108            }
109        }
110    };
111}

118impl ReadIpOption for i32 {
119    fn read_ip_option(addr: Vaddr, max_len: u32) -> Result<Self> {
120        read_ip_int(addr, max_len)
121    }
122}
123
124impl ReadIpOption for bool {
125    fn read_ip_option(addr: Vaddr, max_len: u32) -> Result<Self> {
126        Ok(read_ip_int(addr, max_len)? != 0)
127    }
128}
129
130impl ReadIpOption for IpTtl {
131    fn read_ip_option(addr: Vaddr, max_len: u32) -> Result<Self> {
132        let val = read_ip_int(addr, max_len)?;
133
134        let ttl_value = match val {
135            -1 => None,
136            1..=255 => Some(NonZeroU8::new(val as u8).unwrap()),
137            _ => return_errno_with_message!(Errno::EINVAL, "invalid ttl value"),
138        };
139
140        Ok(IpTtl::new(ttl_value))
141    }
142}

144// Reference: <https://elixir.bootlin.com/linux/v7.2.5/source/net/ipv4/ip_sockglue.c#L927-L936>.
145fn read_ip_int(addr: Vaddr, max_len: u32) -> Result<i32> {
146    if max_len >= size_of::<i32>() as u32 {
147        Ok(current_userspace!().read_val::<i32>(addr)?)
148    } else if max_len >= 1 {
149        Ok(current_userspace!().read_val::<u8>(addr)? as i32)
150    } else {
151        Ok(0)
152    }
