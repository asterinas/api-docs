16const TMPFS_MAGIC: u64 = 0x0102_1994;
17
18/// The temporary file system (tmpfs) structure.
19//
20// TODO: `TmpFs` currently aliases `RamFs` and relies on `RamFs::new_tmpfs()`
21// to create tmpfs-flavored ramfs instances. In the future we need to
22// implement a dedicated tmpfs with tmpfs-specific features such as memory
23// limits and swap support.
24pub(crate) type TmpFs = RamFs;
25
26impl TmpFs {
27    /// Creates a tmpfs filesystem instance.
28    pub(in crate::fs) fn new_tmpfs() -> Arc<Self> {
29        Self::new_tmpfs_backing("tmpfs", RevalidationPolicy::empty())
30    }
31
32    /// Creates a tmpfs-backed filesystem instance with a custom filesystem name
33    /// and directory-entry revalidation policy.
34    pub(in crate::fs) fn new_tmpfs_backing(
35        name: &'static str,
36        revalidation_policy: RevalidationPolicy,
37    ) -> Arc<Self> {
38        let anon_device_id = AnonDeviceId::acquire().expect("no device ID is available for tmpfs");
39        let sb = {
40            let mut super_block = SuperBlock::new(
41                TMPFS_MAGIC,
42                ramfs::BLOCK_SIZE,
43                ramfs::NAME_MAX,
44                anon_device_id.id(),
45            );
46            let max_blocks = default_max_blocks();
47            let max_inodes = default_max_inodes();
48            super_block.blocks = max_blocks;
49            super_block.bfree = max_blocks;
50            super_block.bavail = max_blocks;
51            super_block.files = max_inodes;
52            super_block.ffree = max_inodes;
53            super_block
54        };
55        Self::new_with_sb(name, anon_device_id, sb, revalidation_policy)
56    }
57}
58
59// FIXME: These defaults are only a rough approximation for tmpfs-over-ramfs.
60// A dedicated tmpfs implementation should replace them with real tmpfs limit
61// and accounting semantics.
62fn default_max_blocks() -> usize {
63    crate::vm::mem_total() / PAGE_SIZE / 2
64}
65fn default_max_inodes() -> usize {
66    crate::vm::mem_total() / PAGE_SIZE / 2
67}
68
69pub(super) struct TmpFsType;
70
71impl FsType for TmpFsType {
72    type Key = ();
73
74    fn name(&self) -> &'static str {
75        "tmpfs"
76    }
77
78    fn properties(&self) -> FsProperties {
79        FsProperties::empty()
80    }
81
82    fn create(&self, _fs_creation_ctx: &mut FsCreationCtx) -> Result<Arc<dyn FileSystem>> {
83        Ok(TmpFs::new_tmpfs())
84    }
85
86    fn sysnode(&self) -> Option<Arc<dyn aster_systree::SysNode>> {
87        None
