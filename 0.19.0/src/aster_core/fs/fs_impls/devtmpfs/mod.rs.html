3//! The device temporary filesystem.
4//!
5//! This module implements `devtmpfs` as a singleton filesystem backed by
6//! `tmpfs`. Device subsystems submit node-management requests through
7//! [`create_node`] and [`delete_node`]; the requests are serialized and handled
8//! by the dedicated kernel thread `devtmpfsd`. This gives node operations a
9//! single VFS execution context instead of mutating the filesystem tree directly
10//! from arbitrary device-registration contexts, and keeps these operations
11//! independent of the credentials of the device-registration caller, following
12//! Linux's devtmpfs design.
13//!
14//! Mounting policy follows the selected init path. If an initramfs init is
15//! selected, either by `rdinit=` or by the default `/init` lookup, the kernel
16//! does not mount devtmpfs automatically; the selected initramfs init program is
17//! responsible for mounting it if needed. Otherwise, Asterinas boots from the
18//! configured root filesystem and the kernel mounts this singleton on `/dev`
19//! during first-process initialization.

59    fn device_id(major: u16, minor: u32) -> DeviceId {
60        DeviceId::new(MajorId::new(major), MinorId::new(minor))
61    }
62
63    fn char_node(path: &'static str, major: u16, minor: u32) -> DevtmpfsNode {
64        DevtmpfsNode::new(
65            DeviceType::Char,
66            device_id(major, minor),
67            DevtmpfsNodeMeta::with_mode(path, mkmod!(a+rw)).unwrap(),
68        )
69    }
70
71    #[track_caller]
72    fn assert_missing(path: &str) {
73        match tree::lookup_path(path) {
74            Err(error) => assert_eq!(error.error(), Errno::ENOENT),
75            Ok(_) => panic!("path {path} should not exist"),
76        }
77    }
78
79    fn to_be_revalidated(inode: &dyn Inode) -> bool {
80        inode
81            .downcast_ref::<RamInode>()
82            .unwrap()
83            .to_be_revalidated()
84    }
85
86    #[ktest]
87    fn create_node_marks_ancestor_dirs_and_device_node() {
88        worker::init_for_ktest();
89
90        let path = "__devtmpfs_ktest_create/input/event0";
91        create_node(char_node(path, 240, 1)).unwrap();
92
93        let top_dir = tree::lookup_path("__devtmpfs_ktest_create").unwrap();
94        let input_dir = tree::lookup_path("__devtmpfs_ktest_create/input").unwrap();
95        let event_node = tree::lookup_path(path).unwrap();
96
97        assert_eq!(top_dir.type_(), InodeType::Dir);
98        assert_eq!(input_dir.type_(), InodeType::Dir);
99        assert_eq!(event_node.type_(), InodeType::CharDevice);
100        assert!(to_be_revalidated(top_dir.as_ref()));
101        assert!(to_be_revalidated(input_dir.as_ref()));
102        assert!(to_be_revalidated(event_node.as_ref()));
103
104        let metadata = event_node.metadata().unwrap();
105        assert_eq!(metadata.self_dev_id, Some(device_id(240, 1)));
106        assert_eq!(metadata.mode.bits(), mkmod!(a+rw).bits());
107
108        delete_node(char_node(path, 240, 1)).unwrap();
109        assert_missing("__devtmpfs_ktest_create");
110    }

112    #[ktest]
113    fn delete_node_keeps_nonempty_parent_dir() {
114        worker::init_for_ktest();
115
116        let node0 = char_node("__devtmpfs_ktest_nonempty/input/event0", 240, 3);
117        let node1 = char_node("__devtmpfs_ktest_nonempty/input/event1", 240, 4);
118        create_node(node0).unwrap();
119        create_node(node1).unwrap();
120
121        delete_node(char_node("__devtmpfs_ktest_nonempty/input/event0", 240, 3)).unwrap();
122        assert_missing("__devtmpfs_ktest_nonempty/input/event0");
123        assert!(tree::lookup_path("__devtmpfs_ktest_nonempty/input").is_ok());
124        assert!(tree::lookup_path("__devtmpfs_ktest_nonempty/input/event1").is_ok());
125
126        delete_node(char_node("__devtmpfs_ktest_nonempty/input/event1", 240, 4)).unwrap();
127        assert_missing("__devtmpfs_ktest_nonempty");
128    }

130    #[ktest]
131    fn delete_node_keeps_unmarked_node() {
132        worker::init_for_ktest();
133
134        let path = "__devtmpfs_ktest_unmarked";
135        let root = singleton().root_inode();
136        let root_dentry = crate::fs::vfs::path::Dentry::new_root(root.clone());
137        root.mknod(
138            &root_dentry,
139            path,
140            mkmod!(a+rw),
141            MknodType::CharDevice(device_id(240, 5).as_encoded_u64()),
142        )
143        .unwrap();
144        let inode = root.lookup(path).unwrap();
145        assert!(!to_be_revalidated(inode.as_ref()));
146
147        delete_node(char_node(path, 240, 5)).unwrap();
148
149        let inode = root.lookup(path).unwrap();
150        assert!(!to_be_revalidated(inode.as_ref()));
151        let inode_dentry = root_dentry
152            .as_dir_dentry_or_err()
153            .unwrap()
154            .lookup_child(path)
155            .unwrap();
156        root.unlink(&inode_dentry).unwrap();
157    }
