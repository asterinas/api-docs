3//! Request queue and kernel thread for serialized devtmpfs node operations.
4
5use ostd::sync::{Waiter, Waker};
6use spin::Once;
7
8use super::{DevtmpfsNode, tree};
9use crate::{prelude::*, thread::kernel_thread::ThreadOptions};

11/// Creates a device node through `devtmpfsd`.
12///
13/// The request is queued to the dedicated devtmpfs kernel thread and this
14/// function waits until the node has been created or the creation fails.
15pub(crate) fn create_node(node: DevtmpfsNode) -> Result<()> {
16    submit(Request::CreateNode(node))
17}
18
19/// Deletes a device node through `devtmpfsd`.
20///
21/// The request is queued to the dedicated devtmpfs kernel thread and this
22/// function waits until the deletion has completed or failed. The deletion only
23/// unlinks nodes that were created by `devtmpfsd` and still match the requested
24/// device type and device ID.
25pub(crate) fn delete_node(node: DevtmpfsNode) -> Result<()> {
26    submit(Request::DeleteNode(node))
27}
28
29pub(super) fn init_in_first_kthread() {
30    ThreadOptions::new(devtmpfsd).spawn();
31}
32
33fn submit(request: Request) -> Result<()> {
34    let (waiter, waker) = Waiter::new_pair();
35    let request = Arc::new(PendingRequest::new(request, waker));
36
37    REQUEST_QUEUE.requests.lock().push_back(request.clone());
38    if let Some(waker) = REQUEST_QUEUE.waker.get() {
39        waker.wake_up();
40    } else {
41        // `devtmpfsd` will check the requests after initializing the waker.
42    }
43
44    waiter.wait();
45
46    // `result` is guaranteed to be available after `waiter.wait()`.
47    *request.result.get().unwrap()
48}
49
50fn devtmpfsd() {
51    let (waiter, waker) = Waiter::new_pair();
52
53    REQUEST_QUEUE.waker.call_once(|| waker);
54
55    loop {
56        let request = REQUEST_QUEUE.requests.lock().pop_front();
57        let Some(request) = request else {
58            waiter.wait();
59            continue;
60        };
61
62        let result = match &request.request {
63            Request::CreateNode(node) => tree::create_node(node),
64            Request::DeleteNode(node) => tree::delete_node(node),
65        };
66        request.result.call_once(|| result);
67        request.waker.wake_up();
68    }
69}
70
71struct RequestQueue {
72    requests: SpinLock<VecDeque<Arc<PendingRequest>>>,
73    waker: Once<Arc<Waker>>,
74}
75
76static REQUEST_QUEUE: RequestQueue = RequestQueue {
77    requests: SpinLock::new(VecDeque::new()),
78    waker: Once::new(),
79};
80
81struct PendingRequest {
82    request: Request,
83    result: Once<Result<()>>,
84    waker: Arc<Waker>,
85}
86
87impl PendingRequest {
88    fn new(request: Request, waker: Arc<Waker>) -> Self {
89        Self {
90            request,
91            result: Once::new(),
92            waker,
93        }
94    }
95}
96
97enum Request {
98    CreateNode(DevtmpfsNode),
99    DeleteNode(DevtmpfsNode),
100}
