3//! The logical overlay inode and its VFS trait surface.
4//!
5//! [`OverlayInode`] is the published logical inode shared by every name bound
6//! to the same overlay object. It owns the per-object real-object facts, the
7//! per-inode transaction lock, the one-way name-taken latch, and the
8//! precomputed published identity.
9//!
10//! # Module map
11//!
12//! | Submodule | Responsibility |
13//! |---|---|
14//! | [`copyup`] | the per-object promotion rounds and the temp each round publishes |
15//! | [`data`] | the data path: read, write, seek, resize, sync, and the page cache |
16//! | [`dir`] | the create, link, remove, and rename namespace mutations, and the whiteout publication |
17//! | [`identity`] | the dev/ino identity translation and the durable origin record |
18//! | [`inode_cache`] | the mount-wide real-id inode reuse cache |
19//! | [`lookup`] | upper-first name resolution and inode projection |
20//! | [`metadata`] | the metadata setters |
21//! | [`open`] | the open entry and the per-open directory handle |
22//! | [`readdir`] | the merged-directory merge and its snapshot construction |
23//! | [`xattr`] | the private-record and passthrough name paths |
24//!
25//! # Locking
26//!
27//! [`OverlayInode::lock`] is the per-inode transaction lock that serializes one
28//! object's mutations. For a directory its payload is that directory's readdir
29//! snapshot — a [`readdir::ReaddirCache`], the merged entry table one merge
30//! produced; a non-directory uses the lock as a plain serialization token. The
31//! lock also serializes an object's promotion against the removal or
32//! displacement of its name.
33//! [`OverlayInode::append_write`] holds this lock across the underlying `size()` +
34//! `write_at` so concurrent appends serialize on the post-write size.
35//!
36//! The name-taken latch is written — by a removal, or by a rename that displaces
37//! the name — and read only inside that object's transaction lock; that lock is
38//! what orders a name removal against a promotion round, so the latch needs no
39//! ordering of its own.
40//!
41//! Two overlay-internal locks exist: the per-inode transaction lock and the
42//! [`InodeCache`] guard. A per-open snapshot slot sits outside both of them and
43//! is the outer lock of its pair: the directory transaction lock is taken and
44//! released inside its guard.

92pub(super) struct OverlayInode {
93    fs: Weak<OverlayFs>,
94    /// A directory keeps its whole merge stack; a non-directory only its visible topmost lower.
95    lowers: Vec<RealObject>,
96    upper: Once<RealObject>,
97    /// One-way name-taken latch, set under this object's lock: its name binding was taken away.
98    name_taken: AtomicBool,
99    object_id: ObjectVisibleId,
100    /// The current readdir snapshot slot; `None` means it is not built or was invalidated.
101    lock: Mutex<OverlayInodeLockPayload>,
102    extension: Extension,
103    this: Weak<OverlayInode>,
104}

106impl OverlayInode {
107    /// Two directories of one mount publish the same pair exactly when they are one real directory.
108    fn same_directory_as(&self, other: &OverlayInode) -> bool {
109        debug_assert!(self.type_().is_directory() && other.type_().is_directory());
110        self.object_id == other.object_id
111    }
112
113    /// Returns the read-only take point: the real object this logical object currently shows.
114    ///
115    /// It is the upper when there is one, else the topmost lower, and reads go through the returned
116    /// object's real inode. The object may be a lower: handing it to a write method breaks the write
117    /// paths' discipline, which the type does not prevent.
118    fn real_object(&self) -> &RealObject {
119        match self.upper.get() {
120            Some(upper) => upper,
121            // A real-object stack is never empty, so a lower-only object has a topmost lower.
122            None => self
123                .lowers
124                .first()
125                .expect("a real-object stack is never empty"),
126        }
127    }
128
129    /// Returns the real object a write may land on.
130    ///
131    /// Refuses an effectively read-only mount, returns the receiver's own upper when it has one, and
132    /// otherwise promotes the receiver into `copyup_dentry` first — so the answer is always an upper.
133    fn writable_real_object(&self, copyup_dentry: &Dentry) -> Result<&RealObject> {
134        let fs = self.fs_arc();
135        if fs.policy().is_effective_read_only() {
136            return_errno_with_message!(Errno::EROFS, "the overlay mount is read-only");
137        }
138        if let Some(upper) = self.upper.get() {
139            return Ok(upper);
140        }
141        fs.copy_up_at(copyup_dentry)?;
142        Ok(self.writable_upper())
143    }
144
145    /// Sets the name-taken latch; the caller must hold this object's transaction lock.
146    fn mark_name_taken(&self) {
147        // Only the transaction lock orders this against `is_name_taken`: no lock-free read exists.
148        self.name_taken.store(true, Ordering::Relaxed);
149    }
150
151    /// Reads the name-taken latch; the caller must hold this object's transaction lock.
152    fn is_name_taken(&self) -> bool {
153        self.name_taken.load(Ordering::Relaxed)
154    }
155
156    /// Returns the receiver's own upper real object.
157    ///
158    /// The caller must establish that the receiver has an upper; a write path that has not promoted the
159    /// receiver should enter through `writable_real_object` instead.
160    fn writable_upper(&self) -> &RealObject {
161        self.upper
162            .get()
163            .expect("an overlay object on a write path has an upper real object")
164    }
165
166    /// Takes the per-inode transaction lock.
167    fn lock(&self) -> OverlayInodeLockGuard<'_> {
168        Mutex::lock(&self.lock)
169    }
170
171    fn fs_arc(&self) -> Arc<OverlayFs> {
172        self.fs
173            .upgrade()
174            .expect("the owning fs outlives every overlay inode")
175    }
176
177    fn self_arc(&self) -> Arc<OverlayInode> {
178        self.this
179            .upgrade()
180            .expect("the overlay inode is still owned by an Arc")
181    }
182}

184/// One create-family request; `mode` rides the op and the borrowed fields pass the VFS entry on.
185#[derive(Clone, Copy)]
186enum CreateOp<'a> {
187    General {
188        kind: InodeType,
189        mode: InodeMode,
190    },
191    Symlink {
192        target: &'a str,
193        mode: InodeMode,
194    },
195    Mknod {
196        mode: InodeMode,
197        node: &'a MknodType,
198    },
199}
200
201impl<'a> CreateOp<'a> {
202    fn general(kind: InodeType, mode: InodeMode) -> Self {
203        debug_assert!(
204            kind != InodeType::SymLink,
205            "overlay symlinks are created only through create_symlink",
206        );
207        Self::General { kind, mode }
208    }
209
210    fn symlink(target: &'a str, mode: InodeMode) -> Self {
211        Self::Symlink { target, mode }
212    }
213
214    fn mknod(node: &'a MknodType, mode: InodeMode) -> Result<Self> {
215        // The read side reads a raw `0:0` char device back as a whiteout, so a caller must not
216        // be able to create one and forge the name-level shadow this mount publishes itself.
217        if matches!(node, MknodType::CharDevice(0)) {
218            return_errno_with_message!(
219                Errno::EPERM,
220                "a raw 0:0 whiteout char device must not be user-creatable"
221            );
222        }
223        Ok(Self::Mknod { mode, node })
224    }
225
226    /// Returns the inode type this request creates.
227    fn object_type(&self) -> InodeType {
228        match self {
229            Self::General { kind, .. } => *kind,
230            Self::Symlink { .. } => InodeType::SymLink,
231            Self::Mknod { node, .. } => match *node {
232                MknodType::NamedPipe => InodeType::NamedPipe,
233                MknodType::CharDevice(_) => InodeType::CharDevice,
234                MknodType::BlockDevice(_) => InodeType::BlockDevice,
235            },
236        }
237    }
238
239    /// Runs the one real create call this request selects, in `dir`, under `name`.
240    fn create_child_in(&self, dir: &Dentry, name: &str) -> Result<Arc<Dentry>> {
241        let dir = dir.as_dir_dentry_or_err()?;
242        match self {
243            Self::General { kind, mode } => {
244                dir.create_child(name, || dir.inode().create(&dir, name, *kind, *mode))
245            }
246            Self::Symlink { target, mode } => dir.create_child(name, || {
247                dir.inode().create_symlink(&dir, name, target, *mode)
248            }),
249            Self::Mknod { mode, node } => dir.mknod(
250                name,
251                *mode,
252                match *node {
253                    MknodType::NamedPipe => MknodType::NamedPipe,
254                    MknodType::CharDevice(device) => MknodType::CharDevice(*device),
255                    MknodType::BlockDevice(device) => MknodType::BlockDevice(*device),
256                },
257            ),
258        }
259    }
260}

326    fn lookup(&self, name: &str) -> Result<Arc<dyn Inode>> {
327        // The guard is still taken here: it orders this lookup against this directory's mutations.
328        let _dir_guard = self.lock();
329        // The directory bit comes from the real object alone; the payload says nothing about it.
330        if !self.type_().is_directory() {
331            return Err(Error::with_message(
332                Errno::ENOTDIR,
333                "lookup is supported on overlay directories only",
334            ));
335        }
336        let fs = self.fs.upgrade().ok_or_else(|| {
337            Error::with_message(Errno::EIO, "the overlay mount is no longer alive")
338        })?;
339        match fs.lookup(self, name)? {
340            Lookup::Positive(inode) => Ok(inode),
341            Lookup::Negative(_) => Err(Error::new(Errno::ENOENT)),
342        }
343    }
