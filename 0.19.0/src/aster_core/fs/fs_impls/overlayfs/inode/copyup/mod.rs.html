3//! The copy-up authority: per-object coordination, winner/waiter trigger,
4//! and object-kind promotion.
5//!
6//! Key concepts:
7//! - **copy-up chain**: the lower-only objects from the operation's target up
8//!   to the first upper-backed ancestor, collected from the overlay dentry
9//!   parent chain. That ancestor is the **publication parent**: it publishes
10//!   the object below it instead of being promoted, and each round's promoted
11//!   object becomes the next round's publication parent. Every collected
12//!   element carries the name it publishes under its publication parent.
13//! - **winner/waiter**: the tasks racing on one object's per-inode transaction
14//!   lock — for a non-directory that lock is a plain serialization token,
15//!   because the sole published-state fact is `upper: Once` being set. The
16//!   winner performs the promotion; the waiters re-observe `upper` under that
17//!   lock and skip.
18//! - **promotion round**: one step promoting one object, from the `upper`
19//!   re-check through commit.
20//!
21//! [`OverlayFs::copy_up_at`] is the promotion entry that drives these rounds.
22//!
23//! # Module map
24//!
25//! | Submodule | Responsibility |
26//! |---|---|
27//! | [`workdir`] | the workdir temp create, publish, and cleanup lifecycle |
28//!
29//! # Locking
30//!
31//! A copy-up collects its chain from the VFS dcache alone — reading the dentry's
32//! parent and name are NameAndParent leaf reads. Each promotion round then nests in
33//! this order — the publication parent's transaction lock, then the promoted
34//! object's transaction lock — and releases both at the round's end, so every lock
35//! edge runs ancestor-to-descendant.
36//!
37//! Both name-taken latches of the round are read under those two locks and before
38//! anything is staged, so a name taken away aborts the round with nothing staged.
39//! The winner holds the promoted object's lock continuously from the `upper`
40//! re-check through the commit tail, and the publication parent's lock across that
41//! same span.
42//!
43//! Below the two transaction locks a round takes the commit tail's cache guard,
44//! because the staging and publication calls are real-layer calls. That guard is
45//! the round's single `InodeCache` touch: the commit tail rekeys the promoted
46//! instance onto the key its identity now resolves to, and the round's own latch
47//! reads answer whether its coordinate still denotes it.

68impl OverlayFs {
69    /// Promotes every lower-only ancestor above `dentry`, then the object behind it.
70    pub(super) fn copy_up_at(&self, dentry: &Dentry) -> Result<()> {
71        let promoted = Arc::downcast::<OverlayInode>(dentry.inode().clone())
72            .expect("the copy-up operation dentry is not an overlay inode");
73
74        // The dentry chain is walked without an overlay lock; the name is bound here as well.
75        let mut chain: Vec<(Arc<OverlayInode>, String)> = Vec::new();
76        let mut object = promoted;
77        let mut name = dentry.name();
78        let mut parent_dentry = dentry.parent();
79        let mut publication_parent = loop {
80            let Some(parent) = parent_dentry else {
81                return Err(Error::with_message(
82                    Errno::EIO,
83                    "the copy-up operation dentry has no overlay parent",
84                ));
85            };
86            let parent_object = Arc::downcast::<OverlayInode>(parent.inode().clone())
87                .expect("the copy-up publication parent is not an overlay inode");
88            chain.push((object, name));
89            // The first upper-backed ancestor is the boundary: it publishes, it is not promoted.
90            if parent_object.upper.get().is_some() {
91                break parent_object;
92            }
93            object = parent_object;
94            name = parent.name();
95            parent_dentry = parent.parent();
96        };
97
98        // Top-down: each round publishes into the object promoted by the round above it.
99        for (promoted_object, name) in chain.iter().rev() {
100            self.promote_level(&publication_parent, promoted_object, name)?;
101            publication_parent = Arc::clone(promoted_object);
102        }
103        Ok(())
104    }

106    /// Promotes a rename's target in one round under the publication parent the caller has already
107    /// promoted; this entry is the target's own take point and carries the read-only gate.
108    pub(super) fn promote_rename_target(
109        &self,
110        publication_parent: &Arc<OverlayInode>,
111        promoted_object: &Arc<OverlayInode>,
112        name: &str,
113    ) -> Result<()> {
114        if self.policy().is_effective_read_only() {
115            return_errno_with_message!(Errno::EROFS, "the overlay mount is read-only");
116        }
117        self.promote_level(publication_parent, promoted_object, name)
118    }

120    /// Promotes one level of a copy-up chain under the publication parent's lock, then this object's.
121    fn promote_level(
122        &self,
123        publication_parent: &Arc<OverlayInode>,
124        promoted_object: &Arc<OverlayInode>,
125        name: &str,
126    ) -> Result<()> {
127        // The publication parent's lock precedes the object's: never a child-to-parent edge.
128        let _parent_guard = publication_parent.lock();
129        let _object_guard = promoted_object.lock();
130
131        // Both latches are read under both locks; a set latch fails the round before it stages.
132        if promoted_object.is_name_taken() {
133            return Err(Error::with_message(
134                Errno::ENOENT,
135                "the copy-up target name was taken away before the promotion round",
136            ));
137        }
138        if publication_parent.is_name_taken() {
139            return Err(Error::with_message(
140                Errno::ENOENT,
141                "the copy-up publication parent is no longer reachable",
142            ));
143        }
144
145        // Resolved before staging so a failed probe can never leave a workdir temp behind.
146        let upper_parent = publication_parent.writable_upper();
147
148        // The upper entry at this name decides the round: our own published object is a no-op, and
149        // any other entry is a name this round must not publish over.
150        let upper_dir = upper_parent.dentry().as_dir_dentry_or_err()?;
151        let existing = match upper_dir.lookup_child(name) {
152            Ok(child) => Some(child),
153            Err(err) if err.error() == Errno::ENOENT => None,
154            Err(err) => return Err(err),
155        };
156        match (&existing, promoted_object.upper.get()) {
157            (Some(child), Some(ours)) if Arc::ptr_eq(child.inode(), ours.real_inode()) => {
158                return Ok(());
159            }
160            (None, Some(_)) => return Ok(()),
161            (Some(_), _) => {
162                return Err(Error::with_message(
163                    Errno::ESTALE,
164                    "the promotion name is already held in the upper",
165                ));
166            }
167            (None, None) => {}
168        }
169
170        let source = promoted_object.real_object();
171        let staged = promoted_object.stage_in_workdir(name)?;
172        // An early `?` here drops the temp unconsumed, so its guard removes it from the workspace.
173        let origin_record_landed = promoted_object.stage_temp_contents(&staged, source)?;
174        self.commit_copyup_publication(
175            promoted_object,
176            name,
177            upper_parent.dentry(),
178            staged,
179            origin_record_landed,
180        )?;
181        Ok(())
182    }
183
184    /// Commits one staged temp as the published copy-up object.
185    fn commit_copyup_publication(
186        &self,
187        promoted_object: &Arc<OverlayInode>,
188        name: &str,
189        upper_parent: &Arc<Dentry>,
190        staged: WorkdirTemp,
191        origin_record_landed: bool,
192    ) -> Result<()> {
193        // The impure marker is best-effort: a failure leaves a missing mirror, not an abort.
194        let namespace = self.policy().xattr_namespace();
195        if let Ok(false) = OverlayXattrType::Impure.is_positive_on(upper_parent, namespace) {
196            let _ = OverlayXattrType::Impure.set_value_on(upper_parent, namespace, None);
197        }
198        let published = staged.dentry().clone();
199        staged.publish(upper_parent, name, RenameMode::Replace)?;
200        let published_upper = RealObject::new_upper(published);
201        // Without a durable record the identity is the upper's own pair, so the entry keys on it.
202        let upper_key = self.identity().real_id_of(&published_upper);
203        promoted_object.upper.call_once(|| published_upper);
204        // A record the round did not land leaves the instance's identity on its own upper.
205        if !origin_record_landed
206            && !promoted_object.type_().is_directory()
207            && let Some(lower) = promoted_object.lowers.first()
208        {
209            self.inodes().rekey(
210                self.identity().real_id_of(lower),
211                upper_key,
212                promoted_object,
213            );
214        }
215        Ok(())
216    }
217}

219impl OverlayInode {
220    /// Creates this object's staged temp: an empty real object in the workdir that mirrors this
221    /// object's kind, its mode, and, for a symlink, its target.
222    fn stage_in_workdir(&self, name: &str) -> Result<WorkdirTemp> {
223        let fs = self.fs_arc();
224        let upper_workdir = fs.upper_workdir_inuse();
225        let source = self.real_object();
226        let temp = match source.real_inode().type_() {
227            InodeType::Dir => {
228                // Atomic `Replace` replaces a stale upper entry instead of failing with `EEXIST`.
229                let mode = source.real_inode().mode()?;
230                upper_workdir.create_workdir_temp(
231                    name,
232                    CreateOp::General {
233                        kind: InodeType::Dir,
234                        mode,
235                    },
236                )?
237            }
238            InodeType::File => {
239                // Data is synced before the atomic rename, so the published object is durable.
240                let mode = source.real_inode().mode()?;
241                upper_workdir.create_workdir_temp(
242                    name,
243                    CreateOp::General {
244                        kind: InodeType::File,
245                        mode,
246                    },
247                )?
248            }
249            InodeType::SymLink => {
250                // The symlink is created with its target; the target object itself is not copied.
251                let mode = source.real_inode().mode()?;
252                let target = match source.real_inode().read_link()? {
253                    SymbolicLink::Plain(target) => target,
254                    SymbolicLink::Path(_) => {
255                        return_errno_with_message!(
256                            Errno::EOPNOTSUPP,
257                            "a path-style symlink target cannot be copied up"
258                        );
259                    }
260                };
261                upper_workdir.create_workdir_temp(
262                    name,
263                    CreateOp::Symlink {
264                        target: &target,
265                        mode,
266                    },
267                )?
268            }
269            InodeType::CharDevice
270            | InodeType::BlockDevice
271            | InodeType::NamedPipe
272            | InodeType::Socket => {
273                let mknod_type = match source.real_inode().type_() {
274                    InodeType::NamedPipe => MknodType::NamedPipe,
275                    InodeType::CharDevice => {
276                        let rdev = source
277                            .real_inode()
278                            .metadata()?
279                            .self_dev_id
280                            .ok_or_else(|| {
281                                Error::with_message(
282                                    Errno::EINVAL,
283                                    "the lower char device has no device id",
284                                )
285                            })?
286                            .as_encoded_u64();
287                        MknodType::CharDevice(rdev)
288                    }
289                    InodeType::BlockDevice => {
290                        let rdev = source
291                            .real_inode()
292                            .metadata()?
293                            .self_dev_id
294                            .ok_or_else(|| {
295                                Error::with_message(
296                                    Errno::EINVAL,
297                                    "the lower block device has no device id",
298                                )
299                            })?
300                            .as_encoded_u64();
301                        MknodType::BlockDevice(rdev)
302                    }
303                    _ => {
304                        return Err(Error::with_message(
305                            Errno::EOPNOTSUPP,
306                            "socket nodes cannot be copied up",
307                        ));
308                    }
309                };
310                let mode = source.real_inode().mode()?;
311                upper_workdir.create_workdir_temp(
312                    name,
313                    CreateOp::Mknod {
314                        mode,
315                        node: &mknod_type,
316                    },
317                )?
318            }
319            InodeType::Unknown => {
320                return Err(Error::with_message(
321                    Errno::EINVAL,
322                    "cannot promote an overlay object of unknown type",
323                ));
324            }
325        };
326        Ok(temp)
327    }

329    /// Fills one staged temp from `source` and reports whether the origin-record step landed; `source`
330    /// is the source-side object, while the temp and its dentry are the destination of the copy.
331    fn stage_temp_contents(&self, temp: &WorkdirTemp, source: &RealObject) -> Result<bool> {
332        source.copy_metadata_to(temp.dentry())?;
333
334        self.copy_eligible_xattrs(source, temp, false)?;
335
336        if temp.inode().type_().is_regular_file() {
337            source.copy_data_to(temp.dentry())?;
338        }
339
340        // A hard-linked name keys two names alike; a denying mount cannot hold one.
341        let origin_record_landed = if source.real_inode().type_().is_directory()
342            || source.real_inode().metadata()?.nr_hard_links <= 1
343        {
344            self.fs_arc().record_origin(temp, source)?
345        } else {
346            false
347        };
348
349        source.copy_timestamps_to(temp.dentry())?;
350
351        if temp.inode().type_().is_regular_file() {
352            temp.inode().sync(SyncMode::Full)?;
353        }
354
355        Ok(origin_record_landed)
356    }
