3//! The workdir temporary lifecycle.
4//!
5//! [`WorkdirTemp`] is the guard of one staged object: it holds the name the object was created
6//! under, its dentry, an owned handle of the workspace it lives in, and whether an outlet has
7//! already consumed it. Two consuming outlets discharge the guard — [`WorkdirTemp::publish`]
8//! renames the object into the upper, and [`WorkdirTemp::into_whiteout`] detaches it into a
9//! whiteout handle the workspace never cleans up — and [`Drop`](WorkdirTemp) is the backstop for
10//! an early `?` that took neither, removing the temp. Because the workspace handle is owned and
11//! has no accessor, those outlets are the only paths that move a guarded name.
12//!
13//! [`UpperWorkdirInuse::create_workdir_temp`] and
14//! [`UpperWorkdirInuse::create_workdir_link_temp`] each generate one counter-derived name and
15//! perform one explicit creation; the naming itself belongs to the owner
16//! ([`UpperWorkdirInuse::next_temp_name`]).
17//!
18//! Invariants: the workspace is pinned at mount time and lives outside every layer root; workdir
19//! temps are never visible entries of any layer.
20
21use crate::{
22    fs::{
23        fs_impls::overlayfs::{fs::mount::inuse::UpperWorkdirInuse, inode::CreateOp},
24        vfs::{
25            inode::{Inode, RenameMode},
26            path::Dentry,
27        },
28    },
29    prelude::*,
30};

32/// The guard of one staged object.
33///
34/// It holds the name the object was created under, its dentry, an owned handle of the workspace it
35/// lives in, and whether an outlet has already consumed it.
36pub(in super::super) struct WorkdirTemp {
37    name: String,
38    dentry: Arc<Dentry>,
39    workspace: Arc<Dentry>,
40    consumed: bool,
41}
42
43impl WorkdirTemp {
44    pub(in super::super) fn name(&self) -> &str {
45        &self.name
46    }
47
48    pub(in super::super) fn inode(&self) -> &Arc<dyn Inode> {
49        self.dentry.inode()
50    }
51
52    /// The temp's own dentry, paired with [`Self::inode`] for real-layer setter calls.
53    pub(in super::super) fn dentry(&self) -> &Arc<Dentry> {
54        &self.dentry
55    }
56
57    /// Marks this guard consumed without touching the workspace entry.
58    ///
59    /// The outlet that detaches the staged object (a shared whiteout outlives the guard) calls
60    /// this, so `Drop` leaves the entry where it is.
61    pub(in super::super) fn mark_consumed(&mut self) {
62        self.consumed = true;
63    }
64
65    /// Publishes this staged temp into `upper_parent` under `name`, consuming the guard.
66    ///
67    /// A successful rename moves the entry out of the workspace, and a failed one leaves the
68    /// guard unconsumed so `Drop` removes the temp — the same best-effort cleanup the callers
69    /// used to write by hand.
70    pub(in super::super) fn publish(
71        mut self,
72        upper_parent: &Arc<Dentry>,
73        name: &str,
74        mode: RenameMode,
75    ) -> Result<()> {
76        self.workspace.as_dir_dentry_or_err()?.rename(
77            &self.name,
78            &upper_parent.as_dir_dentry_or_err()?,
79            name,
80            mode,
81        )?;
82        self.consumed = true;
83        Ok(())
84    }
85}
86
87impl Drop for WorkdirTemp {
88    /// Removes a temp that was neither published nor detached.
89    fn drop(&mut self) {
90        if self.consumed {
91            return;
92        }
93        self.consumed = true;
94        let dir = match self.workspace.as_dir_dentry_or_err() {
95            Ok(dir) => dir,
96            Err(_) => return,
97        };
98        let _ = if self.dentry.inode().type_().is_directory() {
99            dir.rmdir(&self.name)
100        } else {
101            dir.unlink(&self.name)
102        };
103    }
104}
105
106impl UpperWorkdirInuse {
107    /// Creates one staged temp under a counter-derived name with the real call `op` selects.
108    pub(in super::super) fn create_workdir_temp(
109        &self,
110        target_name: &str,
111        op: CreateOp<'_>,
112    ) -> Result<WorkdirTemp> {
113        let workspace = self.workdir_workspace()?;
114        let name = self.next_temp_name(target_name);
115        let dentry = op.create_child_in(workspace, &name)?;
116        Ok(WorkdirTemp {
117            name,
118            dentry,
119            workspace: Arc::clone(workspace),
120            consumed: false,
121        })
122    }
123
124    /// Creates one staged temp that is a hard link to `source`.
125    pub(in super::super) fn create_workdir_link_temp(
126        &self,
127        target_name: &str,
128        source: &Arc<Dentry>,
129    ) -> Result<WorkdirTemp> {
130        let workspace = self.workdir_workspace()?;
131        let name = self.next_temp_name(target_name);
132        let dir = workspace.as_dir_dentry_or_err()?;
133        dir.link(source, &name)?;
134        let dentry = dir.lookup_child(&name)?;
135        Ok(WorkdirTemp {
136            name,
137            dentry,
138            workspace: Arc::clone(workspace),
139            consumed: false,
140        })
141    }
142}
