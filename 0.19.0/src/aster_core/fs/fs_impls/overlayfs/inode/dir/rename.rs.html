3//! The rename recipe: the one namespace mutation this module owns, entered only
4//! through [`OverlayInode::rename_impl`].
5//!
6//! A rename arrives as a request on the merged view, while every name it changes
7//! lives in the upper alone. The source name may still be held by a lower layer
8//! after the source leaves it, and the target name is one of three things:
9//! nothing the merged view can see, a name an upper whiteout keeps negative, or a
10//! visible object that this move overwrites or exchanges.
11//!
12//! Everything the move touches has to exist in the upper before the real rename
13//! can express it, so the objects involved are promoted first and the rename then
14//! runs on the upper alone. The upper has to hold the state the merged view
15//! expects afterwards: an upper whiteout at the old name wherever a lower layer
16//! would otherwise keep that name alive, a whiteout carried away from the target
17//! name by the exchange that takes a name there, and an opaque record on a moved
18//! directory whose new name meets a lower directory.
19

36/// Records how the old name's visibility barrier is settled by the move, and what becomes of a
37/// whiteout standing at the target name.
38#[derive(Clone, Copy, Debug, Eq, PartialEq)]
39enum WhiteoutAction {
40    /// Sets no new barrier: no lower entry hides there, or the exchange keeps the name visible.
41    None,
42    /// Creates a fresh whiteout at the old name, hiding the lower entry the source leaves behind.
43    Create,
44    /// Exchanges the target's whiteout onto the old name, where it serves as the barrier.
45    Exchange,
46    /// Exchanges the target's whiteout onto the old name and deletes it: nothing hides below.
47    ExchangeThenDelete,
48}
49
50/// Classifies the target name once for one move: the merged view's answer for that name
51/// crossed with what the upper physically holds there.
52enum RenameTarget {
53    /// Applies where the name has no upper entry, a lower whiteout hiding it landing here too.
54    Absent,
55    /// Applies where an upper whiteout alone holds the name, invisible yet physically taken.
56    UpperWhiteout,
57    /// Applies where the name resolves in the merged view, to a directory or to anything else.
58    Visible {
59        /// Holds the object the name resolves to, which a move that overwrites this name displaces.
60        inode: Arc<OverlayInode>,
61    },
62}
63
64/// Holds one rename's whole input, settled in a single pass before the lock set is taken.
65struct RenameFacts<'x> {
66    /// Reports whether a lower layer holds the new name as a directory.
67    lower_layer_holds_new_name_as_directory: bool,
68    /// Holds the mode the real rename is issued with: the physical form of the caller's request.
69    mode: RenameMode,
70    whiteout: WhiteoutAction,
71    new_name: &'x str,
72    old_parent: Arc<OverlayInode>,
73    old_parent_dentry: Arc<Dentry>,
74    new_parent: Arc<OverlayInode>,
75    new_parent_dentry: &'x Dentry,
76    source: Arc<OverlayInode>,
77    source_dentry: &'x Dentry,
78    target: RenameTarget,
79}

81impl OverlayInode {
82    /// Serves as the rename entry the VFS calls: it freezes the facts of both names, decides the
83    /// move they allow, and performs it.
84    pub(in super::super) fn rename_impl(
85        &self,
86        old_child_dentry: &Dentry,
87        new_dir_dentry: &Dentry,
88        new_name: &str,
89        replaced_inode: Option<&Arc<dyn Inode>>,
90        mode: RenameMode,
91    ) -> Result<()> {
92        let facts = RenameFacts::classify(
93            self.self_arc(),
94            old_child_dentry,
95            new_dir_dentry,
96            new_name,
97            replaced_inode,
98            mode,
99        )?;
100        facts.probe_target_empty_early()?;
101
102        facts.promote_members()?;
103        let Some(mut locks) = RenameLocks::acquire(&facts)? else {
104            return_errno!(Errno::ESTALE)
105        };
106
107        facts.sweep_replaced_whiteouts_if_needed(&mut locks)?;
108        facts.set_opaque_marker_if_needed(&mut locks)?;
109        facts.set_impure_marker_if_needed()?;
110        facts.commit(&mut locks)
111    }
112}

114impl<'x> RenameFacts<'x> {
115    /// Builds the facts of one rename — the target name classified once and the two answers the
116    /// lower layers give — together with the plan they decide.
117    fn classify(
118        old_parent: Arc<OverlayInode>,
119        old_child_dentry: &'x Dentry,
120        new_dir_dentry: &'x Dentry,
121        new_name: &'x str,
122        replaced_inode: Option<&'x Arc<dyn Inode>>,
123        request: RenameMode,
124    ) -> Result<Self> {
125        let old_parent_dentry = old_child_dentry
126            .parent()
127            .expect("the renamed child has no parent dentry");
128        let source = Arc::downcast::<OverlayInode>(old_child_dentry.inode().clone())
129            .expect("the rename source is not an overlay inode");
130        let new_parent = Arc::downcast::<OverlayInode>(new_dir_dentry.inode().clone())
131            .expect("the rename target parent is not an overlay inode");
132        let fs = old_parent.fs_arc();
133        let handed_inode = replaced_inode.map(|inode| {
134            Arc::downcast::<OverlayInode>((*inode).clone())
135                .expect("the rename target is not an overlay inode")
136        });
137        let target = match &handed_inode {
138            Some(inode) => RenameTarget::Visible {
139                inode: inode.clone(),
140            },
141            None => {
142                // The name the VFS found free is held by an upper whiteout or by nothing at all.
143                let upper_dir = new_parent
144                    .upper
145                    .get()
146                    .map(|upper| upper.dentry().as_dir_dentry_or_err())
147                    .transpose()?;
148                let upper_holds_whiteout = match upper_dir {
149                    Some(dir) => match dir.lookup_child(new_name) {
150                        Ok(child) => fs.is_whiteout(&child)?,
151                        Err(err) if err.error() == Errno::ENOENT => false,
152                        Err(err) => return Err(err),
153                    },
154                    None => false,
155                };
156                if upper_holds_whiteout {
157                    RenameTarget::UpperWhiteout
158                } else {
159                    RenameTarget::Absent
160                }
161            }
162        };
163
164        // An unanswerable probe counts as a hit, so the rename never acts on an unknown.
165        let lower_layer_holds_old_name = match old_parent.lower_entry(&old_child_dentry.name()) {
166            Ok(found) => found.is_some(),
167            Err(_) => true,
168        };
169        let lower_layer_holds_new_name_as_directory = match new_parent.lower_entry(new_name) {
170            Ok(found) => found.is_some_and(|type_| type_.is_directory()),
171            Err(_) => true,
172        };
173        let (mode, whiteout) = Self::decide_rename_plan(
174            request,
175            &source,
176            &target,
177            lower_layer_holds_old_name,
178            lower_layer_holds_new_name_as_directory,
179            fs.policy(),
180        )?;
181
182        Ok(Self {
183            lower_layer_holds_new_name_as_directory,
184            mode,
185            whiteout,
186            new_name,
187            old_parent,
188            old_parent_dentry,
189            new_parent,
190            new_parent_dentry: new_dir_dentry,
191            source,
192            source_dentry: old_child_dentry,
193            target,
194        })
195    }

197    /// Decides the rename plan the facts allow: the refusals taken before any lock, the mode the
198    /// real rename is issued with, and the old name's whiteout verdict.
199    fn decide_rename_plan(
200        request: RenameMode,
201        source: &OverlayInode,
202        target: &RenameTarget,
203        lower_layer_holds_old_name: bool,
204        lower_layer_holds_new_name_as_directory: bool,
205        policy: &MountPolicy,
206    ) -> Result<(RenameMode, WhiteoutAction)> {
207        let source_is_dir = source.type_().is_directory();
208        let source_is_merged_dir = source_is_dir && !source.lowers.is_empty();
209        let target_is_dir = matches!(
210            target,
211            RenameTarget::Visible { inode } if inode.type_().is_directory()
212        );
213        let target_is_visible = matches!(target, RenameTarget::Visible { .. });
214        let target_is_merged_dir = target_is_dir
215            && matches!(target, RenameTarget::Visible { inode } if !inode.lowers.is_empty());
216        if source_is_merged_dir
217            || (request == RenameMode::Exchange && target_is_dir && target_is_merged_dir)
218        {
219            return Err(Error::with_message(
220                Errno::EXDEV,
221                "the overlay cross-directory rename of a lower-backed or merged directory \
222                 requires the not-yet-implemented redirect_dir policy",
223            ));
224        }
225        if request != RenameMode::Exchange && source_is_dir && target_is_visible && !target_is_dir {
226            return Err(Error::with_message(
227                Errno::ENOTDIR,
228                "a directory cannot be renamed onto a non-directory name",
229            ));
230        }
231        if request != RenameMode::Exchange && !source_is_dir && target_is_visible && target_is_dir {
232            return Err(Error::with_message(
233                Errno::EISDIR,
234                "a non-directory cannot be renamed onto a directory name",
235            ));
236        }
237        if request == RenameMode::NoReplace && target_is_visible {
238            return Err(Error::with_message(
239                Errno::EEXIST,
240                "the rename target already exists and is visible",
241            ));
242        }
243
244        let target_whiteout_is_swapped = matches!(target, RenameTarget::UpperWhiteout)
245            && (source_is_dir || lower_layer_holds_old_name);
246        let mode = if request == RenameMode::Exchange || target_whiteout_is_swapped {
247            RenameMode::Exchange
248        } else {
249            RenameMode::Replace
250        };
251        // An exchange swaps the two names, so the target's whiteout travels with its name.
252        let whiteout = if request == RenameMode::Exchange {
253            WhiteoutAction::None
254        } else {
255            match target {
256                RenameTarget::UpperWhiteout if lower_layer_holds_old_name => {
257                    WhiteoutAction::Exchange
258                }
259                RenameTarget::UpperWhiteout if source_is_dir => WhiteoutAction::ExchangeThenDelete,
260                RenameTarget::UpperWhiteout => WhiteoutAction::None,
261                RenameTarget::Absent | RenameTarget::Visible { .. } => {
262                    if lower_layer_holds_old_name {
263                        WhiteoutAction::Create
264                    } else {
265                        WhiteoutAction::None
266                    }
267                }
268            }
269        };
270
271        if whiteout == WhiteoutAction::Create && !policy.can_express_whiteout() {
272            return Err(Error::with_message(
273                Errno::EOPNOTSUPP,
274                "the upper filesystem supports no whiteout form; the rename cannot publish one",
275            ));
276        }
277        if source_is_dir
278            && lower_layer_holds_new_name_as_directory
279            && !policy.can_store_private_xattr()
280        {
281            return Err(Error::with_message(
282                Errno::EOPNOTSUPP,
283                "the upper filesystem cannot store the opaque marker required for a rename",
284            ));
285        }
286        Ok((mode, whiteout))
287    }

289    /// Returns the emptiness verdict taken before the lock set: a directory source replacing a
290    /// visible directory must find it empty, the only gate when the target has no upper entry.
291    fn probe_target_empty_early(&self) -> Result<()> {
292        if !self.needs_empty_check() {
293            return Ok(());
294        }
295        let RenameTarget::Visible { inode, .. } = &self.target else {
296            return Ok(());
297        };
298        let target_guard = inode.lock();
299        if inode.is_empty_dir(&target_guard)? {
300            return Ok(());
301        }
302        Err(Error::with_message(
303            Errno::ENOTEMPTY,
304            "the overlay rename target directory is not empty",
305        ))
306    }

308    /// Promotes the objects this move will touch — both parents, the source, and an exchange's
309    /// target without an upper.
310    ///
311    /// The real rename can be expressed on the upper alone.
312    fn promote_members(&self) -> Result<()> {
313        self.old_parent
314            .writable_real_object(&self.old_parent_dentry)?;
315        self.new_parent
316            .writable_real_object(self.new_parent_dentry)?;
317        self.source.writable_real_object(self.source_dentry)?;
318        let RenameTarget::Visible { inode } = &self.target else {
319            return Ok(());
320        };
321        if self.mode != RenameMode::Exchange || inode.upper.get().is_some() {
322            return Ok(());
323        }
324        let fs = self.old_parent.fs_arc();
325        fs.promote_rename_target(&self.new_parent, inode, self.new_name)?;
326        Ok(())
327    }
328
329    /// Clears the whiteout residue out of the upper directory this move replaces.
330    fn sweep_replaced_whiteouts_if_needed(&self, _locks: &mut RenameLocks<'_>) -> Result<()> {
331        if self.mode == RenameMode::Exchange {
332            return Ok(());
333        }
334        let RenameTarget::Visible { inode, .. } = &self.target else {
335            return Ok(());
336        };
337        if !inode.type_().is_directory() {
338            return Ok(());
339        }
340        let Some(target_upper_dir) = inode.upper.get() else {
341            return Ok(());
342        };
343        self.old_parent
344            .fs_arc()
345            .sweep_whiteouts(target_upper_dir.dentry())
346    }

348    /// Writes the opaque record on the source's upper entry.
349    ///
350    /// The moved directory's new name stops merging a lower directory into it.
351    fn set_opaque_marker_if_needed(&self, _locks: &mut RenameLocks<'_>) -> Result<()> {
352        if !self.needs_opaque() {
353            return Ok(());
354        }
355        let fs = self.old_parent.fs_arc();
356        let namespace = fs.policy().xattr_namespace();
357        let source_upper = self.source.writable_upper();
358        OverlayXattrType::Opaque.set_value_on(source_upper.dentry(), namespace, None)
359    }
360
361    /// Writes the impure record on the new parent from either origin direction — the source's own
362    /// or, for an exchange, the target's.
363    ///
364    /// A failure here still stops the move before it commits.
365    fn set_impure_marker_if_needed(&self) -> Result<()> {
366        if !self.is_cross_parent() {
367            return Ok(());
368        }
369        let fs = self.old_parent.fs_arc();
370        let source_has_origin = fs
371            .origin_of(self.source.writable_upper().dentry())
372            .is_some();
373        let target_has_origin = match &self.target {
374            RenameTarget::Visible { inode, .. } if self.mode == RenameMode::Exchange => {
375                fs.origin_of(inode.writable_upper().dentry()).is_some()
376            }
377            _ => false,
378        };
379        if !source_has_origin && !target_has_origin {
380            return Ok(());
381        }
382        if !fs.policy().can_store_private_xattr() {
383            return Err(Error::with_message(
384                Errno::EOPNOTSUPP,
385                "the upper filesystem cannot store the impure marker required for a rename",
386            ));
387        }
388        let namespace = fs.policy().xattr_namespace();
389        let new_parent_upper = self.new_parent.writable_upper().dentry();
390        if OverlayXattrType::Impure.is_positive_on(new_parent_upper, namespace)? {
391            return Ok(());
392        }
393        OverlayXattrType::Impure.set_value_on(new_parent_upper, namespace, None)
394    }

396    /// Commits the move: the real rename in the upper, then the old name's whiteout action.
397    ///
398    /// The consequences the rename already fixed — the displaced target's latch and each parent's
399    /// snapshot drop — hold whether or not the whiteout step succeeded; its error is reported, not
400    /// rolled back.
401    fn commit(&self, locks: &mut RenameLocks<'_>) -> Result<()> {
402        let old_name = self.source_dentry.name();
403        let upper_parent = self.old_parent.writable_upper().dentry();
404        let new_upper_parent = self.new_parent.writable_upper().dentry();
405        upper_parent.as_dir_dentry_or_err()?.rename(
406            &old_name,
407            &new_upper_parent.as_dir_dentry_or_err()?,
408            self.new_name,
409            self.mode,
410        )?;
411
412        let fs = self.old_parent.fs_arc();
413        let outcome = match self.whiteout {
414            WhiteoutAction::None | WhiteoutAction::Exchange => Ok(()),
415            WhiteoutAction::Create => fs.publish_whiteout(upper_parent, &old_name, None),
416            WhiteoutAction::ExchangeThenDelete => fs.remove_whiteout(upper_parent, &old_name),
417        };
418
419        if self.mode == RenameMode::Replace
420            && let RenameTarget::Visible { inode, .. } = &self.target
421        {
422            inode.mark_name_taken();
423        }
424        *locks.old_parent_guard = None;
425        if let Some(new_parent_guard) = locks.new_parent_guard.as_deref_mut() {
426            *new_parent_guard = None;
427        }
428        outcome
429    }
430
431    fn source_is_dir(&self) -> bool {
432        self.source.type_().is_directory()
433    }
434
435    fn is_cross_parent(&self) -> bool {
436        !self.old_parent.same_directory_as(&self.new_parent)
437    }
438
439    fn needs_empty_check(&self) -> bool {
440        if self.mode == RenameMode::Exchange {
441            return false;
442        }
443        let RenameTarget::Visible { inode, .. } = &self.target else {
444            return false;
445        };
446        self.source_is_dir() && inode.type_().is_directory()
447    }
448
449    fn needs_opaque(&self) -> bool {
450        self.source_is_dir() && self.lower_layer_holds_new_name_as_directory
451    }
452}

454/// Records the role a member plays in the lock set.
455///
456/// It makes the set readable by role and derives its key.
457#[derive(Clone, Copy, Debug, Eq, PartialEq)]
458enum LockRole {
459    OldParent,
460    NewParent,
461    Target,
462}
463
464/// Holds the lock set of one rename: both parents, plus the target of an overwrite, taken in one
465/// pass in ascending `(depth, address)` order, handed to a step inside it as a token, not a guard.
466struct RenameLocks<'a> {
467    old_parent_guard: OverlayInodeLockGuard<'a>,
468    new_parent_guard: Option<OverlayInodeLockGuard<'a>>,
469    #[expect(
470        dead_code,
471        reason = "the guard is held so the overwritten target is latched under its own lock"
472    )]
473    target_guard: Option<OverlayInodeLockGuard<'a>>,
474}
475
476impl<'a> RenameLocks<'a> {
477    /// Returns the key that orders one lock-set member: the depth of the directory that hosts the
478    /// object, then the object's own address.
479    fn member_key(
480        role: LockRole,
481        old_parent_depth: usize,
482        new_parent_depth: usize,
483        address: usize,
484    ) -> (usize, usize) {
485        let depth = match role {
486            LockRole::OldParent => old_parent_depth,
487            LockRole::NewParent => new_parent_depth,
488            LockRole::Target => new_parent_depth + 1,
489        };
490        (depth, address)
491    }
492
493    /// Counts the dcache parent steps from this dentry to the mount root.
494    fn dentry_depth(dentry: &Dentry) -> usize {
495        let mut depth = 0;
496        let mut current = dentry.parent();
497        while let Some(parent) = current {
498            depth += 1;
499            current = parent.parent();
500        }
501        depth
502    }
503
504    /// Takes the lock set in ascending key order, then re-checks both name bindings under it.
505    ///
506    /// The facts predate the set, so `Ok(None)` is stale and the caller answers `ESTALE`.
507    fn acquire(facts: &'a RenameFacts<'_>) -> Result<Option<Self>> {
508        let fs = facts.old_parent.fs_arc();
509        let old_parent_depth = Self::dentry_depth(&facts.old_parent_dentry);
510        let new_parent_depth = Self::dentry_depth(facts.new_parent_dentry);
511        let same_parent = Arc::ptr_eq(&facts.old_parent, &facts.new_parent);
512        let target_member = match &facts.target {
513            RenameTarget::Visible { inode, .. } if facts.mode == RenameMode::Replace => {
514                Some(inode.as_ref())
515            }
516            _ => None,
517        };
518
519        let old_parent_address = core::ptr::from_ref(facts.old_parent.as_ref()) as usize;
520        let new_parent_address = core::ptr::from_ref(facts.new_parent.as_ref()) as usize;
521        let target_address = target_member.map(|object| core::ptr::from_ref(object) as usize);
522        let mut ordered: [((usize, usize), LockRole); 3] = [
523            ((usize::MAX, usize::MAX), LockRole::OldParent),
524            ((usize::MAX, usize::MAX), LockRole::NewParent),
525            ((usize::MAX, usize::MAX), LockRole::Target),
526        ];
527        for slot in &mut ordered {
528            let role = slot.1;
529            let member_address = match role {
530                LockRole::OldParent => Some(old_parent_address),
531                LockRole::NewParent => (!same_parent).then_some(new_parent_address),
532                LockRole::Target => target_address,
533            };
534            let Some(address) = member_address else {
535                continue;
536            };
537            *slot = (
538                Self::member_key(role, old_parent_depth, new_parent_depth, address),
539                role,
540            );
541        }
542        for index in 1..3 {
543            let mut position = index;
544            while position > 0 && ordered[position - 1].0 > ordered[position].0 {
545                ordered.swap(position - 1, position);
546                position -= 1;
547            }
548        }
549
550        let mut old_parent_guard = None;
551        let mut new_parent_guard = None;
552        let mut target_guard = None;
553        for (_, role) in ordered {
554            match role {
555                LockRole::OldParent => old_parent_guard = Some(facts.old_parent.lock()),
556                LockRole::NewParent => {
557                    if !same_parent {
558                        new_parent_guard = Some(facts.new_parent.lock());
559                    }
560                }
561                LockRole::Target => {
562                    if let Some(object) = target_member {
563                        target_guard = Some(object.lock());
564                    }
565                }
566            }
567        }
568        let Some(old_parent_guard) = old_parent_guard else {
569            unreachable!("the rename lock set always holds the old parent");
570        };
571
572        let source_is_fresh = match fs.lookup(&facts.old_parent, &facts.source_dentry.name())? {
573            Lookup::Positive(rebound) => Arc::ptr_eq(&rebound, &facts.source),
574            Lookup::Negative(_) => false,
575        };
576        let fresh = fs.lookup(&facts.new_parent, facts.new_name)?;
577        let upper_whiteout = match &fresh {
578            Lookup::Negative(NegativeLookup::HiddenByWhiteout) => {
579                let upper_dir = facts
580                    .new_parent
581                    .upper
582                    .get()
583                    .map(|upper| upper.dentry().as_dir_dentry_or_err())
584                    .transpose()?;
585                match upper_dir {
586                    Some(dir) => match dir.lookup_child(facts.new_name) {
587                        Ok(child) => fs.is_whiteout(&child)?,
588                        Err(err) if err.error() == Errno::ENOENT => false,
589                        Err(err) => return Err(err),
590                    },
591                    None => false,
592                }
593            }
594            Lookup::Negative(NegativeLookup::Absent) | Lookup::Positive(_) => false,
595        };
596        let target_is_fresh = match (&facts.target, &fresh) {
597            (RenameTarget::Visible { inode, .. }, Lookup::Positive(rebound)) => {
598                Arc::ptr_eq(inode, rebound)
599            }
600            (RenameTarget::Absent, Lookup::Negative(NegativeLookup::Absent)) => true,
601            (RenameTarget::Absent, Lookup::Negative(NegativeLookup::HiddenByWhiteout)) => {
602                !upper_whiteout
603            }
604            (RenameTarget::UpperWhiteout, Lookup::Negative(NegativeLookup::HiddenByWhiteout)) => {
605                upper_whiteout
606            }
607            _ => false,
608        };
609        if !source_is_fresh || !target_is_fresh {
610            return Ok(None);
611        }
612
613        Ok(Some(Self {
614            old_parent_guard,
615            new_parent_guard,
616            target_guard,
617        }))
618    }
