3//! The link recipe.
4//!
5//! The recipe gives the linked name a real upper entry: it classifies the name
6//! once, persists the parent's impurity when the source carries an origin
7//! record, and then either links the source into place directly or, where an
8//! upper whiteout holds the name, takes the name over with a prepared hard-link
9//! temp published with [`RenameMode::Replace`], since a real-layer `link` would
10//! fail with `EEXIST` there. Temp cleanup on failure is explicit and fallible.
11//!
12//! Degradation note: without a persistent origin index (a lower-origin
13//! identity map used to deduplicate copy-up targets), two lower aliases of
14//! one lower inode that copy up separately may become two distinct upper
15//! inodes. This split is accepted: each copy-up is independent, and no
16//! origin-index lookup is consulted to share the same upper inode.
17//! Upper-authoritative sources already share one upper inode because they
18//! are not split by separate copy-up.
19
20use crate::{
21    fs::{
22        fs_impls::overlayfs::{
23            fs::OverlayFs,
24            inode::{Lookup, NegativeLookup, OverlayInode, OverlayXattrType},
25        },
26        vfs::{inode::RenameMode, path::Dentry},
27    },
28    prelude::*,
29};
30
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32enum LinkTarget {
33    Whiteout,
34    Vacant,
35}
36
37impl OverlayInode {
38    /// Serves as the link entry the VFS calls: it promotes both objects, then composes the link
39    /// under the parent lock.
40    pub(in super::super) fn link_impl(
41        &self,
42        self_dentry: &Dentry,
43        old_dentry: &Dentry,
44        name: &str,
45    ) -> Result<()> {
46        let fs = self.fs_arc();
47        // The receiver and source promote before any lock: the linked name needs a real upper.
48        self.writable_real_object(self_dentry)?;
49        let source = self.promote_link_source(old_dentry)?;
50        let source_upper = source.writable_upper().dentry();
51        let mut dir_guard = self.lock();
52        let target = self.classify_link_target(&fs, name)?;
53        self.set_link_impure_marker(&fs, source_upper)?;
54        self.link_name(&fs, source_upper, name, target)?;
55        // The linked name belongs to the parent's next merge: the snapshot lacking it is stale.
56        *dir_guard = None;
57        Ok(())
58    }
59
60    fn promote_link_source(&self, old_dentry: &Dentry) -> Result<Arc<OverlayInode>> {
61        let source = Arc::downcast::<OverlayInode>(old_dentry.inode().clone())
62            .expect("the link source is not an overlay inode");
63        // TODO(VFS gap): the source-side admission (access, regular file, set-id) is the VFS's; its
64        // `check_hardlink_source` probes the initial user namespace; overlay keeps no source check.
65        source.writable_real_object(old_dentry)?;
66        Ok(source)
67    }
68
69    fn classify_link_target(&self, fs: &OverlayFs, name: &str) -> Result<LinkTarget> {
70        let target_lookup = fs.lookup(self, name)?;
71        if matches!(target_lookup, Lookup::Positive(_)) {
72            return_errno!(Errno::ESTALE);
73        }
74        if matches!(
75            target_lookup,
76            Lookup::Negative(NegativeLookup::HiddenByWhiteout)
77        ) {
78            return Ok(LinkTarget::Whiteout);
79        }
80        Ok(LinkTarget::Vacant)
81    }
82
83    fn set_link_impure_marker(&self, fs: &OverlayFs, source_upper: &Arc<Dentry>) -> Result<()> {
84        // The source upper is what the source name denotes; its record decides parent impurity.
85        let source_is_origin_backed = fs.origin_of(source_upper).is_some();
86        // An origin-preserved source makes the parent impure; persist the marker before the link.
87        if source_is_origin_backed {
88            let upper_parent = self.writable_upper();
89            if !fs.policy().can_store_private_xattr() {
90                return Err(Error::with_message(
91                    Errno::EOPNOTSUPP,
92                    "the upper filesystem cannot store the impure marker required for a link",
93                ));
94            }
95            if !OverlayXattrType::Impure
96                .is_positive_on(upper_parent.dentry(), fs.policy().xattr_namespace())?
97            {
98                OverlayXattrType::Impure.set_value_on(
99                    upper_parent.dentry(),
100                    fs.policy().xattr_namespace(),
101                    None,
102                )?;
103            }
104        }
105        Ok(())
106    }
107
108    fn link_name(
109        &self,
110        fs: &OverlayFs,
111        source_upper: &Arc<Dentry>,
112        name: &str,
113        target: LinkTarget,
114    ) -> Result<()> {
115        match target {
116            LinkTarget::Whiteout => self.link_over_whiteout(fs, name, source_upper),
117            LinkTarget::Vacant => self
118                .writable_upper()
119                .dentry()
120                .as_dir_dentry_or_err()?
121                .link(source_upper, name),
122        }
123    }
124
125    fn link_over_whiteout(&self, fs: &OverlayFs, name: &str, source: &Arc<Dentry>) -> Result<()> {
126        let upper_parent = self.writable_upper();
127        let upper_workdir = fs.upper_workdir_inuse();
128        let temp = upper_workdir.create_workdir_link_temp(name, source)?;
129        temp.publish(upper_parent.dentry(), name, RenameMode::Replace)
130    }
131}
