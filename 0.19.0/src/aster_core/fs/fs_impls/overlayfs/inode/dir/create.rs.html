3//! The create-object recipe family.
4//!
5//! One dispatcher routes on the fresh layer lookup of the target name: an
6//! absent or opaque-hidden name creates directly in the upper parent, a
7//! whiteout-hidden name replaces the whiteout through a completely prepared
8//! private workdir temp, and a fresh positive target fails with `ESTALE`.
9//! All three VFS create-family entries flow through one [`CreateOp`], and
10//! `SymLink` can only enter through the atomic `create_symlink` entry.
11
12use crate::{
13    fs::{
14        file::InodeType,
15        fs_impls::overlayfs::{
16            fs::OverlayFs,
17            inode::{
18                CreateOp, Lookup, NegativeLookup, OverlayInode, OverlayInodeLockGuard,
19                OverlayXattrType, copyup::workdir::WorkdirTemp,
20            },
21            real::{RealObject, RealObjectStack},
22        },
23        vfs::{
24            inode::{Inode, RenameMode},
25            path::Dentry,
26        },
27    },
28    prelude::*,
29};
30
31impl OverlayInode {
32    /// Serves as the single create-family entry; `CreateOp` encodes the VFS entry that produced it.
33    pub(in super::super) fn create_impl(
34        &self,
35        self_dentry: &Dentry,
36        name: &str,
37        op: CreateOp<'_>,
38    ) -> Result<Arc<dyn Inode>> {
39        let fs = self.fs_arc();
40        // The receiver promotes before its lock, so the created name gets a real upper entry.
41        self.writable_real_object(self_dentry)?;
42        let projected: Arc<dyn Inode> = self.create_object(&fs, name, op)?;
43        Ok(projected)
44    }
45
46    fn create_object(
47        &self,
48        fs: &OverlayFs,
49        name: &str,
50        op: CreateOp<'_>,
51    ) -> Result<Arc<OverlayInode>> {
52        let mut dir_guard = self.lock();
53        match fs.lookup(self, name)? {
54            Lookup::Negative(NegativeLookup::Absent) => {
55                let upper = self.writable_upper();
56                // `create_child_in` selects the single real-layer syscall per op.
57                let new_upper = op.create_child_in(upper.dentry(), name)?;
58                let inode = fs.project_inode(RealObjectStack::upper_only(
59                    RealObject::new_upper(new_upper),
60                ))?;
61                // The new name belongs to the parent's next merge, so the snapshot that lacks it
62                // is stale.
63                *dir_guard = None;
64                Ok(inode)
65            }
66            Lookup::Negative(NegativeLookup::HiddenByWhiteout) => {
67                self.create_over_whiteout(fs, name, op, &mut dir_guard)
68            }
69            Lookup::Positive(_) => return_errno!(Errno::ESTALE),
70        }
71    }
72
73    fn create_over_whiteout(
74        &self,
75        fs: &OverlayFs,
76        name: &str,
77        op: CreateOp<'_>,
78        dir_guard: &mut OverlayInodeLockGuard<'_>,
79    ) -> Result<Arc<OverlayInode>> {
80        let upper_workdir = fs.upper_workdir_inuse();
81        let object_type = op.object_type();
82        let temp = upper_workdir.create_workdir_temp(name, op)?;
83        let temp_name = if object_type == InodeType::Dir {
84            Some(String::from(temp.name()))
85        } else {
86            None
87        };
88        let published = self.commit_create_over_whiteout(fs, temp, name)?;
89        // The published name belongs to the parent's next merge, so the snapshot that lacks it
90        // is stale.
91        **dir_guard = None;
92        if let Some(temp_name) = temp_name {
93            upper_workdir
94                .workdir_workspace()?
95                .as_dir_dentry_or_err()?
96                .unlink(&temp_name)?;
97        }
98        fs.project_inode(RealObjectStack::upper_only(RealObject::new_upper(
99            published,
100        )))
101    }
102
103    /// Commits a whiteout-replacing temp into the upper parent.
104    fn commit_create_over_whiteout(
105        &self,
106        fs: &OverlayFs,
107        temp: WorkdirTemp,
108        name: &str,
109    ) -> Result<Arc<Dentry>> {
110        let upper_parent = self.writable_upper().dentry();
111        let is_directory = temp.inode().type_() == InodeType::Dir;
112        // The opaque marker is written before the swap, so the whiteout is not deleted first.
113        if is_directory {
114            if !fs.policy().can_store_private_xattr() {
115                return Err(Error::with_message(
116                    Errno::EOPNOTSUPP,
117                    "the upper filesystem cannot store the opaque marker \
118                     required for a directory over a whiteout",
119                ));
120            }
121            OverlayXattrType::Opaque.set_value_on(
122                temp.dentry(),
123                fs.policy().xattr_namespace(),
124                None,
125            )?;
126        }
127        let published = temp.dentry().clone();
128        if is_directory {
129            temp.publish(upper_parent, name, RenameMode::Exchange)?;
130        } else {
131            temp.publish(upper_parent, name, RenameMode::Replace)?;
132        }
133        Ok(published)
134    }
135}
