3//! The remove recipes: the shared unlink/rmdir recipe on [`OverlayInode`],
4//! parameterized by [`RemoveKind`].
5//!
6//! An unlink or rmdir removes a name from the merged view: afterwards that name resolves
7//! negatively for every lookup, while the lower layers keep whatever objects they held.
8//!
9//! Where the upper entry is the name's only witness, the removal deletes it outright; everywhere
10//! else a published whiteout hides the name, with the upper directory first replaced through a
11//! prepared empty workdir temp whenever it still holds entries.
12
13use crate::{
14    fs::{
15        file::InodeType,
16        fs_impls::overlayfs::{
17            inode::{
18                CreateOp, Lookup, OverlayInode, OverlayInodeLockGuard, OverlayXattrType,
19                copyup::workdir::WorkdirTemp,
20            },
21            real::{RealObject, read_child_names},
22        },
23        vfs::{
24            inode::{Inode, RenameMode},
25            path::Dentry,
26        },
27    },
28    prelude::*,
29};
30
31#[derive(Clone, Copy, Debug, Eq, PartialEq)]
32enum RemoveKind {
33    Unlink,
34    Rmdir,
35}
36
37/// Records the action a removal performs: one variant per upper state that survives the gates.
38#[derive(Clone, Copy, Debug)]
39enum RemoveAction {
40    /// Applies where the upper entry is the name's only witness: the removal deletes it.
41    DeletePureUpper,
42    /// Applies where the upper holds no entry: a whiteout hides the name the lower layers carry.
43    WhiteoutLowerOnly,
44    /// Applies where the upper entry is not the name's only witness: a whiteout must displace it.
45    ReplaceUpperWithWhiteout,
46}
47
48/// Fixes the projection and the name of one removal as one pair.
49///
50/// It is built inside each step that needs it, never carried across a lock boundary, and holds no
51/// layer fact, so nothing it pairs can go stale while the locks are taken.
52struct RemoveTarget {
53    /// Holds the name's fresh projection, resolved by this request's own lookup.
54    inode: Arc<OverlayInode>,
55    /// Carries the name, derived once from the caller's dentry and owned here.
56    name: String,
57}
58
59impl OverlayInode {
60    /// Serves as the unlink entry the VFS calls: it resolves the removed child's parent and runs
61    /// the removal.
62    pub(in super::super) fn unlink_impl(&self, child_dentry: &Dentry) -> Result<()> {
63        // The take point promotes the parent, sourcing its target from the removed entry's parent.
64        let parent_dentry = child_dentry
65            .parent()
66            .expect("the unlinked child has no parent dentry");
67        self.refuse_removal_early(child_dentry, RemoveKind::Unlink)?;
68        self.writable_real_object(&parent_dentry)?;
69        self.remove_target(child_dentry, RemoveKind::Unlink)
70    }
71
72    /// Serves as the rmdir entry the VFS calls: it resolves the removed child's parent and runs
73    /// the removal.
74    pub(in super::super) fn rmdir_impl(&self, child_dentry: &Dentry) -> Result<()> {
75        let parent_dentry = child_dentry
76            .parent()
77            .expect("the removed directory has no parent dentry");
78        self.refuse_removal_early(child_dentry, RemoveKind::Rmdir)?;
79        self.writable_real_object(&parent_dentry)?;
80        self.remove_target(child_dentry, RemoveKind::Rmdir)
81    }
82
83    /// Answers the verdicts before the parent is promoted.
84    ///
85    /// A doomed removal pays no copy-up. The name keeps the family's historical `removal` infix,
86    /// its one spelling exception.
87    fn refuse_removal_early(&self, child_dentry: &Dentry, kind: RemoveKind) -> Result<()> {
88        let target = self.resolve_remove_target(child_dentry)?;
89        let target_guard = target.inode.lock();
90        let _ = self.decide_remove_action(&target, kind, &target_guard)?;
91        Ok(())
92    }
93
94    /// Sends the removal of one name to the arm its verdict names.
95    fn remove_target(&self, child_dentry: &Dentry, kind: RemoveKind) -> Result<()> {
96        let mut dir_guard = self.lock();
97        let target = self.resolve_remove_target(child_dentry)?;
98        let target_guard = target.inode.lock();
99        let action = self.decide_remove_action(&target, kind, &target_guard)?;
100        match action {
101            RemoveAction::DeletePureUpper => self.delete_pure_upper(&target, kind, &mut dir_guard),
102            RemoveAction::WhiteoutLowerOnly => self.whiteout_lower_only(&target, &mut dir_guard),
103            RemoveAction::ReplaceUpperWithWhiteout => {
104                self.replace_upper_with_whiteout(&target, kind, &mut dir_guard)
105            }
106        }
107    }
108
109    fn resolve_remove_target(&self, child_dentry: &Dentry) -> Result<RemoveTarget> {
110        let name = child_dentry.name();
111        let lookup = self.fs_arc().lookup(self, &name)?;
112        let inode = match lookup {
113            Lookup::Positive(inode) => inode,
114            Lookup::Negative(_) => return_errno!(Errno::ENOENT),
115        };
116        // The caller's dentry and the fresh projection disagree on the name's upper entry, so the
117        // caller's view of the name is stale and the removal stops here instead of acting on it.
118        if inode.is_stale_upper_target(child_dentry) {
119            return Err(Error::with_message(
120                Errno::ESTALE,
121                "the upper object at the target name became stale",
122            ));
123        }
124        Ok(RemoveTarget { inode, name })
125    }
126
127    /// Decides the action one removal performs from the target's upper state and its lower probe.
128    fn decide_remove_action(
129        &self,
130        target: &RemoveTarget,
131        kind: RemoveKind,
132        target_guard: &OverlayInodeLockGuard<'_>,
133    ) -> Result<RemoveAction> {
134        let fs = self.fs_arc();
135        let namespace = fs.policy().xattr_namespace();
136        let target_upper = target.inode.upper.get();
137        // Classified before the gates: the verdict decides both the gate and the whiteout branch.
138        let lower_positive = if target_upper.is_some() {
139            // An unanswerable probe counts as a hit: a removal must never leave a name behind.
140            match self.lower_entry(&target.name) {
141                Ok(found) => found.is_some(),
142                Err(_) => true,
143            }
144        } else {
145            // Without an upper the name cannot be pure upper, and no gate reads the probe.
146            false
147        };
148        let is_pure_upper = match target_upper {
149            Some(upper_obj) => {
150                !lower_positive
151                    // The opaque marker only ever sits on a directory, so a non-directory target
152                    // asks no opaque question; the conjunct is the precondition, not a re-check.
153                    && !(target.inode.type_().is_directory()
154                        && OverlayXattrType::Opaque.is_positive_on(upper_obj.dentry(), namespace)?)
155            }
156            None => false,
157        };
158
159        // The target's own lock covers the gate and removal, so a promotion round sees it.
160        if kind == RemoveKind::Rmdir {
161            if !target.inode.type_().is_directory() {
162                return Err(Error::with_message(
163                    Errno::ENOTDIR,
164                    "a non-directory cannot be removed as a directory",
165                ));
166            }
167            if lower_positive || !is_pure_upper {
168                // The merged view's emptiness, whiteouts excluded, decides `ENOTEMPTY`.
169                // The guard is already held, so the emptiness answer takes and releases no lock.
170                if !target.inode.is_empty_dir(target_guard)? {
171                    return Err(Error::with_message(
172                        Errno::ENOTEMPTY,
173                        "the overlay directory is not empty",
174                    ));
175                }
176            }
177        } else if target.inode.type_().is_directory() {
178            return Err(Error::with_message(
179                Errno::EISDIR,
180                "a directory cannot be unlinked",
181            ));
182        }
183
184        if !is_pure_upper && !fs.policy().can_express_whiteout() {
185            return Err(Error::with_message(
186                Errno::EOPNOTSUPP,
187                "the upper filesystem supports no whiteout form; the removal \
188                 cannot publish one",
189            ));
190        }
191        Ok(if is_pure_upper {
192            RemoveAction::DeletePureUpper
193        } else if target_upper.is_some() {
194            RemoveAction::ReplaceUpperWithWhiteout
195        } else {
196            RemoveAction::WhiteoutLowerOnly
197        })
198    }
199
200    /// Deletes the upper entry itself, all a removal needs where the upper alone holds the name.
201    fn delete_pure_upper(
202        &self,
203        target: &RemoveTarget,
204        kind: RemoveKind,
205        dir_guard: &mut OverlayInodeLockGuard<'_>,
206    ) -> Result<()> {
207        let fs = self.fs_arc();
208        let target_upper = target.inode.upper.get();
209        let upper_parent = self.writable_upper();
210        // The real layer counts whiteouts, so the residue they leave is swept before the rmdir.
211        if kind == RemoveKind::Rmdir {
212            let target_upper_dir = target_upper.ok_or_else(|| {
213                Error::with_message(
214                    Errno::EIO,
215                    "the pure-upper rmdir target has no upper real directory",
216                )
217            })?;
218            fs.sweep_whiteouts(target_upper_dir.dentry())?;
219        }
220        let upper_dir = upper_parent.dentry().as_dir_dentry_or_err()?;
221        let result = if kind == RemoveKind::Rmdir {
222            upper_dir.rmdir(&target.name)
223        } else {
224            upper_dir.unlink(&target.name)
225        };
226        result.map_err(translate_stale_upper_enoent)?;
227        target.inode.mark_name_taken();
228        // The removed name is gone from the next merge, so the parent's snapshot is stale.
229        **dir_guard = None;
230        Ok(())
231    }
232
233    /// Publishes a whiteout at a name the upper holds no entry for.
234    fn whiteout_lower_only(
235        &self,
236        target: &RemoveTarget,
237        dir_guard: &mut OverlayInodeLockGuard<'_>,
238    ) -> Result<()> {
239        let fs = self.fs_arc();
240        let upper_parent = self.writable_upper().dentry();
241        let published = fs.publish_whiteout(upper_parent, &target.name, None);
242        if published.is_ok() {
243            target.inode.mark_name_taken();
244            // The taken name is gone from the parent's next merge, so its snapshot is stale.
245            **dir_guard = None;
246        }
247        published
248    }
249
250    /// Replaces the upper entry behind the name with a whiteout.
251    fn replace_upper_with_whiteout(
252        &self,
253        target: &RemoveTarget,
254        kind: RemoveKind,
255        dir_guard: &mut OverlayInodeLockGuard<'_>,
256    ) -> Result<()> {
257        let fs = self.fs_arc();
258        let upper_parent = self.writable_upper().dentry();
259        let occupant = target
260            .inode
261            .upper
262            .get()
263            .expect("an upper entry replaced by a whiteout has an upper real object");
264
265        let cleared =
266            kind == RemoveKind::Rmdir && !read_child_names(occupant.real_inode())?.is_empty();
267        if cleared {
268            let empty_copy = fs.upper_workdir_inuse().create_workdir_temp(
269                &target.name,
270                CreateOp::General {
271                    kind: InodeType::Dir,
272                    mode: occupant.real_inode().mode()?,
273                },
274            )?;
275
276            let empty_copy_name = String::from(empty_copy.name());
277
278            // The publication displaces this empty copy rather than the occupant: a rename cannot
279            // replace a directory with a whiteout, and the displaced directory has to be empty.
280            self.exchange_occupant_for_empty_copy(target, occupant, upper_parent, empty_copy)?;
281
282            let workspace = fs.upper_workdir_inuse().workdir_workspace()?;
283            let workdir_dir = workspace.as_dir_dentry_or_err()?;
284            let cleanup_failure = match workdir_dir.lookup_child(&empty_copy_name) {
285                // The displaced directory counts whiteouts, so they are swept before its rmdir.
286                Ok(displaced) => {
287                    let sweep_failure = fs.sweep_whiteouts(&displaced).err();
288                    let rmdir_failure = workdir_dir.rmdir(&empty_copy_name).err();
289                    sweep_failure.or(rmdir_failure)
290                }
291                Err(err) => Some(err),
292            };
293            if let Some(err) = cleanup_failure {
294                warn!(
295                    "clear-empty: the displaced-directory cleanup left residue: {:?}",
296                    err
297                );
298            }
299        }
300
301        let published = fs
302            .publish_whiteout(upper_parent, &target.name, Some(target.inode.type_()))
303            .map_err(translate_stale_upper_enoent);
304        if cleared || published.is_ok() {
305            target.inode.mark_name_taken();
306            // The name was taken over here, by the empty copy or by the whiteout, so the parent's
307            // snapshot of it is stale.
308            **dir_guard = None;
309        }
310        published
311    }
312
313    /// Gives the name an empty copy of the occupant to hold it, the real directory landing in the
314    /// workdir, outside the merged view.
315    fn exchange_occupant_for_empty_copy(
316        &self,
317        target: &RemoveTarget,
318        occupant: &RealObject,
319        upper_parent: &Arc<Dentry>,
320        empty_copy: WorkdirTemp,
321    ) -> Result<()> {
322        let fs = self.fs_arc();
323        let namespace = fs.policy().xattr_namespace();
324        // The opaque marker is written before the exchange so the name stays a search barrier.
325        if !fs.policy().can_store_private_xattr() {
326            return Err(Error::with_message(
327                Errno::EOPNOTSUPP,
328                "the upper filesystem cannot store the opaque marker \
329                 required for the clear-empty directory exchange",
330            ));
331        }
332        OverlayXattrType::Opaque.set_value_on(empty_copy.dentry(), namespace, None)?;
333        // Xattrs are copied before owner/group/mode, so a non-owner rmdir cannot fail `EACCES`.
334        self.copy_eligible_xattrs(occupant, &empty_copy, true)?;
335        occupant.copy_metadata_to(empty_copy.dentry())?;
336        occupant.copy_timestamps_to(empty_copy.dentry())?;
337        empty_copy
338            .publish(upper_parent, &target.name, RenameMode::Exchange)
339            .map_err(translate_stale_upper_enoent)?;
340        Ok(())
341    }
342}
343
344fn translate_stale_upper_enoent(err: Error) -> Error {
345    if err.error() == Errno::ENOENT {
346        Error::with_message(
347            Errno::ESTALE,
348            "the upper object at the target name became stale",
349        )
350    } else {
351        err
352    }
