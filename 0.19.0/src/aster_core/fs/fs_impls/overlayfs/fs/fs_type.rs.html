3//! The VFS registration type of overlayfs.
4//!
5//! [`OverlayFsType`] is the carrier the VFS registry sees: it publishes the
6//! filesystem name `overlay`, and each mount request under that name is
7//! answered by constructing one [`OverlayFs`] from the mount's creation
8//! context. Registration happens once at filesystem initialization; every
9//! later mount builds its state through that construction call.
10
11use super::OverlayFs;
12use crate::{
13    fs::vfs::{
14        file_system::FileSystem,
15        registry::{FsCreationCtx, FsProperties, FsType},
16    },
17    prelude::*,
18};
19
20pub(super) const OVERLAY_FS_NAME: &str = "overlay";
21
22pub(super) struct OverlayFsType;
23
24impl FsType for OverlayFsType {
25    type Key = ();
26
27    fn name(&self) -> &'static str {
28        OVERLAY_FS_NAME
29    }
30
31    fn properties(&self) -> FsProperties {
32        FsProperties::empty()
33    }
34
35    fn create(&self, fs_creation_ctx: &mut FsCreationCtx) -> Result<Arc<dyn FileSystem>> {
36        let overlay_fs = OverlayFs::new(fs_creation_ctx)?;
37        Ok(overlay_fs)
38    }
39
40    fn sysnode(&self) -> Option<Arc<dyn aster_systree::SysNode>> {
41        None
42    }
43}
