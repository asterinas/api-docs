3//! Upper-filesystem capability probes.
4//!
5//! After the upper/workdir pair is claimed, this module measures the upper
6//! filesystem's capabilities — on the upper itself and on the workdir
7//! workspace — used to decide whiteout and UUID support for a writable
8//! overlay mount.
9
10#![short_vis_path::add(overlayfs)]
11
12use super::{
13    inuse::{OVERLAY_UUID_SIZE, UpperWorkdirInuse},
14    options::UuidMode,
15};
16use crate::{
17    fs::{
18        file::{InodeMode, InodeType},
19        fs_impls::overlayfs::inode::OverlayXattrType,
20        utils::DirentVisitor,
21        vfs::{
22            inode::MknodType,
23            path::{Dentry, is_dot_or_dotdot},
24            xattr::XattrNamespace,
25        },
26    },
27    prelude::*,
28};
29
30#[derive(Clone, Copy, Debug, Eq, PartialEq)]
31pub(super) struct UpperFilesystemCapabilities {
32    /// Reports whether the upper can persist the private xattr records this mount writes.
33    can_store_private_xattr: bool,
34    /// Reports whether the upper types every non-dot directory entry it lists.
35    can_report_directory_type: bool,
36    /// Reports whether the upper accepts a character-device `mknod`.
37    can_mknod_char: bool,
38}
39
40/// The whiteout form the upper can publish.
41#[derive(Clone, Copy, Debug, Eq, PartialEq)]
42pub(in overlayfs) enum WhiteoutCapability {
43    /// Publishes a char device numbered `0:0`, the form every mount reads as a whiteout.
44    CharDevice,
45    /// Publishes a zero-size regular file carrying this mount's whiteout record.
46    Xattr,
47    /// Blocks new whiteouts.
48    Unsupported,
49}
50
51impl UpperFilesystemCapabilities {
52    /// Probes the upper's private-xattr, d_type, and char-device support at mount time.
53    pub(super) fn probe(
54        upper_workdir: &UpperWorkdirInuse,
55        namespace: XattrNamespace,
56    ) -> Result<Self> {
57        let can_store_private_xattr =
58            Self::probe_private_xattr(upper_workdir.upper_dentry(), namespace)?;
59        let can_report_directory_type = Self::probe_d_type(upper_workdir)?;
60        let can_mknod_char = Self::probe_mknod_char(upper_workdir)?;
61        Ok(Self {
62            can_store_private_xattr,
63            can_report_directory_type,
64            can_mknod_char,
65        })
66    }
67
68    pub(super) fn can_store_private_xattr(&self) -> bool {
69        self.can_store_private_xattr
70    }
71
72    pub(super) fn can_report_directory_type(&self) -> bool {
73        self.can_report_directory_type
74    }
75
76    /// Derives the upper's published whiteout form, preferring the char-device `0:0` form.
77    pub(super) fn whiteout_capability(&self) -> WhiteoutCapability {
78        if self.can_mknod_char {
79            WhiteoutCapability::CharDevice
80        } else if self.can_store_private_xattr() {
81            WhiteoutCapability::Xattr
82        } else {
83            WhiteoutCapability::Unsupported
84        }
85    }
86
87    pub(super) fn validate_uuid_support(&self, uuid_mode: UuidMode) -> Result<bool> {
88        match uuid_mode {
89            UuidMode::On => {
90                if !self.can_store_private_xattr() {
91                    return_errno_with_message!(
92                        Errno::EOPNOTSUPP,
93                        "the upper filesystem cannot persist the overlay uuid"
94                    );
95                }
96                Ok(true)
97            }
98            UuidMode::Auto => Ok(self.can_store_private_xattr()),
99            UuidMode::Off | UuidMode::Null => Ok(false),
100        }
101    }
102
103    fn probe_private_xattr(upper_dentry: &Dentry, namespace: XattrNamespace) -> Result<bool> {
104        let mut value = [0u8; OVERLAY_UUID_SIZE];
105        match OverlayXattrType::Uuid.get_value_from(upper_dentry, namespace, &mut value) {
106            Ok(_) => Ok(true),
107            Err(err) if err.error() == Errno::ENODATA => Ok(true),
108            Err(err) if err.error() == Errno::ERANGE => Ok(true),
109            Err(err) if err.error() == Errno::EOPNOTSUPP => Ok(false),
110            Err(err) => Err(err),
111        }
112    }
113
114    fn probe_d_type(upper_workdir: &UpperWorkdirInuse) -> Result<bool> {
115        /// The name prefix of the workspace entry the d_type probe creates.
116        const D_TYPE_PROBE_PREFIX: &str = ".overlay-dtype-probe-";
117        let workspace = upper_workdir.workdir_workspace()?;
118        let workspace_dir = workspace.as_dir_dentry_or_err()?;
119        let d_type_probe_name = upper_workdir.next_temp_name(D_TYPE_PROBE_PREFIX);
120        workspace_dir.create_child(&d_type_probe_name, || {
121            workspace_dir.inode().create(
122                &workspace_dir,
123                &d_type_probe_name,
124                InodeType::File,
125                InodeMode::empty(),
126            )
127        })?;
128        let mut d_type_probe = DTypeProbeVisitor::new();
129        let mut offset = 0;
130        let d_type_scan_result = loop {
131            match workspace.inode().readdir_at(offset, &mut d_type_probe) {
132                Ok(0) => break Ok(()),
133                Ok(visited) => offset += visited,
134                Err(err) => break Err(err),
135            }
136        };
137        match d_type_scan_result {
138            Ok(()) => {
139                workspace_dir.unlink(&d_type_probe_name)?;
140                Ok(!d_type_probe.saw_unknown_non_dot)
141            }
142            Err(err) => {
143                let _ = workspace_dir.unlink(&d_type_probe_name);
144                Err(err)
145            }
146        }
147    }
148
149    fn probe_mknod_char(upper_workdir: &UpperWorkdirInuse) -> Result<bool> {
150        /// The name prefix of the workspace entry the char-device probe creates.
151        const CHAR_DEVICE_PROBE_PREFIX: &str = ".overlay-char-device-probe-";
152        let workspace = upper_workdir.workdir_workspace()?;
153        let workspace_dir = workspace.as_dir_dentry_or_err()?;
154        let probe_name = upper_workdir.next_temp_name(CHAR_DEVICE_PROBE_PREFIX);
155        match workspace_dir.mknod(&probe_name, InodeMode::empty(), MknodType::CharDevice(0)) {
156            Ok(_) => {
157                workspace_dir.unlink(&probe_name)?;
158                Ok(true)
159            }
160            Err(err)
161                if matches!(
162                    err.error(),
163                    Errno::EOPNOTSUPP | Errno::EPERM | Errno::EACCES
164                ) =>
165            {
166                Ok(false)
167            }
168            Err(err) => Err(err),
169        }
170    }
171}
172
173struct DTypeProbeVisitor {
174    saw_unknown_non_dot: bool,
175}
176
177impl DTypeProbeVisitor {
178    fn new() -> Self {
179        Self {
180            saw_unknown_non_dot: false,
181        }
182    }
183}
184
185impl DirentVisitor for DTypeProbeVisitor {
186    fn visit(&mut self, name: &str, _ino: u64, type_: InodeType, _offset: usize) -> Result<()> {
187        if !is_dot_or_dotdot(name) && type_ == InodeType::Unknown {
188            self.saw_unknown_non_dot = true;
189        }
190        Ok(())
191    }
