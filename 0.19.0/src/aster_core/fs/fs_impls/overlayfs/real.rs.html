3//! The real-object reference model beneath the overlay namespace.
4//!
5//! One [`RealObject`] is one underlying filesystem entry as seen from one layer: the mount-level
6//! layer index it carries and the real dentry that anchors it. [`RealObject::new_upper`] builds
7//! the upper's object and pins the index to `UPPER_LAYER_INDEX`, and [`RealObject::new_lower`]
8//! takes the index the layer stack assigned, so the side a value stands on is carried by the value
9//! and by its enumerated construction points.
10//!
11//! The type offers two exits to the real side: [`RealObject::real_inode`] hands out the real inode
12//! of the anchored dentry, and [`RealObject::dentry`] hands out that dentry itself, which some
13//! real-inode writes take as their anchor. Neither exit is side-restricted: the read take point
14//! `OverlayInode::real_object` also hands out lower objects, and the type does not stop a caller
15//! from writing through one. Review visibility of the write paths is the only thing keeping those
16//! writes honest, so a new write site must issue from a take point that has already promoted the
17//! object.
18//!
19//! Real-directory enumeration lives here: it walks only the real side and holds no overlay state.
20//!
21//! Anchor validity follows the overlay lifetime: the owning layer strongly
22//! holds its root dentry together with the filesystem it is rooted on, so a
23//! reachable logical object never observes a dead anchor.
24

39/// The layer index every upper real object carries: layer 0 is reserved for the upper.
40const UPPER_LAYER_INDEX: usize = 0;
41
42/// One underlying filesystem entry as seen from one layer: its layer index and dentry.
43#[derive(Debug)]
44pub(super) struct RealObject {
45    layer_index: usize,
46    dentry: Arc<Dentry>,
47}
48
49impl RealObject {
50    /// Returns this real object's mount-level layer index.
51    pub(super) fn layer_index(&self) -> usize {
52        self.layer_index
53    }
54
55    pub(super) fn dentry(&self) -> &Arc<Dentry> {
56        &self.dentry
57    }
58
59    pub(super) fn real_inode(&self) -> &Arc<dyn Inode> {
60        self.dentry.inode()
61    }
62
63    /// Returns the status flags a read of this real object carries.
64    ///
65    /// A lower read takes `O_NOATIME`, so it never updates the lower's atime.
66    pub(super) fn status_flags_for_read(&self, status_flags: StatusFlags) -> StatusFlags {
67        if self.layer_index == UPPER_LAYER_INDEX {
68            status_flags
69        } else {
70            status_flags | StatusFlags::O_NOATIME
71        }
72    }
73
74    /// Builds the upper-side object of one upper entry; the upper is layer 0 by construction.
75    pub(super) fn new_upper(dentry: Arc<Dentry>) -> Self {
76        Self {
77            layer_index: UPPER_LAYER_INDEX,
78            dentry,
79        }
80    }
81
82    /// Builds the lower-side object; `layer_index` is the mount-level index the stack assigned.
83    pub(super) fn new_lower(layer_index: usize, dentry: Arc<Dentry>) -> Self {
84        debug_assert!(layer_index != UPPER_LAYER_INDEX);
85        Self {
86            layer_index,
87            dentry,
88        }
89    }
90
91    /// Writes `self`'s owner, group, mode, and (regular files) size onto `temp_dentry`.
92    ///
93    /// A symlink's mode is left alone, as in Linux `ovl_set_attr`.
94    ///
95    /// Reference:
96    /// <https://elixir.bootlin.com/linux/latest/source/fs/overlayfs/copy_up.c#L392-L416>.
97    pub(super) fn copy_metadata_to(&self, temp_dentry: &Dentry) -> Result<()> {
98        let source_inode = self.real_inode();
99        let temp_inode = temp_dentry.inode();
100        temp_inode.set_owner(temp_dentry, source_inode.owner()?)?;
101        temp_inode.set_group(temp_dentry, source_inode.group()?)?;
102        if !matches!(source_inode.type_(), InodeType::SymLink) {
103            temp_inode.set_mode(temp_dentry, source_inode.mode()?)?;
104        }
105        if source_inode.type_().is_regular_file() {
106            temp_inode.resize(temp_dentry, source_inode.size())?;
107        }
108        Ok(())
109    }
110
111    /// Copies `self`'s regular-file bytes into `temp_dentry`'s object; a short copy is `EIO`, while
112    /// a short read advances the loop.
113    pub(super) fn copy_data_to(&self, temp_dentry: &Dentry) -> Result<()> {
114        /// The bytes one copy-up read moves; a short read only advances the loop.
115        const COPY_CHUNK_SIZE: usize = 64 * 1024;
116        let size = self.real_inode().size();
117        let mut offset = 0usize;
118        let mut buffer = vec![0u8; COPY_CHUNK_SIZE];
119        while offset < size {
120            let chunk = min(COPY_CHUNK_SIZE, size - offset);
121            let mut writer = VmWriter::from(&mut buffer[..chunk]).to_fallible();
122            let read_len = self
123                .real_inode()
124                .read_at(offset, &mut writer, StatusFlags::empty())?;
125            // A zero-length read before the declared size ends the copy; the completeness check
126            // below reports it, because a short copy is the failure this loop must not hide.
127            if read_len == 0 {
128                break;
129            }
130            let mut reader = VmReader::from(&buffer[..read_len]).to_fallible();
131            let write_len =
132                temp_dentry
133                    .inode()
134                    .write_at(offset, &mut reader, StatusFlags::empty())?;
135            if write_len != read_len {
136                return_errno_with_message!(
137                    Errno::EIO,
138                    "the workdir temp accepted a short write during copy-up"
139                );
140            }
141            offset += write_len;
142        }
143        // The invariant: every byte the lower declared must have been copied.
144        if offset != size {
145            return_errno_with_message!(
146                Errno::EIO,
147                "the copy-up copied fewer bytes than the lower file declares"
148            );
149        }
150        Ok(())
151    }
152
153    /// Writes `self`'s three timestamps onto `temp_dentry`.
154    pub(super) fn copy_timestamps_to(&self, temp_dentry: &Dentry) -> Result<()> {
155        let source_inode = self.real_inode();
156        let temp_inode = temp_dentry.inode();
157        temp_inode.set_atime(temp_dentry, source_inode.atime());
158        temp_inode.set_mtime(temp_dentry, source_inode.mtime());
159        temp_inode.set_ctime(temp_dentry, source_inode.ctime());
160        Ok(())
161    }
162}

164/// The ordered real objects behind one logical overlay object, and which supplies its metadata.
165#[derive(Debug)]
166pub(super) struct RealObjectStack {
167    pub(super) upper: Option<RealObject>,
168    pub(super) lowers: Vec<RealObject>,
169}
170
171impl RealObjectStack {
172    pub(super) fn new(upper: Option<RealObject>, lowers: Vec<RealObject>) -> Self {
173        debug_assert!(upper.is_some() || !lowers.is_empty());
174        Self { upper, lowers }
175    }
176
177    pub(super) fn upper_only(upper: RealObject) -> Self {
178        Self {
179            upper: Some(upper),
180            lowers: Vec::new(),
181        }
182    }
183
184    pub(super) fn lower_only(lower: RealObject) -> Self {
185        Self {
186            upper: None,
187            lowers: vec![lower],
188        }
189    }
190
191    pub(super) fn visible_source(&self) -> &RealObject {
192        match &self.upper {
193            Some(upper) => upper,
194            None => &self.lowers[0],
195        }
196    }
197}

199/// Drives one real directory through `visitor` until it reports exhaustion.
200pub(super) fn read_all_dirents(
201    real_dir: &Arc<dyn Inode>,
202    visitor: &mut dyn DirentVisitor,
203) -> Result<()> {
204    // A single `readdir_at` may stop short of the end, so the loop advances by what it reported.
205    let mut offset = 0;
206    loop {
207        match real_dir.readdir_at(offset, visitor)? {
208            0 => break,
209            visited => offset += visited,
210        }
211    }
212    Ok(())
213}
214
215pub(super) fn read_child_names(real_dir: &Arc<dyn Inode>) -> Result<Vec<String>> {
216    let mut names: Vec<String> = Vec::new();
217    read_all_dirents(real_dir, &mut names)?;
218    names.retain(|name| !path::is_dot_or_dotdot(name));
219    Ok(names)
220}
