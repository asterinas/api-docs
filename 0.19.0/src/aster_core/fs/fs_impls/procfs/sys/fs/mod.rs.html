21/// Represents the inode at `/proc/sys/fs`.
22pub(super) struct FsDirOps;
23
24impl FsDirOps {
25    pub(super) fn new_inode(parent: Weak<dyn Inode>) -> Arc<dyn Inode> {
26        // Reference:
27        // <https://elixir.bootlin.com/linux/v6.16.5/source/fs/file_table.c#L139>
28        // <https://elixir.bootlin.com/linux/v6.16.5/source/fs/proc/proc_sysctl.c#L978>
29        ProcDir::new(Self, parent, mkmod!(a+rx))
30    }
31
32    const STATIC_ENTRIES: &'static [StaticEntry] =
33        &[("nr_open", InodeType::File, NrOpenFileOps::new_inode)];
34}
35
36impl ProcDirOps for FsDirOps {
37    fn lookup_child(&self, this_dir: &ProcDir<Self>, name: &str) -> Result<Arc<dyn Inode>> {
38        if let Some(child) = lookup_child_from_table(name, Self::STATIC_ENTRIES, |f| {
39            (f)(this_dir.this_weak().clone())
40        }) {
41            return Ok(child);
42        }
43
44        return_errno_with_message!(Errno::ENOENT, "the file does not exist");
45    }
46
47    fn visit_entries_from_offset<'a, F>(&'a self, offset: usize, visit_fn: F) -> Result<()>
48    where
49        F: FnMut(ReaddirEntry<'a>) -> Result<()>,
50    {
51        visit_listed_entries(
52            offset,
53            listed_entries_from_table(Self::STATIC_ENTRIES),
54            visit_fn,
55        )
56    }
57}
