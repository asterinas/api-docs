3//! This module offers `/proc/uptime` file support, which provides
4//! information about the system uptime and idle time.
5//!
6//! Reference: <https://man7.org/linux/man-pages/man5/proc_uptime.5.html>
7
8use core::time::Duration;
9
10use aster_util::printer::VmPrinter;
11
12use crate::{
13    fs::{
14        file::mkmod,
15        procfs::template::{ProcFile, ProcFileOps},
16        vfs::inode::Inode,
17    },
18    prelude::*,
19    time::{NSEC_PER_SEC, cpu_time_stats::CpuTimeStatsManager},
20};
21
22/// Represents the inode at `/proc/uptime`.
23pub(super) struct UptimeFileOps;
24
25impl UptimeFileOps {
26    pub(super) fn new_inode(parent: Weak<dyn Inode>) -> Arc<dyn Inode> {
27        // Reference:
28        // <https://elixir.bootlin.com/linux/v6.16.5/source/fs/proc/uptime.c#L45>
29        // <https://elixir.bootlin.com/linux/v6.16.5/source/fs/proc/generic.c#L549-L550>
30        ProcFile::new(Self, parent, mkmod!(a+r))
31    }
32
33    fn print_uptime(printer: &mut VmPrinter) -> Result<()> {
34        let (uptime_secs, uptime_centis) =
35            duration_to_seconds_and_centiseconds(aster_time::read_monotonic_time());
36
37        let cpustat = CpuTimeStatsManager::singleton();
38        let (idle_secs, idle_centis) = duration_to_seconds_and_centiseconds(
39            cpustat.collect_stats_on_all_cpus().idle.as_duration(),
40        );
41
42        writeln!(
43            printer,
44            "{uptime_secs}.{uptime_centis:02} {idle_secs}.{idle_centis:02}"
45        )?;
46        Ok(())
47    }
48}
49
50/// Splits a duration into seconds and the centiseconds printed by `/proc/uptime`.
51fn duration_to_seconds_and_centiseconds(duration: Duration) -> (u64, u32) {
52    let centiseconds = duration.subsec_nanos() / (NSEC_PER_SEC as u32 / 100);
53    (duration.as_secs(), centiseconds)
54}
55
56impl ProcFileOps for UptimeFileOps {
57    fn read_at(&self, offset: usize, writer: &mut VmWriter) -> Result<usize> {
58        let mut printer = VmPrinter::new_skip(writer, offset);
59
60        Self::print_uptime(&mut printer)?;
61
62        Ok(printer.bytes_written())
63    }
64}

66#[cfg(ktest)]
67mod tests {
68    use core::time::Duration;
69
70    use ostd::prelude::ktest;
71
72    use super::duration_to_seconds_and_centiseconds;
73
74    #[ktest]
75    fn uptime_centiseconds_are_truncated() {
76        assert_eq!(
77            duration_to_seconds_and_centiseconds(Duration::new(42, 0)),
78            (42, 0)
79        );
80        assert_eq!(
81            duration_to_seconds_and_centiseconds(Duration::new(42, 9_999_999)),
82            (42, 0)
83        );
84        assert_eq!(
85            duration_to_seconds_and_centiseconds(Duration::new(42, 10_000_000)),
86            (42, 1)
87        );
88        assert_eq!(
89            duration_to_seconds_and_centiseconds(Duration::new(42, 999_999_999)),
90            (42, 99)
91        );
92    }
