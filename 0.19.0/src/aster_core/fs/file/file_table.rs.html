16/// Represents a validated, non-negative file descriptor.
17///
18/// The value is guaranteed to be in the range `[0, i32::MAX]`.
19/// Use [`RawFileDesc`] at syscall boundaries,
20/// then convert to `FileDesc` via `TryFrom` for kernel-internal use.
21///
22/// Some system calls (e.g., `fcntl`) reinterpret
23/// values of types other than [`RawFileDesc`] (e.g., `u64` or `usize`)
24/// as file descriptors. Linux typically truncates the high bits
25/// without checking whether the full argument fits in range.
26/// To avoid accidental misuse, we do not implement `TryFrom` for
27/// those types. The syscall layer should first convert them to
28/// a [`RawFileDesc`] in an explicit, syscall-specific way,
29/// and then convert that value to a `FileDesc`.
30#[derive(Clone, Copy, Debug, Eq, Ord, PartialEq, PartialOrd)]
31pub(crate) struct FileDesc(RangedU32<0, { i32::MAX as _ }>);
32
33/// A raw file descriptor as received from or returned to user space.
34///
35/// This is the `int` type from Linux syscall signatures.
36/// It may hold negative sentinel values like `AT_FDCWD` (-100).
37/// Convert to [`FileDesc`] via `TryFrom` before use.
38pub(crate) type RawFileDesc = i32;
39
40impl FileDesc {
41    /// File descriptor 0.
42    pub(crate) const ZERO: Self = Self(RangedU32::new(0));
43}
44
45impl Display for FileDesc {
46    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
47        write!(f, "{}", self.0.get())
48    }
49}

81// Intentionally, `TryFrom<RawFileDesc>` is
82// the only `TryFrom` implementation for `FileDesc`.
83//
84// We do not implement conversions from wider integer types
85// for Linux compatibility reasons; see the `FileDesc` type docs.
86// We also do not implement `TryFrom<u32>` directly,
87// to encourage callers to name `RawFileDesc` explicitly
88// instead of using a plain `u32`.
89impl TryFrom<RawFileDesc> for FileDesc {
90    type Error = Error;
91
92    fn try_from(value: RawFileDesc) -> Result<Self> {
93        if value < 0 {
94            return_errno_with_message!(Errno::EBADF, "negative FDs are not valid");
95        }
96        Ok(Self(RangedU32::new(value.cast_unsigned())))
97    }
98}

100/// A file table.
101///
102/// A file table is created within an [`RwArc`] and remains within it (see [`Self::new`] and
103/// [`Self::fork_from`]). The file table's address is used as a [`RangeLockOwner`], so users should
104/// not try to move the file table to a different address.
105pub(crate) struct FileTable {
106    table: SlotVec<FileTableEntry>,
107}
108
109impl FileTable {
110    /// Creates a new file table.
111    pub(crate) fn new() -> RwArc<Self> {
112        RwArc::new(Self {
113            table: SlotVec::new(),
114        })
115    }
116
117    /// Creates a new file table containing clones of all entries in `parent`.
118    pub(crate) fn fork_from(parent: &Self) -> RwArc<Self> {
119        RwArc::new(Self {
120            table: parent.table.clone(),
121        })
122    }
123
124    /// Returns the range-lock owner of `file_table`.
125    pub(crate) fn range_lock_owner(file_table: &RwArc<Self>) -> RangeLockOwner {
126        RangeLockOwner::from_address(file_table.as_ptr().addr())
127    }
128
129    fn as_range_lock_owner(&self) -> RangeLockOwner {
130        RangeLockOwner::from_address(self as *const Self as usize)
131    }
132
133    pub(crate) fn len(&self) -> usize {
134        self.table.slots_len()
135    }
136
137    /// Duplicates `fd` onto the lowest-numbered available descriptor equal to
138    /// or greater than `ceil_fd`.
139    pub(crate) fn dup_ceil(
140        &mut self,
141        fd: FileDesc,
142        ceil_fd: FileDesc,
143        flags: FdFlags,
144    ) -> Result<FileDesc> {
145        let entry = self.duplicate_entry(fd, flags)?;
146
147        // Get the lowest-numbered available fd equal to or greater than `ceil_fd`.
148        let get_min_free_fd = || -> usize {
149            let ceil_fd = ceil_fd.into();
150            if self.table.get(ceil_fd).is_none() {
151                return ceil_fd;
152            }
153
154            for idx in ceil_fd + 1..self.len() {
155                if self.table.get(idx).is_none() {
156                    return idx;
157                }
158            }
159            self.len()
160        };
161
162        let min_free_fd = get_min_free_fd();
163        self.table.put_at(min_free_fd, entry);
164        // Resource limits guarantee the table never exceeds `i32::MAX` entries.
165        Ok((min_free_fd as RawFileDesc).try_into().unwrap())
166    }
167
168    /// Duplicates `fd` onto the exact descriptor number `new_fd`.
169    pub(crate) fn dup_exact(
170        &mut self,
171        fd: FileDesc,
172        new_fd: FileDesc,
173        flags: FdFlags,
174    ) -> Result<Option<ClosedFile>> {
175        let entry = self.duplicate_entry(fd, flags)?;
176        let closed_file = self.close_file(new_fd);
177        self.table.put_at(new_fd.into(), entry);
178        Ok(closed_file)
179    }
180
181    fn duplicate_entry(&self, fd: FileDesc, flags: FdFlags) -> Result<FileTableEntry> {
182        let file = self
183            .table
184            .get(fd.into())
185            .map(|entry| entry.file.clone())
186            .ok_or_else(|| Error::with_message(Errno::EBADF, "the FD does not exist"))?;
187        Ok(FileTableEntry::new(file, flags))
188    }
189
190    pub(crate) fn insert(&mut self, item: Arc<dyn FileLike>, flags: FdFlags) -> FileDesc {
191        let entry = FileTableEntry::new(item, flags);
192        // Resource limits guarantee the table never exceeds `i32::MAX` entries.
193        (self.table.put(entry) as RawFileDesc).try_into().unwrap()
194    }
195
196    pub(crate) fn close_file(&mut self, fd: FileDesc) -> Option<ClosedFile> {
197        let removed_entry = self.table.remove(fd.into())?;
198        // POSIX record locks are process-associated and Linux drops them when any fd for the inode is
199        // closed by that process, even if duplicated descriptors still exist.
200        //
201        // Reference: <https://man7.org/linux/man-pages/man2/fcntl_locking.2.html>
202        Some(ClosedFile::new(
203            removed_entry.file,
204            self.as_range_lock_owner(),
205        ))
206    }
207
208    pub(crate) fn close_files_on_exec(&mut self) -> Vec<ClosedFile> {
209        self.close_files(|entry| entry.flags().contains(FdFlags::CLOEXEC))
210    }
211
212    fn close_files<F>(&mut self, should_close: F) -> Vec<ClosedFile>
213    where
214        F: Fn(&FileTableEntry) -> bool,
215    {
216        let mut closed_files = Vec::new();
217        let closed_fds: Vec<FileDesc> = self
218            .table
219            .idxes_and_items()
220            .filter_map(|(idx, entry)| {
221                if should_close(entry) {
222                    // Resource limits guarantee the table never exceeds `i32::MAX` entries.
223                    Some((idx as RawFileDesc).try_into().unwrap())
224                } else {
225                    None
226                }
227            })
228            .collect();
229
230        for fd in closed_fds {
231            closed_files.push(self.close_file(fd).unwrap());
232        }
233
234        closed_files
235    }
236
237    pub(crate) fn get_file(&self, fd: FileDesc) -> Result<&Arc<dyn FileLike>> {
238        self.table
239            .get(fd.into())
240            .map(|entry| entry.file())
241            .ok_or_else(|| Error::with_message(Errno::EBADF, "the FD does not exist"))
242    }
243
244    pub(crate) fn get_entry(&self, fd: FileDesc) -> Result<&FileTableEntry> {
245        self.table
246            .get(fd.into())
247            .ok_or_else(|| Error::with_message(Errno::EBADF, "the FD does not exist"))
248    }
249
250    pub(crate) fn get_entry_mut(&mut self, fd: FileDesc) -> Result<&mut FileTableEntry> {
251        self.table
252            .get_mut(fd.into())
253            .ok_or_else(|| Error::with_message(Errno::EBADF, "the FD does not exist"))
254    }
255
256    pub(crate) fn fds_and_files(&self) -> impl Iterator<Item = (FileDesc, &'_ Arc<dyn FileLike>)> {
257        // Resource limits guarantee the table never exceeds `i32::MAX` entries.
258        self.table
259            .idxes_and_items()
260            .map(|(idx, entry)| ((idx as RawFileDesc).try_into().unwrap(), entry.file()))
261    }
262}
263
264impl Drop for FileTable {
265    fn drop(&mut self) {
266        for (_, file) in self.fds_and_files() {
267            if let Ok(inode_handle) = file.as_inode_handle_or_err() {
268                inode_handle.release_range_locks(self.as_range_lock_owner());
269            }
270        }
271    }
272}

274/// A file removed from a [`FileTable`] whose close cleanup is pending.
275///
276/// Dropping this value may block, so it must be dropped after releasing the file-table lock.
277#[must_use = "close cleanup runs when this value is dropped"]
278pub(crate) struct ClosedFile {
279    file: Arc<dyn FileLike>,
280    range_lock_owner: RangeLockOwner,
281}
282
283impl ClosedFile {
284    fn new(file: Arc<dyn FileLike>, range_lock_owner: RangeLockOwner) -> Self {
285        Self {
286            file,
287            range_lock_owner,
288        }
289    }
290}
291
292impl Drop for ClosedFile {
293    fn drop(&mut self) {
294        if let Ok(inode_handle) = self.file.as_inode_handle_or_err() {
295            inode_handle.release_range_locks(self.range_lock_owner);
296        }
297    }

300/// A helper trait that provides methods to operate the file table.
301pub(crate) trait WithFileTable {
302    /// Calls `f` with the file table.
303    ///
304    /// This method is lockless if the file table is not shared. Otherwise, `f` is called while
305    /// holding the read lock on the file table.
306    fn read_with<R>(&mut self, f: impl FnOnce(&FileTable) -> R) -> R;
307}
308
309impl WithFileTable for FileTableRefMut<'_> {
310    fn read_with<R>(&mut self, f: impl FnOnce(&FileTable) -> R) -> R {
311        let file_table = self.unwrap();
312
313        if let Some(inner) = file_table.get() {
314            f(inner)
315        } else {
316            f(&file_table.read())
317        }
318    }
319}

321/// Gets a file from a file descriptor as fast as possible.
322///
323/// `file_table` should be a mutable borrow of the file table contained in the `file_table` field
324/// (which is a [`RefCell`]) in [`ThreadLocal`]. A mutable borrow is required because its
325/// exclusivity can be useful for achieving lockless file lookups.
326///
327/// If the file table is not shared with another thread, this macro will be free of locks
328/// ([`RwArc::read`]) and free of reference counting ([`Arc::clone`]).
329///
330/// If the file table is shared, the read lock is taken, the file is cloned, and then the read lock
331/// is released. Cloning and releasing the lock is necessary because we cannot hold such locks when
332/// operating on files, since many operations on files can block.
333///
334/// Note: This has to be a macro due to a limitation in the Rust borrow check implementation. Once
335/// <https://github.com/rust-lang/rust/issues/58910> is fixed, we can try to convert this macro to
336/// a function.
337///
338/// [`RefCell`]: core::cell::RefCell
339/// [`ThreadLocal`]: crate::process::posix_thread::ThreadLocal
340/// [`RwArc::read`]: ostd::sync::RwArc::read
341macro_rules! get_file_fast {
342    ($file_table:expr, $file_desc:expr) => {{
343        use alloc::borrow::Cow;
344
345        use ostd::sync::RwArc;
346        use $crate::{
347            fs::file::file_table::{FileDesc, FileTable},
348            process::posix_thread::FileTableRefMut,
349        };
350
351        let file_table: &mut FileTableRefMut<'_> = $file_table;
352        let file_table: &mut RwArc<FileTable> = file_table.unwrap();
353        let file_desc: FileDesc = $file_desc;
354
355        if let Some(inner) = file_table.get() {
356            // Fast path: The file table is not shared, we can get the file in a lockless way.
357            Cow::Borrowed(inner.get_file(file_desc)?)
358        } else {
359            // Slow path: The file table is shared, we need to hold the lock and clone the file.
360            Cow::Owned(file_table.read().get_file(file_desc)?.clone())
361        }
362    }};
363}

365pub(crate) use get_file_fast;

367pub(crate) struct FileTableEntry {
368    file: Arc<dyn FileLike>,
369    flags: AtomicU8,
370}
371
372impl FileTableEntry {
373    pub(crate) fn new(file: Arc<dyn FileLike>, flags: FdFlags) -> Self {
374        Self {
375            file,
376            flags: AtomicU8::new(flags.bits()),
377        }
378    }
379
380    pub(crate) fn file(&self) -> &Arc<dyn FileLike> {
381        &self.file
382    }
383
384    pub(crate) fn flags(&self) -> FdFlags {
385        FdFlags::from_bits(self.flags.load(Ordering::Relaxed)).unwrap()
386    }
387
388    pub(crate) fn set_flags(&self, flags: FdFlags) {
389        self.flags.store(flags.bits(), Ordering::Relaxed);
390    }
391}
392
393impl Clone for FileTableEntry {
394    fn clone(&self) -> Self {
395        Self {
396            file: self.file.clone(),
397            flags: AtomicU8::new(self.flags.load(Ordering::Relaxed)),
398        }
399    }
400}
