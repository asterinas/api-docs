43/// Unpacks the boot initramfs into the bootstrap root filesystem.
44///
45/// Returns successfully without changing the filesystem when no initramfs was supplied.
46pub(crate) fn init_in_first_kthread(path_resolver: &PathResolver) -> Result<()> {
47    let Some(initramfs_buf) = boot_info().initramfs else {
48        return Ok(());
49    };
50
51    match &initramfs_buf[..4] {
52        // Gzip magic number: 0x1F 0x8B
53        &[0x1F, 0x8B, _, _] => {
54            println!("[kernel] unpacking initramfs.cpio.gz to rootfs ...");
55            unpack_to_rootfs(
56                CpioDecoder::new(GzipReader::new(initramfs_buf)),
57                path_resolver,
58            )?;
59        }
60        _ => {
61            println!("[kernel] unpacking initramfs.cpio to rootfs ...");
62            unpack_to_rootfs(CpioDecoder::new(Cursor::new(initramfs_buf)), path_resolver)?;
63        }
64    };
65
66    ensure_dev_console(path_resolver)?;
67
68    println!("[kernel] initramfs is ready");
69    Ok(())
70}

72/// Unpacks every entry of a CPIO archive into the rootfs.
73fn unpack_to_rootfs<R: Read>(
74    mut decoder: CpioDecoder<R>,
75    path_resolver: &PathResolver,
76) -> Result<()> {
77    while let Some(entry_result) = decoder.next() {
78        let mut entry = entry_result?;
79        if let Err(e) = try_append_entry_to_rootfs(&mut entry, path_resolver) {
80            warn!("failed to add entry {} to rootfs: {:?}", entry.name(), e);
81        }
82    }
83    Ok(())
84}

86fn ensure_dev_console(path_resolver: &PathResolver) -> Result<()> {
87    // Linux's default built-in initramfs provides /dev and /dev/console so
88    // early userspace can open the console even if the external initramfs does
89    // not carry this node. Asterinas provides the same default entries after
90    // unpacking the supplied initramfs.
91    // Reference: <https://elixir.bootlin.com/linux/v6.13/source/usr/default_cpio_list>.
92    let dev_path = super::lookup_or_create_dev(path_resolver)?;
93    match path_resolver.lookup(&FsPath::try_from("/dev/console")?) {
94        Ok(_) => Ok(()),
95        Err(error) if error.error() == Errno::ENOENT => {
96            dev_path.mknod(
97                "console",
98                mkmod!(u+rw),
99                MknodType::CharDevice(tty::CONSOLE_DEVICE_ID.as_encoded_u64()),
100            )?;
101            Ok(())
102        }
103        Err(error) => Err(error),
104    }
105}

107/// Finds the init program to run from the initramfs.
108///
109/// Resolves the path specified by `rdinit`, or `/init` when `rdinit` is not provided, and returns
110/// the resolved path together with the original pathname. Returns an error if the pathname is
111/// invalid or cannot be resolved.
112pub(crate) fn find_init(path_resolver: &PathResolver) -> Result<(Path, &'static str)> {
113    const DEFAULT_INITRAMFS_INIT_PATH: &str = "/init";
114
115    let init_path = RDINIT_PATH
116        .get()
117        .map(String::as_str)
118        .unwrap_or(DEFAULT_INITRAMFS_INIT_PATH);
119    let path = path_resolver.lookup(&FsPath::try_from(init_path)?)?;
120    Ok((path, init_path))
121}

123fn try_append_entry_to_rootfs<R: Read>(
124    entry: &mut CpioEntry<R>,
125    path_resolver: &PathResolver,
126) -> Result<()> {
127    // Make sure the name is a relative path, and is not end with "/".
128    let entry_name = entry.name().trim_start_matches('/').trim_end_matches('/');
129    if entry_name.is_empty() {
130        return_errno_with_message!(Errno::EINVAL, "invalid entry name");
131    }
132    if is_dot(entry_name) {
133        return Ok(());
134    }
135
136    // Here we assume that the directory referred by "prefix" must has been created.
137    // The basis of this assumption is：
138    // The mkinitramfs script uses `find` command to ensure that the entries are
139    // sorted that a directory always appears before its child directories and files.
140    let (parent, name) = if let Some((prefix, last)) = entry_name.rsplit_once('/') {
141        (path_resolver.lookup(&FsPath::try_from(prefix)?)?, last)
142    } else {
143        (path_resolver.root().clone(), entry_name)
144    };
145
146    let metadata = entry.metadata();
147    let mode = InodeMode::from_bits_truncate(metadata.permission_mode());
148    match metadata.file_type() {
149        FileType::File => {
150            let path = parent.new_child(name, InodeType::File, mode)?;
151            let writer = InodeWriter {
152                inner: path.inode().as_ref(),
153                offset: 0,
154            };
155            entry.read_all(writer)?;
156        }
157        FileType::Dir => {
158            let _ = parent.new_child(name, InodeType::Dir, mode)?;
159        }
160        FileType::Link => {
161            // Obtain the owned name here. Otherwise, the later mutable borrow of `entry`
162            // will conflict with the immutable borrow of `name` here.
163            let child_name = name.to_owned();
164
165            let mut link_data: Vec<u8> = Vec::new();
166            entry.read_all(&mut link_data)?;
167            let link_content = core::str::from_utf8(&link_data)?;
168
169            parent.new_symlink_child(&child_name, link_content, mode)?;
170        }
171        FileType::Char => {
172            let device_id = try_device_id_from_metadata(metadata)?;
173            parent.mknod(name, mode, MknodType::CharDevice(device_id))?;
174        }
175        FileType::Block => {
176            let device_id = try_device_id_from_metadata(metadata)?;
177            parent.mknod(name, mode, MknodType::BlockDevice(device_id))?;
178        }
179        FileType::FiFo => {
180            parent.mknod(name, mode, MknodType::NamedPipe)?;
181        }
182        FileType::Socket => {
183            return_errno_with_message!(Errno::EINVAL, "socket files are not supported in initramfs")
184        }
185    }
186
187    Ok(())
188}

190/// A streaming gzip decompressor over an in-memory compressed buffer.
191struct GzipReader<'a> {
192    // The DEFLATE body not yet consumed, shrinking as it is read.
193    deflate_body: &'a [u8],
194    state: Box<InflateState>,
195    done: bool,
196}
197
198impl<'a> GzipReader<'a> {
199    /// Creates a decompressor, parsing and skipping the gzip header of `buf`.
200    fn new(buf: &'a [u8]) -> Self {
201        Self {
202            deflate_body: strip_gzip_header(buf),
203            state: InflateState::new_boxed(DataFormat::Raw),
204            done: false,
205        }
206    }
207}
208
209impl Read for GzipReader<'_> {
210    fn read(&mut self, out: &mut [u8]) -> io::Result<usize> {
211        if out.is_empty() {
212            return Ok(0);
213        }
214
215        while !self.done {
216            let result = inflate(&mut self.state, self.deflate_body, out, MZFlush::None);
217            self.deflate_body = &self.deflate_body[result.bytes_consumed..];
218            match result.status {
219                Ok(MZStatus::StreamEnd) => self.done = true,
220                Ok(_) => {}
221                Err(_) => {
222                    return Err(io::Error::new(
223                        io::ErrorKind::InvalidData,
224                        "gzip decompression failed",
225                    ));
226                }
227            }
228            if result.bytes_written > 0 {
229                return Ok(result.bytes_written);
230            }
231            // No output and no input consumed means no further progress is
232            // possible (truncated stream); stop to avoid spinning forever.
233            if result.bytes_consumed == 0 {
234                break;
235            }
236        }
237
238        Ok(0)
239    }
240}

242/// Parses the gzip header and returns the remaining bytes.
243///
244/// # Panics
245///
246/// Panics if `buf` contains an invalid or incomplete gzip header.
247fn strip_gzip_header(buf: &[u8]) -> &[u8] {
248    // The gzip header flag bits.
249    // Reference: <https://datatracker.ietf.org/doc/html/rfc1952>.
250    const FLG_FHCRC: u8 = 0x02;
251    const FLG_FEXTRA: u8 = 0x04;
252    const FLG_FNAME: u8 = 0x08;
253    const FLG_FCOMMENT: u8 = 0x10;
254
255    // Fixed 10-byte header: ID1, ID2, CM, FLG, MTIME(4), XFL, OS.
256    let flg = buf[3];
257    let mut pos = 10;
258
259    if flg & FLG_FEXTRA != 0 {
260        let xlen = u16::from_le_bytes([buf[pos], buf[pos + 1]]) as usize;
261        pos += 2 + xlen;
262    }
263    if flg & FLG_FNAME != 0 {
264        pos += buf[pos..].iter().position(|&b| b == 0).unwrap() + 1;
265    }
266    if flg & FLG_FCOMMENT != 0 {
267        pos += buf[pos..].iter().position(|&b| b == 0).unwrap() + 1;
268    }
269    if flg & FLG_FHCRC != 0 {
270        pos += 2;
271    }
272
273    &buf[pos..]
274}

276struct InodeWriter<'a> {
277    inner: &'a dyn Inode,
278    offset: usize,
279}
280
281impl io::Write for InodeWriter<'_> {
282    fn write(&mut self, buf: &[u8]) -> io::Result<usize> {
283        let mut reader = VmReader::from(buf).to_fallible();
284        let write_len = self
285            .inner
286            .write_at(self.offset, &mut reader, StatusFlags::empty())
287            .map_err(|_| io::ErrorKind::WriteZero)?;
288        self.offset += write_len;
289        Ok(write_len)
290    }
291
292    fn flush(&mut self) -> io::Result<()> {
293        Ok(())
294    }
295}

297fn try_device_id_from_metadata(metadata: &FileMetadata) -> Result<u64> {
298    let major = {
299        let dev_maj = u16::try_from(metadata.rdev_maj())?;
300        MajorId::try_from(dev_maj).map_err(|msg| Error::with_message(Errno::EINVAL, msg))?
301    };
302    let minor = MinorId::try_from(metadata.rdev_min())
303        .map_err(|msg| Error::with_message(Errno::EINVAL, msg))?;
304    Ok(DeviceId::new(major, minor).as_encoded_u64())
305}
