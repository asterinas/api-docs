48/// A `Path` is used to represent an exact location in the VFS tree.
49///
50/// Each `Path` corresponds to a node in the VFS tree, and a single node
51/// may have multiple `Path` instances referencing it due to mount operations.
52#[derive(Clone, Debug)]
53pub struct Path {
54    mount: Arc<Mount>,
55    dentry: Arc<Dentry>,
56}
57
58impl PartialEq for Path {
59    fn eq(&self, other: &Self) -> bool {
60        self.mount.id() == other.mount.id() && Arc::ptr_eq(&self.dentry, &other.dentry)
61    }
62}

68    pub(crate) fn new_root(mount: Arc<Mount>) -> Self {
69        let inner = mount.root_dentry().clone();
70        Self::new(mount, inner)

73    /// Creates a new child `Path` in a file system.
74    ///
75    /// This API is for non-symbolic-link children.
76    /// Use [`Self::new_symlink_child`] to create a symbolic link, because a
77    /// symbolic link should be created atomically with its target.
78    pub(crate) fn new_child(&self, name: &str, type_: InodeType, mode: InodeMode) -> Result<Self> {
79        debug_assert_ne!(type_, InodeType::SymLink);
80
81        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
82        self.check_dir_entry_mutation()?;
83        let new_child_dentry = dir_dentry.create_child(name, || {
84            dir_dentry.inode().create(&dir_dentry, name, type_, mode)
85        })?;
86        Ok(Self::new(self.mount.clone(), new_child_dentry))
87    }

89    /// Creates a new symbolic-link child `Path` in a file system.
90    ///
91    /// Use this API instead of [`Self::new_child`] when creating symbolic
92    /// links so the link target is supplied at creation time.
93    pub(crate) fn new_symlink_child(
94        &self,
95        name: &str,
96        target: &str,
97        mode: InodeMode,
98    ) -> Result<Self> {
99        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
100        self.check_dir_entry_mutation()?;
101        let new_child_dentry = dir_dentry.create_child(name, || {
102            dir_dentry
103                .inode()
104                .create_symlink(&dir_dentry, name, target, mode)
105        })?;
106        Ok(Self::new(self.mount.clone(), new_child_dentry))
107    }

110    ///
111    /// This does not interpret special names, check search permissions, or traverse mount points.
112    pub(crate) fn lookup_child(&self, name: &str) -> Result<Self> {
113        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;

118    /// Creates a new `Path` to represent an unnamed temporary file.
119    ///
120    /// The returned inode has no directory entry and is invisible to `readdir`.
121    /// If created hard-linkable, it may later be given a name via a link
122    /// operation; otherwise it can never be linked.
123    pub(crate) fn create_tmpfile(
124        &self,
125        mode: InodeMode,
126        hard_linkability: HardLinkability,
127    ) -> Result<Self> {
128        let dir_dentry = self.dentry.as_dir_dentry_or_err()?;
129        self.check_dir_entry_mutation()?;
130        let tmp_inode = self
131            .inode()
132            .create_tmpfile(&self.dentry, mode, hard_linkability)?;
133        let tmp_dentry = Dentry::new_anonymous(tmp_inode, &dir_dentry);
134        Ok(Self::new(self.mount.clone(), tmp_dentry))
135    }

166    /// Opens the `Path` with the given `OpenArgs`.
167    ///
168    /// Returns an `InodeHandle` on success.
169    pub(crate) fn open(&self, open_args: OpenArgs) -> Result<InodeHandle> {
170        let inode = self.inode().as_ref();
171        let inode_type = inode.type_();
172        let creation_flags = &open_args.creation_flags;
173        let status_flags = &open_args.status_flags;
174
175        if creation_flags.contains(CreationFlags::O_CREAT)
176            && creation_flags.contains(CreationFlags::O_EXCL)
177        {
178            return_errno_with_message!(Errno::EEXIST, "the file already exists");
179        }
180        if creation_flags.contains(CreationFlags::O_CREAT) && inode_type == InodeType::Dir {
181            return_errno_with_message!(
182                Errno::EISDIR,
183                "O_CREAT is specified but the file is a directory"
184            );
185        }
186
187        if inode_type == InodeType::SymLink
188            && creation_flags.contains(CreationFlags::O_NOFOLLOW)
189            && !status_flags.contains(StatusFlags::O_PATH)
190        {
191            return_errno_with_message!(Errno::ELOOP, "the file is a symlink");
192        }
193        if creation_flags.contains(CreationFlags::O_DIRECTORY) && inode_type != InodeType::Dir {
194            return_errno_with_message!(
195                Errno::ENOTDIR,
196                "O_DIRECTORY is specified but the file is not a directory"
197            );
198        }
199
200        if inode_type.is_regular_file()
201            && creation_flags.contains(CreationFlags::O_TRUNC)
202            && !status_flags.contains(StatusFlags::O_PATH)
203        {
204            self.resize(0)?;
205        }
206
207        InodeHandle::new(self.clone(), open_args.access_mode, *status_flags)
208    }

210    /// Gets the parent `Path` within the same mount.
211    ///
212    /// This method returns the parent path within the same filesystem/mount.
213    /// It does NOT cross mount boundaries. If the current path is the root of a mount,
214    /// it will return `None`.
215    ///
216    /// For cross-filesystem parent lookup, use `effective_parent()` instead.
217    pub(super) fn parent_within_mount(&self) -> Option<Self> {
218        let parent = self.dentry.parent()?;
219        Some(Self::new(self.mount.clone(), parent))
220    }

222    /// Gets the top `Path` of the current.
223    ///
224    /// Used when different file systems are mounted on the same mount point.
225    ///
226    /// For example, first `mount /dev/sda1 /mnt` and then `mount /dev/sda2 /mnt`.
227    /// After the second mount is completed, the content of the first mount will be overridden.
228    /// We need to recursively obtain the top `Path`.
229    pub(crate) fn get_top_path(mut self) -> Self {
230        while self.dentry.is_mountpoint() {
231            if let Some(child_mount) = self.mount.get(&self.dentry) {
232                let inner = child_mount.root_dentry().clone();
233                self = Self::new(child_mount, inner);
234            } else {
235                break;
236            }
237        }
238
239        self
240    }

254    /// Checks if this path is reachable from the given `root` path.
255    ///
256    /// A path is considered reachable if it is the same as or a descendant
257    /// of the `root` path. The check traverses upwards from the current path,
258    /// crossing mount point boundaries as necessary, until it either finds
259    /// the `root` path or reaches the global root.
260    fn is_reachable_from(&self, root: &Path, _topology: &MountTopology) -> bool {
261        let mut owned;
262        let mut current = self;
263
264        loop {
265            if current.mount.id() == root.mount.id() {
266                return current.dentry.is_equal_or_descendant_of(&root.dentry);
267            }
268
269            let Some(parent_mount) = current.mount.parent().and_then(|mount| mount.upgrade())
270            else {
271                // `upgrade()` may fail if `current` is in a mount namespace that has been
272                // dropped. In this situation, `current` is not reachable from `root` because
273                // `root` holds strong reference counts for all its descendant mounts.
274                return false;
275            };
276
277            owned = Path::new(
278                parent_mount,
279                current
280                    .mount
281                    .mountpoint()
282                    .expect("Mounts with parents must have a mount point")
283                    .clone(),
284            );
285            current = &owned;
286        }
287    }

298    /// Checks whether this path is a searchable directory.
299    fn check_dir_search_permission(&self) -> Result<()> {
300        if self.type_() != InodeType::Dir {
301            return_errno_with_message!(Errno::ENOTDIR, "must be directory");
302        }
303
304        self.inode()
305            .check_permission(Permission::MAY_EXEC | Permission::MAY_CHDIR)
306    }

308    /// Checks whether the path is on a writable mount and filesystem.
309    fn check_mount_writable(&self) -> Result<()> {
310        if self.mount.flags().contains(PerMountFlags::RDONLY)
311            || self.fs().flags().contains(FsFlags::RDONLY)
312        {
313            return_errno_with_message!(Errno::EROFS, "the mount or filesystem is read-only");
314        }
315
316        Ok(())
317    }

319    /// Checks whether a directory entry may be modified in this directory.
320    fn check_dir_entry_mutation(&self) -> Result<()> {
321        self.check_mount_writable()?;
322        self.inode()
323            .check_permission(Permission::MAY_WRITE | Permission::MAY_EXEC)
324    }

326    /// Checks whether an inode may be used as the source of a hard link.
327    fn check_hardlink_source(&self) -> Result<()> {
328        let current_thread = current_thread!();
329        let Some(posix_thread) = current_thread.as_posix_thread() else {
330            return Ok(());
331        };
332
333        let metadata = self.metadata()?;
334        let fsuid = posix_thread.credentials().fsuid();
335
336        // The source inode owner can hardlink all they like.
337        if fsuid == metadata.uid {
338            return Ok(());
339        }
340
341        // Hardlinking to unreadable or unwritable sources is dangerous.
342        if self
343            .inode()
344            .check_permission(Permission::MAY_READ | Permission::MAY_WRITE)
345            .is_err()
346        {
347            return_errno_with_message!(
348                Errno::EPERM,
349                "the source is not permitted to be hard-linked"
350            );
351        }
352
353        // Non-owners may only hard-link regular files.
354        if self.type_() != InodeType::File {
355            return_errno_with_message!(
356                Errno::EPERM,
357                "the source is not permitted to be hard-linked"
358            );
359        }
360
361        // Hardlinking to setuid or executable setgid files requires CAP_FOWNER.
362        if (metadata.mode.has_set_uid()
363            || (metadata.mode.has_set_gid() && metadata.mode.is_group_executable()))
364            && lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
365                UserNamespace::get_init_singleton().as_ref(),
366                posix_thread,
367                CapSet::FOWNER,
368            ))
369            .is_err()
370        {
371            return_errno_with_message!(
372                Errno::EPERM,
373                "the source is not permitted to be hard-linked"
374            );
375        }
376
377        Ok(())
378    }

388impl Path {
389    /// Mounts a filesystem at the current path.
390    ///
391    /// This method attaches a given filesystem to the VFS tree at the location
392    /// represented by `self`. The current path becomes the mountpoint for the new
393    /// filesystem.
394    ///
395    /// Returns the newly created child mount on success.
396    ///
397    /// # Errors
398    ///
399    /// Returns `ENOTDIR` if the path is not a directory.
400    /// Returns `EINVAL` if attempting to mount on root or if the path is not
401    /// in the current mount namespace.
402    pub(crate) fn mount(
403        &self,
404        fs_and_root: FsAndRoot,
405        flags: PerMountFlags,
406        source: Option<String>,
407        ctx: &Context,
408    ) -> Result<Arc<Mount>> {
409        if self.type_() != InodeType::Dir {
410            return_errno_with_message!(Errno::ENOTDIR, "the path is not a directory");
411        }
412
413        let fs_ref = ctx.thread_local.borrow_fs();
414        let path_resolver = fs_ref.resolver().read();
415
416        if path_resolver.root() == self {
417            return_errno_with_message!(Errno::EINVAL, "the root cannot be mounted on");
418        }
419
420        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
421        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
422        if !current_mnt_ns.owns(&self.mount) {
423            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
424        }
425
426        let mut topology_guard = MountTopology::write_lock();
427        self.mount.do_mount(
428            fs_and_root,
429            flags,
430            &self.dentry,
431            source,
432            &mut topology_guard,
433        )
434    }

436    /// Unmounts the filesystem mounted at the current path.
437    ///
438    /// Returns the unmounted child mount on success.
439    ///
440    /// # Errors
441    ///
442    /// Returns `EINVAL` in the following cases:
443    /// - The current path is not a mount root.
444    /// - The mount of the current path is the root mount.
445    /// - The current path is not in the current mount namespace.
446    pub(crate) fn unmount(&self, ctx: &Context) -> Result<Arc<Mount>> {
447        if !self.is_mount_root() {
448            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
449        }
450
451        let Some(_mountpoint) = self.mount.mountpoint() else {
452            return_errno_with_message!(Errno::EINVAL, "the root mount cannot be unmounted");
453        };
454
455        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
456        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
457        if !current_mnt_ns.owns(&self.mount) {
458            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
459        }
460
461        self.mount.sync()?;
462
463        let mut topology_guard = MountTopology::write_lock();
464        let Some(mountpoint) = self.mount.mountpoint() else {
465            return_errno_with_message!(Errno::EINVAL, "the mount has been detached");
466        };
467        let parent_mount = self.mount.parent().unwrap().upgrade().unwrap();
468        let child_mount = parent_mount.do_unmount(&mountpoint, &mut topology_guard)?;
469
470        Ok(child_mount)
471    }

505    /// Creates a bind mount from the current path to the destination path.
506    ///
507    /// Creates a new mount tree that mirrors either the root mount (non-recursive)
508    /// or the entire mount subtree (recursive), and attaches it to the destination path.
509    ///
510    /// A recursive bind omits unbindable subtrees.
511    /// If the destination parent is shared,
512    /// the new tree is also attached below its propagation receivers.
513    ///
514    /// # Errors
515    ///
516    /// Returns `ENOTDIR` if one of the source and destination is a directory
517    /// and the other is not.
518    ///
519    /// Returns `EINVAL` if any of the following holds:
520    /// - The destination path is not in the current mount namespace.
521    /// - The source path is a mount namespace file that would create a namespace loop.
522    /// - The source mount is unbindable.
523    pub(crate) fn bind_mount_to(
524        &self,
525        dst_path: Self,
526        recursive: bool,
527        ctx: &Context,
528    ) -> Result<()> {
529        let can_bind = {
530            let src_is_dir = self.type_() == InodeType::Dir;
531            let dst_is_dir = dst_path.type_() == InodeType::Dir;
532            (src_is_dir && dst_is_dir) || (!src_is_dir && !dst_is_dir)
533        };
534        if !can_bind {
535            return_errno_with_message!(
536                Errno::ENOTDIR,
537                "the source and destination must both be directories or both be non-directories"
538            );
539        }
540
541        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
542        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
543
544        // Linux does not require the source path to belong to the current
545        // mount namespace, but still rejects source mount namespace files that
546        // would form loops.
547        if !current_mnt_ns.owns(&dst_path.mount) {
548            return_errno_with_message!(
549                Errno::EINVAL,
550                "the destination path is not in this mount namespace"
551            );
552        }
553        if current_mnt_ns.would_form_mnt_ns_loop(&self.dentry) {
554            return_errno_with_message!(
555                Errno::EINVAL,
556                "the source is a mount namespace file that would create a namespace loop"
557            );
558        }
559
560        let mut topology_guard = MountTopology::write_lock();
561        let current_mnt_ns_weak = Arc::downgrade(current_mnt_ns);
562        let mut pending_changes = PendingPropagationChanges::default();
563        let new_mount = self.mount.clone_mount_tree_deferred(
564            &self.dentry,
565            &current_mnt_ns_weak,
566            recursive,
567            MountTreeCloneMode::Bind,
568            &mut pending_changes,
569            &topology_guard,
570        )?;
571        let attachment =
572            new_mount.prepare_mount_tree_attachment(dst_path, pending_changes, &topology_guard)?;
573        attachment.attach(&mut topology_guard);
574        Ok(())
575    }

577    /// Moves a mount tree from the current path to the destination path.
578    ///
579    /// If the destination parent is shared,
580    /// corresponding clones are also attached below its propagation receivers.
581    ///
582    /// # Errors
583    ///
584    /// Returns `EINVAL` in the following cases:
585    /// - The current path is not a mount root.
586    /// - The mount of the current path is the root mount.
587    /// - The source is a non-root mount in a detached mount tree.
588    /// - The destination is in a detached mount tree while the source is not.
589    /// - The source and destination are in the same detached mount tree.
590    /// - One of the source and destination is a directory and the other is not.
591    /// - The source mount's current parent is shared.
592    /// - The destination mount is shared and the source tree contains an
593    ///   unbindable mount.
594    ///
595    /// Returns `ELOOP` in the following cases:
596    /// - The destination path is inside the subtree being moved.
597    /// - The mount tree contains a mount namespace file that would create a namespace loop.
598    pub(crate) fn move_mount_to(&self, dst_path: Self, ctx: &Context) -> Result<()> {
599        if !self.is_mount_root() {
600            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
601        };
602
603        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
604        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
605        if self.mount.id() == current_mnt_ns.root().id() {
606            return_errno_with_message!(Errno::EINVAL, "the root mount can not be moved");
607        }
608        if !current_mnt_ns.owns(&self.mount) {
609            return_errno_with_message!(
610                Errno::EINVAL,
611                "the source path is not in this mount namespace"
612            );
613        }
614        if !current_mnt_ns.owns(&dst_path.mount) {
615            return_errno_with_message!(
616                Errno::EINVAL,
617                "the destination path is not in this mount namespace"
618            );
619        }
620
621        let mut topology_guard = MountTopology::write_lock();
622        let source_tree_root = mount_tree_root(self.mount_node(), &topology_guard);
623        let target_tree_root = mount_tree_root(dst_path.mount_node(), &topology_guard);
624        let current_tree_root = current_mnt_ns.root();
625        let source_is_detached = source_tree_root.id() != current_tree_root.id();
626        let target_is_detached = target_tree_root.id() != current_tree_root.id();
627
628        // Since Linux v6.15, a detached anonymous-namespace root can be
629        // moved onto either the current mount namespace or an acceptable
630        // anonymous namespace. Asterinas does not model anonymous mount
631        // namespaces yet, so compare detached mount tree roots instead.
632        // TODO: Once anonymous mount namespaces are modeled, determine
633        // detachedness from the mount namespace directly instead of inferring it
634        // from whether the mount tree root is the current namespace root.
635        if source_is_detached && self.mount.id() != source_tree_root.id() {
636            return_errno_with_message!(
637                Errno::EINVAL,
638                "only a detached mount tree root can be moved out of the detached tree"
639            );
640        }
641        if !source_is_detached && target_is_detached {
642            return_errno_with_message!(
643                Errno::EINVAL,
644                "the destination path is in a detached mount tree"
645            );
646        }
647        if source_is_detached && source_tree_root.id() == target_tree_root.id() {
648            return_errno_with_message!(
649                Errno::EINVAL,
650                "the source and destination are in the same detached mount tree"
651            );
652        }
653        let source_is_dir = self.type_() == InodeType::Dir;
654        let target_is_dir = dst_path.type_() == InodeType::Dir;
655        if source_is_dir != target_is_dir {
656            return_errno_with_message!(
657                Errno::EINVAL,
658                "the source and destination must both be directories or both be non-directories"
659            );
660        }
661
662        current_mnt_ns.check_no_mnt_ns_loop_in_tree(self.mount_node(), &topology_guard)?;
663        if dst_path
664            .mount_node()
665            .is_equal_or_descendant_of(self.mount_node(), &topology_guard)
666        {
667            // Reject moves that would place a mount beneath itself, because the mount tree
668            // must remain acyclic.
669            return_errno_with_message!(
670                Errno::ELOOP,
671                "the destination path is inside the mount subtree being moved"
672            );
673        }
674        if self
675            .mount
676            .parent()
677            .and_then(|parent| parent.upgrade())
678            .is_some_and(|parent| parent.is_shared())
679        {
680            return_errno_with_message!(
681                Errno::EINVAL,
682                "moving a mount under a shared parent is not supported"
683            );
684        }
685        if dst_path.mount_node().is_shared() {
686            self.mount.try_walk_tree(|mount| {
687                if mount.is_unbindable() {
688                    return_errno_with_message!(
689                        Errno::EINVAL,
690                        "an unbindable mount tree cannot be moved under a shared mount"
691                    );
692                }
693                Ok(())
694            })?;
695        }
696
697        let pending_changes = PendingPropagationChanges::default();
698        let attachment =
699            self.mount
700                .prepare_mount_tree_attachment(dst_path, pending_changes, &topology_guard)?;
701        attachment.attach(&mut topology_guard);
702
703        Ok(())
704    }
705
706    /// Sets the propagation policy of this path's mount.
707    ///
708    /// If `recursive` is `true`,
709    /// the same requested transition is applied independently
710    /// to every mount in the subtree.
711    ///
712    /// # Errors
713    ///
714    /// Returns `EINVAL` if this path is not a mount root
715    /// or its mount does not belong to the current mount namespace.
716    /// Returns `ENOMEM` if the transition requires a new peer group
717    /// and the peer-group ID pool is exhausted.
718    pub(crate) fn set_propagation(
719        &self,
720        prop: MountPropType,
721        recursive: bool,
722        ctx: &Context,
723    ) -> Result<()> {
724        if !self.is_mount_root() {
725            return_errno_with_message!(Errno::EINVAL, "the path is not a mount root");
726        };
727
728        let current_ns_proxy = ctx.thread_local.borrow_ns_proxy();
729        let current_mnt_ns = current_ns_proxy.unwrap().mnt_ns();
730        if !current_mnt_ns.owns(&self.mount) {
731            return_errno_with_message!(Errno::EINVAL, "the path is not in this mount namespace");
732        }
733
734        let mut topology_guard = MountTopology::write_lock();
735        let mut pending_changes = PendingPropagationChanges::default();
736        self.mount.set_propagation_deferred(
737            prop,
738            recursive,
739            &mut pending_changes,
740            &topology_guard,
741        )?;
742        pending_changes.commit(&mut topology_guard);
743        Ok(())
744    }
745}

747fn mount_tree_root(mount: &Arc<Mount>, _topology: &MountTopology) -> Arc<Mount> {
748    let mut root = mount.clone();
749    while let Some(parent) = root.parent().and_then(|parent| parent.upgrade()) {
750        root = parent;
751    }
752
753    root
754}

913fn capable_modify_file_cap(inode: &dyn Inode, ctx: &Context) -> Result<()> {
914    lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
915        ctx.thread_local.borrow_user_ns().as_ref(),
916        ctx.posix_thread,
917        CapSet::SETFCAP,
918    ))?;
919
920    // With the currently supported identity user-namespace mapping, only invalid IDs are
921    // unmapped. Replace this with mount-idmap-aware namespace mapping checks when supported.
922    let metadata = inode.metadata()?;
923    if metadata.uid == Uid::INVALID || metadata.gid == Gid::INVALID {
924        return_errno_with_message!(
925            Errno::EPERM,
926            "the inode owner or group is not mapped in the current user namespace"
927        );
928    }
929
930    Ok(())
931}

933/// Checks if the file name is ".", indicating it's the current directory.
934pub(crate) const fn is_dot(filename: &str) -> bool {
935    let name_bytes = filename.as_bytes();
936    name_bytes.len() == 1 && name_bytes[0] == DOT_BYTE
937}
938
939/// Checks if the file name is "..", indicating it's the parent directory.
940pub(crate) const fn is_dotdot(filename: &str) -> bool {
941    let name_bytes = filename.as_bytes();
942    name_bytes.len() == 2 && name_bytes[0] == DOT_BYTE && name_bytes[1] == DOT_BYTE
943}
944
945/// Checks if the file name is "." or "..", indicating it's the current or parent directory.
946pub(crate) const fn is_dot_or_dotdot(filename: &str) -> bool {
947    let name_bytes = filename.as_bytes();
948    let name_len = name_bytes.len();
949    if name_len == 1 {
950        name_bytes[0] == DOT_BYTE
951    } else if name_len == 2 {
952        name_bytes[0] == DOT_BYTE && name_bytes[1] == DOT_BYTE
953    } else {
954        false
955    }
956}
