32/// A `Mount` represents a mounted filesystem instance in the VFS.
33///
34/// Each `Mount` can be viewed as a node in the mount tree, maintaining
35/// mount-related information and the structure of the mount tree.
36pub(crate) struct Mount {
37    /// Pair of recyclable and unique mount identifiers; the recyclable
38    /// half is returned to the pool when the `Mount` drops.
39    id: MountId,
40    /// Root dentry.
41    root_dentry: Arc<Dentry>,
42    /// Mountpoint dentry. A mount node can be mounted on one dentry of another mount node,
43    /// which makes the mount being the child of the mount node.
44    mountpoint: RwLock<Option<Arc<Dentry>>>,
45    /// The associated mounted filesystem.
46    fs: Arc<dyn FileSystem>,
47    /// The mount source (e.g., a device path like "/dev/vda" or a filesystem name like "proc").
48    ///
49    /// The source is stored in `Mount` instead of requiring each filesystem to provide it.
50    /// If a filesystem does not provide a source, it falls back to the value stored in `Mount`.
51    /// This behavior aligns with that of Linux. Concrete examples can be found here:
52    /// <https://github.com/asterinas/asterinas/pull/2929#discussion_r2729739818>.
53    ///
54    /// Reference: <https://elixir.bootlin.com/linux/v6.17/source/fs/mount.h#L68>
55    source: Option<String>,
56    /// The parent mount node.
57    parent: RwLock<Option<Weak<Mount>>>,
58    /// Child mount nodes which are mounted on one dentry of self.
59    pub(super) children: RwLock<HashMap<DentryKey, Arc<Self>>>,
60    /// The associated mount namespace.
61    mnt_ns: Weak<MountNamespace>,
62    /// Propagation state of this mount.
63    propagation: RwLock<MountPropagation>,
64    /// The flags of this mount.
65    flags: AtomicPerMountFlags,
66    /// Reference to self.
67    this: Weak<Self>,
68}

70impl Mount {
71    /// Visits this mount subtree in depth-first order.
72    pub(super) fn try_walk_tree(
73        self: &Arc<Self>,
74        mut visit_fn: impl FnMut(Arc<Self>) -> Result<()>,
75    ) -> Result<()> {
76        let mut pending = vec![self.clone()];
77
78        while let Some(mount) = pending.pop() {
79            let children = mount.children.read();
80            pending.extend(children.values().cloned());
81            drop(children);
82
83            visit_fn(mount)?;
84        }
85
86        Ok(())
87    }

89    /// Creates a root mount node with an associated FS.
90    ///
91    /// The root mount node is not mounted on other mount nodes (which means it has no
92    /// parent). The root inode of the fs will form the inner root dentry.
93    ///
94    /// It is allowed to create a mount node even if the fs has been provided to another
95    /// mount node. It is the fs's responsibility to ensure the data consistency.
96    pub(in crate::fs) fn new_root(
97        fs: Arc<dyn FileSystem>,
98        flags: PerMountFlags,
99        mnt_ns: Weak<MountNamespace>,
100    ) -> Result<Arc<Self>> {
101        let source = fs.name().to_string();
102        let root_dentry = Dentry::new_root(fs.root_inode());
103        Self::new(root_dentry, fs, flags, None, mnt_ns, Some(source))
104    }

139    /// The internal constructor.
140    ///
141    /// A root mount node has no mountpoint, while other mount nodes must have one.
142    ///
143    /// Here, a `Mount` is instantiated without an initial mountpoint,
144    /// avoiding fixed mountpoint limitations. This allows the root mount node to
145    /// exist without a mountpoint, ensuring uniformity and security, while all other
146    /// mount nodes must be explicitly assigned a mountpoint to maintain structural integrity.
147    fn new(
148        root_dentry: Arc<Dentry>,
149        fs: Arc<dyn FileSystem>,
150        flags: PerMountFlags,
151        parent_mount: Option<Weak<Mount>>,
152        mnt_ns: Weak<MountNamespace>,
153        source: Option<String>,
154    ) -> Result<Arc<Self>> {
155        let id = MountId::alloc()
156            .ok_or_else(|| Error::with_message(Errno::ENOMEM, "mount ID space exhausted"))?;
157
158        let mount = Arc::new_cyclic(|weak_self| Self {
159            id,
160            root_dentry,
161            mountpoint: RwLock::new(None),
162            fs,
163            source,
164            parent: RwLock::new(parent_mount),
165            children: RwLock::new(HashMap::new()),
166            mnt_ns,
167            propagation: RwLock::new(MountPropagation::default()),
168            flags: AtomicPerMountFlags::new(flags),
169            this: weak_self.clone(),
170        });
171        // `upgrade` returns `None` for pseudo mounts (no namespace) and for
172        // mounts still being built by `MountNamespace::new_with_root` (the
173        // namespace is a `UniqueArc` at that point). The latter are
174        // registered by `new_with_root` after the namespace is finalized.
175        if let Some(ns) = mount.mnt_ns.upgrade() {
176            ns.register_mount(&mount);
177        }
178        Ok(mount)
179    }

196    /// Mounts a file system at `mountpoint` and returns the new child mount.
197    ///
198    /// The same file system may be mounted at multiple locations.
199    /// The file system remains responsible for keeping its data consistent.
200    ///
201    /// A user-provided source is retained by the new mount.
202    /// If this mount is shared,
203    /// the attachment is also created below its propagation receivers.
204    ///
205    /// The caller must hold the mount-topology write lock in `topology`.
206    pub(super) fn do_mount(
207        self: &Arc<Self>,
208        fs_and_root: FsAndRoot,
209        flags: PerMountFlags,
210        mountpoint: &Arc<Dentry>,
211        source: Option<String>,
212        topology: &mut MountTopology,
213    ) -> Result<Arc<Self>> {
214        if mountpoint.type_() != InodeType::Dir {
215            return_errno!(Errno::ENOTDIR);
216        }
217
218        let child_mount = Self::new_detached(fs_and_root, flags, self.mnt_ns.clone(), source)?;
219        let target_path = Path::new(self.clone(), mountpoint.clone());
220        let pending_changes = PendingPropagationChanges::default();
221        let attachment =
222            child_mount.prepare_mount_tree_attachment(target_path, pending_changes, topology)?;
223        attachment.attach(topology);
224
225        Ok(child_mount)
226    }

228    /// Unmounts the child at `mountpoint` and returns the requested child mount.
229    ///
230    /// If this mount is shared,
231    /// corresponding children below its propagation receivers are also unmounted.
232    ///
233    /// The operation is atomic: an error leaves the mount tree and propagation topology
234    /// unchanged.
235    /// The caller must hold the mount-topology write lock in `topology`.
236    pub(super) fn do_unmount(
237        self: &Arc<Self>,
238        mountpoint: &Arc<Dentry>,
239        topology: &mut MountTopology,
240    ) -> Result<Arc<Self>> {
241        let child_mount = self
242            .get(mountpoint)
243            .ok_or_else(|| Error::with_message(Errno::ENOENT, "can not find child mount"))?;
244        let parent = child_mount
245            .parent()
246            .and_then(|parent| parent.upgrade())
247            .ok_or_else(|| {
248                Error::with_message(Errno::EINVAL, "the root mount cannot be unmounted")
249            })?;
250
251        let mut unmounted_mounts = vec![child_mount];
252        for receiver in topology.propagation_receivers(&parent, mountpoint) {
253            if let Some(child) = receiver.get(mountpoint) {
254                unmounted_mounts.push(child);
255            }
256        }
257
258        for target in &unmounted_mounts {
259            target
260                .try_walk_tree(|mount| {
261                    mount.clear_propagation(topology);
262                    Ok(())
263                })
264                .unwrap();
265            target.detach_from_parent(topology);
266        }
267
268        Ok(unmounted_mounts.into_iter().next().unwrap())
269    }

271    /// Clones this mount as an unpublished mount rooted at `root_dentry`.
272    ///
273    /// The clone belongs to `new_ns` and shares this mount's file system,
274    /// source, and per-mount flags.
275    /// It has no parent, children, mountpoint, or published propagation links.
276    fn clone_mount(
277        &self,
278        root_dentry: &Arc<Dentry>,
279        propagation: MountPropagation,
280        new_ns: &Weak<MountNamespace>,
281    ) -> Result<Arc<Self>> {
282        let id = MountId::alloc()
283            .ok_or_else(|| Error::with_message(Errno::ENOMEM, "mount ID space exhausted"))?;
284
285        let mount = Arc::new_cyclic(|weak_self| Self {
286            id,
287            root_dentry: root_dentry.clone(),
288            mountpoint: RwLock::new(None),
289            fs: self.fs.clone(),
290            source: self.source.clone(),
291            parent: RwLock::new(None),
292            children: RwLock::new(HashMap::new()),
293            mnt_ns: new_ns.clone(),
294            propagation: RwLock::new(propagation),
295            flags: AtomicPerMountFlags::new(self.flags.load(Ordering::Relaxed)),
296            this: weak_self.clone(),
297        });
298        if let Some(ns) = mount.mnt_ns.upgrade() {
299            ns.register_mount(&mount);
300        }
301        Ok(mount)
302    }

304    /// Clones this mount tree and defers propagation updates.
305    ///
306    /// The returned tree belongs to `new_ns` and is rooted at `root_dentry`.
307    /// A non-recursive clone contains only this mount;
308    /// a recursive clone reproduces the eligible descendants selected by `clone_mode`.
309    ///
310    /// [`MountTreeCloneMode::Bind`] preserves propagation state,
311    /// rejects an unbindable root,
312    /// and omits unbindable child subtrees.
313    /// [`MountTreeCloneMode::Namespace`] preserves propagation state
314    /// and unbindable subtrees,
315    /// but omits mounts rooted at mount-namespace files.
316    /// [`MountTreeCloneMode::SharedPropagation`] makes corresponding source
317    /// and clone mounts peers.
318    /// [`MountTreeCloneMode::SlavePropagation`] makes each clone
319    /// a slave of its source mount.
320    /// All modes except [`MountTreeCloneMode::Bind`] reject a root at a
321    /// mount-namespace file and omit children rooted at such files.
322    ///
323    /// The caller must retain `pending_changes` until every fallible step of the enclosing
324    /// operation succeeds, then call [`PendingPropagationChanges::commit`] to apply these operations.
325    pub(super) fn clone_mount_tree_deferred(
326        self: &Arc<Self>,
327        root_dentry: &Arc<Dentry>,
328        new_ns: &Weak<MountNamespace>,
329        recursive: bool,
330        clone_mode: MountTreeCloneMode,
331        pending_changes: &mut PendingPropagationChanges,
332        _topology: &MountTopology,
333    ) -> Result<Arc<Self>> {
334        let skip_unbindable = clone_mode == MountTreeCloneMode::Bind;
335        if skip_unbindable && self.is_unbindable() {
336            return_errno_with_message!(Errno::EINVAL, "the source mount is unbindable");
337        }
338        if clone_mode != MountTreeCloneMode::Bind && try_get_mnt_ns_inode(root_dentry).is_some() {
339            return_errno_with_message!(
340                Errno::EINVAL,
341                "a mount namespace file cannot be the root of this mount tree clone"
342            );
343        }
344
345        let mut clone_mount_fn = |source_mount: &Mount,
346                                  cloned_root_dentry: &Arc<Dentry>|
347         -> Result<Arc<Mount>> {
348            let cloned_propagation = match clone_mode {
349                MountTreeCloneMode::SharedPropagation => {
350                    pending_changes.make_shared_for_clone(source_mount)?
351                }
352                MountTreeCloneMode::SlavePropagation => {
353                    let source_propagation = pending_changes.make_shared_for_clone(source_mount)?;
354                    let source_peer_group_id = source_propagation
355                        .peer_group_id()
356                        .expect("the propagation source must be shared");
357                    MountPropagation::slave(source_peer_group_id)
358                }
359                MountTreeCloneMode::Bind | MountTreeCloneMode::Namespace => {
360                    source_mount.propagation()
361                }
362            };
363
364            let cloned_mount =
365                source_mount.clone_mount(cloned_root_dentry, cloned_propagation, new_ns)?;
366            pending_changes.add_register_op(cloned_mount.clone());
367            Ok(cloned_mount)
368        };
369
370        let new_root_mount = clone_mount_fn(self, root_dentry)?;
371        let mut cloned_mounts = vec![(self.this(), new_root_mount.clone())];
372        if recursive {
373            let mut work_idx = 0;
374            while work_idx < cloned_mounts.len() {
375                let (old_parent_mount, new_parent_mount) = cloned_mounts[work_idx].clone();
376                let old_children = old_parent_mount
377                    .children
378                    .read()
379                    .values()
380                    .cloned()
381                    .collect::<Vec<_>>();
382                for old_child_mount in old_children {
383                    if skip_unbindable && old_child_mount.is_unbindable() {
384                        continue;
385                    }
386                    if clone_mode != MountTreeCloneMode::Bind
387                        && try_get_mnt_ns_inode(old_child_mount.root_dentry()).is_some()
388                    {
389                        continue;
390                    }
391
392                    let mountpoint = old_child_mount.mountpoint().unwrap();
393                    if !mountpoint.is_equal_or_descendant_of(new_parent_mount.root_dentry()) {
394                        continue;
395                    }
396
397                    let new_child_mount =
398                        clone_mount_fn(&old_child_mount, old_child_mount.root_dentry())?;
399
400                    let key = mountpoint.key();
401                    new_parent_mount
402                        .children
403                        .write()
404                        .insert(key, new_child_mount.clone());
405                    new_child_mount.set_parent(Some(&new_parent_mount));
406                    new_child_mount.set_mountpoint(mountpoint);
407                    cloned_mounts.push((old_child_mount, new_child_mount));
408                }
409                work_idx += 1;
410            }
411        }
412
413        Ok(new_root_mount)
414    }

416    /// Queues a propagation-policy change for this mount.
417    ///
418    /// If `recursive` is `true`,
419    /// the requested transition is queued independently for every mount in this subtree.
420    /// The changes become visible only when `pending_changes` is committed, either directly
421    /// or by transferring it to [`Self::prepare_mount_tree_attachment`].
422    pub(super) fn set_propagation_deferred(
423        self: &Arc<Self>,
424        prop: MountPropType,
425        recursive: bool,
426        pending_changes: &mut PendingPropagationChanges,
427        _topology: &MountTopology,
428    ) -> Result<()> {
429        let mut queue_transition_fn = |mount: Arc<Mount>| -> Result<()> {
430            let new_propagation = match prop {
431                MountPropType::Shared => {
432                    let old_propagation = mount.propagation();
433                    if old_propagation.is_shared() {
434                        return Ok(());
435                    } else {
436                        let peer_group = PeerGroup::new()?;
437                        let new_propagation = old_propagation.make_shared(peer_group.id());
438                        pending_changes.add_peer_group(mount.unique_id(), peer_group);
439                        new_propagation
440                    }
441                }
442                MountPropType::Slave => {
443                    pending_changes.add_make_slave_op(mount);
444                    return Ok(());
445                }
446                MountPropType::Private => MountPropagation::private(),
447                MountPropType::Unbindable => MountPropagation::unbindable(),
448            };
449            pending_changes.add_set_op(mount, new_propagation);
450            Ok(())
451        };
452        if recursive {
453            self.try_walk_tree(queue_transition_fn)?;
454        } else {
455            queue_transition_fn(self.this())?;
456        }
457
458        Ok(())
459    }

461    /// Returns the space-prefixed propagation fields for `/proc/PID/mountinfo`.
462    pub(in crate::fs) fn propagation_mountinfo_fields(&self) -> String {
463        let propagation = self.propagation.read();
464        let peer_group_id = propagation.peer_group_id();
465        let master_id = propagation.master_id();
466        match (peer_group_id, master_id) {
467            (Some(peer_group_id), Some(master_id)) => {
468                alloc::format!(" shared:{} master:{}", peer_group_id, master_id)
469            }
470            (Some(peer_group_id), None) => alloc::format!(" shared:{}", peer_group_id),
471            (None, Some(master_id)) => alloc::format!(" master:{}", master_id),
472            (None, None) => String::new(),
473        }
474    }

517    /// Detaches the mount node from the parent mount node.
518    pub(super) fn detach_from_parent(&self, topology: &mut MountTopology) {
519        if let Some(parent) = self.parent() {
520            let parent = parent.upgrade().unwrap();
521            let child = parent
522                .children
523                .write()
524                .remove(&self.mountpoint().unwrap().key());
525
526            if let Some(child) = child {
527                child.clear_topology_link(topology);
528            }
529        }
530    }

548            .children

561    /// Prepares this mount tree's attachment and its propagation copies.
562    ///
563    /// If the destination parent is shared,
564    /// the tree is also cloned below every reachable peer and slave receiver.
565    /// A receiver participates only if the target dentry is at or below its root dentry;
566    /// a bind mount rooted elsewhere cannot address the event location.
567    /// Corresponding mounts below peer receivers join the same peer groups;
568    /// copies below slave receivers become slaves of the corresponding source mounts.
569    ///
570    /// This method takes ownership of any propagation-topology changes already prepared by
571    /// the enclosing operation without changing existing mount-tree links or publishing
572    /// propagation changes. The caller must retain the mount-topology write lock until
573    /// the returned attachment is attached or dropped.
574    pub(super) fn prepare_mount_tree_attachment(
575        self: &Arc<Self>,
576        target_path: Path,
577        mut pending_changes: PendingPropagationChanges,
578        topology: &MountTopology,
579    ) -> Result<PendingMountTreeAttachment> {
580        let mut copies = Vec::new();
581        let mut propagation_layers = Vec::new();
582        if let Some(peer_group_id) = target_path.mount_node().propagation().peer_group_id() {
583            propagation_layers.push((
584                target_path.mount_node().clone(),
585                self.clone(),
586                peer_group_id,
587            ));
588        }
589
590        while let Some((source_parent, source_mount, peer_group_id)) = propagation_layers.pop() {
591            let copies_before_layer = copies.len();
592
593            for receiver in topology.group_peers_of(peer_group_id) {
594                if receiver.unique_id() == source_parent.unique_id()
595                    || !target_path
596                        .dentry()
597                        .is_equal_or_descendant_of(receiver.root_dentry())
598                {
599                    continue;
600                }
601                let copy = source_mount.clone_mount_tree_deferred(
602                    source_mount.root_dentry(),
603                    receiver.mnt_ns(),
604                    true,
605                    MountTreeCloneMode::SharedPropagation,
606                    &mut pending_changes,
607                    topology,
608                )?;
609                copies.push((receiver, copy));
610            }
611
612            let mut seen_slave_peer_groups = HashSet::new();
613            for receiver in topology.slave_mounts_of(peer_group_id) {
614                if !target_path
615                    .dentry()
616                    .is_equal_or_descendant_of(receiver.root_dentry())
617                {
618                    continue;
619                }
620                let receiver_peer_group_id = receiver.propagation().peer_group_id();
621                if receiver_peer_group_id
622                    .is_some_and(|peer_group_id| !seen_slave_peer_groups.insert(peer_group_id))
623                {
624                    continue;
625                }
626
627                let copy = source_mount.clone_mount_tree_deferred(
628                    source_mount.root_dentry(),
629                    receiver.mnt_ns(),
630                    true,
631                    MountTreeCloneMode::SlavePropagation,
632                    &mut pending_changes,
633                    topology,
634                )?;
635                if let Some(peer_group_id) = receiver_peer_group_id {
636                    propagation_layers.push((receiver.clone(), copy.clone(), peer_group_id));
637                }
638                copies.push((receiver, copy));
639            }
640
641            // Each propagation layer has its own source tree. Cloning any receiver makes
642            // that tree shared as a side effect. If this layer has no eligible receiver,
643            // perform the transition explicitly; copies from earlier layers do not count.
644            if copies.len() == copies_before_layer {
645                source_mount.set_propagation_deferred(
646                    MountPropType::Shared,
647                    true,
648                    &mut pending_changes,
649                    topology,
650                )?;
651            }
652        }
653
654        Ok(PendingMountTreeAttachment {
655            mount: self.clone(),
656            target_path,
657            copies,
658            pending_changes,
659        })
660    }

704    pub(super) fn remount(
705        &self,
706        mount_flags: PerMountFlags,
707        fs_flags: Option<FsFlags>,
708        data: Option<&str>,
709        ctx: &Context,
710        _topology: &mut MountTopology,
711    ) -> Result<()> {
712        if let Some(flags) = fs_flags {
713            self.fs.set_fs_flags(flags, data, ctx)?;
714        }
715
716        // The logics here are consistent with Linux.
717        // In Linux, `NOATIME`, `RELATIME`, and `STRICTATIME` are mutually exclusive.
718        // If none of them nor `NODIRATIME` are set, the atime policy will be inherited
719        // from the old flags.
720        // Reference: https://elixir.bootlin.com/linux/v6.17/source/fs/namespace.c#L4097
721        const ATIME_MASK: PerMountFlags = PerMountFlags::NOATIME
722            .union(PerMountFlags::RELATIME)
723            .union(PerMountFlags::STRICTATIME);
724
725        let need_inherit_atime = !mount_flags.intersects(ATIME_MASK | PerMountFlags::NODIRATIME);
726
727        if need_inherit_atime {
728            let old_flags = self.flags.load(Ordering::Relaxed);
729            let new_flags = mount_flags | (old_flags & ATIME_MASK);
730            self.flags.store(new_flags, Ordering::Relaxed);
731        } else {
732            self.flags.store(mount_flags, Ordering::Relaxed);
733        }
734
735        Ok(())
736    }

743    /// Returns whether `self` is `ancestor` or a descendant of it in the mount tree.
744    pub(super) fn is_equal_or_descendant_of(
745        &self,
746        ancestor: &Arc<Self>,
747        _topology: &MountTopology,
748    ) -> bool {
749        let mut current = self.this();
750        loop {
751            if Arc::ptr_eq(&current, ancestor) {
752                return true;
753            }
754
755            let Some(parent) = current.parent().and_then(|parent| parent.upgrade()) else {
756                return false;
757            };
758            current = parent;
759        }
760    }

786    /// Finds the corresponding `Mount` in the given mount namespace.
787    pub(super) fn find_corresponding_mount(
788        &self,
789        mnt_ns: &Arc<MountNamespace>,
790        _topology: &MountTopology,
791    ) -> Option<Arc<Self>> {
792        // Collect the ancestors from self to the root mount (The root mount is not included).
793        let mut ancestors = VecDeque::new();
794        let mut current = self.this();
795        while let Some(weak_parent) = current.parent() {
796            ancestors.push_front(current.clone());
797            current = weak_parent.upgrade().unwrap();
798        }
799
800        let mut target_mount = mnt_ns.root().clone();
801        while let Some(ancestor) = ancestors.pop_front() {
802            // Find the child mount that matches the mountpoint of ancestor.
803            let mount_point = ancestor.mountpoint().unwrap();
804            let child_mount = target_mount
805                .children
806                .read()
807                .get(&mount_point.key())
808                .cloned()?;
809            target_mount = child_mount;
810        }
811
812        Some(target_mount)
813    }

816        self.this.upgrade().unwrap()
817    }

830impl Drop for Mount {
831    fn drop(&mut self) {
832        self.clear_mountpoint();
833        // `upgrade` returns `None` for pseudo mounts (no namespace) and for
834        // mounts whose namespace is itself being dropped (the `mounts` map
835        // will be freed in a moment).
836        if let Some(ns) = self.mnt_ns.upgrade() {
837            ns.deregister_mount(self.id.unique_id());
838        }
839        // The recyclable ID is returned to the pool by `MountId`'s `Drop`.
840    }
841}

843/// A prepared mount-tree attachment whose remaining steps cannot fail.
844///
845/// Dropping it leaves existing mount-tree links and propagation relationships unchanged.
846pub(super) struct PendingMountTreeAttachment {
847    mount: Arc<Mount>,
848    target_path: Path,
849    copies: Vec<(Arc<Mount>, Arc<Mount>)>,
850    pending_changes: PendingPropagationChanges,
851}
852
853impl PendingMountTreeAttachment {
854    /// Attaches the prepared trees and publishes their propagation changes.
855    ///
856    /// The caller must retain the mount-topology write lock from preparation through attachment.
857    pub(super) fn attach(self, topology: &mut MountTopology) {
858        self.mount.detach_from_parent(topology);
859
860        for (receiver, copy) in self.copies {
861            let mount_point = self.target_path.dentry.clone();
862            debug_assert!(Arc::ptr_eq(&receiver.fs, &mount_point.inode().fs()));
863            // Propagation stacks the new copy below an existing mount at this
864            // dentry. Preserve the existing mount so it can be uncovered again
865            // when the propagated copy is unmounted, as on Linux.
866            if let Some(existing_mount) = receiver.get(&mount_point) {
867                let copy_root = Path::new(copy.clone(), copy.root_dentry().clone()).get_top_path();
868                existing_mount.graft_mount_tree(copy_root, topology);
869            }
870            let target_path = Path::new(receiver, mount_point);
871
872            copy.attach_to_path(target_path, topology);
873        }
874        self.mount.attach_to_path(self.target_path, topology);
875        self.pending_changes.commit(topology);
876    }
877}

879/// A policy for cloning a mount tree.
880#[derive(Clone, Copy, Debug, Eq, PartialEq)]
881pub(super) enum MountTreeCloneMode {
882    /// A bind-mount clone that omits unbindable subtrees.
883    Bind,
884    /// A namespace clone that omits mounts rooted at mount-namespace files.
885    Namespace,
886    /// A propagation clone whose mounts are peers of the source mounts.

892/// A recyclable mount ID exposed by legacy interfaces such as `/proc/*/mountinfo`.
893pub(in crate::fs) type RecyclableMountId = u32;
894
895/// A mount ID that remains unique until reboot.
896pub(in crate::fs) type UniqueMountId = u64;
897
898/// 32-bit recyclable mount IDs.
899///
900/// IDs start at 1; 0 is never issued and denotes an invalid mount.
901/// Exhaustion returns `ENOMEM`.
902/// Reference: <https://elixir.bootlin.com/linux/v6.17/source/fs/namespace.c#L282>
903static MOUNT_ID_ALLOCATOR: SpinLock<SparseIdAlloc> =
904    SpinLock::new(SparseIdAlloc::new(1, i32::MAX as u32));
905
906/// The first 64-bit unique mount ID.
907///
908/// The minimum keeps unique IDs above the recyclable ID range, ensuring
909/// the two ID spaces never overlap numerically.
910///
911/// This value is chosen to align with Linux.
912/// Reference: <https://elixir.bootlin.com/linux/v6.17/source/fs/namespace.c#L73>
913pub(crate) const MNT_UNIQUE_ID_MIN: u64 = (1 << 31) + 1;
914
915/// Monotonically increasing 64-bit unique mount IDs.
916static NEXT_FREE_UNIQUE_ID: AtomicU64 = AtomicU64::new(MNT_UNIQUE_ID_MIN);
917
918pub(super) fn init() {}
919
920/// A mount's pair of identifiers.
921///
922/// Allocates from [`MOUNT_ID_ALLOCATOR`] and [`NEXT_FREE_UNIQUE_ID`] on
923/// construction; on drop, releases the recyclable ID back to the pool.
924/// The unique ID is monotonic and is not freed.
925pub(super) struct MountId {
926    recyclable_id: RecyclableMountId,
927    unique_id: UniqueMountId,
928}
929
930impl MountId {
931    /// Allocates a new pair of mount identifiers.
932    ///
933    /// Returns `None` if the recyclable ID range is exhausted.
934    pub(super) fn alloc() -> Option<Self> {
935        let recyclable_id = MOUNT_ID_ALLOCATOR.lock().alloc()?;
936        let unique_id = NEXT_FREE_UNIQUE_ID.fetch_add(1, Ordering::Relaxed);
937        Some(Self {
938            recyclable_id,
939            unique_id,
940        })
941    }
942
943    pub(super) fn recyclable_id(&self) -> RecyclableMountId {
944        self.recyclable_id
945    }
946
947    pub(super) fn unique_id(&self) -> UniqueMountId {
948        self.unique_id
949    }
950}
951
952impl Drop for MountId {
953    fn drop(&mut self) {
954        MOUNT_ID_ALLOCATOR.lock().free(self.recyclable_id);
955    }
956}

958bitflags! {
959    pub(crate) struct PerMountFlags: u32 {
960        /// Mount read-only.
961        const RDONLY         = 1 << 0;
962        /// Ignore suid and sgid bits.
963        const NOSUID         = 1 << 1;
964        /// Disallow access to device special files.
965        const NODEV          = 1 << 2;
966        /// Disallow program execution.
967        const NOEXEC         = 1 << 3;
968        /// Do not follow symlinks.
969        const NOSYMFOLLOW    = 1 << 8;
970        /// Do not update access times.
971        const NOATIME        = 1 << 10;
972        /// Do not update directory access times.
973        const NODIRATIME     = 1 << 11;
974        /// Update atime relative to mtime/ctime.
975        const RELATIME       = 1 << 21;
976        /// Kernel (pseudo) mount.
977        const KERNMOUNT      = 1 << 22;
978        /// Always perform atime updates.
979        const STRICTATIME    = 1 << 24;
980    }
981}
982
983impl Default for PerMountFlags {
984    fn default() -> Self {
985        let empty = Self::empty();
986        empty | Self::RELATIME
987    }
988}

990impl PerMountFlags {
991    /// Gets the atime policy.
992    fn atime_policy(&self) -> AtimePolicy {
993        if self.contains(PerMountFlags::STRICTATIME) {
994            AtimePolicy::Strictatime
995        } else if self.contains(PerMountFlags::NOATIME) {
996            AtimePolicy::Noatime
997        } else {
998            AtimePolicy::Relatime
999        }
1000    }
1001}

1003/// The policy for updating access times (atime).
1004///
1005/// A Mount can only have one of the following atime policies.
1006#[derive(Clone, Copy, Debug, Eq, PartialEq)]
1007enum AtimePolicy {
1008    Relatime,
1009    Noatime,
1010    Strictatime,
1011}

1013impl core::fmt::Display for PerMountFlags {
1014    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
1015        if self.contains(PerMountFlags::RDONLY) {
1016            write!(f, "ro")?;
1017        } else {
1018            write!(f, "rw")?;
1019        };
1020        if self.contains(PerMountFlags::NOSUID) {
1021            write!(f, ",nosuid")?;
1022        }
1023        if self.contains(PerMountFlags::NODEV) {
1024            write!(f, ",nodev")?;
1025        }
1026        if self.contains(PerMountFlags::NOEXEC) {
1027            write!(f, ",noexec")?;
1028        }
1029        if self.contains(PerMountFlags::NODIRATIME) {
1030            write!(f, ",nodiratime")?;
1031        }
1032        let atime_policy = match self.atime_policy() {
1033            AtimePolicy::Relatime => "relatime",
1034            AtimePolicy::Noatime => "noatime",
1035            AtimePolicy::Strictatime => "strictatime",
1036        };
1037        write!(f, ",{}", atime_policy)
1038    }
1039}
