20pub trait NetDeviceCallback = Fn() + Send + Sync + 'static;
21
22pub fn register_device(
23    name: String,
24    device: Arc<SpinLock<dyn AnyNetworkDevice, BottomHalfDisabled>>,
25) {
26    COMPONENT
27        .get()
28        .unwrap()
29        .network_device_table
30        .lock()
31        .insert(name, NetworkDeviceIrqCallbackSet::new(device));
32}
33
34pub fn get_device(str: &str) -> Option<Arc<SpinLock<dyn AnyNetworkDevice, BottomHalfDisabled>>> {
35    let table = COMPONENT.get().unwrap().network_device_table.lock();
36    let callbacks = table.get(str)?;
37    Some(callbacks.device.clone())
38}

40/// Registers callback which will be called when receiving message.
41///
42/// Since the callback will be called in softirq context,
43/// the callback function should _not_ sleep.
44pub fn register_recv_callback(name: &str, callback: impl NetDeviceCallback) {
45    let device_table = COMPONENT.get().unwrap().network_device_table.lock();
46    let Some(callbacks) = device_table.get(name) else {
47        return;
48    };
49    callbacks.recv_callbacks.lock().push(Arc::new(callback));
50}
51
52/// Registers a callback that will be invoked
53/// when the device has completed sending a packet.
54///
55/// Since this callback is executed in a softirq context,
56/// the callback function should _not_ block or sleep.
57///
58/// Please note that the callback may not be called every time a packet is sent.
59/// The driver may skip certain callbacks for performance optimization.
60pub fn register_send_callback(name: &str, callback: impl NetDeviceCallback) {
61    let device_table = COMPONENT.get().unwrap().network_device_table.lock();
62    let Some(callbacks) = device_table.get(name) else {
63        return;
64    };
65    callbacks.send_callbacks.lock().push(Arc::new(callback));
66}
67
68fn handle_rx_softirq() {
69    let device_table = COMPONENT.get().unwrap().network_device_table.lock();
70    // TODO: We should handle network events for just one device per softirq,
71    // rather than processing events for all devices.
72    // This issue should be addressed once new network devices are added.
73    for callback_set in device_table.values() {
74        let recv_callbacks = callback_set.recv_callbacks.lock();
75        for callback in recv_callbacks.iter() {
76            callback();
77        }
78    }
79}
80
81fn handle_tx_softirq() {
82    let device_table = COMPONENT.get().unwrap().network_device_table.lock();
83    // TODO: We should handle network events for just one device per softirq,
84    // rather than processing events for all devices.
85    // This issue should be addressed once new network devices are added.
86    for callback_set in device_table.values() {
87        let can_send = {
88            let mut device = callback_set.device.lock();
89            device.free_processed_tx_buffers();
90            device.can_send()
91        };
92
93        if !can_send {
94            continue;
95        }
96
97        let send_callbacks = callback_set.send_callbacks.lock();
98        for callback in send_callbacks.iter() {
99            callback();
100        }
101    }
102}

104/// Raises softirq for handling transmission events
105pub fn raise_send_softirq() {
106    SoftIrqLine::get(NETWORK_TX_SOFTIRQ_ID).raise();
107}
108
109/// Raises softirq for handling reception events
110pub fn raise_receive_softirq() {
111    SoftIrqLine::get(NETWORK_RX_SOFTIRQ_ID).raise();
112}
113
114pub fn all_devices() -> Vec<(String, NetworkDeviceRef)> {
115    let network_devs = COMPONENT.get().unwrap().network_device_table.lock();
116    network_devs
117        .iter()
118        .map(|(name, callbacks)| (name.clone(), callbacks.device.clone()))
119        .collect()
120}

122static COMPONENT: Once<Component> = Once::new();
123
124#[init_component]
125fn init() -> Result<(), ComponentInitError> {
126    let component = Component::init()?;
127    COMPONENT.call_once(|| component);
128
129    SoftIrqLine::get(NETWORK_TX_SOFTIRQ_ID).enable(handle_tx_softirq);
130    SoftIrqLine::get(NETWORK_RX_SOFTIRQ_ID).enable(handle_rx_softirq);
131
132    Ok(())
133}

135type NetDeviceCallbackListRef = Arc<SpinLock<Vec<Arc<dyn NetDeviceCallback>>, BottomHalfDisabled>>;
136type NetworkDeviceRef = Arc<SpinLock<dyn AnyNetworkDevice, BottomHalfDisabled>>;
137
138struct Component {
139    /// Device list, the key is device name, value is (callbacks, device);
140    network_device_table:
141        SpinLock<BTreeMap<String, NetworkDeviceIrqCallbackSet>, BottomHalfDisabled>,
142}
143
144/// The send callbacks and recv callbacks for a network device
145struct NetworkDeviceIrqCallbackSet {
146    device: NetworkDeviceRef,
147    recv_callbacks: NetDeviceCallbackListRef,
148    send_callbacks: NetDeviceCallbackListRef,
149}
150
151impl NetworkDeviceIrqCallbackSet {
152    fn new(device: NetworkDeviceRef) -> Self {
153        Self {
154            device,
155            recv_callbacks: Arc::new(SpinLock::new(Vec::new())),
156            send_callbacks: Arc::new(SpinLock::new(Vec::new())),
157        }
158    }
159}
160
161impl Component {
162    fn init() -> Result<Self, ComponentInitError> {
163        Ok(Self {
164            network_device_table: SpinLock::new(BTreeMap::new()),
165        })
166    }
