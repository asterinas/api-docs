17/// Maximum number of colormap entries (standard 8-bit palette)
18pub const MAX_CMAP_SIZE: usize = 256;

20/// The framebuffer used for text or graphical output.
21///
22/// # Notes
23///
24/// It is highly recommended to use a synchronization primitive, such as a `SpinLock`, to
25/// lock the framebuffer before performing any operation on it.
26/// Failing to properly synchronize access can result in corrupted framebuffer content
27/// or unspecified behavior during rendering.
28#[derive(Debug)]
29pub struct FrameBuffer {
30    io_mem: IoMem,
31    width: usize,
32    height: usize,
33    line_size: usize,
34    pixel_format: PixelFormat,
35    cmap: Mutex<FbCmap>,
36}
37
38/// A single entry in the color map with 16-bit color values.
39///
40/// Linux framebuffer colormap uses 16-bit values (0-65535) for each color channel
41/// to support high precision color mapping.
42#[derive(Clone, Copy, Debug, Eq, PartialEq)]
43pub struct ColorMapEntry {
44    /// Red color value (16-bit)
45    pub red: u16,
46    /// Green color value (16-bit)
47    pub green: u16,
48    /// Blue color value (16-bit)
49    pub blue: u16,
50    /// Transparency value (16-bit)
51    pub transp: u16,
52}
53
54/// Internal framebuffer colormap structure.
55#[derive(Clone, Debug)]
56struct FbCmap {
57    /// Color map entries
58    entries: Vec<ColorMapEntry>,
59}

61pub static FRAMEBUFFER: Once<Arc<FrameBuffer>> = Once::new();

63pub(crate) fn init() {
64    let Some(framebuffer_arg) = boot_info().framebuffer_arg else {
65        ostd::warn!("Framebuffer not found");
66        return;
67    };
68
69    if framebuffer_arg.address == 0 {
70        ostd::error!("Framebuffer address is zero");
71        return;
72    }
73
74    if framebuffer_arg.address % PAGE_SIZE != 0 {
75        ostd::error!("Framebuffer address is not page-aligned");
76        return;
77    }
78
79    // FIXME: There are several pixel formats that have the same BPP. We lost the information
80    // during the boot phase, so here we guess the pixel format on a best effort basis.
81    let pixel_format = match framebuffer_arg.bpp {
82        8 => PixelFormat::Grayscale8,
83        16 => PixelFormat::Rgb565,
84        24 => PixelFormat::Rgb888,
85        32 => PixelFormat::BgrReserved,
86        _ => {
87            ostd::error!(
88                "Unsupported framebuffer pixel format: {} bpp",
89                framebuffer_arg.bpp
90            );
91            return;
92        }
93    };
94
95    let framebuffer = {
96        // FIXME: There can be more than `width` pixels per framebuffer line due to alignment
97        // purposes. We need to collect this information during the boot phase.
98        let line_size = framebuffer_arg
99            .width
100            .checked_mul(pixel_format.nbytes())
101            .unwrap();
102        let fb_size = framebuffer_arg
103            .height
104            .checked_mul(line_size)
105            .unwrap()
106            // The framebuffer should cover an entire set of pages. These pages can be mapped to
107            // userspace upon request.
108            .align_up(PAGE_SIZE);
109
110        let fb_base = framebuffer_arg.address;
111        // Use write-combining for framebuffer to enable faster write operations.
112        // Write-combining allows the CPU to combine multiple writes into fewer bus transactions,
113        // which is ideal for framebuffer access patterns (sequential writes).
114        let io_mem = IoMem::acquire_with_cache_policy(
115            fb_base..fb_base.checked_add(fb_size).unwrap(),
116            CachePolicy::WriteCombining,
117        )
118        .unwrap();
119
120        let default_cmap = FbCmap {
121            entries: Vec::new(),
122        };
123
124        FrameBuffer {
125            io_mem,
126            width: framebuffer_arg.width,
127            height: framebuffer_arg.height,
128            line_size,
129            pixel_format,
130            cmap: Mutex::new(default_cmap),
131        }
132    };
133
134    framebuffer.clear();
135    FRAMEBUFFER.call_once(|| Arc::new(framebuffer));
136}

138impl FrameBuffer {
139    /// Returns the width of the framebuffer in pixels.
140    pub fn width(&self) -> usize {
141        self.width
142    }
143
144    /// Returns the height of the framebuffer in pixels.
145    pub fn height(&self) -> usize {
146        self.height
147    }
148
149    /// Returns the line size of the framebuffer in bytes.
150    pub fn line_size(&self) -> usize {
151        self.line_size
152    }
153
154    /// Returns a reference to the `IoMem` instance of the framebuffer.
155    pub fn io_mem(&self) -> &IoMem {
156        &self.io_mem
157    }
158
159    /// Returns the pixel format of the framebuffer.
160    pub fn pixel_format(&self) -> PixelFormat {
161        self.pixel_format
162    }
163
164    /// Renders the pixel according to the pixel format of the framebuffer.
165    pub fn render_pixel(&self, pixel: Pixel) -> RenderedPixel {
166        pixel.render(self.pixel_format)
167    }
168
169    /// Calculates the offset of a pixel at the specified position.
170    pub fn calc_offset(&self, x: usize, y: usize) -> PixelOffset<'_> {
171        PixelOffset {
172            fb: self,
173            offset: (x * self.pixel_format.nbytes() + y * self.line_size) as isize,
174        }
175    }
176
177    /// Writes a pixel at the specified position.
178    pub fn write_pixel_at(&self, offset: PixelOffset, pixel: RenderedPixel) -> Result<()> {
179        self.io_mem.write_bytes(offset.as_usize(), pixel.as_slice())
180    }
181
182    /// Writes raw bytes at the specified offset.
183    pub fn write_bytes_at(&self, offset: usize, bytes: &[u8]) -> Result<()> {
184        self.io_mem.write_bytes(offset, bytes)
185    }
186
187    /// Clears the framebuffer with default color (black).
188    pub fn clear(&self) {
189        let frame = alloc::vec![0u8; self.io_mem().size()];
190        self.write_bytes_at(0, &frame).unwrap();
191    }
192
193    /// Sets color map entries starting from the given index.
194    ///
195    /// For efifb devices, hardware color map is not supported, so we maintain
196    /// an in-memory map for software emulation.
197    pub fn set_color_map(&self, start: usize, entries: &[ColorMapEntry]) -> Result<()> {
198        if start > MAX_CMAP_SIZE || entries.len() > MAX_CMAP_SIZE - start {
199            return Err(Error::InvalidArgs);
200        }
201
202        let mut cmap = self.cmap.lock();
203        let required_len = start + entries.len();
204
205        // Ensure the colormap has enough space
206        if cmap.entries.len() < required_len {
207            cmap.entries.resize(
208                required_len,
209                ColorMapEntry {
210                    red: 0,
211                    green: 0,
212                    blue: 0,
213                    transp: 0,
214                },
215            );
216        }
217
218        // Copy the entries
219        cmap.entries[start..start + entries.len()].copy_from_slice(entries);
220
221        Ok(())
222    }
223
224    /// Gets color map entries from the given range.
225    pub fn get_color_map(&self, start: usize, len: usize) -> Option<Vec<ColorMapEntry>> {
226        let cmap = self.cmap.lock();
227
228        if start >= cmap.entries.len() || len > cmap.entries.len() - start {
229            return None;
230        }
231
232        Some(cmap.entries[start..start + len].to_vec())
233    }
234}

236/// The offset of a pixel in the framebuffer.
237#[derive(Clone, Copy, Debug)]
238pub struct PixelOffset<'a> {
239    fb: &'a FrameBuffer,
240    offset: isize,
241}
242
243impl PixelOffset<'_> {
244    /// Adds the specified delta to the x coordinate.
245    pub fn x_add(&mut self, x_delta: isize) {
246        let delta = x_delta * self.fb.pixel_format.nbytes() as isize;
247        self.offset += delta;
248    }
249
250    /// Adds the specified delta to the y coordinate.
251    pub fn y_add(&mut self, y_delta: isize) {
252        let delta = y_delta * self.fb.line_size as isize;
253        self.offset += delta;
254    }
255
256    /// Returns the offset value as a `usize`.
257    pub fn as_usize(&self) -> usize {
258        self.offset as usize
259    }
260}
