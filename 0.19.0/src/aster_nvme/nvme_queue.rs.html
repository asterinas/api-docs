25/// Number of entries in each submission and completion ring.
26pub(crate) const QUEUE_DEPTH: usize = 64;
27
28/// Number of queue pairs the driver allocates (admin plus I/O).
29//
30// TODO: This value should be changed when supporting more than 1 I/O queue pairs.
31pub(crate) const QUEUE_NUM: usize = 2;

33/// Completion Queue.
34#[derive(Debug)]
35pub(crate) struct NvmeCompletionQueue {
36    cqueue: SafePtr<Cqring, DmaCoherent>,
37    head: u16,
38    phase: bool,
39}
40
41struct Cqring {
42    ring: [NvmeCompletion; QUEUE_DEPTH],
43}
44
45impl NvmeCompletionQueue {
46    /// Creates a new completion ring.
47    ///
48    /// Returns `None` if DMA memory for the completion ring cannot be allocated.
49    pub(crate) fn new() -> Option<Self> {
50        let dma = DmaCoherent::alloc(1, true).ok()?;
51        Some(Self {
52            cqueue: SafePtr::new(dma, 0),
53            head: 0,
54            phase: true,
55        })
56    }
57
58    /// Returns the DMA physical address of the completion ring.
59    pub(crate) fn cq_daddr(&self) -> usize {
60        self.cqueue.daddr()
61    }
62
63    /// Consumes the next completion entry if its phase tag matches the expected phase.
64    ///
65    /// Returns the new head index (for the CQ head doorbell) and the completion, or `None` if no
66    /// entry is ready.
67    fn complete(&mut self) -> Option<(u16, NvmeCompletion)> {
68        let ring_ptr: SafePtr<[NvmeCompletion; QUEUE_DEPTH], &DmaCoherent> =
69            field_ptr!(&self.cqueue, Cqring, ring);
70        let mut ring_slot_ptr = ring_ptr.cast::<NvmeCompletion>();
71        ring_slot_ptr.add(self.head as usize);
72
73        let phase_tag = NvmeCompletion::read_phase_tag(&ring_slot_ptr);
74        if phase_tag != self.phase {
75            return None;
76        }
77
78        // Read barrier.
79        fence(Ordering::SeqCst);
80
81        let entry = ring_slot_ptr
82            .read()
83            .expect("CQ slot pointer must be valid within allocated DMA ring");
84        self.head = (self.head + 1) % (QUEUE_DEPTH as u16);
85        if self.head == 0 {
86            self.phase = !self.phase;
87        }
88        Some((self.head, entry))
89    }
90}

92/// Submission Queue.
93///
94/// The type parameter `C` is optional driver-side state for in-flight commands:
95/// [`NoContext`] when none is needed (e.g., the admin queue),
96/// or [`Context`] to keep one value per CID until the matching completion is taken.
97pub(crate) struct NvmeSubmissionQueue<C = NoContext> {
98    inner: NvmeSubmissionQueueInner,
99    context: C,
100}
101
102#[derive(Default)]
103pub(crate) struct NoContext;
104
105pub(crate) struct Context<T>([Option<T>; QUEUE_DEPTH]);
106
107impl<T> Default for Context<T> {
108    fn default() -> Self {
109        Self(core::array::from_fn(|_| None))
110    }
111}

113impl<C> core::fmt::Debug for NvmeSubmissionQueue<C> {
114    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
115        f.debug_struct("NvmeSubmissionQueue")
116            .field("inner", &self.inner)
117            .finish_non_exhaustive()
118    }
119}

121impl<C: Default> NvmeSubmissionQueue<C> {
122    /// Creates a new submission queue.
123    ///
124    /// Returns `None` if DMA memory for the submission ring cannot be allocated.
125    pub(crate) fn new() -> Option<Self> {
126        Some(Self {
127            inner: NvmeSubmissionQueueInner::new()?,
128            context: Default::default(),
129        })
130    }
131}

133impl<C> NvmeSubmissionQueue<C> {
134    /// Updates the mirrored SQ head from the SQ head pointer in `completion`.
135    pub(crate) fn update_sq_head(&mut self, completion: &NvmeCompletion) {
136        self.inner.head = completion.sq_head() % (QUEUE_DEPTH as u16);
137    }
138
139    /// Returns the DMA physical address of the submission ring.
140    pub(crate) fn sq_daddr(&self) -> usize {
141        self.inner.squeue.daddr()
142    }
143
144    /// Returns the number of SQ slots available for new commands starting at the current tail.
145    pub(crate) fn free_slots(&self) -> usize {
146        let used = if self.inner.tail >= self.inner.head {
147            (self.inner.tail - self.inner.head) as usize
148        } else {
149            QUEUE_DEPTH - (self.inner.head - self.inner.tail) as usize
150        };
151        (QUEUE_DEPTH - 1) - used
152    }
153}

157    pub(crate) fn with_items<T>(self) -> NvmeSubmissionQueue<Context<T>> {
158        NvmeSubmissionQueue {
159            inner: self.inner,

165impl<T> NvmeSubmissionQueue<Context<T>> {
166    /// Removes and returns the outstanding context for `cid`.
167    ///
168    /// Returns `None` if `cid` is out of range or the slot has no outstanding context.

174/// Hardware submission ring.
175#[derive(Debug)]
176struct NvmeSubmissionQueueInner {
177    squeue: SafePtr<Sqring, DmaCoherent>,
178    tail: u16,
179    head: u16,
180}
181
182struct Sqring {
183    ring: [NvmeCommand; QUEUE_DEPTH],
184}
185
186impl NvmeSubmissionQueueInner {
187    /// Creates a new submission ring.
188    ///
189    /// Returns `None` if DMA memory for the submission ring cannot be allocated.
190    fn new() -> Option<Self> {
191        let dma = DmaCoherent::alloc(1, true).ok()?;
192        Some(Self {
193            squeue: SafePtr::new(dma, 0),
194            tail: 0,
195            head: 0,
196        })
197    }
198
199    /// Enqueues a command into the submission ring.
200    ///
201    /// Does nothing when the queue is full (`(tail + 1) % size == head`).
202    ///
203    /// Returns the new tail index for the SQ Tail doorbell, or `None` if full.
204    fn submit(&mut self, entry: NvmeCommand) -> Option<u16> {
205        let next_tail = (self.tail + 1) % (QUEUE_DEPTH as u16);
206        if next_tail == self.head {
207            return None;
208        }
209
210        let ring_ptr: SafePtr<[NvmeCommand; QUEUE_DEPTH], &DmaCoherent> =
211            field_ptr!(&self.squeue, Sqring, ring);
212        let mut ring_slot_ptr = ring_ptr.cast::<NvmeCommand>();
213        ring_slot_ptr.add(self.tail as usize);
214        ring_slot_ptr
215            .write(&entry)
216            .expect("SQ slot pointer must be valid within allocated DMA ring");
217
218        self.tail = next_tail;
219        Some(self.tail)
220    }
221}

223pub(crate) struct NvmeCompletionQueueAccess<'a, Q> {
224    qid: u16,
225    dstrd: u16,
226    queue: Q,
227    dbregs: DbregAccess<'a>,
228}
229
230impl<'a, Q> NvmeCompletionQueueAccess<'a, Q>
231where
232    Q: DerefMut<Target = NvmeCompletionQueue>,
233{
234    /// Binds queue `qid` and doorbell stride `dstrd` to `queue` and `dbregs` for locked poll.
235    pub(crate) fn new(qid: u16, dstrd: u16, queue: Q, dbregs: DbregAccess<'a>) -> Self {
236        Self {
237            qid,
238            dstrd,
239            queue,
240            dbregs,
241        }
242    }
243
244    /// Consumes all ready completions and updates the CQ head doorbell once.
245    ///
246    /// Returns an empty vector when no completion is ready.
247    pub(crate) fn complete(&mut self) -> Vec<NvmeCompletion> {
248        let mut entries = Vec::new();
249        let mut last_head = None;
250        while let Some((entry, new_head)) = self.poll() {
251            last_head = Some(new_head);
252            entries.push(entry);
253        }
254        if let Some(new_head) = last_head {
255            self.ring_doorbell(new_head);
256        }
257        entries
258    }
259
260    /// Polls one completion without ringing the CQ doorbell.
261    fn poll(&mut self) -> Option<(NvmeCompletion, u16)> {
262        let (new_head, entry) = self.queue.complete()?;
263        if entry.has_error() {
264            warn!(
265                "completion queue {}: command failed (CID={}, status={:04X}, SC={:#04x}, SQID={})",
266                self.qid,
267                entry.cid(),
268                entry.status(),
269                entry.status_code(),
270                entry.sq_id(),
271            );
272        }
273        Some((entry, new_head))
274    }
275
276    /// Rings the CQ head doorbell after one or more [`Self::poll`] calls.
277    fn ring_doorbell(&mut self, new_head: u16) {
278        // Full barrier: do not update the doorbell until the completion entry reads finish.
279        fence(Ordering::SeqCst);
280        self.dbregs.write_racy(
281            NvmeDoorbellRegs::Cqhdbl,
282            self.qid,
283            self.dstrd,
284            new_head as u32,
285        );
286    }
287}

296impl<'a, Q, C> NvmeSubmissionQueueAccess<'a, Q>
297where
298    Q: DerefMut<Target = NvmeSubmissionQueue<C>>,
299{
300    /// Binds queue `qid` and doorbell stride `dstrd` to `queue` and `dbregs` for locked submit.
301    pub(crate) fn new(qid: u16, dstrd: u16, queue: Q, dbregs: DbregAccess<'a>) -> Self {
302        Self {
303            qid,
304            dstrd,
305            queue,
306            dbregs,
307        }
308    }
309
310    /// Returns the number of free SQ slots available for new commands.
311    pub(crate) fn free_slots(&self) -> usize {
312        self.queue.free_slots()
313    }
314
315    /// Rings the SQ tail doorbell after one or more enqueued commands.
316    fn ring_doorbell(&mut self) {
317        let new_tail = self.queue.inner.tail;
318        // Write barrier: do not update the doorbell until the submit entry writes finish.
319        fence(Ordering::SeqCst);
320        self.dbregs.write_racy(
321            NvmeDoorbellRegs::Sqtdbl,
322            self.qid,
323            self.dstrd,
324            new_tail as u32,
325        );
326    }
327}

329impl<'a, Q> NvmeSubmissionQueueAccess<'a, Q>
330where
331    Q: DerefMut<Target = NvmeSubmissionQueue<NoContext>>,
332{
333    /// Submits a command and rings the SQ tail doorbell.
334    ///
335    /// Writes at the current tail, sets the command identifier to that slot index, advances the
336    /// tail, then updates the doorbell.
337    ///
338    /// Returns the command identifier used (same as the tail before enqueue), or `None` if the
339    /// queue is full.
340    pub(crate) fn submit(&mut self, mut entry: NvmeCommand) -> Option<u16> {
341        let cid = self.queue.inner.tail;
342        entry.set_cid(cid);
343        self.queue.inner.submit(entry)?;
344        self.ring_doorbell();
345        Some(cid)
346    }
347}
348
349impl<'a, Q, T> NvmeSubmissionQueueAccess<'a, Q>
350where
351    Q: DerefMut<Target = NvmeSubmissionQueue<Context<T>>>,
352{
353    /// Submits a batch of commands with contexts and rings the SQ doorbell once.
354    pub(crate) fn submit_with_items(
355        &mut self,
356        commands: impl IntoIterator<Item = (NvmeCommand, T)>,
357    ) -> Option<usize> {
358        let mut count = 0;
359        for (entry, item) in commands {
360            self.enqueue_one(entry, item)?;
361            count += 1;
362        }
363        if count > 0 {
364            self.ring_doorbell();
365        }
366        Some(count)
367    }
368
369    /// Enqueues one command and context without ringing the SQ doorbell.
370    fn enqueue_one(&mut self, mut entry: NvmeCommand, item: T) -> Option<u16> {
371        let cid = self.queue.inner.tail;
372        if self.queue.context.0[cid as usize].is_some() {
373            warn!(
374                "submission queue {} slot {} is still outstanding",
375                self.qid, cid
376            );
377            return None;
378        }
379        entry.set_cid(cid);
380        self.queue.inner.submit(entry)?;
381        self.queue.context.0[cid as usize] = Some(item);
382        Some(cid)
383    }
