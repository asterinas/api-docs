11pub(in crate::iface) fn parse(
12    mut pkt: RxPacket<TransportLayer>,
13    ip_repr: &IpRepr,
14    csum: bool,
15) -> Option<(RxPacket<ApplicationLayer>, UdpRepr)> {
16    let header = pkt.reader().read_val::<Header>().ok()?;
17
18    let packet_len = u16::from(header.len) as usize;
19    if packet_len < size_of::<Header>() || packet_len > pkt.len() {
20        return None;
21    }
22    pkt.truncate(packet_len);
23
24    let dst_port = u16::from(header.dst);
25    if dst_port == 0 {
26        // In practice, zero port numbers are not valid because none of the operating systems in
27        // production can bind to them. They are reserved by the IANA:
28        // <https://www.iana.org/assignments/service-names-port-numbers/service-names-port-numbers.xhtml>.
29        //
30        // However, UDP allows the source port to be omitted. "Source Port is an optional field [..]
31        // If not used, a value of zero is inserted."
32        // Reference: <https://datatracker.ietf.org/doc/html/rfc768>.
33        return None;
34    }
35
36    if header.csum == 0 {
37        if csum && matches!(ip_repr, IpRepr::Ipv6(_)) {
38            // "IPv6 receivers must discard UDP packets containing a zero checksum."
39            // Reference: <https://datatracker.ietf.org/doc/html/rfc8200#section-8.1>.
40            return None;
41        }
42    } else if csum
43        && Checksum::new()
44            .with_pseudo_header(ip_repr, packet_len)
45            .with_reader(pkt.reader())
46            .finish()
47            != u16::MAX
48    {
49        return None;
50    }
51
52    let repr = UdpRepr {
53        src_port: header.src.into(),
54        dst_port,
55    };
56    Some((pkt.peel(size_of::<Header>()), repr))
57}

59pub(in crate::iface) fn emit(
60    pkt: TxPacket<ApplicationLayer>,
61    ip_repr: &IpRepr,
62    udp_repr: &UdpRepr,
63    csum: bool,
64) -> TxPacket<TransportLayer> {
65    debug_assert_eq!(ip_repr.next_header(), IpProtocol::Udp);
66
67    let packet_len = size_of::<Header>() + pkt.len();
68    debug_assert_eq!(ip_repr.payload_len(), packet_len);
69
70    let mut header = Header {
71        src: udp_repr.src_port.into(),
72        dst: udp_repr.dst_port.into(),
73        len: (packet_len as u16).into(),
74        csum: 0,
75    };
76
77    if csum {
78        let csum_val = !Checksum::new()
79            .with_pseudo_header(ip_repr, packet_len)
80            .with_bytes(header.as_bytes())
81            .with_reader(pkt.reader())
82            .finish();
83        header.csum = if csum_val == 0 { u16::MAX } else { csum_val };
84    }
85
86    pkt.prepend_and_pack(&header)
87}
88
89#[repr(C)]
90#[derive(Clone, Copy, Debug, Pod)]
91struct Header {
92    src: Be16,
93    dst: Be16,
94    len: Be16,
95    csum: u16,
96}
