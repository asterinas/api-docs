15/// A network interface.
16///
17/// A network interface (abbreviated as iface) is a hardware or software component that connects a
18/// computer to a network. Network interfaces can be physical components like Ethernet ports or
19/// wireless adapters. They can also be virtual interfaces created by software, such as virtual
20/// private network (VPN) connections.
21pub trait Iface<E>: internal::IfaceInternal<E> + Send + Sync {
22    /// Returns the Ethernet address, if the interface uses Ethernet framing.
23    fn ethernet_addr(&self) -> Option<EthernetAddress>;
24
25    /// Transmits or receives packets queued in the iface, and updates socket status accordingly.
26    fn poll(&self);
27
28    /// Returns the maximum transmission unit.
29    fn mtu(&self) -> usize;
30}
31
32impl<E: Ext> dyn Iface<E> {
33    // FIXME: The reason for binding the socket and the iface together is because there are
34    // limitations inside smoltcp. See discussion at
35    // <https://github.com/smoltcp-rs/smoltcp/issues/779>.
36
37    /// Binds a TCP port to the iface.
38    ///
39    /// If no specific port is given in [`BindPortConfig`], the iface will pick up an ephemeral
40    /// port.
41    pub fn bind_tcp(
42        self: &Arc<Self>,
43        config: BindPortConfig,
44    ) -> Result<BoundTcpPort<E>, BindError> {
45        let common = self.common();
46        common.bind_tcp(self.clone(), config)
47    }
48
49    /// Binds a UDP port to the iface.
50    ///
51    /// If no specific port is given in [`BindPortConfig`], the iface will pick up an ephemeral
52    /// port.
53    pub fn bind_udp(
54        self: &Arc<Self>,
55        config: BindPortConfig,
56    ) -> Result<BoundUdpPort<E>, BindError> {
57        let common = self.common();
58        common.bind_udp(self.clone(), config)
59    }
60
61    /// Returns the interface index.
62    pub fn index(&self) -> u32 {
63        self.common().index()
64    }
65
66    /// Gets the name of the iface.
67    ///
68    /// In Linux, the name is usually the driver name followed by a unit number.
69    pub fn name(&self) -> &InterfaceName {
70        self.common().name()
71    }
72
73    /// Returns the interface type.
74    pub fn type_(&self) -> InterfaceType {
75        self.common().type_()
76    }
77
78    /// Returns the interface flags.
79    pub fn flags(&self) -> InterfaceFlags {
80        self.common().flags()
81    }
82
83    // FIXME: Linux and smoltcp allow multiple IP CIDRs per interface, while the
84    // address-related APIs below only account for the first CIDR of each family.
85
86    /// Gets the IPv4 CIDR of the iface, if any.
87    pub fn ipv4_cidr(&self) -> Option<Ipv4Cidr> {
88        self.common().ipv4_cidr()
89    }
90
91    /// Gets the IPv6 CIDR of the iface, if any.
92    pub fn ipv6_cidr(&self) -> Option<Ipv6Cidr> {
93        self.common().ipv6_cidr()
94    }
95
96    /// Gets the IPv4 broadcast address of the iface, if any.
97    ///
98    /// IPv6 does not define broadcast addresses and uses multicast instead.
99    pub fn broadcast_addr(&self) -> Option<Ipv4Address> {
100        self.common().ipv4_cidr()?.broadcast()
101    }
102
103    /// Returns routes currently installed in the interface.
104    pub fn routes(&self) -> Vec<Route> {
105        self.common().routes()
106    }
107
108    /// Returns a reference to the associated [`ScheduleNextPoll`].
109    ///
110    /// [`ScheduleNextPoll`]: crate::iface::sched::ScheduleNextPoll
111    pub fn sched_poll(&self) -> &E::ScheduleNextPoll {
112        self.common().sched_poll()
113    }
114}
