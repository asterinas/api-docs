24/// Initializes the unaccepted-memory table from the EFI stub boot parameters.
25pub(crate) fn init(boot_params: &BootParams) {
26    let efi_info = boot_params.efi_info;
27    assert_eq!(
28        efi_info.efi_loader_signature,
29        EfiInfo::TDX_LOADER_SIGNATURE,
30        "TDX guests must be booted with `TDX_LOADER_SIGNATURE`"
31    );
32    assert_eq!(
33        efi_info.efi_systab_hi, 0,
34        "TDX unaccepted-memory table must be below 4 GiB"
35    );
36
37    let table_addr = efi_info.efi_systab;
38    if table_addr == 0 {
39        return;
40    }
41    let table_ptr =
42        crate::mm::kspace::paddr_to_vaddr(table_addr as usize) as *const EfiUnacceptedMemory;
43
44    crate::info!("Found unaccepted memory table at {:p}", table_ptr);
45    // SAFETY:
46    // - The table is safe to read because of the contract with the EFI stub.
47    // - We reserve the table region in `table_memory_range`, so it will live as an immutable
48    //   reference for `'static`.
49    let table: &'static EfiUnacceptedMemory = unsafe { &*table_ptr };
50    UNACCEPTED_TABLE.call_once(|| table);
51}

53/// Returns the physical range occupied by the table storage.
54pub(crate) fn table_memory_range() -> Option<Range<Paddr>> {
55    let table = UNACCEPTED_TABLE.get().copied()?;
56    let base = core::ptr::from_ref(table).addr() - crate::mm::kspace::LINEAR_MAPPING_BASE_VADDR;
57    let size = size_of::<EfiUnacceptedMemory>() + table.bitmap_size_bytes() as usize;
58    Some(base..base + size)
59}

61/// Accepts the BSP's slice of unaccepted memory, waits for all APs to finish,
62/// and publishes the accepted memory to the global frame allocator.
63///
64/// # Safety
65///
66/// This function must be called only once in the boot context of the BSP, after booting APs.
67pub(crate) unsafe fn accept_memory_on_bsp() {
68    let Some(table) = UNACCEPTED_TABLE.get().copied() else {
69        return;
70    };
71
72    let num_cpus = crate::cpu::num_cpus();
73    let range = range_to_accept_on_cpu(table, CpuId::bsp().as_usize(), num_cpus);
74    if !range.is_empty() {
75        // SAFETY:
76        // - This table and bitmap describe pending private-memory units.
77        // - The BSP and the APs accept pages concurrently, but their ranges are disjoint.
78        unsafe { table.accept_range(range.start, range.end) }.expect("failed to accept memory");
79    }
80    FINISHED_CPU_COUNT.fetch_add(1, Ordering::Release);
81
82    while FINISHED_CPU_COUNT.load(Ordering::Acquire) != num_cpus {
83        core::hint::spin_loop();
84    }
85    publish_accepted_memory(table);
86}

88/// Accepts the calling AP's disjoint slice of unaccepted memory and marks completion.
89///
90/// # Safety
91///
92/// This function must be called only once in the boot context of the AP.
93pub(crate) unsafe fn accept_memory_on_ap() {
94    let Some(table) = UNACCEPTED_TABLE.get().copied() else {
95        return;
96    };
97
98    let num_cpus = crate::cpu::num_cpus();
99    let range = range_to_accept_on_cpu(table, CpuId::current_racy().as_usize(), num_cpus);
100    if !range.is_empty() {
101        // SAFETY:
102        // - This table and bitmap describe pending private-memory units.
103        // - The BSP and the APs accept pages concurrently, but their ranges are disjoint.
104        unsafe { table.accept_range(range.start, range.end) }.expect("failed to accept memory");
105    }
106    FINISHED_CPU_COUNT.fetch_add(1, Ordering::Release);
107}

109/// Records early-allocated ranges and returns the unaccepted-memory table's coverage range.
110/// It may include memory that has already been accepted.
111pub(super) fn init_unaccepted_memory_state(
112    early_allocated_ranges: &[Range<Paddr>; 2],
113) -> Range<Paddr> {
114    let Some(table) = UNACCEPTED_TABLE.get().copied() else {
115        return 0..0;
116    };
117    EARLY_ALLOCATED_RANGES.call_once(|| early_allocated_ranges.clone());
118
119    let coverage_range = table.coverage_range();
120    coverage_range.start as Paddr..coverage_range.end as Paddr
121}

123/// Accepts memory that must be accessed before parallel early acceptance starts.
124///
125/// # Safety
126///
127/// This function must not be called concurrently with any other functions that accept pages.
128pub(super) unsafe fn accept_early_allocated_range(start: Paddr, size: usize) {
129    let Some(table) = UNACCEPTED_TABLE.get().copied() else {
130        return;
131    };
132    let start_aligned = start.align_down(crate::mm::PAGE_SIZE);
133    let end_aligned = (start + size).align_up(crate::mm::PAGE_SIZE);
134
135    // SAFETY:
136    // - This table and bitmap describe pending private-memory units.
137    // - There are no concurrent operations, as guaranteed by the caller.
138    unsafe { table.accept_range(start_aligned as u64, end_aligned as u64) }
139        .expect("failed to accept boot memory");
140}

142/// Publishes accepted memory regions to the global frame allocator.
143fn publish_accepted_memory(table: &EfiUnacceptedMemory) {
144    let coverage_range = table.coverage_range();
145    let coverage_range = coverage_range.start as Paddr..coverage_range.end as Paddr;
146    let early_allocated_ranges = EARLY_ALLOCATED_RANGES
147        .get()
148        .expect("early allocated ranges are unavailable");
149
150    let regions = &crate::boot::EARLY_INFO.get().unwrap().memory_regions;
151    let [early_range_1, early_range_2] = early_allocated_ranges;
152
153    for region in regions
154        .iter()
155        .filter(|region| region.typ() == crate::boot::memory_region::MemoryRegionType::Usable)
156    {
157        let region_range = region.base()..region.end();
158        for accepted_range in range_difference(&region_range, early_range_1)
159            .flat_map(|range| range_difference(&range, early_range_2))
160            .filter_map(|range| range_intersection(&range, &coverage_range))
161        {
162            crate::info!(
163                "Adding accepted free frames to the allocator: {:x?}",
164                accepted_range
165            );
166            super::allocator::get_global_frame_allocator()
167                .add_free_memory(accepted_range.start, accepted_range.len());
168        }
169    }
170}

172/// Returns the GPA range assigned to one CPU within the table coverage.
173fn range_to_accept_on_cpu(
174    table: &EfiUnacceptedMemory,
175    index: usize,
176    num_cpus: usize,
177) -> Range<u64> {
178    debug_assert!(index < num_cpus);
179
180    let coverage_range = table.coverage_range();
181    let unit_size = u64::from(table.unit_size_bytes());
182    let num_units = (coverage_range.end - coverage_range.start) / unit_size;
183    let (start_unit, end_unit) = partition_units(num_units, index, num_cpus);
184
185    let start = coverage_range.start + start_unit * unit_size;
186    let end = coverage_range.start + end_unit * unit_size;
187    start..end
188}

190/// Uniformly partitions `total_units` across `num_cpus`, aligned to 8 units (1 bitmap byte).
191fn partition_units(total_units: u64, index: usize, num_cpus: usize) -> (u64, u64) {
192    const UNITS_PER_BYTE: u64 = 8;
193
194    let total_bytes = total_units / UNITS_PER_BYTE;
195    let base_bytes = total_bytes / num_cpus as u64;
196    let remain_bytes = total_bytes % num_cpus as u64;
197
198    let index = index as u64;
199    let start_byte = base_bytes * index + remain_bytes.min(index);
200    let end_byte = base_bytes * (index + 1) + remain_bytes.min(index + 1);
201
202    let start_unit = start_byte * UNITS_PER_BYTE;
203    let mut end_unit = end_byte * UNITS_PER_BYTE;
204    if index == (num_cpus as u64 - 1) {
205        end_unit = total_units;
206    }
207    (start_unit, end_unit)
208}
