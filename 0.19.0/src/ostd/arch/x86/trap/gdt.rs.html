33/// Initializes and loads the GDT and TSS.
34///
35/// The caller should only call this method once in the boot context for each available processor.
36/// This is not a safety requirement, however, because calling this method again will do nothing
37/// more than load the GDT and TSS with the same contents.
38///
39/// # Safety
40///
41/// The caller must ensure that no preemption can occur during the method, otherwise we may
42/// accidentally load a wrong GDT and TSS that actually belongs to another CPU.
43pub(super) unsafe fn init_on_cpu() {
44    // No races because the caller guarantees that no preemption can occur.
45    let gdt = GDT.get_on_cpu(CpuId::current_racy());
46
47    let tss_ptr = UnsafeCell::raw_get(LOCAL_TSS.as_ptr());
48
49    // FIXME: The segment limit in the descriptor created by `tss_segment_unchecked` does not
50    // include the I/O port bitmap.
51
52    // SAFETY: As a CPU-local variable, the TSS lives for `'static`.
53    let tss_desc = unsafe { Descriptor::tss_segment_unchecked(tss_ptr) };
54    let (tss0, tss1) = match tss_desc {
55        Descriptor::SystemSegment(tss0, tss1) => (tss0, tss1),
56        _ => unreachable!(),
57    };
58    gdt.tss0.store(tss0, Ordering::Relaxed);
59    gdt.tss1.store(tss1, Ordering::Relaxed);
60
61    // The kernel CS is considered a global invariant set by the boot GDT. This method is not
62    // intended for switching to a new kernel CS.
63    assert_eq!(CS::get_reg(), KERNEL_CS);
64
65    // Load the new GDT.
66    let gdtr = DescriptorTablePointer {
67        limit: (size_of_val(gdt) - 1) as u16,
68        base: VirtAddr::new(core::ptr::from_ref(gdt).addr() as u64),
69    };
70    // SAFETY: The GDT is valid to load because:
71    //  - It lives for `'static`.
72    //  - It contains correct entries at correct indexes: the kernel code/data segments, the user
73    //    code/data segments, and the TSS segment.
74    //  - Specifically, the TSS segment points to the CPU-local TSS of the current CPU.
75    unsafe { lgdt(&gdtr) };
76
77    // Load the TSS.
78    //
79    // SAFETY: The selector points to the TSS descriptors in the GDT.
80    unsafe { load_tss(TSS_SEL) };
81}

83/// Returns the current CPU's GDT base address.

88/// Returns the GDT limit.

93/// Returns the current CPU's TSS base address.

95    UnsafeCell::raw_get(LOCAL_TSS.as_ptr()) as Vaddr

98// The linker script makes sure that the `.cpu_local_tss` section is at the beginning of the area
99// that stores CPU-local variables. This is important because `trap.S` and `syscall.S` will assume
100// this and treat the beginning of the CPU-local area as a TSS for loading and saving the kernel
101// stack!
102//
103// No other special initialization is required because the kernel stack information is stored in
104// the TSS when we start the userspace program. However, to ensure the store's safety, we should
105// wrap the TSS in an `UnsafeCell`. See `syscall.S` for details.
106//
107// SAFETY: This is properly handled in the linker script.
108#[unsafe(link_section = ".cpu_local_tss")]
109static LOCAL_TSS: StaticCpuLocal<UnsafeCell<TaskStateSegment>> = {
110    let tss = UnsafeCell::new(TaskStateSegment::new());
111    // SAFETY: The `.cpu_local_tss` section is part of the CPU-local area.
112    unsafe { CpuLocal::__new_static(tss) }
113};

115#[repr(C)]
116struct Gdt {
117    // Immutable part: They are constant because they are either set to zero or all of the used bits
118    // are set at compile time, including the ACCESSED bit. So they will not be modified later.
119    _reserved: u64,
120    kcode64: u64,
121    kdata: u64,
122    _kcode32: u64,
123    _ucode32: u64,
124    udata: u64,
125    ucode64: u64,
126    // Mutable part: They are initialized at runtime; the LTR instruction will set the BUSY bit.
127    tss0: AtomicU64,
128    tss1: AtomicU64,
129}
130
131cpu_local! {
132    static GDT: Gdt = Gdt {
133        _reserved: 0,
134        kcode64: KCODE64,
135        kdata: KDATA,
136        _kcode32: 0, // Not used.
137        _ucode32: 0, // Not used.
138        udata: UDATA,
139        ucode64: UCODE64,
140        tss0: AtomicU64::new(0), // To be set in `init_on_cpu`.
141        tss1: AtomicU64::new(0), // To be set in `init_on_cpu`.
142    };
143}

145// ========= Descriptors =========
146
147// Kernel code and data descriptors.
148//
149// These are the exact, unique values that satisfy the requirements of the `syscall` instruction.
150// The Intel manual says: "It is the responsibility of OS software to ensure that the descriptors
151// (in GDT or LDT) referenced by those selector values correspond to the fixed values loaded into
152// the descriptor caches; the SYSCALL instruction does not ensure this correspondence."
153pub(in crate::arch) const KCODE64: u64 = 0x00AF_9B00_0000_FFFF;
154pub(in crate::arch) const KDATA: u64 = 0x00CF_9300_0000_FFFF;
155
156// A 32-bit code descriptor that is used in the boot stage only. See `boot/bsp_boot.S`.
157pub(in crate::arch) const KCODE32: u64 = 0x00CF_9B00_0000_FFFF;
158
159// User code and data descriptors.
160//
161// These are the exact, unique values that satisfy the requirements of the `sysret` instruction.
162// The Intel manual says: "It is the responsibility of OS software to ensure that the descriptors
163// (in GDT or LDT) referenced by those selector values correspond to the fixed values loaded into
164// the descriptor caches; the SYSRET instruction does not ensure this correspondence."
165const UCODE64: u64 = 0x00AF_FB00_0000_FFFF;
166const UDATA: u64 = 0x00CF_F300_0000_FFFF;
167
168// ========== Selectors ==========
169
170pub(in crate::arch) const KERNEL_CS: SegmentSelector =
171    SegmentSelector::new(1, PrivilegeLevel::Ring0);
172const KERNEL_SS: SegmentSelector = SegmentSelector::new(2, PrivilegeLevel::Ring0);
173const_assert!(KERNEL_CS.0 & OFFSET_MASK == offset_of!(Gdt, kcode64) as u16);
174const_assert!(KERNEL_SS.0 & OFFSET_MASK == offset_of!(Gdt, kdata) as u16);
175
176pub(super) const USER_CS: SegmentSelector = SegmentSelector::new(6, PrivilegeLevel::Ring3);
177pub(super) const USER_SS: SegmentSelector = SegmentSelector::new(5, PrivilegeLevel::Ring3);
178const_assert!(USER_CS.0 & OFFSET_MASK == offset_of!(Gdt, ucode64) as u16);
179const_assert!(USER_SS.0 & OFFSET_MASK == offset_of!(Gdt, udata) as u16);
180
181pub(in crate::arch) const TSS_SEL: SegmentSelector = SegmentSelector::new(7, PrivilegeLevel::Ring0);
182const_assert!(TSS_SEL.0 & OFFSET_MASK == offset_of!(Gdt, tss0) as u16);
183const_assert!(TSS_SEL.0 & OFFSET_MASK == (offset_of!(Gdt, tss1) - size_of::<u64>()) as u16);
184
185pub(in crate::arch) const SYSRET_SEL: SegmentSelector =
186    SegmentSelector::new(4, PrivilegeLevel::Ring3);
187pub(in crate::arch) const SYSCALL_SEL: SegmentSelector =
188    SegmentSelector::new(1, PrivilegeLevel::Ring0);
189const_assert!(SYSRET_SEL.0 & OFFSET_MASK == (offset_of!(Gdt, udata) - size_of::<u64>()) as u16);
190const_assert!(
191    SYSRET_SEL.0 & OFFSET_MASK == (offset_of!(Gdt, ucode64) - size_of::<u64>() * 2) as u16
192);
193const_assert!(SYSCALL_SEL.0 & OFFSET_MASK == offset_of!(Gdt, kcode64) as u16);
194const_assert!(SYSCALL_SEL.0 & OFFSET_MASK == (offset_of!(Gdt, kdata) - size_of::<u64>()) as u16);
195
196// Masking out the lower 3 bits of a selector yields the offset to the GDT entry from the GDT base.
197const OFFSET_MASK: u16 = !7;
