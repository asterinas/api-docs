3use super::super::types::VcpuRegs;
4use crate::irq::DisabledLocalIrqGuard;

6/// Enters or resumes the guest and returns after a VM exit.
7///
8/// # Safety
9///
10/// 1. VMX must be enabled with an exclusively owned current VMCS whose launch
11///    state matches `is_launched`.
12/// 2. The VMCS must isolate guest memory and privileged operations, and restore
13///    the current host's state, including RIP at `vm_exit_handler_virtaddr()`.
14///    All referenced resources must remain alive through the VM exit.
15pub(super) unsafe fn vcpu_run(
16    regs: &mut VcpuRegs,
17    is_launched: bool,
18    _irq_guard: &DisabledLocalIrqGuard,
19) -> u64 {
20    // SAFETY: The caller ensures the VMX execution contract. The exclusive
21    // reference also provides valid register storage for the assembly routine.
22    unsafe { __vcpu_run(regs, u64::from(is_launched)) }
23}
24
25pub(in crate::arch::vm) fn vm_exit_handler_virtaddr() -> usize {
26    __vm_exit_handler as *const () as usize
27}
28
29unsafe extern "C" {
30    fn __vcpu_run(regs: *mut VcpuRegs, launched: u64) -> u64;
31    fn __vm_exit_handler();
32}
