9sensitive_io_port! {
10    unsafe {
11        static MASTER_CMD: IoPort<u8, WriteOnlyAccess> = IoPort::new(0x20);
12        static MASTER_DATA: IoPort<u8, ReadWriteAccess> = IoPort::new(0x21);
13        static SLAVE_CMD: IoPort<u8, WriteOnlyAccess> = IoPort::new(0xA0);
14        static SLAVE_DATA: IoPort<u8, WriteOnlyAccess> = IoPort::new(0xA1);
15    }
16}
17
18const IRQ_OFFSET: u8 = 0x20;
19
20/// Probes whether a legacy 8259A PIC is present.
21///
22/// Reference: <https://elixir.bootlin.com/linux/v7.0/source/arch/x86/kernel/i8259.c#L306>
23pub(super) fn probe_legacy_pic() -> bool {
24    // Mask all except cascade (IRQ 2) on master IMR.
25    const PROBE_VAL: u8 = 0xFB;
26
27    // Mask slave PIC first to avoid interrupts during probe.
28    SLAVE_DATA.write(0xFF);
29
30    // Mask master PIC except cascade (IRQ 2).
31    MASTER_DATA.write(PROBE_VAL);
32
33    // Read back from master PIC. If the port exists, it returns what we wrote.
34    // If it's a memory hole (no PIC), it usually returns 0xFF or 0x00.
35    MASTER_DATA.read() == PROBE_VAL
36}

38/// Initializes and disables the 8259 Programmable Interrupt Controller (PIC).
39pub(super) fn init_and_disable() {
40    info!("Initializing PIC as disabled");
41
42    set_mask(0xff, 0xff);
43}
44
45fn set_mask(master_mask: u8, slave_mask: u8) {
46    // Start initialization.
47    MASTER_CMD.write(0x11);
48    SLAVE_CMD.write(0x11);
49
50    // Set offsets.
51    // - Map master PIC vector 0x00~0x07 to IRQ number 0x20~0x27.
52    MASTER_DATA.write(IRQ_OFFSET);
53    // - Map slave PIC vector 0x00~0x07 to IRQ number 0x28~0x2f.
54    SLAVE_DATA.write(IRQ_OFFSET + 0x08);
55
56    // Set up cascade (there is a slave PIC at IRQ2).
57    MASTER_DATA.write(4);
58    SLAVE_DATA.write(2);
59
60    // Set up interrupt mode (1 is 8086/88 mode, 2 is auto EOI).
61    MASTER_DATA.write(1);
62    SLAVE_DATA.write(1);
63
64    // Mask interrupts.
65    MASTER_DATA.write(master_mask);
66    SLAVE_DATA.write(slave_mask);
67}
