3//! TCPv4 Protocol
4//!
5//! This module provides the TCPv4 Protocol interface definitions. The
6//! TCPv4 Protocol provides services to send and receive data streams
7//! over IPv4 networks.
8//!
9//! The protocol is defined in the [UEFI Specification, Section 28.1](https://uefi.org/specs/UEFI/2.11/28_Network_Protocols_TCP_IP_and_Configuration.html#efi-tcpv4-protocol).
10

17#[derive(Debug)]
18#[repr(C)]
19pub struct Tcp4Protocol {
20    /// Get the current operational status.
21    ///
22    /// `get_mode_data` copies the current operational settings of
23    /// this instance into user-supplied structs. This function can
24    /// also be used to retrieve the operational setting of underlying
25    /// drivers such as IPv4, MNP, or SNP.
26    #[allow(clippy::type_complexity)]
27    pub get_mode_data: unsafe extern "efiapi" fn(
28        this: *mut Self,
29        connection_state: *mut Tcp4ConnectionState,
30        config_data: *mut Tcp4ConfigData,
31        ip4_mode_data: *mut Ip4ModeData,
32        managed_network_config_data: *mut c_void,
33        simple_network_mode: *mut NetworkMode,
34    ) -> Status,
35
36    /// Initialize or brutally reset the operational parameters for
37    /// this instance.
38    ///
39    /// No other [`Tcp4Protocol`] operation can be executed by this
40    /// instance until it is configured properly. For an active
41    /// [`Tcp4Protocol`] instance, after a proper configuration it may
42    /// call [`Tcp4Protocol::connect`] to initiates the three-way
43    /// handshake. For a passive [`Tcp4Protocol`] instance, its state
44    /// will transition to [`Tcp4ConnectionState::LISTEN`] after
45    /// configuration, and [`accept`][Self::accept] may be called to
46    /// listen for incoming TCP connection requests. If `config_data`
47    /// is set to `NULL`, the instance is reset. Resetting process
48    /// will be done brutally; the state machine will be set to
49    /// [`Tcp4ConnectionState::CLOSED`] directly, the receive queue
50    /// and transmit queue will be flushed, and no traffic will be
51    /// allowed through this instance.
52    pub configure:
53        unsafe extern "efiapi" fn(this: *mut Self, config_data: *const Tcp4ConfigData) -> Status,
54
55    /// Add or delete routing entries.
56    ///
57    /// The most specific route is selected by comparing the
58    /// `subnet_address` with the destination IP address
59    /// arithmetically ANDed with the `subnet_mask`.
60    ///
61    /// The default route is added with both `subnet_address` and
62    /// `subnet_mask` set to `0.0.0.0`. The default route matches all
63    /// destination IP addresses if there is no more specific route.
64    ///
65    /// A direct route is added with `gateway_address` set to
66    /// `0.0.0.0`. Packets are sent to the destination host if its
67    /// address can be found in the Address Resolution Protocol (ARP)
68    /// cache or it is on the local subnet. If the instance is
69    /// configured to use default address, a direct route to the local
70    /// network will be added automatically.
71    ///
72    /// Each TCP instance has its own independent routing table. An
73    /// instance that uses the default IP address will have a copy of
74    /// the
75    /// [`Ipv4Config2Protocol`][super::ip4_config2::Ip4Config2Protocol]'s
76    /// routing table. The copy will be updated automatically whenever
77    /// the IP driver reconfigures its instance. As a result, the
78    /// previous modification to the instance's local copy will be
79    /// lost.
80    ///
81    /// The priority of checking the route table is specific to the IP
82    /// implementation, and every IP implementation must comply with
83    /// RFC 1122.
84    ///
85    /// Note: There is no way to set up routes to other network
86    /// interface cards (NICs) because each NIC has its own
87    /// independent network stack that shares information only through
88    /// EFI TCPv4 variable.
89    pub routes: unsafe extern "efiapi" fn(
90        this: *mut Self,
91        delete_route: Boolean,
92        subnet_address: *const Ipv4Address,
93        subnet_mask: *const Ipv4Address,
94        gateway_address: *const Ipv4Address,
95    ) -> Status,
96
97    /// Initiate a nonblocking TCP connection request for an active
98    /// TCP instance.
99    ///
100    /// `connect` initiates an active open to the remote peer
101    /// configured in the current TCP instance if it is configured as
102    /// active. If the connection succeeds or fails due to any error,
103    /// the [`token.completion_token.event`][Tcp4CompletionToken::event]
104    /// will be signaled and `token.completion_token.status` will be
105    /// updated accordingly. This function can only be called for the
106    /// TCP instance in the [`Tcp4ConnectionState::CLOSED`] state. The
107    /// instance will transition to [`Tcp4ConnectionState::SYN_SENT`]
108    /// if the function returns [`Status::SUCCESS`]. If the TCP
109    /// three-way handshake succeeds, its state will become
110    /// [`Tcp4ConnectionState::ESTABLISHED`], otherwise, the state
111    /// will return to [`Tcp4ConnectionState::CLOSED`].
112    pub connect:
113        unsafe extern "efiapi" fn(this: *mut Self, token: *mut Tcp4ConnectionToken) -> Status,
114
115    /// Listen on the passive instance to accept an incoming
116    /// connection request. This is a nonblocking operation.
117    ///
118    /// The `accept` function initiates an asynchronous accept request
119    /// to wait for an incoming connection on the passive TCP
120    /// instance. If a remote peer successfully establishes a
121    /// connection with this instance, a new TCP instance will be
122    /// created and its handle will be returned in
123    /// [`new_child_handle`][Tcp4ListenToken::new_child_handle]. The
124    /// newly created instance is configured by inheriting the passive
125    /// instance's configuration and is ready for use upon return. The
126    /// instance is in the [`Tcp4ConnectionState::ESTABLISHED`]
127    /// state.
128    ///
129    /// The [`new_child_handle`][Tcp4ListenToken::new_child_handle]
130    /// will be signaled when a new connection is accepted, the user
131    /// aborts the listen, or the connection is reset.
132    ///
133    /// This function can only be called when the current TCP instance
134    /// is in [`Tcp4ConnectionState::LISTEN`] state.
135    pub accept:
136        unsafe extern "efiapi" fn(this: *mut Self, listen_token: *mut Tcp4ListenToken) -> Status,
137
138    /// Queues outgoing data into the transmit queue.
139    ///
140    /// The `transmit` function queues a sending request to this
141    /// instance along with the user data. The status of the token is
142    /// updated and the event in the token will be signaled once the
143    /// data is sent out or some error occurs.
144    pub transmit: unsafe extern "efiapi" fn(this: *mut Self, token: *mut Tcp4IoToken) -> Status,
145
146    /// Places an asynchronous receive request into the receiving
147    /// queue.
148    ///
149    /// `receive` places a completion token into the receive packet
150    /// queue. This function is always asynchronous. The caller must
151    /// allocate the
152    /// [`token.completion_token.event`][Tcp4CompletionToken::event]
153    /// and the `fragment_buffer` used to receive data. They also must
154    /// fill [`data_length`][Tcp4ReceiveData::data_length] which
155    /// represents the whole length of all `fragment_buffer`. When the
156    /// receive operation completes, the driver updates the
157    /// [`token.completion_token.status`][Tcp4CompletionToken::status]
158    /// and [`token.packet.rx_data`][Tcp4Packet::rx_data] fields and
159    /// and the `token.completion_token.event` is signaled. If data
160    /// was received, the data and its length will be copied into the
161    /// `fragment_table`. At the same time, the full length of
162    /// received data will be recorded in the `data_length`
163    /// fields. Providing a proper notification function and context
164    /// for the event will enable the user to receive the notification
165    /// and receiving status. That notification function is guaranteed
166    /// to not be re-entered.
167    pub receive: unsafe extern "efiapi" fn(this: *mut Self, token: *mut Tcp4IoToken) -> Status,
168
169    /// Disconnect a TCP connection gracefully or reset a TCP
170    /// connection. This function is a nonblocking operation.
171    ///
172    /// Initiate an asynchronous close token to TCP driver. After
173    /// `close` is called, any buffered transmission data will be sent
174    /// by TCP driver and the current instance will have a graceful
175    /// close working flow described as RFC 793 if `abort_on_close` is
176    /// set to `false`, otherwise, a reset packet will be sent by the
177    /// TCP driver to quickly disconnect this connection. When the
178    /// close operation completes successfully the TCP instance is in
179    /// [`Tcp4ConnectionState::CLOSED`] state, all pending
180    /// asynchronous operations are signaled, and any buffers used for
181    /// TCP network traffic are flushed.
182    pub close:
183        unsafe extern "efiapi" fn(this: *mut Self, close_token: *mut Tcp4CloseToken) -> Status,
184
185    /// Abort an asynchronous connection, listen, transmission or
186    /// receive request.
187    ///
188    /// The `cancel` function aborts a pending connection, listen,
189    /// transmit or receive request. If `completion_token` is not
190    /// `NULL` and the token is in the connection, listen,
191    /// transmission or receive queue when it is being cancelled, its
192    /// `status` will be set to [`Status::ABORTED`] and then `event`
193    /// will be signaled. If the token is not in one of the queues,
194    /// which usually means that the asynchronous operation has
195    /// completed, [`Status::NOT_FOUND`] is returned. If
196    /// `completion_token` is `NULL`, all asynchronous tokens issued
197    /// by [`Tcp4Protocol::connect`], [`Tcp4Protocol::accept`],
198    /// [`Tcp4Protocol::transmit`] and [`Tcp4Protocol::receive`]
199    /// will be aborted.
200    pub cancel: unsafe extern "efiapi" fn(
201        this: *mut Self,
202        completion_token: *mut Tcp4CompletionToken,
203    ) -> Status,
204
205    /// Poll to receive incoming data and transmit outgoing segments.
206    ///
207    /// The `poll` function increases the rate that data is moved
208    /// between the network and application and can be called when the
209    /// TCP instance is created successfully. Its use is optional.
210    ///
211    /// In some implementations, the periodical timer in the MNP
212    /// driver may not poll the underlying communications device fast
213    /// enough to avoid dropping packets. Drivers and applications
214    /// that are experiencing packet loss should try calling the
215    /// `poll` function at a high frequency.
216    pub poll: unsafe extern "efiapi" fn(this: *mut Self) -> Status,
217}

233        CLOSED = 0,
234        LISTEN = 1,
235        SYN_SENT = 2,
236        SYN_RECEIVED = 3,

383#[derive(Debug)]
384#[repr(C)]
385pub struct Tcp4ReceiveData {
386    /// When `TRUE`, the instance is in urgent mode. The
387    /// implementations of this specification should follow RFC793 to
388    /// process urgent data, and should NOT mix the data across the
389    /// urgent point in one token.
390    pub urgent: Boolean,
391    /// When calling [`receive`][Tcp4Protocol::receive], the caller
392    /// is responsible for setting it to the byte counts of all
393    /// fragment buffers. When the token is signaled by the driver
394    /// it is the length of received data in the fragments.
395    pub data_length: u32,
396    /// Number of fragments in the following fragment table.
397    pub fragment_count: u32,
398    /// Variable-length array of fragment descriptors.
399    ///
400    /// NOTE: this is a flexible array member.
401    pub fragment_table: [Tcp4FragmentData; 0],
402}

404#[derive(Debug)]
405#[repr(C)]
406pub struct Tcp4TransmitData {
407    /// If `TRUE`, data must be transmitted promptly, and the PUSH bit
408    /// in the last TCP segment created will be set. If `FALSE`, data
409    /// transmission may be delayed to combine with data from
410    /// subsequent [`transmit`][Tcp4Protocol::transmit] for
411    /// efficiency.
412    pub push: Boolean,
413    /// The data in the fragment table are urgent and urgent point is
414    /// in effect if `TRUE`. Otherwise those data are NOT considered
415    /// urgent.
416    pub urgent: Boolean,
417    /// Total length of data to transmit.
418    pub data_length: u32,
419    /// Number of fragments in the following fragment table.
420    pub fragment_count: u32,
421    /// Variable-length array of fragment descriptors.
422    ///
423    /// NOTE: this is a flexible array member.
424    pub fragment_table: [Tcp4FragmentData; 0],
425}
