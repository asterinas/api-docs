23pub(super) fn init() {
24    #[cfg(target_arch = "x86_64")]
25    ostd::if_tdx_enabled!({
26        // TODO: support virtio-mmio devices on TDX.
27        //
28        // Currently, virtio-mmio devices need to acquire sub-page MMIO regions,
29        // which are not supported by `IoMem::acquire` in the TDX environment.
30    } else {
31        arch::probe_for_device();
32    });
33    #[cfg(not(target_arch = "x86_64"))]
34    arch::probe_for_device();
35}

37/// Validates a potential VirtIO-MMIO device.
38fn validate_mmio_device(io_mem: &IoMem) -> Result<(), MmioValidateError> {
39    // We now check the requirements specified in Virtual I/O Device (VIRTIO) Version 1.3,
40    // Section 4.2.2.2 Driver Requirements: MMIO Device Register Layout.
41
42    // "The driver MUST ignore a device with MagicValue which is not 0x74726976, although it
43    // MAY report an error."
44    if !mmio_check_magic(io_mem) {
45        debug!(
46            "Abort MMIO detection at {:#x} because the magic number does not match",
47            io_mem.paddr()
48        );
49        return Err(MmioValidateError::MagicMismatch);
50    }
51
52    // TODO: "The driver MUST ignore a device with Version which is not 0x2, although it MAY
53    // report an error."
54
55    // "The driver MUST ignore a device with DeviceID 0x0, but MUST NOT report any error."
56    match mmio_read_device_id(io_mem) {
57        Err(_) | Ok(0) => Err(MmioValidateError::NoDevice),
58        Ok(_) => Ok(()),
59    }
60}
61
62#[derive(Clone, Copy, Debug)]
63enum MmioValidateError {
64    /// Not a VirtIO-MMIO slot.
65    MagicMismatch,
66    /// No device present.
67    NoDevice,
68}
