29/// IRQ line for i8042 keyboard.

32/// Registered device instance for event submission.
33static REGISTERED_DEVICE: Once<RegisteredInputDevice> = Once::new();

35/// ISA interrupt number for i8042 keyboard.
36const ISA_INTR_NUM: u8 = 1;
37
38/// Translation mode for keyboard scancodes.
39static TRANSLATION_MODE: AtomicU8 = AtomicU8::new(TranslationMode::Hardware as u8);
40
41pub(super) fn init(
42    controller: &mut I8042Controller,
43    translation_mode: TranslationMode,
44) -> Result<(), I8042ControllerError> {
45    // Reference: <https://elixir.bootlin.com/linux/v6.17.9/source/drivers/input/serio/libps2.c#L184>
46    const DEVICE_ID_REGULAR_KEYBOARD: u8 = 0xAB;
47
48    TRANSLATION_MODE.store(translation_mode as u8, Ordering::Relaxed);
49
50    let mut init_ctx = InitCtx(controller);
51
52    if ATKBD_RESET.load(Ordering::Relaxed) {
53        // Linux treats a failed keyboard reset as non-fatal. It just warns and continues.
54        // Reference: <https://elixir.bootlin.com/linux/v7.0/source/drivers/input/keyboard/atkbd.c#L820>
55        if let Err(err) = init_ctx.reset() {
56            ostd::warn!("PS/2 keyboard reset failed: {:?}", err);
57        }
58    }
59
60    // If the get-ID command fails, Linux falls back to the set-LEDs command as a probe. If that
61    // succeeds, the device is still treated as a valid keyboard. We take a simpler approach:
62    // just tolerate the failure and continue.
63    // Reference: <https://elixir.bootlin.com/linux/v7.0/source/drivers/input/keyboard/atkbd.c#L838>
64    match init_ctx.get_device_id() {
65        Ok((device_id, _)) => {
66            ostd::info!("PS/2 keyboard device ID: 0x{:02X}", device_id);
67            if device_id != DEVICE_ID_REGULAR_KEYBOARD {
68                // TODO: Support other kinds of keyboards.
69                return Err(I8042ControllerError::DeviceUnknown);
70            }
71        }
72        Err(err) => {
73            ostd::warn!("PS/2 keyboard device ID query failed: {:?}", err);
74        }
75    }
76
77    let mut irq_line = IrqLine::alloc()
78        .and_then(|irq_line| {
79            IRQ_CHIP
80                .get()
81                .unwrap()
82                .map_isa_pin_to(irq_line, ISA_INTR_NUM)
83        })
84        .map_err(|_| I8042ControllerError::DeviceAllocIrqFailed)?;
85    irq_line.on_active(handle_keyboard_input);
86    IRQ_LINE.call_once(|| irq_line);
87
88    // Create and register the i8042 keyboard device.
89    let keyboard_device = Arc::new(I8042Keyboard::new());
90    let registered_device = aster_input::register_device(keyboard_device);
91    REGISTERED_DEVICE.call_once(|| registered_device);
92
93    Ok(())
94}

96/// How Scan Code Set 2 scancodes are translated to Set 1.
97#[derive(Clone, Copy, Debug, PartialEq, Eq)]
98#[repr(u8)]
99pub(super) enum TranslationMode {
100    /// The i8042 controller translates Set 2 to Set 1 in hardware.
101    Hardware = 0,
102    /// No hardware translation; the driver translates Set 2 to Set 1 in software.
103    Software = 1,
104}
105
106impl TranslationMode {
107    fn from_u8(val: u8) -> Self {
108        match val {
109            0 => Self::Hardware,
110            _ => Self::Software,
111        }
112    }
113}

115struct InitCtx<'a>(&'a mut I8042Controller);
116
117impl InitCtx<'_> {
118    fn get_device_id(&mut self) -> Result<(u8, u8), I8042ControllerError> {
119        let mut buf = [0u8; cmd::GetDeviceId::RES_LEN];
120        self.command::<cmd::GetDeviceId>(&[], &mut buf)?;
121        Ok((buf[0], buf[1]))
122    }
123}
124
125impl CommandCtx for InitCtx<'_> {
126    fn controller(&mut self) -> &mut I8042Controller {
127        self.0
128    }
129
130    fn write_to_port(&mut self, data: u8) -> Result<(), I8042ControllerError> {
131        self.0.wait_and_send_data(data)
132    }
133}

143#[derive(Debug)]
144struct I8042Keyboard {
145    name: String,
146    phys: String,
147    uniq: String,
148    id: InputId,
149    capability: InputCapability,
150}

152impl I8042Keyboard {
153    fn new() -> Self {
154        let mut capability = InputCapability::new();
155
156        capability.set_supported_event_type(aster_input::event_type_codes::EventTypes::KEY);
157        capability.set_supported_event_type(aster_input::event_type_codes::EventTypes::SYN);
158
159        // Add all standard keyboard keys.
160        capability.add_standard_keyboard_keys();
161
162        Self {
163            // Standard name for i8042 PS/2 keyboard devices.
164            name: "i8042_keyboard".to_string(),
165
166            // Physical path describing the device's connection topology
167            // isa0060: ISA bus port 0x60 (i8042 data port)
168            // serio0: Serial I/O device 0 (PS/2 is a serial protocol)
169            // input0: Input device 0 (first input device on this controller)
170            phys: "isa0060/serio0/input0".to_string(),
171
172            // Unique identifier - empty string because traditional i8042 keyboards
173            // don't have unique hardware identifiers like serial numbers or MAC addresses
174            uniq: "".to_string(),
175
176            // Device ID with standard values for i8042 keyboards
177            // BUS_I8042 (0x11): PS/2 interface bus type
178            // vendor (0x0001): Generic vendor ID for standard keyboards
179            // product (0x0001): Generic product ID for standard keyboards
180            // version (0x0001): Version 1.0 - standard PS/2 keyboard protocol
181            id: InputId::new(InputId::BUS_I8042, 0x0001, 0x0001, 0x0001),
182
183            capability,
184        }
185    }
186}

210fn handle_keyboard_input(_trap_frame: &TrapFrame) {
211    let Some(scancode_event) = ScancodeInfo::read() else {
212        return;
213    };
214
215    // Dispatch the input event.
216    if let Some(key_code) = scancode_event.to_key_code() {
217        if let Some(registered_device) = REGISTERED_DEVICE.get() {
218            let events = [
219                InputEvent::from_key_and_status(key_code, scancode_event.key_status),
220                InputEvent::from_sync_event(SynEvent::Report),
221            ];
222            registered_device.submit_events(&events);
223        }
224    } else {
225        ostd::debug!(
226            "PS/2 keyboard unmapped scancode {:?} dropped",
227            scancode_event.scancode
228        );
229    }
230}

232#[derive(Clone, Copy, Debug)]
233struct ScanCode(u8);
234
235/// Scancode value and relevant modifier states.
236#[derive(Clone, Debug)]
237struct ScancodeInfo {
238    scancode: ScanCode,
239    key_status: KeyStatus,
240    extended: bool,
241}
242
243impl ScancodeInfo {
244    /// Reads the keyboard scancode from the i8042 controller.
245    fn read() -> Option<Self> {
246        let Some(data) = I8042_CONTROLLER.get()?.lock().receive_data() else {
247            ostd::warn!("PS/2 keyboard has no input data");
248            return None;
249        };
250
251        let mode = TranslationMode::from_u8(TRANSLATION_MODE.load(Ordering::Relaxed));
252        match mode {
253            TranslationMode::Hardware => Self::read_set1(data),
254            TranslationMode::Software => Self::read_set2(data),
255        }
256    }
257
258    /// Interprets a byte as a Scan Code Set 1 scancode (hardware translation enabled).
259    fn read_set1(data: u8) -> Option<Self> {
260        static EXTENDED_KEY: AtomicBool = AtomicBool::new(false);
261
262        let code = ScanCodeSet1(data);
263
264        if code.has_error() {
265            ostd::warn!("PS/2 keyboard key detection error or internal buffer overrun");
266            return None;
267        }
268
269        // Handle the extension code.
270        if code.is_extension() {
271            EXTENDED_KEY.store(true, Ordering::Relaxed);
272            return None;
273        }
274
275        let key_status = code.key_status();
276
277        let extended = EXTENDED_KEY.load(Ordering::Relaxed);
278        EXTENDED_KEY.store(false, Ordering::Relaxed);
279
280        Some(Self {
281            scancode: ScanCode(code.key()),
282            key_status,
283            extended,
284        })
285    }
286
287    /// Interprets a byte as a Scan Code Set 2 scancode (no hardware translation).
288    fn read_set2(data: u8) -> Option<Self> {
289        static EXTENDED_KEY: AtomicBool = AtomicBool::new(false);
290        static RELEASE_NEXT: AtomicBool = AtomicBool::new(false);
291
292        let code = ScanCodeSet2(data);
293
294        if code.has_error() {
295            ostd::warn!("PS/2 keyboard key detection error or internal buffer overrun");
296            return None;
297        }
298
299        // Handle the extension code.
300        if code.is_extension() {
301            EXTENDED_KEY.store(true, Ordering::Relaxed);
302            return None;
303        }
304
305        // Handle the release prefix.
306        if code.is_release_prefix() {
307            RELEASE_NEXT.store(true, Ordering::Relaxed);
308            return None;
309        }
310
311        let releasing = RELEASE_NEXT.load(Ordering::Relaxed);
312        RELEASE_NEXT.store(false, Ordering::Relaxed);
313        let extended = EXTENDED_KEY.load(Ordering::Relaxed);
314        EXTENDED_KEY.store(false, Ordering::Relaxed);
315
316        let set1_code = SET2_TO_SET1[code.raw() as usize];
317        if set1_code == 0xFF {
318            return None;
319        }
320
321        let key_status = if releasing {
322            KeyStatus::Released
323        } else {
324            KeyStatus::Pressed
325        };
326
327        Some(Self {
328            scancode: ScanCode(set1_code),
329            key_status,
330            extended,
331        })
332    }
333
334    /// Maps the keyboard scancode to a [`KeyCode`] in the input subsystem.
335    fn to_key_code(&self) -> Option<KeyCode> {
336        let ScanCode(code) = self.scancode;
337
338        // Handle extended keys.
339        if self.extended {
340            return match code {
341                0x47 => Some(KeyCode::Home),
342                0x48 => Some(KeyCode::Up),
343                0x49 => Some(KeyCode::PageUp),
344                0x4B => Some(KeyCode::Left),
345                0x4D => Some(KeyCode::Right),
346                0x4F => Some(KeyCode::End),
347                0x50 => Some(KeyCode::Down),
348                0x51 => Some(KeyCode::PageDown),
349                0x52 => Some(KeyCode::Insert),
350                0x53 => Some(KeyCode::Delete),
351                _ => None,
352            };
353        }
354
355        // Standard key mapping.
356        Some(match code {
357            // Letters - handle shift/caps lock
358            0x1E => KeyCode::A,
359            0x30 => KeyCode::B,
360            0x2E => KeyCode::C,
361            0x20 => KeyCode::D,
362            0x12 => KeyCode::E,
363            0x21 => KeyCode::F,
364            0x22 => KeyCode::G,
365            0x23 => KeyCode::H,
366            0x17 => KeyCode::I,
367            0x24 => KeyCode::J,
368            0x25 => KeyCode::K,
369            0x26 => KeyCode::L,
370            0x32 => KeyCode::M,
371            0x31 => KeyCode::N,
372            0x18 => KeyCode::O,
373            0x19 => KeyCode::P,
374            0x10 => KeyCode::Q,
375            0x13 => KeyCode::R,
376            0x1F => KeyCode::S,
377            0x14 => KeyCode::T,
378            0x16 => KeyCode::U,
379            0x2F => KeyCode::V,
380            0x11 => KeyCode::W,
381            0x2D => KeyCode::X,
382            0x15 => KeyCode::Y,
383            0x2C => KeyCode::Z,
384
385            // Digits
386            0x02 => KeyCode::Num1,
387            0x03 => KeyCode::Num2,
388            0x04 => KeyCode::Num3,
389            0x05 => KeyCode::Num4,
390            0x06 => KeyCode::Num5,
391            0x07 => KeyCode::Num6,
392            0x08 => KeyCode::Num7,
393            0x09 => KeyCode::Num8,
394            0x0A => KeyCode::Num9,

398            0x39 => KeyCode::Space,
399            0x0F => KeyCode::Tab,
400            0x1C => KeyCode::Enter,
401            0x0E => KeyCode::Backspace,
402
403            // Punctuation
404            0x0C => KeyCode::Minus,
405            0x0D => KeyCode::Equal,
406            0x29 => KeyCode::Grave,
407            0x2B => KeyCode::Backslash,
408            0x33 => KeyCode::Comma,
409            0x34 => KeyCode::Dot,
410            0x35 => KeyCode::Slash,
411            0x27 => KeyCode::Semicolon,
412            0x28 => KeyCode::Apostrophe,
413            0x1A => KeyCode::LeftBrace,
414            0x1B => KeyCode::RightBrace,
415
416            // Modifier keys
417            0x1D => KeyCode::LeftCtrl,
418            0x2A => KeyCode::LeftShift,
419            0x38 => KeyCode::LeftAlt,
420            0x3A => KeyCode::CapsLock,
421
422            // Function keys
423            0x3B => KeyCode::F1,
424            0x3C => KeyCode::F2,
425            0x3D => KeyCode::F3,
426            0x3E => KeyCode::F4,
427            0x3F => KeyCode::F5,
428            0x40 => KeyCode::F6,
429            0x41 => KeyCode::F7,
430            0x42 => KeyCode::F8,
431            0x43 => KeyCode::F9,
432            0x44 => KeyCode::F10,
433            0x57 => KeyCode::F11,
434            0x58 => KeyCode::F12,
435
436            // Escape
437            0x01 => KeyCode::Esc,
438
439            // Unhandled mappings -> None
440            _ => return None,
441        })
442    }

445/// A raw scan code byte in Scan Code Set 1.
446///
447/// Reference: <https://wiki.osdev.org/PS/2_Keyboard#Scan_Code_Set_1>.
448#[derive(Clone, Copy, Debug)]
449struct ScanCodeSet1(u8);
450
451impl ScanCodeSet1 {
452    const CODE_ERROR: u8 = 0xFF;
453    const CODE_EXT_PREFIX: u8 = 0xE0;
454    const RELEASE_MASK: u8 = 0x80;
455
456    fn has_error(&self) -> bool {
457        // Key detection error or internal buffer overrun.
458        self.0 == Self::CODE_ERROR
459    }
460
461    fn key_status(&self) -> KeyStatus {
462        if self.0 & Self::RELEASE_MASK == 0 {
463            KeyStatus::Pressed
464        } else {
465            KeyStatus::Released
466        }
467    }
468
469    fn is_extension(&self) -> bool {
470        self.0 == Self::CODE_EXT_PREFIX
471    }
472
473    fn key(&self) -> u8 {
474        self.0 & !Self::RELEASE_MASK
475    }
476}

487    const CODE_RELEASE_PREFIX: u8 = 0xF0;
488
489    fn has_error(&self) -> bool {
490        // Key detection error or internal buffer overrun.
491        self.0 == Self::CODE_ERROR
492    }
493

496    }

504    }
505}
