39/// A memory mapping for a range of virtual addresses in a [`Vmar`].
40///
41/// A `VmMapping` can be bound with a [`Vmo`] which can provide physical pages
42/// for the mapping. Such mappings are called VMO-backed mappings. Generally, a
43/// VMO-backed mapping is a file-backed mapping. There are also some exceptions
44/// where the pages that need to be mapped are not in a file, but the mappings
45/// are still considered as VMO-backed, e.g.,
46///  - Mappings to the vDSO data.
47///  - Shared anonymous mappings, where the mapped pages need to be retained and
48///    shared with other processes.
49///
50/// Otherwise, the mapping may not be backed by a VMO, in which case it will be
51/// private and anonymous. Such mappings will map to newly allocated, zeroed
52/// physical pages.
53///
54/// This type controls the actual mapping in the [`VmSpace`]. It is a linear
55/// type and cannot be [`Drop`]. To remove a mapping, use [`Self::unmap`].
56///
57/// [`Vmar`]: crate::vm::vmar::Vmar
58pub(crate) struct VmMapping {
59    /// The size of mapping, in bytes. The map size can even be larger than the
60    /// size of VMO. Those pages outside VMO range cannot be read or write.
61    ///
62    /// Zero sized mapping is not allowed. So this field is always non-zero.
63    map_size: NonZeroUsize,
64    /// The base address relative to the VMAR where the VMO is mapped.
65    map_to_addr: Vaddr,
66    /// The mapped memory object. This field specifies what type of memory is
67    /// mapped and provides access to the underlying memory object (VMO, device
68    /// memory, or anonymous memory).
69    ///
70    /// The start of the virtual address maps to the start of the range
71    /// specified in the mapped object.
72    mapped_mem: MappedMemory,
73    /// The file that backs the mapping.
74    ///
75    /// If the mapping is VMO-backed, the `mapped_mem` field should be the page
76    /// cache of the inode referenced by the file's path.
77    file: Option<Arc<dyn FileLike>>,
78    /// Whether the mapping is shared.
79    ///
80    /// The updates to a shared mapping are visible among processes, or carried
81    /// through to the underlying file for file-backed shared mappings.
82    is_shared: bool,
83    /// Whether the mapping needs to handle surrounding pages when handling
84    /// page fault.
85    handle_page_faults_around: bool,
86    /// The permissions of pages in the mapping.
87    ///
88    /// All pages within the same `VmMapping` have the same permissions.
89    perms: VmPerms,
90}
91
92impl Debug for VmMapping {
93    fn fmt(&self, f: &mut Formatter<'_>) -> FmtResult {
94        f.debug_struct("VmMapping")
95            .field("map_size", &self.map_size)
96            .field("map_to_addr", &self.map_to_addr)
97            .field("mapped_mem", &self.mapped_mem)
98            .field("file", &self.file.as_ref().map(|file| file.path()))
99            .field("is_shared", &self.is_shared)
100            .field("handle_page_faults_around", &self.handle_page_faults_around)
101            .field("perms", &self.perms)
102            .finish()
103    }
104}

108        self.map_to_addr..self.map_to_addr + self.map_size.get()
109    }

223    /// Returns the shared futex backing identity for the address if available.
224    pub(crate) fn futex_backing(&self, addr: Vaddr) -> Result<Option<(Weak<Vmo>, usize)>> {
225        if !self.is_shared {
226            return Ok(None);
227        }
228
229        let mapped_vmo = match &self.mapped_mem {
230            MappedMemory::Vmo(mapped_vmo) => mapped_vmo,
231            MappedMemory::Anonymous => return Ok(None),
232            MappedMemory::Device => {
233                return_errno_with_message!(
234                    Errno::EFAULT,
235                    "shared futexes on device mappings are not supported"
236                );
237            }
238        };
239
240        let offset = mapped_vmo.offset() + (addr - self.map_to_addr);
241
242        Ok(Some((Arc::downgrade(mapped_vmo.vmo()), offset)))
243    }

253    /// Populates device memory for this mapping.
254    ///
255    /// This method should only be called for device memory mappings. It maps
256    /// the provided I/O memory region into the virtual address space.
257    ///
258    /// # Panics
259    ///
260    /// In debug builds, this method panics if the mapping is not a device
261    /// memory mapping.
262    pub(super) fn populate_device(&self, vm_space: &VmSpace, io_mem: IoMem, vmo_offset: usize) {
263        debug_assert!(matches!(self.mapped_mem, MappedMemory::Device));
264
265        let preempt_guard = disable_preempt();
266        let map_range = self.map_to_addr..self.map_to_addr + self.map_size.get();
267        let mut cursor = vm_space.cursor_mut(&preempt_guard, &map_range).unwrap();
268        let io_page_prop =
269            PageProperty::new_user(PageFlags::from(self.perms), io_mem.cache_policy());
270        cursor.map_iomem(io_mem, io_page_prop, self.map_size.get(), vmo_offset);
271    }

273    /// Prints the mapping information in the format of `/proc/[pid]/maps`.
274    ///
275    /// Reference: <https://elixir.bootlin.com/linux/v6.16.5/source/fs/proc/task_mmu.c#L304-L359>
276    pub(crate) fn print_to_maps(
277        &self,
278        printer: &mut VmPrinter,
279        parent_vmar: &Vmar,
280        parent_heap_guard: &LockedHeap,
281        path_resolver: &PathResolver,
282    ) -> Result<()> {
283        let start = self.map_to_addr;
284        let end = self.map_end();
285        let read_char = if self.perms.contains(VmPerms::READ) {
286            'r'
287        } else {
288            '-'
289        };
290        let write_char = if self.perms.contains(VmPerms::WRITE) {
291            'w'
292        } else {
293            '-'
294        };
295        let exec_char = if self.perms.contains(VmPerms::EXEC) {
296            'x'
297        } else {
298            '-'
299        };
300        let shared_char = if self.is_shared { 's' } else { 'p' };
301        let offset = self.vmo().map(|vmo| vmo.offset).unwrap_or(0);
302        let (dev_major, dev_minor) = self
303            .inode()
304            .and_then(|inode| {
305                let metadata = inode.metadata().ok()?;
306                Some(device_id::decode_device_numbers(
307                    metadata.container_dev_id.as_encoded_u64(),
308                ))
309            })
310            .unwrap_or((0, 0));
311        let ino = self.inode().map(|inode| inode.ino()).unwrap_or(0);
312
313        let line = format!(
314            "{:x}-{:x} {}{}{}{} {:08x} {:02x}:{:02x} {} ",
315            start,
316            end,
317            read_char,
318            write_char,
319            exec_char,
320            shared_char,
321            offset,
322            dev_major,
323            dev_minor,
324            ino,
325        );
326
327        let name = || {
328            let process_vm = parent_vmar.process_vm();
329            let user_stack_top = process_vm.init_stack().user_stack_top();
330            if self.map_to_addr <= user_stack_top && self.map_end() > user_stack_top {
331                return Some(Cow::Borrowed("[stack]"));
332            }
333
334            let heap_range = parent_heap_guard.heap_range();
335            if self.map_to_addr >= heap_range.start && self.map_end() <= heap_range.end {
336                return Some(Cow::Borrowed("[heap]"));
337            }
338
339            #[cfg(any(target_arch = "x86_64", target_arch = "riscv64"))]
340            if let Some(vmo) = self.vmo() {
341                use crate::vdso::{VDSO_VMO_LAYOUT, vdso_vmo};
342
343                if let Some(vdso_vmo) = vdso_vmo()
344                    && Arc::ptr_eq(vmo.vmo(), &vdso_vmo)
345                {
346                    let offset = vmo.offset();
347                    if offset == VDSO_VMO_LAYOUT.data_segment_offset {
348                        return Some(Cow::Borrowed("[vvar]"));
349                    } else if offset == VDSO_VMO_LAYOUT.text_segment_offset {
350                        return Some(Cow::Borrowed("[vdso]"));
351                    }
352                }
353            }
354
355            if let Some(file) = &self.file {
356                return Some(Cow::Owned(
357                    path_resolver.make_abs_path(file.path()).into_string(),
358                ));
359            }
360
361            // Reference: <https://github.com/google/gvisor/blob/38123b53da96ff6983fcc103dfe2a9cc4e0d80c8/test/syscalls/linux/proc.cc#L1158-L1172>
362            if matches!(&self.mapped_mem, MappedMemory::Vmo(_)) && self.is_shared {
363                return Some(Cow::Borrowed("/dev/zero (deleted)"));
364            }
365
366            // Common anonymous mappings do not have names.
367            None
368        };
369
370        let name = name();
371
372        if let Some(name) = name {
373            writeln!(printer, "{:<72} {}", line, name)?;
374        } else {
375            writeln!(printer, "{}", line)?;
376        }
377
378        Ok(())
379    }

391impl VmMapping {
392    /// Handles a page fault.
393    pub(super) fn handle_page_fault(
394        &self,
395        vm_space: &VmSpace,
396        page_fault_info: &PageFaultInfo,
397        rss_delta: &mut RssDelta,
398    ) -> Result<()> {
399        self.check_perms_for_page_fault(page_fault_info)?;
400
401        let page_aligned_addr = page_fault_info.address.align_down(PAGE_SIZE);
402        let is_write = page_fault_info.required_perms.contains(VmPerms::WRITE);
403
404        if !is_write
405            && matches!(&self.mapped_mem, MappedMemory::Vmo(_))
406            && self.handle_page_faults_around
407        {
408            let res = self.handle_page_faults_around(
409                vm_space,
410                page_aligned_addr,
411                page_fault_info.required_perms,
412                rss_delta,
413            );
414
415            // Errors caused by the "around" pages should be ignored, so here we
416            // only return the error if the faulting page is still not mapped.
417            if res.is_err() {
418                let preempt_guard = disable_preempt();
419                let mut cursor = vm_space.cursor(
420                    &preempt_guard,
421                    &(page_aligned_addr..page_aligned_addr + PAGE_SIZE),
422                )?;
423                if let (_, Some(_)) = cursor.query().unwrap() {
424                    return Ok(());
425                }
426            }
427
428            return res;
429        }
430
431        self.handle_single_page_fault(
432            vm_space,
433            page_aligned_addr,
434            page_fault_info.required_perms,
435            rss_delta,
436        )
437    }
438
439    fn check_perms_for_page_fault(&self, page_fault_info: &PageFaultInfo) -> Result<()> {
440        debug!(
441            "self.perms {:?}, page_fault_info.required_perms {:?}, self.range {:?}",
442            self.perms,
443            page_fault_info.required_perms,
444            self.range()
445        );
446
447        let mut perms = self.perms;
448
449        // Reference: <https://elixir.bootlin.com/linux/v6.16.5/source/mm/gup.c#L1282-L1311>
450        if page_fault_info.is_forced() {
451            if perms.contains(VmPerms::MAY_READ) {
452                perms.insert(VmPerms::READ);
453            }
454            if self.is_cow() {
455                perms.insert(VmPerms::WRITE);
456            }
457        }
458
459        if !perms.contains(page_fault_info.required_perms) {
460            return_errno_with_message!(
461                Errno::EACCES,
462                "the mapping does not have the required permissions"
463            );
464        }
465
466        Ok(())
467    }
468
469    fn handle_single_page_fault(
470        &self,
471        vm_space: &VmSpace,
472        page_aligned_addr: Vaddr,
473        required_perms: VmPerms,
474        rss_delta: &mut RssDelta,
475    ) -> Result<()> {
476        'retry: loop {
477            let preempt_guard = disable_preempt();
478            let mut cursor = vm_space.cursor_mut(
479                &preempt_guard,
480                &(page_aligned_addr..page_aligned_addr + PAGE_SIZE),
481            )?;
482
483            let (va, item) = cursor.query().unwrap();
484            let is_write = required_perms.contains(VmPerms::WRITE);
485            match item {
486                Some(VmQueriedItem::MappedRam { frame, mut prop }) => {
487                    if VmPerms::from(prop.flags).contains(required_perms) {
488                        // The page fault is already handled maybe by other threads.
489                        // Just flush the TLB and return.
490                        TlbFlushOp::for_range(va).perform_on_current();
491                        return Ok(());
492                    }
493                    assert!(is_write);
494
495                    // For shared mappings, ensure that the VMO allows the page to be written.
496                    if self.is_shared
497                        && let MappedMemory::Vmo(vmo) = &self.mapped_mem
498                        && let Err(err) = vmo.get_committed_frame(
499                            page_aligned_addr - self.map_to_addr,
500                            VmoMapMode::SharedWrite,
501                        )
502                    {
503                        let index = err.pending_index()?;
504                        drop(cursor);
505                        drop(preempt_guard);
506                        vmo.commit_on(index, VmoMapMode::SharedWrite)?;
507                        continue 'retry;
508                    }
509
510                    // If the forked child or parent immediately unmaps the page after
511                    // the fork without accessing it, we are the only reference to the
512                    // frame. We can directly map the frame as writable without copying.
513                    let only_reference = frame.reference_count() == 1;
514
515                    let new_flags = PageFlags::W | PageFlags::ACCESSED | PageFlags::DIRTY;
516
517                    if self.is_shared || only_reference {
518                        cursor.protect_next(PAGE_SIZE, |flags, _cache| {
519                            *flags |= new_flags;
520                        });
521                        cursor.flusher().issue_tlb_flush(TlbFlushOp::for_range(va));
522                        cursor.flusher().dispatch_tlb_flush();
523                    } else {
524                        // Perform COW if it is a write access to a private mapping.
525                        let new_frame = duplicate_frame(&frame)?;
526                        prop.flags |= new_flags;
527                        cursor.unmap(PAGE_SIZE);
528                        cursor.jump(va.start).unwrap();
529                        cursor.map(new_frame.into(), prop);
530                        // FIXME: Linux re-classifies the page from `File` to `Anon` in RSS,
531                        // when a COW on a file-backed mapping happens.
532                        // We currently do not support this re-classification,
533                        // since it will introduce some complexity when unmapping this page.
534                    }
535                    cursor.flusher().sync_tlb_flush();
536                }
537                Some(VmQueriedItem::MappedIoMem { .. }) => {
538                    // The page of I/O memory is populated when the memory
539                    // mapping is created.
540                    return_errno_with_message!(
541                        Errno::EFAULT,
542                        "device memory page faults cannot be resolved"
543                    );
544                }
545                None => {
546                    // Map a new frame to the page fault address.
547                    let (frame, is_readonly) = match self.prepare_page(page_aligned_addr, is_write)
548                    {
549                        Ok((frame, is_readonly)) => (frame, is_readonly),
550                        Err(err) => {
551                            let index = err.pending_index()?;
552                            drop(cursor);
553                            drop(preempt_guard);
554                            self.vmo()
555                                .unwrap()
556                                .commit_on(index, self.vmo_map_mode_from_is_write(is_write))?;
557                            continue 'retry;
558                        }
559                    };
560
561                    let vm_perms = {
562                        let mut perms = self.perms;
563                        if is_readonly {
564                            // COW pages are forced to be read-only.
565                            perms -= VmPerms::WRITE;
566                        }
567                        perms
568                    };
569
570                    let mut page_flags = vm_perms.into();
571                    page_flags |= PageFlags::ACCESSED;
572                    if is_write {
573                        page_flags |= PageFlags::DIRTY;
574                    }
575                    let map_prop = PageProperty::new_user(page_flags, CachePolicy::Writeback);
576
577                    cursor.map(frame, map_prop);
578                    rss_delta.add(self.rss_type(), 1);
579                }
580            }
581            break 'retry;
582        }
583
584        Ok(())
585    }
586
587    fn prepare_page(
588        &self,
589        page_aligned_addr: Vaddr,
590        is_write: bool,
591    ) -> Result<(UFrame, bool), VmoCommitError> {
592        let mut is_readonly = false;
593
594        let vmo = match &self.mapped_mem {
595            MappedMemory::Vmo(vmo) => vmo,
596            MappedMemory::Anonymous => {
597                // Anonymous mapping. Allocate a new frame.
598                return Ok((FrameAllocOptions::new().alloc_frame()?.into(), is_readonly));
599            }
600            MappedMemory::Device => {
601                // Device memory is populated when the memory mapping is created.
602                return Err(VmoCommitError::Err(Error::with_message(
603                    Errno::EFAULT,
604                    "device memory page faults cannot be resolved",
605                )));
606            }
607        };
608
609        let page_offset = page_aligned_addr - self.map_to_addr;
610        if !self.is_shared && page_offset >= vmo.valid_size() {
611            // The page index is outside the VMO. This is only allowed in private mapping.
612            return Ok((FrameAllocOptions::new().alloc_frame()?.into(), is_readonly));
613        }
614
615        let (page, mode) =
616            vmo.get_committed_frame(page_offset, self.vmo_map_mode_from_is_write(is_write))?;
617        if !self.is_shared && is_write {
618            // Write access to private VMO-backed mapping. Performs COW directly.
619            Ok((duplicate_frame(&page.into())?.into(), is_readonly))
620        } else {
621            // Operations to shared mapping or read access to private VMO-backed mapping.
622            // If read access to private VMO-backed mapping triggers a page fault,
623            // the map should be readonly. If user next tries to write to the frame,
624            // another page fault will be triggered which will performs a COW (Copy-On-Write).
625            is_readonly = !self.is_shared || mode != VmoMapMode::SharedWrite;
626            Ok((page.into(), is_readonly))
627        }
628    }
629
630    fn vmo_map_mode_from_is_write(&self, is_write: bool) -> VmoMapMode {
631        if !self.is_shared {
632            VmoMapMode::Private
633        } else if !is_write {
634            VmoMapMode::SharedRead
635        } else {
636            VmoMapMode::SharedWrite
637        }
638    }

640    /// Handles a page fault and maps additional surrounding pages.
641    fn handle_page_faults_around(
642        &self,
643        vm_space: &VmSpace,
644        page_aligned_addr: Vaddr,
645        required_perms: VmPerms,
646        mut rss_delta: &mut RssDelta,
647    ) -> Result<()> {
648        const SURROUNDING_PAGE_NUM: usize = 16;
649        const SURROUNDING_PAGE_ADDR_MASK: usize = !(SURROUNDING_PAGE_NUM * PAGE_SIZE - 1);
650
651        let vmo = self.vmo().unwrap();
652        let around_page_addr = page_aligned_addr & SURROUNDING_PAGE_ADDR_MASK;
653        let size = min(vmo.valid_size(), self.map_size.get());
654        let mut start_addr = max(around_page_addr, self.map_to_addr);
655        let end_addr = min(
656            start_addr + SURROUNDING_PAGE_NUM * PAGE_SIZE,
657            self.map_to_addr + size,
658        );
659
660        // The page fault address falls outside the VMO bounds.
661        // Only a single page fault is handled in this situation.
662        if end_addr <= page_aligned_addr {
663            return self.handle_single_page_fault(
664                vm_space,
665                page_aligned_addr,
666                required_perms,
667                rss_delta,
668            );
669        }
670
671        let vm_perms = self.perms - VmPerms::WRITE;
672        let mode = if !self.is_shared {
673            VmoMapMode::Private
674        } else {
675            VmoMapMode::SharedRead
676        };
677
678        'retry: loop {
679            let preempt_guard = disable_preempt();
680            let mut cursor = vm_space.cursor_mut(&preempt_guard, &(start_addr..end_addr))?;
681
682            let rss_delta_ref = &mut rss_delta;
683            let operate =
684                move |commit_fn: &mut dyn FnMut() -> Result<(usize, CachePage), VmoCommitError>| {
685                    if let (_, None) = cursor.query().unwrap() {
686                        // We regard all the surrounding pages as accessed, no matter
687                        // if it is really so. Then the hardware won't bother to update
688                        // the accessed bit of the page table on following accesses.
689                        let page_flags = PageFlags::from(vm_perms) | PageFlags::ACCESSED;
690                        let page_prop = PageProperty::new_user(page_flags, CachePolicy::Writeback);
691                        let (_, frame) = commit_fn()?;
692                        cursor.map(frame.into(), page_prop);
693                        rss_delta_ref.add(self.rss_type(), 1);
694                    } else {
695                        let next_addr = cursor.virt_addr() + PAGE_SIZE;
696                        if next_addr < end_addr {
697                            let _ = cursor.jump(next_addr);
698                        }
699                    }
700                    Ok(())
701                };
702
703            let start_offset = start_addr - self.map_to_addr;
704            let end_offset = end_addr - self.map_to_addr;
705            match vmo.try_operate_on_range(&(start_offset..end_offset), operate, mode) {
706                Ok(_) => return Ok(()),
707                Err(err) => {
708                    let index = err.pending_index()?;
709                    drop(preempt_guard);
710                    vmo.commit_on(index, mode)?;
711                    start_addr = (index * PAGE_SIZE - vmo.offset()) + self.map_to_addr;
712                    continue 'retry;
713                }
714            }
715        }
716    }
717}

721impl VmMapping {
722    /// Enlarges the mapping by `extra_size` bytes to the high end.
723    pub(crate) fn enlarge(self, extra_size: usize) -> Self {
724        Self {
725            map_size: NonZeroUsize::new(self.map_size.get() + extra_size).unwrap(),
726            ..self
727        }
728    }
729
730    /// Splits the mapping at the specified address.
731    ///
732    /// The address must be within the mapping and page-aligned. The address
733    /// must not be either the start or the end of the mapping.
734    pub(crate) fn split(self, at: Vaddr) -> (Self, Self) {
735        debug_assert!(self.map_to_addr < at && at < self.map_end());
736        debug_assert!(at.is_multiple_of(PAGE_SIZE));
737
738        let left_size = at - self.map_to_addr;
739        let right_size = self.map_size.get() - left_size;
740
741        let r_mapped_mem = self.mapped_mem.dup_at_offset(left_size);
742        let l_mapped_mem = self.mapped_mem;
743
744        let left = Self {
745            map_size: NonZeroUsize::new(left_size).unwrap(),
746            map_to_addr: self.map_to_addr,
747            mapped_mem: l_mapped_mem,
748            file: self.file.clone(),
749            ..self
750        };
751        let right = Self {
752            map_size: NonZeroUsize::new(right_size).unwrap(),
753            map_to_addr: at,
754            mapped_mem: r_mapped_mem,
755            ..self
756        };
757
758        (left, right)
759    }
760
761    /// Splits the mapping at the specified address.
762    ///
763    /// There are four conditions:
764    /// 1. |-outside `range`-| + |------------within `range`------------|
765    /// 2. |------------within `range`------------| + |-outside `range`-|
766    /// 3. |-outside `range`-| + |-within `range`-| + |-outside `range`-|
767    /// 4. |----------------------within `range` -----------------------|
768    ///
769    /// Returns (left outside, within, right outside) if successful.
770    ///
771    /// # Panics
772    ///
773    /// Panics if the mapping does not contain the range, or if the start or
774    /// end of the range is not page-aligned.
775    pub(crate) fn split_range(self, range: &Range<Vaddr>) -> (Option<Self>, Self, Option<Self>) {
776        let mapping_range = self.range();
777        if range.start <= mapping_range.start && mapping_range.end <= range.end {
778            // Condition 4.
779            (None, self, None)
780        } else if mapping_range.start < range.start {
781            let (left, within) = self.split(range.start);
782            if range.end < mapping_range.end {
783                // Condition 3.
784                let (within, right) = within.split(range.end);
785                (Some(left), within, Some(right))
786            } else {
787                // Condition 1.
788                (Some(left), within, None)
789            }
790        } else if mapping_range.contains(&range.end) {
791            // Condition 2.
792            let (within, right) = self.split(range.end);
793            (None, within, Some(right))
794        } else {
795            panic!("The mapping does not contain the splitting range");
796        }
797    }
798
799    /// Attempts to merge `self` with the given `vm_mapping` if they are
800    /// adjacent and compatible.
801    ///
802    /// Two mappings are considered *adjacent* if the end address of `self`
803    /// equals to the start address of `vm_mapping`, or vice versa.
804    ///
805    /// Two mappings are considered *compatible* if all of the following
806    /// conditions are met:
807    /// - They have the same access permissions.
808    /// - They are both anonymous or share the same backing file.
809    /// - Their file offsets are contiguous if file-backed.
810    /// - Other attributes (e.g., shared/private flags, whether need to handle
811    ///   page faults around, etc.) must also match.
812    ///
813    /// This method returns:
814    /// - the merged mapping along with the address of the mapping
815    ///   to be removed if successful.
816    /// - the original `self` and a `None` otherwise.
817    pub(crate) fn try_merge_with(self, vm_mapping: &VmMapping) -> (Self, Option<Vaddr>) {
818        debug_assert!(!is_intersected(&self.range(), &vm_mapping.range()));
819
820        let (left, right) = if self.map_to_addr < vm_mapping.map_to_addr {
821            (&self, vm_mapping)
822        } else {
823            (vm_mapping, &self)
824        };
825
826        if let Some(merged) = try_merge(left, right) {
827            (merged, Some(vm_mapping.map_to_addr))
828        } else {
829            (self, None)
830        }
831    }
832}

834/************************** VM Space operations ******************************/
835
836impl VmMapping {
837    /// Unmaps the mapping from the VM space,
838    /// and returns the number of unmapped pages.
839    pub(super) fn unmap(self, vm_space: &VmSpace) -> usize {
840        let preempt_guard = disable_preempt();
841        let range = self.range();
842        let mut cursor = vm_space.cursor_mut(&preempt_guard, &range).unwrap();
843
844        let num_unmapped = cursor.unmap(range.len());
845        cursor.flusher().dispatch_tlb_flush();
846        cursor.flusher().sync_tlb_flush();
847
848        num_unmapped
849    }
850
851    /// Change the perms of the mapping.
852    pub(super) fn protect(self, vm_space: &VmSpace, perms: VmPerms) -> Self {
853        // We should never convert a page to a writable page directly.
854        //  - For private mappings, we may need to perform Copy-On-Write (COW)
855        //    before doing so.
856        //  - For shared mappings, the page may need to be marked as a dirty
857        //    page in the page cache.
858        let new_flags = PageFlags::from(perms) - PageFlags::W;
859
860        let preempt_guard = disable_preempt();
861        let range = self.range();
862        let mut cursor = vm_space.cursor_mut(&preempt_guard, &range).unwrap();
863
864        let op = |flags: &mut PageFlags, _cache: &mut CachePolicy| *flags = new_flags;
865        while cursor.virt_addr() < range.end {
866            if let Some(va) = cursor.protect_next(range.end - cursor.virt_addr(), op) {
867                cursor.flusher().issue_tlb_flush(TlbFlushOp::for_range(va));
868            } else {
869                break;
870            }
871        }
872        cursor.flusher().dispatch_tlb_flush();
873        cursor.flusher().sync_tlb_flush();
874
875        Self { perms, ..self }
876    }
877}

879/// Memory mapped by a [`VmMapping`].
880#[derive(Debug)]
881pub(super) enum MappedMemory {
882    /// Anonymous memory.
883    ///
884    /// These pages are not associated with any files. On-demand population is possible by enabling
885    /// page fault handlers to allocate zeroed pages.
886    Anonymous,
887    /// Memory in a [`Vmo`].
888    ///
889    /// These pages are associated with regular files that are backed by the page cache. On-demand
890    /// population is possible by enabling page fault handlers to allocate pages and read the page
891    /// content from the disk.
892    Vmo(MappedVmo),
893    /// Device memory.
894    ///
895    /// These pages are associated with special files (typically device memory). They are populated
896    /// when the memory mapping is created via mmap, instead of occurring at page faults.
897    Device,
898}
899
900impl MappedMemory {
901    /// Duplicates the mapped memory capability.
902    pub(super) fn dup(&self) -> Self {
903        match self {
904            MappedMemory::Anonymous => MappedMemory::Anonymous,
905            MappedMemory::Vmo(v) => MappedMemory::Vmo(v.dup()),
906            MappedMemory::Device => MappedMemory::Device,
907        }
908    }
909
910    /// Duplicates the mapped memory capability for a sub-range.
911    ///
912    /// The `offset` is relative to the start of the mapping, and must be
913    /// page-aligned.
914    fn dup_at_offset(&self, offset: usize) -> Self {
915        debug_assert!(offset.is_multiple_of(PAGE_SIZE));
916
917        match self {
918            MappedMemory::Anonymous => MappedMemory::Anonymous,
919            MappedMemory::Vmo(vmo) => {
920                let new_offset = vmo
921                    .offset()
922                    .checked_add(offset)
923                    .expect("mapped VMO offset should not overflow");
924                MappedMemory::Vmo(vmo.dup_at_offset(new_offset))
925            }
926            MappedMemory::Device => MappedMemory::Device,
927        }
928    }
929}

931/// A wrapper that represents a mapped [`Vmo`] and provide required functionalities
932/// that need to be provided to mappings from the VMO.
933#[derive(Debug)]
934pub(super) struct MappedVmo {
935    vmo: Arc<Vmo>,
936    /// Represents the mapped offset in the VMO for the mapping.
937    offset: usize,
938    /// Whether the VMO's writable mappings need to be tracked, and the
939    /// mapping is writable to the VMO.
940    is_writable_tracked: bool,
941}
942
943impl MappedVmo {
944    /// Creates a `MappedVmo` used for the mapping.
945    pub(super) fn new(vmo: Arc<Vmo>, offset: usize, is_writable_tracked: bool) -> Result<Self> {
946        if is_writable_tracked {
947            vmo.writable_mapping_status().map()?;
948        }
949
950        Ok(Self {
951            vmo,
952            offset,
953            is_writable_tracked,
954        })
955    }
956
957    /// Returns the **valid** size of the `MappedVmo`.
958    ///
959    /// The **valid** size of a `MappedVmo` is the size of its accessible range
960    /// that actually falls within the bounds of the underlying VMO.
961    fn valid_size(&self) -> usize {
962        let vmo_size = self.vmo.size();
963        (self.offset..vmo_size).len()
964    }
965
966    /// Gets the committed frame at the input offset in the mapped VMO.
967    ///
968    /// If the VMO has not committed a frame at this index, it will commit
969    /// one first and return it. If the commit operation needs to perform I/O,
970    /// it will return a [`VmoCommitError::NeedIo`].
971    fn get_committed_frame(
972        &self,
973        page_offset: usize,
974        map_mode: VmoMapMode,
975    ) -> Result<(CachePage, VmoMapMode), VmoCommitError> {
976        debug_assert!(page_offset.is_multiple_of(PAGE_SIZE));
977        self.vmo
978            .try_commit_page(self.offset + page_offset, map_mode)
979    }
980
981    /// Commits a page at a specific page index.
982    ///
983    /// This method may involve I/O operations if the VMO needs to fetch
984    /// a page from the underlying page cache.
985    pub(crate) fn commit_on(&self, page_idx: usize, map_mode: VmoMapMode) -> Result<()> {
986        self.vmo.commit_on(page_idx, map_mode)
987    }
988
989    /// Traverses the indices within a specified range of a VMO sequentially.
990    ///
991    /// For each index position, you have the option to commit the page as well as
992    /// perform other operations.
993    ///
994    /// Once a commit operation needs to perform I/O, it will return a
995    /// [`VmoCommitError::NeedIo`].
996    fn try_operate_on_range<F>(
997        &self,
998        range: &Range<usize>,
999        operate: F,
1000        map_mode: VmoMapMode,
1001    ) -> Result<(), VmoCommitError>
1002    where
1003        F: FnMut(
1004            &mut dyn FnMut() -> Result<(usize, CachePage), VmoCommitError>,
1005        ) -> Result<(), VmoCommitError>,
1006    {
1007        let range = self.offset + range.start..self.offset + range.end;
1008        self.vmo.try_operate_on_range(&range, operate, map_mode)
1009    }
1010
1011    /// Gets a reference to the underlying VMO.
1012    pub(crate) fn vmo(&self) -> &Arc<Vmo> {
1013        &self.vmo
1014    }
1015
1016    /// Gets the offset for the mappings.
1017    pub(crate) fn offset(&self) -> usize {
1018        self.offset
1019    }
1020
1021    /// Duplicates the capability.
1022    pub(crate) fn dup(&self) -> Self {
1023        self.dup_at_offset(self.offset)
1024    }
1025
1026    /// Duplicates the capability at a specific offset.
1027    fn dup_at_offset(&self, offset: usize) -> Self {
1028        if self.is_writable_tracked {
1029            self.vmo.writable_mapping_status().increment();
1030        }
1031
1032        Self {
1033            vmo: self.vmo.clone(),
1034            offset,
1035            is_writable_tracked: self.is_writable_tracked,
1036        }
1037    }
1038}

1040impl Drop for MappedVmo {
1041    fn drop(&mut self) {
1042        if self.is_writable_tracked {
1043            self.vmo.writable_mapping_status().decrement();
1044        }
1045    }

1048/// Attempts to merge two [`VmMapping`]s into a single mapping if they are
1049/// adjacent and compatible.
1050///
1051/// - Returns the merged [`VmMapping`] if successful. The caller should
1052///   remove the original mappings before inserting the merged mapping
1053///   into the [`Vmar`].
1054/// - Returns `None` otherwise.
1055///
1056/// [`Vmar`]: crate::vm::vmar::Vmar
1057fn try_merge(left: &VmMapping, right: &VmMapping) -> Option<VmMapping> {
1058    let is_adjacent = left.map_end() == right.map_to_addr();
1059    let is_type_equal = left.is_shared == right.is_shared
1060        && left.handle_page_faults_around == right.handle_page_faults_around
1061        && left.perms == right.perms
1062        && match (&left.file, &right.file) {
1063            (Some(left_file), Some(right_file)) => Arc::ptr_eq(left_file, right_file),
1064            (None, None) => true,
1065            _ => false,
1066        };
1067
1068    if !is_adjacent || !is_type_equal {
1069        return None;
1070    }
1071
1072    let mapped_mem = match (&left.mapped_mem, &right.mapped_mem) {
1073        (MappedMemory::Anonymous, MappedMemory::Anonymous) => {
1074            // Anonymous memory mappings can be merged if they are adjacent
1075            MappedMemory::Anonymous
1076        }
1077        (MappedMemory::Vmo(l_vmo_obj), MappedMemory::Vmo(r_vmo_obj)) => {
1078            let l_vmo = l_vmo_obj.vmo();
1079            let r_vmo = r_vmo_obj.vmo();
1080
1081            if Arc::ptr_eq(l_vmo, r_vmo) {
1082                let is_offset_contiguous =
1083                    l_vmo_obj.offset() + left.map_size() == r_vmo_obj.offset();
1084                if !is_offset_contiguous {
1085                    return None;
1086                }
1087                MappedMemory::Vmo(l_vmo_obj.dup())
1088            } else {
1089                return None;
1090            }
1091        }
1092        // Device memory and other types cannot be merged
1093        _ => return None,
1094    };
1095
1096    let map_size = NonZeroUsize::new(left.map_size() + right.map_size()).unwrap();
1097
1098    Some(VmMapping {
1099        map_size,
1100        mapped_mem,
1101        file: left.file.clone(),
1102        ..*left
1103    })
1104}
