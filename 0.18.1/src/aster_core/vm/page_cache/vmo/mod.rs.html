3//! Lower-level virtual memory object machinery used by the page cache.
4//!
5//! This submodule defines [`Vmo`] and [`VmoOptions`], the generic memory-object
6//! layer underneath [`crate::vm::page_cache::PageCache`]. A [`Vmo`] owns a sparse
7//! array of pages, commits them on demand, and optionally cooperates with a
8//! [`crate::vm::page_cache::PageCacheBackend`] to populate or write back pages with
9//! a backend.
10//!
11//! # When to Use This Module
12//!
13//! - Filesystem buffered I/O, invalidation, and resize should usually stay at
14//!   [`crate::vm::page_cache::PageCache`].
15//! - Memory-mapping, page-fault, and other lower-level VM paths that need
16//!   direct page-oriented access should work with [`Vmo`].

44/// Page-indexed memory object used by the page cache and mapping code.
45///
46/// A `Vmo` represents a contiguous byte range backed by a sparse array of
47/// [`CachePage`]s. Pages are materialized on demand: anonymous VMOs allocate
48/// zero-filled RAM pages, while VMOs with a backend consult a
49/// [`PageCacheBackend`] to populate or write back individual pages.
50///
51/// `Vmo` is the storage engine underneath [`crate::vm::page_cache::PageCache`]. It
52/// understands page indices, per-page state transitions, and optional backend
53/// I/O, but it deliberately stays below filesystem policy. In particular, it
54/// does not define logical EOF semantics, extent mapping policy, or whole-file
55/// synchronization.
56///
57/// # Backing Modes
58///
59/// - **Anonymous VMO.** RAM-backed and zero-filled on first access. This is
60///   used for anonymous memory and in-memory page caches.
61/// - **Backend VMO.** Connected to a [`PageCacheBackend`]. Pages are loaded
62///   lazily and can later be written back.
63///
64/// # What `Vmo` Owns
65///
66/// - the page-aligned byte span returned by [`Vmo::size`];
67/// - the sparse page array and page-commit path;
68/// - per-page initialization, dirtying, and writeback coordination; and
69/// - writable-mapping tracking for callers that opt into it.
70///
71/// # What Stays Outside `Vmo`
72///
73/// Filesystems and higher layers still own:
74///
75/// - logical file size and inode metadata;
76/// - extent or block mapping decisions embodied by the backend; and
77/// - the higher-level locking that excludes conflicting buffered I/O,
78///   invalidation, truncate, or page-fault operations.
79///
80/// # Size Synchronization
81///
82/// A file that uses [`crate::vm::page_cache::PageCache`] has two related sizes:
83///
84/// - the logical file size stored in filesystem metadata; and
85/// - the page-aligned accessible range returned by [`Vmo::size`].
86///
87/// The logical file size defines EOF for filesystem operations. [`Vmo::size`]
88/// defines how far lower-level VMO operations such as reads, page commits, and
89/// page faults may access, and it may be larger because it is rounded up to
90/// whole pages.
91///
92/// These two sizes must be updated while holding the same higher-level
93/// inode- or file-level lock that serializes resize against buffered I/O, page
94/// faults, and invalidation. The filesystem-facing resize API,
95/// [`crate::vm::page_cache::PageCache::resize`], documents the required ordering
96/// between logical file size updates and [`Vmo::size`] changes.
97///
98/// # Backend Page Lifecycle
99///
100/// For pages with a backend, the state machine is:
101///
102/// - `Uninit -> UpToDate`: the page is populated from the backend, or a hole is
103///   materialized as zeros.
104/// - `Uninit -> Dirty`: a full-page overwrite commits and keeps the page
105///   locked until the writer has copied its new contents.
106/// - `UpToDate -> Dirty`: buffered writes modify the page under the page lock,
107///   or the page is mapped writably to userspace.
108/// - `Dirty -> UpToDate`: the VMO writeback path hands the page to the backend.
109/// - `UpToDate -> Evicted`: the page is evicted and the page lock is held until
110///   the page is removed from the VMO.
111///
112/// The auxiliary `is_writing_back` bit is set under the page lock, then cleared
113/// later by the BIO completion callback after the writeback state has been
114/// handed off. Anonymous VMOs stay `UpToDate` in steady state once a page is
115/// committed.
116pub(crate) struct Vmo {
117    /// The backend that provides disk I/O operations, if any.
118    //
119    // TODO: Using `Weak` here is to avoid circular references in exfat file systems.
120    // We should avoid the circular reference by design, and then we can change this to `Arc`.
121    pub(super) backend: Option<Weak<dyn PageCacheBackend>>,
122    /// Flags
123    pub(super) flags: VmoFlags,
124    /// The virtual pages where the VMO resides.
125    pub(super) pages: XArray<CachePage>,
126    /// The size of the VMO.
127    ///
128    /// This is the page-aligned VMO size, which may differ from the
129    /// filesystem's logical file size. Updates must happen after locking the
130    /// [`XArray`] in `pages`, so a size read after taking that lock observes
131    /// the latest resize performed under the required higher-level file-size
132    /// synchronization.
133    pub(super) size: AtomicUsize,
134    /// The status of writable mappings of the VMO.
135    //
136    // TODO: This field is used only by VMOs belonging to memfd (i.e., `MemfdInode`). But VMOs do
137    // not have the knowledge to determine if they belong to memfd. We may want to enhance
138    // `VmoOptions` to make VMOs aware of whether its writable mappings should be tracked.
139    pub(super) writable_mapping_status: WritableMappingStatus,
140    /// Reserve mappings.
141    pub(super) rmap: Mutex<Rmap>,
142}

172/// The error type used for commit operations of [`Vmo`].
173#[derive(Debug)]
174pub(crate) enum VmoCommitError {
175    /// A general error occurred during the commit operation.
176    Err(Error),
177    /// The commit operation requires backend I/O for the page at `index`.
178    NeedIo { index: usize },
179    /// The page exists but is not yet initialized.
180    ///
181    /// The caller should wait for initialization of `page` at `index` to
182    /// complete.
183    WaitUntilInit { index: usize, page: CachePage },
184}
185
186impl VmoCommitError {
187    /// Returns the page index whose commit is pending on I/O or initialization.
188    pub(crate) fn pending_index(&self) -> Result<usize> {
189        match self {
190            Self::NeedIo { index } | Self::WaitUntilInit { index, .. } => Ok(*index),
191            Self::Err(e) => Err(*e),
192        }
193    }
194}
195
196impl From<Error> for VmoCommitError {
197    fn from(e: Error) -> Self {
198        VmoCommitError::Err(e)
199    }
200}

203    fn from(e: ostd::Error) -> Self {

205    }

220/// A mode specified when committing a new [`Vmo`] page to the map in a page table.
221#[derive(Debug, Clone, Copy, PartialEq, Eq)]
222pub(crate) enum VmoMapMode {
223    /// The page is private. It can be a snapshot.
224    Private,
225    /// The page is shared. It needs to be read.
226    SharedRead,
227    /// The page is shared. It needs to be read and written.
228    SharedWrite,
229}
230
231impl VmoMapMode {
232    /// Ensures that the page can be mapped to a page table.
233    fn ensure(self, page: CachePage) {
234        match page.state() {
235            PageState::UpToDate if self.is_write() => {
236                let locked_page = page.lock_guard();
237                if locked_page.state().is_up_to_date() {
238                    locked_page.set_dirty();
239                }
240            }
241            PageState::UpToDate | PageState::Dirty => (),
242            PageState::Evicted if !self.is_shared() => (),
243            state @ (PageState::Uninit | PageState::Evicted) => {
244                // We shouldn't reach this point if the page is not initialized. `CommitMode::Read`
245                // should have been specified earlier to either read the page from the backend
246                // storage or report an error if reading fails.
247                debug_assert_eq!(state, PageState::Evicted);
248                // The page may be evicted. It still contains valid data. However, it cannot be
249                // mapped to a page table. We acquire the page lock here so that, when the caller
250                // retries, it won't see this page anymore.
251                let _ = page.lock_guard();
252            }
253        }
254    }
255
256    /// Tries to ensure that the page can be mapped to a page table without blocking.
257    ///
258    /// On success, returns a mode that satisfies `self` but may grant the caller more permissions
259    /// when mapping `page`. See also the "Map modes" section in [`Vmo::try_commit_page`].
260    fn try_ensure(self, page_idx: usize, page: &CachePage) -> Result<Self, VmoCommitError> {
261        match page.state() {
262            PageState::UpToDate if self.is_write() => {
263                if let Some(locked_page) = page.try_lock_guard()
264                    && locked_page.state().is_up_to_date()
265                {
266                    locked_page.set_dirty();
267                    Ok(Self::SharedWrite)
268                } else {
269                    Err(VmoCommitError::NeedIo { index: page_idx })
270                }
271            }
272            PageState::UpToDate => Ok(Self::SharedRead),
273            PageState::Dirty => Ok(Self::SharedWrite),
274            PageState::Evicted if !self.is_shared() => Ok(Self::Private),
275            PageState::Uninit | PageState::Evicted => {
276                Err(VmoCommitError::NeedIo { index: page_idx })
277            }
278        }
279    }
280
281    fn is_write(self) -> bool {
282        matches!(self, Self::SharedWrite)
283    }
284
285    fn is_shared(self) -> bool {
286        matches!(self, Self::SharedRead | Self::SharedWrite)
287    }
288}

290impl Vmo {
291    /// Commits the page at `page_idx`, blocking on backend I/O if required.
292    ///
293    /// For anonymous VMOs the page is zero-filled on first access.
294    /// For VMOs with a backend this may perform synchronous I/O.
295    pub(crate) fn commit_on(&self, page_idx: usize, map_mode: VmoMapMode) -> Result<()> {
296        let page = self.commit_on_internal(page_idx, CommitMode::Read)?;
297        map_mode.ensure(page);
298        Ok(())
299    }
300
301    fn commit_on_internal(&self, page_idx: usize, commit_mode: CommitMode) -> Result<CachePage> {
302        if let Some(backed_vmo) = self.as_backed_vmo() {
303            return backed_vmo.commit_on_internal(page_idx, commit_mode);
304        }
305
306        self.commit_on_anonymous(page_idx)
307    }
308
309    fn commit_on_anonymous(&self, page_idx: usize) -> Result<CachePage> {
310        let mut locked_pages = self.pages.lock();
311        if page_idx >= self.size().div_ceil(PAGE_SIZE) {
312            return_errno_with_message!(Errno::EINVAL, "the offset is outside the VMO");
313        }
314
315        let mut cursor = locked_pages.cursor_mut(page_idx as u64);
316        if let Some(page) = cursor.load() {
317            return Ok(page.clone());
318        }
319
320        let new_page = CachePage::alloc_zero()?;
321        cursor.store(new_page.clone());
322        Ok(new_page)
323    }
324
325    /// Tries to commit the page covering byte `offset` without blocking on I/O.
326    ///
327    /// If the commit operation needs to perform I/O, it will return a
328    /// [`VmoCommitError::NeedIo`].
329    ///
330    /// If the VMO corresponds to a page cache, callers must hold the page table
331    /// lock and insert the committed page atomically to the page table after
332    /// this method returns.
333    ///
334    /// # Map modes
335    ///
336    /// `map_mode` specifies the requested mapping mode for the page in the page
337    /// table. If successful, this method returns another [`VmoMapMode`] that
338    /// grants at least the requested mode, but may grant additional permissions
339    /// according to the page state. Callers can use this information to prevent
340    /// some future page faults.
341    ///
342    /// For example, if the page is already dirty, mapping it as writable is
343    /// permitted even if only read access is required in `map_mode`. In this
344    /// case, the requested mode (`map_mode`) is [`VmoMapMode::SharedRead`] but
345    /// the granted mode (the return value) is [`VmoMapMode::SharedWrite`].
346    pub(crate) fn try_commit_page(
347        &self,
348        offset: usize,
349        map_mode: VmoMapMode,
350    ) -> Result<(CachePage, VmoMapMode), VmoCommitError> {
351        let page_idx = offset / PAGE_SIZE;
352        if offset >= self.size() {
353            return Err(VmoCommitError::Err(Error::with_message(
354                Errno::EINVAL,
355                "the offset is outside the VMO",
356            )));
357        }
358
359        let (_, page) = {
360            let guard = disable_preempt();
361            let mut cursor = self.pages.cursor(&guard, page_idx as u64);
362            // `VmoMapMode::try_ensure` checks if the page has been initialized.
363            self.try_commit_with_cursor(&mut cursor, CommitMode::Overwrite)?
364        };
365        let mode = map_mode.try_ensure(page_idx, &page)?;
366        Ok((page, mode))
367    }
368
369    fn try_commit_with_cursor(
370        &self,
371        cursor: &mut Cursor<'_, CachePage>,
372        commit_mode: CommitMode,
373    ) -> Result<(usize, CachePage), VmoCommitError> {
374        if let Some(backed_vmo) = self.as_backed_vmo() {
375            return backed_vmo.try_commit_with_cursor(cursor, commit_mode);
376        } else if let Some(page) = cursor.load() {
377            let index = cursor.index() as usize;
378            return Ok((index, page.clone()));
379        }
380
381        // Need to commit. Only Anonymous VMOs can reach here, because VMOs
382        // with a backend will return `Err` if the page is not loaded.
383        let index = cursor.index() as usize;
384        let page = self.commit_on_anonymous(index)?;
385        Ok((index, page))
386    }
387
388    /// Traverses the indices within a specified range of a VMO sequentially.
389    ///
390    /// For each index position, you have the option to commit the page as well as
391    /// perform other operations.
392    ///
393    /// Once a commit operation needs to perform I/O, it will return a
394    /// [`VmoCommitError::NeedIo`].
395    ///
396    /// If the VMO corresponds to a page cache, callers must hold the page table
397    /// lock and insert the committed page atomically to the page table inside
398    /// the operation.
399    pub(crate) fn try_operate_on_range<F>(
400        &self,
401        range: &Range<usize>,
402        mut operate: F,
403        map_mode: VmoMapMode,
404    ) -> Result<(), VmoCommitError>
405    where
406        F: FnMut(
407            &mut dyn FnMut() -> Result<(usize, CachePage), VmoCommitError>,
408        ) -> Result<(), VmoCommitError>,
409    {
410        self.try_operate_on_range_internal(
411            range,
412            |commit_fn| {
413                let mut new_commit_fn = || {
414                    let (idx, page) = commit_fn()?;
415                    map_mode.try_ensure(idx, &page)?;
416                    Ok((idx, page))
417                };
418                operate(&mut new_commit_fn)
419            },
420            // `VmoMapMode::try_ensure` checks if the page has been initialized.
421            CommitMode::Overwrite,
422        )
423    }
424
425    fn try_operate_on_range_internal<F>(
426        &self,
427        range: &Range<usize>,
428        mut operate: F,
429        commit_mode: CommitMode,
430    ) -> Result<(), VmoCommitError>
431    where
432        F: FnMut(
433            &mut dyn FnMut() -> Result<(usize, CachePage), VmoCommitError>,
434        ) -> Result<(), VmoCommitError>,
435    {
436        if range.end > self.size() {
437            return Err(VmoCommitError::Err(Error::with_message(
438                Errno::EINVAL,
439                "the range to operate exceeds the VMO size",
440            )));
441        }
442
443        let page_idx_range = get_page_idx_range(range);
444        let guard = disable_preempt();
445        let mut cursor = self.pages.cursor(&guard, page_idx_range.start as u64);
446        for _ in page_idx_range {
447            let mut commit_fn = || self.try_commit_with_cursor(&mut cursor, commit_mode);
448            operate(&mut commit_fn)?;
449            cursor.next();
450        }
451        Ok(())
452    }

463        // Currently, only VMOs used by `MemfdInode` (anonymous) track writable mapping status.

469    /// Returns reverse mappings of the VMO.
470    ///
471    /// Holding either this lock or the VMAR lock ensures the stability of the
472    /// associated VM mappings. In other words, when adding, removing, or
473    /// moving a VM mapping, both this lock and the VMAR lock must be held.
474    ///
475    /// To avoid deadlocks, acquire this lock after the VMAR lock in cases both
476    /// locks need to be acquired.

481    /// Decommits anonymous pages in the specified byte range.
482    pub(super) fn decommit_anon_pages(
483        &self,
484        mut locked_rmap: MutexGuard<Rmap>,
485        mut locked_pages: LockedXArray<CachePage>,
486        range: Range<usize>,
487    ) -> Result<()> {
488        debug_assert!(!self.has_backend());
489
490        let page_idx_range = get_page_idx_range(&range);
491        let mut cursor = locked_pages.cursor_mut(page_idx_range.start as u64);
492
493        // We remove pages before unmapping them in the page table. So a concurrent page fault will
494        // synchronize via the page table lock. Either the mapped item will be unmapped below or the
495        // page fault handler will commit the page and be blocked on the reverse-mapping lock.
496        loop {
497            cursor.remove();
498            let page_idx = cursor.next_present();
499            if page_idx.is_none_or(|idx| idx >= page_idx_range.end as u64) {
500                break;
501            }
502        }
503
504        drop(locked_pages);
505
506        locked_rmap.unmap(page_idx_range.start * PAGE_SIZE..page_idx_range.end * PAGE_SIZE);
507
508        Ok(())
509    }

511    /// Converts this VMO to a backend VMO wrapper if it has a backend.
512    ///
513    /// Returns `None` if this is an anonymous VMO.
514    pub(super) fn as_backed_vmo(&self) -> Option<BackedVmo<'_>> {
515        self.backend.as_ref().and_then(|weak_backend| {
516            weak_backend
517                .upgrade()
518                .map(|backend| BackedVmo { vmo: self, backend })
519        })
520    }
521
522    /// Returns whether this VMO has a backend.
523    fn has_backend(&self) -> bool {
524        self.backend.is_some()
525    }

528// Implement the read/write methods for `Vmo`.
529impl Vmo {
530    /// Maximum number of pages collected per batch from the `XArray`.
531    const PAGE_BATCH_CAPACITY: usize = 32;
532
533    /// Reads data from the VMO at `offset` into `writer`.
534    pub(crate) fn read(&self, offset: usize, writer: &mut VmWriter) -> Result<()> {
535        let read_len = writer.avail().min(self.size().saturating_sub(offset));
536        if read_len == 0 {
537            return Ok(());
538        }
539        let read_end = offset + read_len;
540
541        let range = offset..read_end;
542        let page_idx_range = get_page_idx_range(&range);
543        let mut current_idx = page_idx_range.start;
544        let mut page_offset = offset % PAGE_SIZE;
545        let mut page_batch =
546            Vec::with_capacity(min(page_idx_range.len(), Self::PAGE_BATCH_CAPACITY));
547
548        while current_idx < page_idx_range.end {
549            self.collect_pages(
550                current_idx,
551                page_idx_range.end,
552                CommitMode::Read,
553                &mut page_batch,
554            )?;
555
556            for (_, page) in page_batch.iter() {
557                page.reader().skip(page_offset).read_fallible(writer)?;
558                page_offset = 0;
559            }
560
561            // `current_idx < page_idx_range.end` guarantees at least one page is successfully
562            // collected here.
563            current_idx = page_batch.last().unwrap().0 + 1;
564        }
565
566        Ok(())
567    }
568
569    /// Writes data from `reader` into the VMO at `offset`.
570    pub(crate) fn write(&self, offset: usize, reader: &mut VmReader) -> Result<()> {
571        let write_len = reader.remain().min(self.size().saturating_sub(offset));
572        if write_len == 0 {
573            return Ok(());
574        }
575        let write_end = offset + write_len;
576
577        let write_range = offset..write_end;
578        let mut page_offset = offset % PAGE_SIZE;
579        let mut page_batch = Vec::with_capacity(min(
580            write_range.len().div_ceil(PAGE_SIZE),
581            Self::PAGE_BATCH_CAPACITY,
582        ));
583
584        if !self.has_backend() {
585            return self.write_anonymous_pages(
586                &write_range,
587                &mut page_offset,
588                reader,
589                &mut page_batch,
590            );
591        }
592
593        // VMOs with a backend require dirty tracking and may skip backend reads
594        // for page-aligned ranges that will be entirely overwritten.
595        if write_range.len() < PAGE_SIZE {
596            return self.write_pages_with_backend(
597                &write_range,
598                &mut page_offset,
599                reader,
600                CommitMode::Read,
601                &mut page_batch,
602            );
603        }
604
605        // Split into head (unaligned), middle (aligned), and tail (unaligned).
606        let up_align_start = write_range.start.align_up(PAGE_SIZE);
607        let down_align_end = write_range.end.align_down(PAGE_SIZE);
608
609        if write_range.start != up_align_start {
610            let head = write_range.start..up_align_start;
611            self.write_pages_with_backend(
612                &head,
613                &mut page_offset,
614                reader,
615                CommitMode::Read,
616                &mut page_batch,
617            )?;
618        }
619        if up_align_start != down_align_end {
620            let mid = up_align_start..down_align_end;
621            self.write_pages_with_backend(
622                &mid,
623                &mut page_offset,
624                reader,
625                CommitMode::Overwrite,
626                &mut page_batch,
627            )?;
628        }
629        if down_align_end != write_range.end {
630            let tail = down_align_end..write_range.end;
631            self.write_pages_with_backend(
632                &tail,
633                &mut page_offset,
634                reader,
635                CommitMode::Read,
636                &mut page_batch,
637            )?;
638        }
639
640        Ok(())
641    }
642
643    /// Fills the specified range with zeros.
644    pub(super) fn fill_zeros(&self, range: Range<usize>) -> Result<()> {
645        if range.is_empty() {
646            return Ok(());
647        }
648        if range.end > self.size() {
649            return_errno_with_message!(Errno::EINVAL, "the range to fill exceeds the VMO size");
650        }
651
652        static ZERO_PAGE: [u8; PAGE_SIZE] = [0; PAGE_SIZE];
653
654        let mut current_offset = range.start;
655        while current_offset < range.end {
656            let page_remaining = PAGE_SIZE - current_offset % PAGE_SIZE;
657            let write_len = (range.end - current_offset).min(page_remaining);
658            let mut reader = VmReader::from(&ZERO_PAGE[..write_len]).to_fallible();
659
660            self.write(current_offset, &mut reader)?;
661            current_offset += write_len;
662        }
663
664        Ok(())
665    }

667    /// Writes data to pages without dirty tracking.
668    ///
669    /// Used for anonymous VMO writes where pages are always in the `UpToDate`
670    /// state.
671    fn write_anonymous_pages(
672        &self,
673        range: &Range<usize>,
674        page_offset: &mut usize,
675        reader: &mut VmReader,
676        page_batch: &mut Vec<(usize, CachePage)>,
677    ) -> Result<()> {
678        let page_idx_range = get_page_idx_range(range);
679        let mut current_idx = page_idx_range.start;
680
681        while current_idx < page_idx_range.end {
682            self.collect_pages(
683                current_idx,
684                page_idx_range.end,
685                CommitMode::Read,
686                page_batch,
687            )?;
688
689            for (_, page) in page_batch.iter() {
690                page.writer().skip(*page_offset).write_fallible(reader)?;
691                *page_offset = 0;
692            }
693
694            // `current_idx < page_idx_range.end` guarantees at least one page is successfully
695            // collected here.
696            current_idx = page_batch.last().unwrap().0 + 1;
697        }
698
699        Ok(())
700    }

702    /// Writes data to pages with backend dirty tracking.
703    ///
704    /// Each page is locked before writing to ensure correct state transitions
705    /// and marked dirty for later writeback.
706    fn write_pages_with_backend(
707        &self,
708        range: &Range<usize>,
709        page_offset: &mut usize,
710        reader: &mut VmReader,
711        commit_mode: CommitMode,
712        page_batch: &mut Vec<(usize, CachePage)>,
713    ) -> Result<()> {
714        let page_idx_range = get_page_idx_range(range);
715        let mut current_idx = page_idx_range.start;
716
717        while current_idx < page_idx_range.end {
718            self.collect_pages(current_idx, page_idx_range.end, commit_mode, page_batch)?;
719
720            // `current_idx < page_idx_range.end` guarantees at least one page is successfully
721            // collected here.
722            let mut next_idx = page_batch.last().unwrap().0 + 1;
723
724            for (idx, page) in page_batch.drain(..) {
725                let locked_page = page.lock();
726
727                let state = locked_page.state();
728                if state.is_evicted() {
729                    // Retry. We will not see this page again because we've acquired the page lock.
730                    next_idx = idx;
731                    continue;
732                }
733
734                let written_size = match locked_page
735                    .writer()
736                    .skip(*page_offset)
737                    .write_fallible(reader)
738                {
739                    Ok(written_size) => written_size,
740                    Err((err, written_size)) => {
741                        // If the page is not initialized, keep it as it is on a partial write.
742                        if written_size > 0 && state.is_up_to_date() {
743                            locked_page.set_dirty();
744                        }
745                        return Err(Error::from(err));
746                    }
747                };
748
749                if written_size > 0 {
750                    locked_page.set_dirty();
751                }
752
753                *page_offset = 0;
754            }
755
756            current_idx = next_idx;
757        }
758
759        Ok(())
760    }
761
762    /// Collects a batch of committed pages from `start_idx` to `end_idx`.
763    ///
764    /// Returns up to a small batch of consecutive committed pages. The
765    /// caller receives only successfully committed pages and never needs to
766    /// deal with [`VmoCommitError`] directly.
767    ///
768    /// This helper keeps the returned pages alive, but it does not pin them in
769    /// the `XArray`. Callers must still serialize against `evict_range()` /
770    /// `invalidate_range()` with a higher-level invalidation lock before they
771    /// take per-page locks, otherwise buffered writes may dirty a page that has
772    /// already been detached from the page cache.
773    fn collect_pages(
774        &self,
775        start_idx: usize,
776        end_idx: usize,
777        commit_mode: CommitMode,
778        pages: &mut Vec<(usize, CachePage)>,
779    ) -> Result<()> {
780        let end_idx = end_idx.min(start_idx + pages.capacity());
781        pages.clear();
782        let range = (start_idx * PAGE_SIZE)..(end_idx * PAGE_SIZE);
783        let mut current_range = range.clone();
784
785        'retry: loop {
786            let mut operate =
787                |commit_fn: &mut dyn FnMut() -> Result<(usize, CachePage), VmoCommitError>| {
788                    let (idx, page) = commit_fn()?;
789                    pages.push((idx, page));
790                    Ok(())
791                };
792            match self.try_operate_on_range_internal(&current_range, &mut operate, commit_mode) {
793                Ok(()) => break 'retry,
794                Err(err) => {
795                    let (idx, page) = self.handle_commit_error(err, commit_mode)?;
796                    pages.push((idx, page));
797                    current_range.start = (idx + 1) * PAGE_SIZE;
798                    if current_range.start >= range.end {
799                        break 'retry;
800                    }
801                }
802            }
803        }
804
805        Ok(())
806    }
807
808    /// Handles a commit error by performing the necessary I/O or initialization.
809    ///
810    /// Returns the page index and page that were recovered.
811    fn handle_commit_error(
812        &self,
813        err: VmoCommitError,
814        commit_mode: CommitMode,
815    ) -> Result<(usize, CachePage)> {
816        match err {
817            VmoCommitError::Err(e) => Err(e),
818            VmoCommitError::NeedIo { index } => {
819                Ok((index, self.commit_on_internal(index, commit_mode)?))
820            }
821            VmoCommitError::WaitUntilInit { index, page } => {
822                page.ensure_init(|locked_page| {
823                    self.as_backed_vmo()
824                        .unwrap()
825                        .backend
826                        .read_page(index, locked_page)
827                })?;
828                Ok((index, page))
829            }
830        }
831    }
832}

834/// A wrapper around a VMO with a backend that provides specialized operations.
835///
836/// This structure is created by calling [`Vmo::as_backed_vmo()`] and provides
837/// access to backend-specific functionality like reading from storage and
838/// managing dirty pages.
839pub(crate) struct BackedVmo<'a> {
840    vmo: &'a Vmo,
841    backend: Arc<dyn PageCacheBackend>,
842}
843
844enum PageSelection {
845    Flush,
846    Evict,
847}
848
849impl PageSelection {
850    fn should_collect(&self, page: &CachePage) -> bool {
851        let state = page.state();
852        match self {
853            PageSelection::Flush if state.is_dirty() => true,
854            PageSelection::Flush if state.is_uninit() => false,
855            PageSelection::Flush if let Some(locked_page) = page.try_lock_guard() => {
856                locked_page.is_writing_back()
857            }
858            // We need to wait for all the pages that are being written back to finish before
859            // returning from `flush()`. But we don't know whether this page is being writing back
860            // because `try_lock_guard()` fails. So let's be conservative and collect this page.
861            PageSelection::Flush => true,
862            PageSelection::Evict => state.is_up_to_date(),
863        }
864    }
865}
866
867impl<'a> BackedVmo<'a> {
868    /// Writes back dirty pages in the specified byte range to the backend storage.
869    pub(super) fn flush_dirty_pages(&self, range: &Range<usize>) -> Result<()> {
870        if range.start >= self.size() {
871            return Ok(());
872        }
873
874        let mut locked_rmap = self.rmap.lock();
875
876        let page_idx_range = get_page_idx_range(range);
877        let pages_to_flush = self.collect_pages_if(&page_idx_range, PageSelection::Flush);
878        if pages_to_flush.is_empty() {
879            return Ok(());
880        }
881
882        // We must lock and mark all pages as up-to-date before freezing them in the page table. A
883        // concurrent page fault caused by a write operation will synchronize via the page table
884        // lock. Either the mapped item will be freezed below or the page fault handler will see the
885        // page as up to date and be blocked on the page lock.
886        let mut locked_dirty_pages = Vec::new();
887        let mut maybe_writing_back_pages = Vec::new();
888        for (idx, page) in pages_to_flush {
889            if !page.state().is_dirty() {
890                maybe_writing_back_pages.push(page);
891                continue;
892            }
893            let locked_page = page.lock();
894            if locked_page.state().is_dirty() {
895                locked_page.set_up_to_date();
896                locked_dirty_pages.push((idx, locked_page));
897            } else if locked_page.is_writing_back() {
898                maybe_writing_back_pages.push(locked_page.unlock());
899            }
900        }
901
902        locked_rmap.freeze(page_idx_range.start * PAGE_SIZE..page_idx_range.end * PAGE_SIZE);
903
904        // As long as the pages are locked and their state is `UpToDate`, no one can write to them.
905        // Therefore, we can release the reverse-mapping lock now.
906        drop(locked_rmap);
907
908        let mut io_batch = IoBatch::with_capacity(locked_dirty_pages.len());
909        for (idx, locked_page) in locked_dirty_pages {
910            self.backend
911                .write_page_async(idx, locked_page, &mut io_batch)?;
912        }
913        io_batch.wait_all()?;
914
915        for page in maybe_writing_back_pages {
916            let locked_page = page.lock();
917            locked_page.wait_until_finish_writing_back();
918        }
919
920        Ok(())
921    }
922
923    /// Removes up-to-date (clean) pages in the specified byte range from the page cache.
924    ///
925    /// Only pages in the `UpToDate` state are removed. Dirty and uninitialized
926    /// pages are left in place.
927    pub(super) fn evict_up_to_date_pages(&self, range: &Range<usize>) -> Result<()> {
928        if range.start >= self.size() {
929            return Ok(());
930        }
931
932        let mut locked_rmap = self.rmap.lock();
933
934        let page_idx_range = get_page_idx_range(range);
935        let up_to_date_pages = self.collect_pages_if(&page_idx_range, PageSelection::Evict);
936        if up_to_date_pages.is_empty() {
937            return Ok(());
938        }
939
940        // We need to lock all pages and ensure they are up to date under the page lock. In
941        // addition, we mark pages to be removed as `Evicted` before unmapping them in the page
942        // table. A concurrent page fault will synchronize via the page table lock. Either the
943        // mapped item will be unmapped below or the page fault handler will see the page as removed
944        // and be blocked on the page lock.
945        let locked_up_to_date_pages = up_to_date_pages
946            .into_iter()
947            .filter_map(|(idx, page)| {
948                let locked_page = page.lock();
949                if !locked_page.state().is_up_to_date() {
950                    return None;
951                }
952                locked_page.set_evicted();
953                Some((idx, locked_page))
954            })
955            .collect::<Vec<_>>();
956        if locked_up_to_date_pages.is_empty() {
957            return Ok(());
958        }
959
960        locked_rmap.unmap(page_idx_range.start * PAGE_SIZE..page_idx_range.end * PAGE_SIZE);
961
962        // As long as the pages are locked and their state is `Evicted`, no one can read from them.
963        // Therefore, we can release the reverse-mapping lock now.
964        drop(locked_rmap);
965
966        for (idx, page) in locked_up_to_date_pages {
967            page.wait_until_finish_writing_back();
968            let mut locked_pages = self.pages.lock();
969            let mut cursor_mut = locked_pages.cursor_mut(idx as u64);
970            let removed_page = cursor_mut.remove();
971            // The page won't be evicted because it has already been marked as `Evicted`. It won't
972            // be decommitted either, as the caller should hold a file system lock that prevents
973            // concurrent resizing.
974            //
975            // TODO: If we allowing pages to be evicted without holding the file system lock, the
976            // old page may be decommitted and replaced by another page. In this case, we must
977            // reload and recheck the page before removing it.
978            debug_assert_eq!(removed_page.unwrap().paddr(), page.paddr());
979        }
980
981        Ok(())
982    }
983
984    /// Decommits pages with a backend in the specified byte range.
985    ///
986    /// Dirty pages in the range are discarded without being written back. If a
987    /// page is already under writeback, decommit waits for that writeback to
988    /// finish before removing the page from the `XArray`.
989    pub(super) fn decommit_pages(
990        &self,
991        mut locked_rmap: MutexGuard<Rmap>,
992        locked_pages: LockedXArray<CachePage>,
993        range: &Range<usize>,
994    ) -> Result<()> {
995        // Do not check `self.size()` here. It contains the new size, however, we want to decommit
996        // pages outside of the new range. See `PageCache::resize` for a concrete example.
997
998        // We remove pages before unmapping them in the page table. So a concurrent page fault will
999        // synchronize via the page table lock. Either the mapped item will be unmapped below or the
1000        // page fault handler will see the new size after acquiring the lock in `self.pages`.
1001        let page_idx_range = get_page_idx_range(range);
1002        let removed_pages = self.remove_pages_in(locked_pages, &page_idx_range);
1003        if removed_pages.is_empty() {
1004            return Ok(());
1005        }
1006
1007        locked_rmap.unmap(page_idx_range.start * PAGE_SIZE..page_idx_range.end * PAGE_SIZE);
1008
1009        drop(locked_rmap);
1010
1011        for (_, page) in removed_pages {
1012            let locked_page = page.lock();
1013            locked_page.wait_until_finish_writing_back();
1014        }
1015
1016        Ok(())
1017    }
1018
1019    /// Commits a page at the given index for a VMO with a backend.
1020    fn commit_on_internal(&self, page_idx: usize, commit_mode: CommitMode) -> Result<CachePage> {
1021        let mut locked_pages = self.pages.lock();
1022        if page_idx >= self.size().div_ceil(PAGE_SIZE) {
1023            return_errno_with_message!(Errno::EINVAL, "the offset is outside the VMO");
1024        }
1025
1026        let mut cursor = locked_pages.cursor_mut(page_idx as u64);
1027
1028        if let Some(page) = cursor.load() {
1029            let page = page.clone();
1030            drop(locked_pages);
1031
1032            if !commit_mode.skips_backend_read() {
1033                page.ensure_init(|locked_page| self.backend.read_page(page_idx, locked_page))?;
1034                return Ok(page);
1035            }
1036
1037            return Ok(page);
1038        };
1039
1040        // The page is within the file bounds - need to allocate a cache page.
1041        let uninit_page = CachePage::alloc_uninit()?;
1042        cursor.store(uninit_page.clone());
1043        drop(locked_pages);
1044
1045        if commit_mode.skips_backend_read() {
1046            // The page will be completely overwritten, no need to read.
1047            Ok(uninit_page)
1048        } else {
1049            // Read the page from the backend storage.
1050            uninit_page.ensure_init(|locked_page| self.backend.read_page(page_idx, locked_page))?;
1051            Ok(uninit_page)
1052        }
1053    }

1055    /// Attempts to commit a page using a cursor, without blocking on I/O.
1056    fn try_commit_with_cursor(
1057        &self,
1058        cursor: &mut Cursor<'_, CachePage>,
1059        commit_mode: CommitMode,
1060    ) -> Result<(usize, CachePage), VmoCommitError> {
1061        let page_idx = cursor.index() as usize;
1062
1063        let Some(page) = cursor.load() else {
1064            return Err(VmoCommitError::NeedIo { index: page_idx });
1065        };
1066
1067        // Check if the page is initialized.
1068        if !commit_mode.skips_backend_read() && page.state().is_uninit() {
1069            return Err(VmoCommitError::WaitUntilInit {
1070                index: page_idx,
1071                page: page.clone(),
1072            });
1073        }
1074
1075        Ok((page_idx, page.clone()))
1076    }

1078    /// Collects pages in the specified page index range.
1079    fn collect_pages_if(
1080        &self,
1081        page_idx_range: &Range<usize>,
1082        cond: PageSelection,
1083    ) -> Vec<(usize, CachePage)> {
1084        if page_idx_range.is_empty() {
1085            return Vec::new();
1086        }
1087
1088        let mut collected_pages = Vec::new();
1089
1090        let preempt_guard = disable_preempt();
1091        let mut cursor = self
1092            .pages
1093            .cursor(&preempt_guard, page_idx_range.start as u64);
1094        let mut index = page_idx_range.start;
1095
1096        loop {
1097            if let Some(page) = cursor.load()
1098                && cond.should_collect(&page)
1099            {
1100                collected_pages.push((index, page.clone()));
1101            }
1102
1103            let Some(next_index) = cursor.next_present() else {
1104                break;
1105            };
1106            index = next_index as usize;
1107            if index >= page_idx_range.end {
1108                break;
1109            }
1110        }
1111
1112        collected_pages
1113    }

1115    /// Removes pages in the specified page index range.
1116    fn remove_pages_in(
1117        &self,
1118        mut locked_pages: LockedXArray<CachePage>,
1119        page_idx_range: &Range<usize>,
1120    ) -> Vec<(usize, CachePage)> {
1121        if page_idx_range.is_empty() {
1122            return Vec::new();
1123        }
1124
1125        let mut removed_pages = Vec::new();
1126
1127        let mut cursor_mut = locked_pages.cursor_mut(page_idx_range.start as u64);
1128        let mut index = page_idx_range.start;
1129
1130        loop {
1131            if let Some(page) = cursor_mut.remove() {
1132                removed_pages.push((index, page.clone()));
1133            }
1134
1135            let Some(next_index) = cursor_mut.next_present() else {
1136                break;
1137            };
1138            index = next_index as usize;
1139            if index >= page_idx_range.end {
1140                break;
1141            }
1142        }
1143
1144        removed_pages
1145    }

1156/// Gets the page index range that contains the offset range of VMO.
1157pub(crate) fn get_page_idx_range(vmo_offset_range: &Range<usize>) -> Range<usize> {
1158    let start = vmo_offset_range.start.align_down(PAGE_SIZE);
1159    let end = vmo_offset_range.end.align_up(PAGE_SIZE);

1163/// The status of writable mappings of a VMO, i.e., shared mappings that may
1164/// include the `PROT_WRITE` permission.
1165///
1166/// Internally, it uses an `AtomicIsize` counter with the following rules:
1167///
1168/// - **Positive values**: number of active writable mappings.
1169/// - **Zero**: no writable mappings, and writable mappings are still allowed.
1170/// - **Negative values**: writable mappings are denied.
1171#[derive(Debug, Default)]
1172pub(crate) struct WritableMappingStatus(AtomicIsize);
1173
1174impl WritableMappingStatus {
1175    /// Builds a new writable mapping.
1176    ///
1177    /// Fails with `EPERM` if writable mappings have been denied.
1178    pub(crate) fn map(&self) -> Result<()> {
1179        // Increase unless negative
1180        self.0
1181            .try_update(Ordering::Relaxed, Ordering::Relaxed, |v| {
1182                (v >= 0).then(|| v + 1)
1183            })
1184            .map_err(|_| Error::with_message(Errno::EPERM, "writable mappings have been denied"))?;
1185        Ok(())
1186    }
1187
1188    /// Denies any future writable mapping.
1189    ///
1190    /// Fails with `EBUSY` if there are still active writable mappings.
1191    pub(crate) fn deny(&self) -> Result<()> {
1192        // Decrease unless positive
1193        self.0
1194            .try_update(Ordering::Relaxed, Ordering::Relaxed, |v| {
1195                (v <= 0).then_some(-1)
1196            })
1197            .map_err(|_| {
1198                Error::with_message(Errno::EBUSY, "there are still active writable mappings")
1199            })?;
1200        Ok(())
1201    }
1202
1203    /// Increments the writable mapping counter.
1204    ///
1205    /// Typically used when splitting an existing mapping, or forking a process.
1206    pub(crate) fn increment(&self) {
1207        self.0.fetch_add(1, Ordering::Relaxed);
1208    }
1209
1210    /// Decrements the writable mapping counter.
1211    ///
1212    /// Typically used when unmapping a region, exiting a process, or merging mappings.
1213    pub(crate) fn decrement(&self) {
1214        self.0.fetch_sub(1, Ordering::Relaxed);
1215    }
1216}
