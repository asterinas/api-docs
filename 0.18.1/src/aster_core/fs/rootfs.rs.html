3//! Root filesystem mounting during boot.
4//!
5//! Asterinas first tries to boot from the initramfs. It checks the path specified by `rdinit`, or
6//! `/init` when `rdinit` is not provided. If the selected path is accessible, Asterinas runs it as
7//! the first userspace process. Otherwise, Asterinas boots from the root filesystem.
8//!
9//! For root filesystem boot, this module mounts a supported filesystem from the block device
10//! specified by `root`. The mounted filesystem replaces the bootstrap root in the initial mount
11//! namespace and becomes the root and working directory of the first userspace process. The
12//! bootstrap root is then detached. The `root` parameter is required when this boot path is
13//! selected, and [`SUPPORTED_ROOTFS_TYPES`] lists the supported filesystem types.
14//!
15//! The following kernel parameters configure root filesystem boot:
16//!
17//! - `root` specifies the block device to mount.
18//! - `rootfstype` specifies a comma-separated list of filesystem type candidates. All supported
19//!   types are tried when it is not provided.
20//! - `ro` and `rw` select read-only or read-write mounting. The root filesystem is read-only by
21//!   default. They use last-wins semantics: the last `ro` or `rw` parameter determines the mount
22//!   mode.
23//! - `init` specifies the init executable on the mounted filesystem. If it is not provided,
24//!   Asterinas tries `/sbin/init`, `/etc/init`, `/bin/init`, and `/bin/sh` in order.
25

51/// Filesystem types supported for the root filesystem.

54/// Mounts and switches to the root filesystem configured by the kernel command line.
55///
56/// The filesystem replaces the bootstrap root mount and becomes the resolver's root and working
57/// directory.
58///
59/// Returns an error if `root` is not provided or the root filesystem cannot be opened or mounted.
60pub(crate) fn switch_to_rootfs(path_resolver: &mut PathResolver) -> Result<()> {
61    let root = ROOT_PATH.get().ok_or_else(|| {
62        Error::with_message(Errno::EINVAL, "the `root` parameter was not provided")
63    })?;
64    let (fs_flags, mount_flags) = rootfs_flags();
65    let fs_and_root = open_rootfs(root, fs_flags)?;
66    let rootfs_mount = Mount::new_detached(
67        fs_and_root,
68        mount_flags,
69        Arc::downgrade(MountNamespace::get_init_singleton()),
70        Some(root.to_string()),
71    )?;
72    path_resolver.switch_root_for_boot(rootfs_mount);
73    println!("[kernel] rootfs is ready (mounted from {})", root);
74    Ok(())
75}

77/// Finds the init program specified for booting from the root filesystem.
78///
79/// Resolves the path specified by `init` and returns the resolved path together with the original
80/// pathname. Returns `Ok(None)` when `init` is not provided, and an error if the configured
81/// pathname is invalid or cannot be resolved.
82pub(crate) fn find_init(path_resolver: &PathResolver) -> Result<Option<(Path, &'static str)>> {
83    let Some(init_path) = INIT_PATH.get().map(String::as_str) else {
84        return Ok(None);
85    };
86
87    let path = path_resolver.lookup(&FsPath::try_from(init_path)?)?;
88    Ok(Some((path, init_path)))
89}

91fn open_rootfs(root: &str, fs_flags: FsFlags) -> Result<FsAndRoot> {
92    // Treat a `/dev/...` value of `root` as a Linux-compatible root device spec, not as a VFS path
93    // lookup.
94    // Reference: <https://elixir.bootlin.com/linux/v6.18/source/drivers/base/devtmpfs.c#L358-L359>.
95    let device_name = root
96        .strip_prefix("/dev/")
97        .ok_or_else(|| Error::with_message(Errno::EINVAL, "root must name a /dev block device"))?;
98    let device = aster_block::lookup_by_name(device_name)
99        .ok_or_else(|| Error::with_message(Errno::ENODEV, "root block device not found"))?;
100    open_rootfs_from_candidates(device, rootfs_types(), fs_flags)
101}
102
103fn open_rootfs_from_candidates(
104    device: Arc<dyn aster_block::BlockDevice>,
105    rootfs_types: &[&dyn DynFsType],
106    fs_flags: FsFlags,
107) -> Result<FsAndRoot> {
108    let mut fs_creation_ctx = FsCreationCtx::from_block_device(device, fs_flags, None);
109    for rootfs_type in rootfs_types {
110        match rootfs_type.get_or_create(&mut fs_creation_ctx) {
111            Ok(fs_and_root) => return Ok(fs_and_root),
112            Err(err) if err.error() == Errno::EINVAL => continue,
113            Err(err) => return Err(err),
114        }
115    }
116
117    return_errno_with_message!(Errno::ENODEV, "no root filesystem type could mount root")
118}

120fn rootfs_flags() -> (FsFlags, PerMountFlags) {
121    let mut fs_flags = FsFlags::empty();
122    let mut mount_flags = PerMountFlags::default();
123    if ROOT_MOUNT_READ_ONLY.load(Ordering::Relaxed) {
124        fs_flags.insert(FsFlags::RDONLY);
125        mount_flags.insert(PerMountFlags::RDONLY);
126    }
127    (fs_flags, mount_flags)
128}
129
130fn rootfs_types() -> &'static [&'static dyn DynFsType] {
131    ROOTFS_TYPES
132        .get()
133        .map(RootFsTypes::as_slice)
134        .unwrap_or(SUPPORTED_ROOTFS_TYPES)
135}

143/// Root filesystem type candidates.
144struct RootFsTypes(Vec<&'static dyn DynFsType>);
145
146impl RootFsTypes {
147    /// Returns the root filesystem type candidates as a slice.
148    fn as_slice(&self) -> &[&'static dyn DynFsType] {
149        self.0.as_slice()
150    }
151}
152
153impl FromStr for RootFsTypes {
154    type Err = core::convert::Infallible;
155
156    fn from_str(value: &str) -> Result<Self, Self::Err> {
157        let candidates = value
158            .split(',')
159            .filter(|type_name| !type_name.is_empty())
160            .filter_map(|type_name| {
161                SUPPORTED_ROOTFS_TYPES
162                    .iter()
163                    .copied()
164                    .find(|fs_type| fs_type.name() == type_name)
165            })
166            .collect();
167
168        Ok(Self(candidates))
169    }
170}

175static ROOT_MOUNT_READ_ONLY: AtomicBool = AtomicBool::new(true);
176
177struct SetRootMountReadOnly;
178struct SetRootMountReadWrite;
179
180impl ParamStorage for SetRootMountReadOnly {
181    type Value = bool;
182
183    fn store_param(&self, value: Self::Value) {
184        if value {
185            ROOT_MOUNT_READ_ONLY.store(true, Ordering::Relaxed);
186        }
187    }
188}
189
190impl ParamStorage for SetRootMountReadWrite {
191    type Value = bool;
192
193    fn store_param(&self, value: Self::Value) {
194        if value {
195            ROOT_MOUNT_READ_ONLY.store(false, Ordering::Relaxed);
196        }
197    }
198}

200static RO_PARAM: SetRootMountReadOnly = SetRootMountReadOnly;
201static RW_PARAM: SetRootMountReadWrite = SetRootMountReadWrite;
202aster_cmdline::define_flag_param!("ro", RO_PARAM);
203aster_cmdline::define_flag_param!("rw", RW_PARAM);
