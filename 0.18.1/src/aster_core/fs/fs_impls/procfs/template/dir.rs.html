31/// Wraps directory-specific procfs operations as a VFS inode.
32///
33/// `ProcDir` owns the directory implementation object,
34/// tracks parent linkage for `.` and `..`,
35/// and forwards inode methods that are common to all procfs directories.
36pub(in procfs) struct ProcDir<D: ProcDirOps> {
37    inner: D,
38    this: Weak<ProcDir<D>>,
39    parent: Option<Weak<dyn Inode>>,
40    common: Common,
41}
42
43impl<D: ProcDirOps> ProcDir<D> {
44    /// Creates the root procfs directory inode.
45    pub(in procfs) fn new_root(
46        dir: D,
47        fs: Weak<dyn FileSystem>,
48        ino: u64,
49        sb: &SuperBlock,
50        mode: InodeMode,
51    ) -> Arc<Self> {
52        let metadata = Metadata::new_dir(ino, mode, BLOCK_SIZE, sb.container_dev_id);
53        let common = Common::new(metadata, fs);
54
55        Arc::new_cyclic(|weak_self| Self {
56            inner: dir,
57            this: weak_self.clone(),
58            parent: None,
59            common,
60        })
61    }
62
63    /// Creates a non-root procfs directory inode under `parent`.
64    pub(in procfs) fn new(dir: D, parent: Weak<dyn Inode>, mode: InodeMode) -> Arc<Self> {
65        let common = {
66            let fs = parent.upgrade().unwrap().fs();
67            let procfs = fs.downcast_ref::<ProcFs>().unwrap();
68            let ino = procfs.alloc_id();
69            let metadata = Metadata::new_dir(ino, mode, BLOCK_SIZE, procfs.sb().container_dev_id);
70            Common::new(metadata, Arc::downgrade(&fs))
71        };
72        Arc::new_cyclic(|weak_self| Self {
73            inner: dir,
74            this: weak_self.clone(),
75            parent: Some(parent),
76            common,
77        })
78    }
79
80    fn this(&self) -> Arc<ProcDir<D>> {
81        self.this.upgrade().unwrap()
82    }
83
84    pub(in procfs) fn this_weak(&self) -> &Weak<ProcDir<D>> {
85        &self.this
86    }
87
88    fn parent(&self) -> Option<Arc<dyn Inode>> {
89        self.parent.as_ref().and_then(|p| p.upgrade())
90    }
91
92    /// Returns the directory-specific procfs operations.
93    pub(in procfs) fn inner(&self) -> &D {
94        &self.inner
95    }
96}

98impl<D: ProcDirOps + 'static> FileOps for ProcDir<D> {
99    fn read_at(
100        &self,
101        _offset: usize,
102        _writer: &mut VmWriter,
103        _status_flags: StatusFlags,
104    ) -> Result<usize> {
105        Err(Error::new(Errno::EISDIR))
106    }
107
108    fn write_at(
109        &self,
110        _offset: usize,
111        _reader: &mut VmReader,
112        _status_flags: StatusFlags,
113    ) -> Result<usize> {
114        Err(Error::new(Errno::EISDIR))
115    }
116
117    fn readdir_at(&self, offset: usize, visitor: &mut dyn DirentVisitor) -> Result<usize> {
118        /// Returns the always-present `.` and `..` entries for a procfs directory.
119        fn special_entries<D: ProcDirOps + 'static>(
120            dir: &ProcDir<D>,
121        ) -> impl Iterator<Item = (&'static str, Arc<dyn Inode>, usize)> {
122            let this_inode: Arc<dyn Inode> = dir.this();
123            let parent_inode = dir.parent().unwrap_or_else(|| this_inode.clone());
124            [(".", this_inode, 1), ("..", parent_inode, 2)].into_iter()
125        }
126
127        let try_readdir = |offset: &mut usize, visitor: &mut dyn DirentVisitor| -> Result<()> {
128            for (name, inode, next_offset) in special_entries(self) {
129                if next_offset <= *offset {
130                    continue;
131                }
132
133                visitor.visit(name, inode.ino(), inode.type_(), next_offset)?;
134                *offset = next_offset;
135            }
136
137            self.inner.visit_entries_from_offset(*offset, |child| {
138                visitor.visit(
139                    child.name.as_ref(),
140                    child.ino,
141                    child.type_,
142                    child.next_offset,
143                )?;
144                *offset = child.next_offset;
145                Ok(())
146            })?;
147            Ok(())
148        };
149
150        let mut iterate_offset = offset;
151        match try_readdir(&mut iterate_offset, visitor) {
152            Err(e) if iterate_offset == offset => Err(e),
153            _ => Ok(iterate_offset - offset),
154        }
155    }
156}

199        Err(Error::new(Errno::EPERM))

203        Err(Error::new(Errno::EPERM))
204    }
205
206    fn rmdir(&self, _name: &str) -> Result<()> {
207        Err(Error::new(Errno::EPERM))
208    }
209
210    fn lookup(&self, name: &str) -> Result<Arc<dyn Inode>> {
211        if is_dot(name) {
212            return Ok(self.this());
213        }
214        if is_dotdot(name) {
215            return Ok(self.parent().unwrap_or(self.this()));
216        }
217
218        self.inner.lookup_child(self, name)
219    }

233    }
234

236        self.inner.revalidate_exists(name, child)
237    }
238
239    fn revalidate_absent(&self, name: &str) -> bool {
240        self.inner.revalidate_absent(name)
241    }
242
243    fn seek_end(&self) -> Option<usize> {
244        // Seeking directories under `/proc` with `SEEK_END` will start from zero.

246    }

249pub(in procfs) trait ProcDirOps: Sync + Send + Sized {
250    /// Returns the thread whose credentials own this procfs inode.
251    fn owner_thread(&self) -> Option<Arc<Thread>> {
252        None
253    }
254
255    /// Looks up a child inode in `this_dir` by name.
256    fn lookup_child(&self, this_dir: &ProcDir<Self>, name: &str) -> Result<Arc<dyn Inode>>;
257
258    /// Visits readdir entries whose continuation offsets are strictly greater than `offset`.
259    ///
260    /// Build entries with helpers from the [`readdir`] submodule and forward to
261    /// [`readdir::visit_readdir_entries`]. Common patterns:
262    ///
263    /// - Static or sequential children: [`readdir::sequential_readdir_entries`]
264    ///   or the [`readdir::visit_listed_entries`] convenience wrapper.
265    /// - Children identified by a stable integer key: [`readdir::keyed_readdir_entries`].
266    /// - Mixed children: invoke [`readdir::visit_readdir_entries`] more than
267    ///   once per call, once per group. See `RootDirOps`.
268    ///
269    /// The default implementation reports no entries.
270    fn visit_entries_from_offset<'a, F>(&'a self, _offset: usize, _visit_fn: F) -> Result<()>
271    where
272        F: FnMut(ReaddirEntry<'a>) -> Result<()>,
273    {
274        Ok(())
275    }
276
277    /// Returns the dentry cache revalidation policy for this directory.
278    fn revalidation_policy(&self) -> RevalidationPolicy {
279        RevalidationPolicy::empty()
280    }
281
282    /// Checks whether a positive lookup result still exists.
283    ///
284    /// This method is only consulted when [`Self::revalidation_policy`] includes
285    /// `REVALIDATE_EXISTS`.
286    #[must_use]
287    fn revalidate_exists(&self, _name: &str, _child: &dyn Inode) -> bool {
288        true
289    }
290
291    /// Checks whether a negative lookup result is still absent.
292    ///
293    /// This method is only consulted when [`Self::revalidation_policy`] includes
294    /// `REVALIDATE_ABSENT`.
295    #[must_use]
296    fn revalidate_absent(&self, _name: &str) -> bool {
297        true
298    }
299}

301/// A statically declared procfs child entry.
302///
303/// The tuple stores the exported filename,
304/// the inode type reported to [`FileOps::readdir_at`],
305/// and the constructor used by lookup paths to instantiate the inode.
306pub(in procfs) type StaticDirEntry<Fp> = (&'static str, InodeType, Fp);
307
308/// Looks up a statically declared child from a `StaticDirEntry` table.
309///
310/// The `constructor_adaptor` receives the stored constructor payload
311/// and can bind any per-directory context that is needed at the call site.
312pub(in procfs) fn lookup_child_from_table<Fp, F>(
313    name: &str,
314    table: &[StaticDirEntry<Fp>],
315    constructor_adaptor: F,
316) -> Option<Arc<dyn Inode>>
317where
318    Fp: Copy,
319    F: FnOnce(Fp) -> Arc<dyn Inode>,
320{
321    table
322        .iter()
323        .find(|(child_name, _, _)| *child_name == name)
324        .map(|(_, _, child_constructor)| (constructor_adaptor)(*child_constructor))
325}
326
327/// Converts a static procfs child table into listed entries.
328pub(in procfs) fn listed_entries_from_table<'a, Fp>(
329    table: &'a [StaticDirEntry<Fp>],
330) -> impl Iterator<Item = ListedEntry<'a>> + 'a
331where
332    Fp: Copy + 'a,
333{
334    table
335        .iter()
336        .map(|(name, type_, _)| ListedEntry::new(*name, *type_))
337}

339mod readdir {
340    //! `readdir` resumption types and strategies.
341    //!
342    //! When user space calls `getdents`, it passes an offset and expects the
343    //! kernel to emit entries strictly past it. Each emitted entry carries a
344    //! continuation offset that user space remembers for the next call.
345    //!
346    //! # Vocabulary
347    //!
348    //! - **Offset** - the cursor passed to `FileOps::readdir_at`. Maps to
349    //!   `dirent::d_off`. Offsets `1` and `2` are reserved for `.` and `..`.
350    //!
351    //! - **Key** - an integer that uniquely identifies one dynamic child
352    //!   (PID, TID, FD). Comes from kernel state, not iteration order.
353    //!
354    //! - **Continuation offset** - the offset stored in [`ReaddirEntry`].
355    //!   Computed by [`sequential_readdir_entries`] from index or
356    //!   [`keyed_readdir_entries`] from key:
357    //!
358    //!   ```text
359    //!   continuation_offset = first_entry_offset + index + 1   // sequential_*
360    //!   continuation_offset = first_entry_offset + key   + 1   // keyed_*
361    //!   ```
362
363    use super::*;
364
365    /// A listed entry that is currently visible in a procfs directory.
366    ///
367    /// Unlike [`ReaddirEntry`], this type does not carry a continuation offset.
368    /// It is intended for higher-level directory descriptions,
369    /// while [`ReaddirEntry`] derives offsets from the iteration strategy.
370    pub(in procfs) struct ListedEntry<'a> {
371        pub(super) name: Cow<'a, str>,
372        pub(super) type_: InodeType,
373    }
374
375    impl<'a> ListedEntry<'a> {
376        /// Creates a listed entry with the given filename and inode type.
377        pub(in procfs) fn new(name: impl Into<Cow<'a, str>>, type_: InodeType) -> Self {
378            Self {
379                name: name.into(),
380                type_,
381            }
382        }
383
384        /// Returns the filename of the listed entry.
385        pub(in procfs) fn name(&self) -> &str {
386            self.name.as_ref()
387        }
388    }
389
390    /// A directory entry emitted by [`ProcDirOps::visit_entries_from_offset`].
391    ///
392    /// The `next_offset` is the continuation offset that should be used as the
393    /// starting point of the next [`FileOps::readdir_at`] call.
394    pub(in procfs) struct ReaddirEntry<'a> {
395        pub(super) name: Cow<'a, str>,
396        pub(super) ino: u64,
397        pub(super) type_: InodeType,
398        pub(super) next_offset: usize,
399    }
400
401    impl<'a> ReaddirEntry<'a> {
402        /// Creates an entry with an explicit continuation offset.
403        pub(super) fn new(
404            name: impl Into<Cow<'a, str>>,
405            ino: u64,
406            type_: InodeType,
407            next_offset: usize,
408        ) -> Self {
409            Self {
410                name: name.into(),
411                ino,
412                type_,
413                next_offset,
414            }
415        }
416    }
417
418    /// A placeholder inode number for procfs entries.
419    ///
420    /// [`FileOps::readdir_at`] reports procfs entries without instantiating child inodes first,
421    /// so the inode number returned here is only a placeholder.
422    /// Lookup still creates the real child inode on demand when the path is opened.
423    //
424    // TODO: Adopt deterministic inode numbers derived from stable entity identity
425    // (for example, PID) as the long-term fix for `d_ino`/`st_ino` consistency.
426    pub(super) const PLACEHOLDER_DIRENT_INO: u64 = 2;
427
428    /// Builds sequential directory entries whose continuation offsets increase by one.
429    ///
430    /// This helper is appropriate for directories whose visible children already
431    /// have a deterministic iteration order.
432    /// It reserves offsets `1` and `2` for `.` and `..`,
433    /// so callers typically pass `2` as `first_entry_offset`.
434    pub(in procfs) fn sequential_readdir_entries<'a, I>(
435        offset: usize,
436        first_entry_offset: usize,
437        entries: I,
438    ) -> impl Iterator<Item = ReaddirEntry<'a>>
439    where
440        I: IntoIterator<Item = ListedEntry<'a>>,
441    {
442        entries
443            .into_iter()
444            .enumerate()
445            .filter_map(move |(idx, entry)| {
446                let next_offset = first_entry_offset.saturating_add(idx).saturating_add(1);
447                let ListedEntry { name, type_ } = entry;
448                (next_offset > offset)
449                    .then(|| ReaddirEntry::new(name, PLACEHOLDER_DIRENT_INO, type_, next_offset))
450            })
451    }
452
453    /// Builds directory entries whose offsets are derived from stable integer keys.
454    ///
455    /// This is useful for procfs directories such as `/proc`, `/proc/[pid]/task`,
456    /// and `/proc/[pid]/fd`, where Linux uses identifiers like PID, TID, or FD to
457    /// keep iteration stable across mutations.
458    ///
459    /// Names are produced only after the corresponding key passes the offset check.
460    /// The resulting continuation offset for key `k` is `first_entry_offset + k + 1`,
461    /// so callers should provide non-negative keys in increasing order.
462    /// Keys do not need to be dense,
463    /// but each key must stay stable for the duration of one directory snapshot.
464    pub(in procfs) fn keyed_readdir_entries<'a, I, F>(
465        offset: usize,
466        first_entry_offset: usize,
467        keys: I,
468        mut entry_fn: F,
469    ) -> impl Iterator<Item = ReaddirEntry<'a>>
470    where
471        I: IntoIterator<Item = usize>,
472        F: FnMut(usize) -> ListedEntry<'a>,
473    {
474        let start_key = offset.saturating_sub(first_entry_offset);
475        keys.into_iter().filter_map(move |key| {
476            let next_offset = first_entry_offset.saturating_add(key).saturating_add(1);
477            (key >= start_key && next_offset > offset).then(|| {
478                let ListedEntry { name, type_ } = entry_fn(key);
479                ReaddirEntry::new(name, PLACEHOLDER_DIRENT_INO, type_, next_offset)
480            })
481        })
482    }
483
484    /// Visits the given [`ReaddirEntry`] values in order, calling `visit_fn` for each one.
485    pub(in procfs) fn visit_readdir_entries<'a, I, F>(entries: I, mut visit_fn: F) -> Result<()>
486    where
487        I: IntoIterator<Item = ReaddirEntry<'a>>,
488        F: FnMut(ReaddirEntry<'a>) -> Result<()>,
489    {
490        for entry in entries {
491            visit_fn(entry)?;
492        }
493
494        Ok(())
495    }
496
497    /// Visits listed entries using sequential continuation offsets.
498    pub(in procfs) fn visit_listed_entries<'a, I, F>(
499        offset: usize,
500        listed: I,
501        visit_fn: F,
502    ) -> Result<()>
503    where
504        I: IntoIterator<Item = ListedEntry<'a>>,
505        F: FnMut(ReaddirEntry<'a>) -> Result<()>,
506    {
507        visit_readdir_entries(sequential_readdir_entries(offset, 2, listed), visit_fn)
508    }
509}
