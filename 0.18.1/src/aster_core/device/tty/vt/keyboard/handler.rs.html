29impl InputHandlerClass for VtKeyboardHandlerClass {
30    fn name(&self) -> &str {
31        "vt_keyboard"
32    }
33
34    fn connect(&self, dev: Arc<dyn InputDevice>) -> Result<Arc<dyn InputHandler>, ConnectError> {
35        let capability = dev.capability();
36        if !capability.look_like_keyboard() {
37            return Err(ConnectError::IncompatibleDevice);
38        }
39        ostd::info!(
40            "Virtual terminal keyboard handler connected to device: {}",
41            dev.name()
42        );
43        Ok(Arc::new(VtKeyboardHandler))
44    }
45
46    fn disconnect(&self, dev: &Arc<dyn InputDevice>) {
47        ostd::info!(
48            "Virtual terminal keyboard handler disconnected from device: {}",
49            dev.name()
50        );
51    }
52}

54/// The state of modifier keys (Shift, Ctrl, Alt).
55///
56/// It is shared across all virtual terminals.
57static MODIFIER_KEYS_STATE: ModifierKeysState = ModifierKeysState::new();
58
59/// A virtual terminal keyboard handler that handles input from a specific device.
60#[derive(Debug)]
61struct VtKeyboardHandler;
62
63impl VtKeyboardHandler {
64    fn handle_key_event(&self, keycode: KeyCode, key_status: KeyStatus) {
65        ostd::debug!(
66            "Virtual terminal keyboard handler received key event: keycode={:?}, status={:?}",
67            keycode,
68            key_status
69        );
70
71        // Lock order: console keyboard -> console backend
72
73        let vtm = VT_MANAGER.get().expect("`VT_MANAGER` is not initialized");
74        let vt = vtm.active_vt();
75        let vt_console = vt.driver().vt_console();
76        let mut vt_keyboard = vt_console.lock_keyboard();
77        let keyboard_mode = vt_keyboard.mode();
78
79        if keyboard_mode == KeyboardMode::MediumRaw {
80            self.handle_medium_mode(keycode, key_status, vt);
81        }
82
83        // Modifier keys are handled based on the keycode instead of the keymap.
84        //
85        // This is important because the keymap lookup depends on the current
86        // modifier state. If a modifier is released while multiple modifiers
87        // are active (e.g. Shift+Alt), the keymap may not contain an entry for
88        // that modifier combination. In that case the lookup would return
89        // `KeySym::NoOp`, and the release event would be ignored, causing the
90        // modifier state to become stuck.
91        //
92        // By detecting modifiers directly from the keycode, we ensure that
93        // modifier press/release events are always processed correctly,
94        // regardless of the current modifier combination.
95        let key_sym = if let Some(modifier) = self.keycode_to_modifier_key(keycode) {
96            KeySym::Modifier(modifier)
97        } else {
98            get_keysym(MODIFIER_KEYS_STATE.flags(), keycode)
99        };
100
101        // In Medium Raw, Raw, or Off mode, ignore non-modifier keys
102        if (keyboard_mode == KeyboardMode::MediumRaw
103            || keyboard_mode == KeyboardMode::Raw
104            || keyboard_mode == KeyboardMode::Off)
105            && !matches!(key_sym, KeySym::Modifier(_))
106        {
107            // FIXME: For now, we don't support SAK (secure attention key).
108            // SAK is allowed even in Raw mode.
109            return;
110        }
111
112        // Ignore key release events except for modifier keys
113        if key_status == KeyStatus::Released && !matches!(key_sym, KeySym::Modifier(_)) {
114            return;
115        }
116
117        match key_sym {
118            KeySym::Char(ch) => self.push_input(ch, vt, &vt_keyboard),
119            KeySym::Letter(ch) => self.handle_letter(ch, vt, &vt_keyboard),
120            KeySym::Meta(ch) => self.handle_meta(ch, vt, &vt_keyboard),
121            KeySym::Modifier(modifier_key) => {
122                self.handle_modifier(modifier_key, key_status == KeyStatus::Pressed)
123            }
124            KeySym::Numpad(numpad_key) => self.handle_numpad(numpad_key, vt, &vt_keyboard),
125            KeySym::Function(id) => self.handle_function(id, vt),
126            KeySym::Cursor(cursor_key) => self.handle_cursor(cursor_key, vt, &vt_keyboard),
127            KeySym::Special(SpecialHandler::DecreaseConsole) => {
128                drop(vt_keyboard);
129                let mut guard = vtm.lock();
130
131                if let Err(e) = guard.dec_console() {
132                    ostd::warn!("VT decreasing failed: {:?}", e);
133                }
134            }
135            KeySym::Special(SpecialHandler::IncreaseConsole) => {
136                drop(vt_keyboard);
137                let mut guard = vtm.lock();
138
139                if let Err(e) = guard.inc_console() {
140                    ostd::warn!("VT increasing failed: {:?}", e);
141                }
142            }
143            KeySym::Special(handler) => self.handle_special(handler, vt, &mut vt_keyboard),
144            KeySym::SwitchVt(index) => {
145                drop(vt_keyboard);
146                let mut guard = vtm.lock();
147
148                if let Err(e) = guard.switch_vt(index) {
149                    ostd::warn!("VT switching failed: {:?}", e);
150                }
151            }
152            KeySym::AltNumpad(_) => {
153                // TODO: Implement Alt+Numpad input
154                ostd::warn!("Alt+Numpad input is not implemented yet");
155            }
156            KeySym::NoOp => {
157                // Do nothing
158            }
159        }
160    }

162    fn keycode_to_modifier_key(&self, keycode: KeyCode) -> Option<ModifierKey> {
163        match keycode {
164            KeyCode::LeftShift | KeyCode::RightShift => Some(ModifierKey::Shift),
165            KeyCode::LeftCtrl | KeyCode::RightCtrl => Some(ModifierKey::Ctrl),
166            KeyCode::LeftAlt | KeyCode::RightAlt => Some(ModifierKey::Alt),
167            _ => None,
168        }
169    }
170
171    /// Handles key events in Medium Raw mode.
172    ///
173    /// Reference: <https://elixir.bootlin.com/linux/v6.13/source/drivers/tty/vt/keyboard.c#L1436-L1454>
174    fn handle_medium_mode(&self, keycode: KeyCode, key_status: KeyStatus, vt: &Tty<VtDriver>) {
175        const UP_FLAG: u8 = 0x80;
176
177        let up_bit = if matches!(key_status, KeyStatus::Pressed) {
178            0
179        } else {
180            UP_FLAG
181        };
182        let kc = keycode as u16;
183
184        let _ = if kc < 128 {
185            vt.push_input(&[(kc as u8) | up_bit])
186        } else {
187            let b0 = up_bit;
188            let b1 = ((kc >> 7) as u8) | UP_FLAG;
189            let b2 = (kc as u8) | UP_FLAG;
190            vt.push_input(&[b0, b1, b2])
191        };
192    }
193
194    fn handle_letter(&self, ch: char, vt: &Tty<VtDriver>, vt_keyboard: &VtKeyboard) {
195        let caps_on = vt_keyboard
196            .lock_key_flags()
197            .contains(LockKeyFlags::CAPS_LOCK);
198
199        let out = if caps_on && ch.is_ascii_alphabetic() {
200            if ch.is_ascii_lowercase() {
201                ch.to_ascii_uppercase()
202            } else {
203                ch.to_ascii_lowercase()
204            }
205        } else {
206            ch
207        };
208
209        self.push_input(out, vt, vt_keyboard);
210    }
211
212    fn handle_meta(&self, ch: char, vt: &Tty<VtDriver>, vt_keyboard: &VtKeyboard) {
213        debug_assert!(ch.is_ascii());
214
215        let mode_flags = vt_keyboard.mode_flags();
216        let _ = if mode_flags.contains(KeyboardModeFlags::META) {
217            vt.push_input(&[0x1b, ch as u8])
218        } else {
219            vt.push_input(&[ch as u8])
220        };
221    }
222
223    fn handle_modifier(&self, modifier_key: ModifierKey, is_pressed: bool) {
224        match modifier_key {
225            ModifierKey::Shift => {
226                if is_pressed {
227                    MODIFIER_KEYS_STATE.press(ModifierKeyFlags::SHIFT);
228                } else {
229                    MODIFIER_KEYS_STATE.release(ModifierKeyFlags::SHIFT);
230                }
231            }
232            ModifierKey::Ctrl => {
233                if is_pressed {
234                    MODIFIER_KEYS_STATE.press(ModifierKeyFlags::CTRL);
235                } else {
236                    MODIFIER_KEYS_STATE.release(ModifierKeyFlags::CTRL);
237                }
238            }
239            ModifierKey::Alt => {
240                if is_pressed {
241                    MODIFIER_KEYS_STATE.press(ModifierKeyFlags::ALT);
242                } else {
243                    MODIFIER_KEYS_STATE.release(ModifierKeyFlags::ALT);
244                }
245            }
246        }
247    }
248
249    fn handle_numpad(&self, numpad_key: NumpadKey, vt: &Tty<VtDriver>, vt_keyboard: &VtKeyboard) {
250        let mode_flags = vt_keyboard.mode_flags();
251
252        let shift_down = MODIFIER_KEYS_STATE
253            .flags()
254            .contains(ModifierKeyFlags::SHIFT);
255        let num_lock_on = vt_keyboard
256            .lock_key_flags()
257            .contains(LockKeyFlags::NUM_LOCK);
258        let application_mode = mode_flags.contains(KeyboardModeFlags::APPLICATION);
259
260        if application_mode && !shift_down {
261            let _ = match numpad_key {
262                NumpadKey::Num0 => vt.push_input(b"\x1bOp"),     // p
263                NumpadKey::Num1 => vt.push_input(b"\x1bOq"),     // q
264                NumpadKey::Num2 => vt.push_input(b"\x1bOr"),     // r
265                NumpadKey::Num3 => vt.push_input(b"\x1bOs"),     // s
266                NumpadKey::Num4 => vt.push_input(b"\x1bOt"),     // t
267                NumpadKey::Num5 => vt.push_input(b"\x1bOu"),     // u
268                NumpadKey::Num6 => vt.push_input(b"\x1bOv"),     // v
269                NumpadKey::Num7 => vt.push_input(b"\x1bOw"),     // w
270                NumpadKey::Num8 => vt.push_input(b"\x1bOx"),     // x
271                NumpadKey::Num9 => vt.push_input(b"\x1bOy"),     // y
272                NumpadKey::Plus => vt.push_input(b"\x1bOl"),     // l
273                NumpadKey::Minus => vt.push_input(b"\x1bOS"),    // S
274                NumpadKey::Asterisk => vt.push_input(b"\x1bOR"), // R
275                NumpadKey::Slash => vt.push_input(b"\x1bOQ"),    // Q
276                NumpadKey::Enter => vt.push_input(b"\x1bOM"),    // M
277                NumpadKey::Dot => vt.push_input(b"\x1bOn"),      // n
278            };
279            return;
280        }
281
282        if !num_lock_on {
283            match numpad_key {
284                NumpadKey::Num0 => self.handle_function(FuncId::Insert, vt),
285                NumpadKey::Num1 => self.handle_function(FuncId::Select, vt),
286                NumpadKey::Num2 => self.handle_cursor(CursorKey::Down, vt, vt_keyboard),
287                NumpadKey::Num3 => self.handle_function(FuncId::Next, vt),
288                NumpadKey::Num4 => self.handle_cursor(CursorKey::Left, vt, vt_keyboard),
289                NumpadKey::Num5 => {
290                    let _ = if application_mode {
291                        vt.push_input(b"\x1bOG")
292                    } else {
293                        vt.push_input(b"\x1b[G")
294                    };
295                }
296                NumpadKey::Num6 => self.handle_cursor(CursorKey::Right, vt, vt_keyboard),
297                NumpadKey::Num7 => self.handle_function(FuncId::Find, vt),
298                NumpadKey::Num8 => self.handle_cursor(CursorKey::Up, vt, vt_keyboard),
299                NumpadKey::Num9 => self.handle_function(FuncId::Prior, vt),
300                NumpadKey::Dot => self.handle_function(FuncId::Remove, vt),
301                _ => {}
302            }
303            return;
304        }
305
306        let ch = match numpad_key {
307            NumpadKey::Num0 => '0',
308            NumpadKey::Num1 => '1',
309            NumpadKey::Num2 => '2',
310            NumpadKey::Num3 => '3',
311            NumpadKey::Num4 => '4',
312            NumpadKey::Num5 => '5',
313            NumpadKey::Num6 => '6',
314            NumpadKey::Num7 => '7',
315            NumpadKey::Num8 => '8',
316            NumpadKey::Num9 => '9',
317            NumpadKey::Dot => '.',
318            NumpadKey::Enter => '\r',
319            NumpadKey::Plus => '+',
320            NumpadKey::Minus => '-',
321            NumpadKey::Asterisk => '*',
322            NumpadKey::Slash => '/',
323        };
324
325        let _ = vt.push_input(&[ch as u8]);
326
327        if matches!(numpad_key, NumpadKey::Enter) && mode_flags.contains(KeyboardModeFlags::CRLF) {
328            let _ = vt.push_input(b"\n");
329        }
330    }
331
332    fn handle_function(&self, id: FuncId, vt: &Tty<VtDriver>) {
333        if let Some(seq) = id.to_function_string() {
334            let _ = vt.push_input(seq);
335        }
336    }
337
338    fn handle_cursor(&self, cursor_key: CursorKey, vt: &Tty<VtDriver>, vt_keyboard: &VtKeyboard) {
339        let cursor_key_mode = vt_keyboard
340            .mode_flags()
341            .contains(KeyboardModeFlags::CURSOR_KEY);
342
343        let _ = if cursor_key_mode {
344            match cursor_key {
345                CursorKey::Up => vt.push_input(b"\x1bOA"),
346                CursorKey::Down => vt.push_input(b"\x1bOB"),
347                CursorKey::Left => vt.push_input(b"\x1bOD"),
348                CursorKey::Right => vt.push_input(b"\x1bOC"),
349            }
350        } else {
351            match cursor_key {
352                CursorKey::Up => vt.push_input(b"\x1b[A"),
353                CursorKey::Down => vt.push_input(b"\x1b[B"),
354                CursorKey::Left => vt.push_input(b"\x1b[D"),
355                CursorKey::Right => vt.push_input(b"\x1b[C"),
356            }
357        };
358    }
359
360    fn handle_special(
361        &self,
362        handler: SpecialHandler,
363        vt: &Tty<VtDriver>,
364        vt_keyboard: &mut VtKeyboard,
365    ) {
366        match handler {
367            SpecialHandler::ToggleCapsLock => {
368                vt_keyboard.toggle_lock_keys(LockKeyFlags::CAPS_LOCK);
369            }
370            SpecialHandler::ToggleNumLock => {
371                let application_mode = vt_keyboard
372                    .mode_flags()
373                    .contains(KeyboardModeFlags::APPLICATION);
374                if application_mode {
375                    let _ = vt.push_input(b"\x1bOP");
376                } else {
377                    let _ = vt.push_input(b"\x1b[O");
378                    vt_keyboard.toggle_lock_keys(LockKeyFlags::NUM_LOCK);
379                }
380            }
381            SpecialHandler::ToggleBareNumLock => {
382                vt_keyboard.toggle_lock_keys(LockKeyFlags::NUM_LOCK);
383            }
384            SpecialHandler::ToggleScrollLock => {
385                // TODO: Scroll Lock will affect the scrolling behavior of the console.
386                // For now, we just toggle the state.
387                vt_keyboard.toggle_lock_keys(LockKeyFlags::SCROLL_LOCK);
388            }
389            SpecialHandler::Enter => {
390                let _ = if vt_keyboard.mode_flags().contains(KeyboardModeFlags::CRLF) {
391                    vt.push_input(b"\r\n")
392                } else {
393                    vt.push_input(b"\r")
394                };
395            }
396            SpecialHandler::Reboot => {
397                ostd::power::restart(ostd::power::ExitCode::Success);
398            }
399            SpecialHandler::DecreaseConsole | SpecialHandler::IncreaseConsole => unreachable!(),
400            SpecialHandler::ScrollBackward
401            | SpecialHandler::ScrollForward
402            | SpecialHandler::ShowMem
403            | SpecialHandler::ShowState
404            | SpecialHandler::Compose => {
405                ostd::warn!("VT keyboard action {:?} is not implemented yet", handler);
406            }
407        }
408    }

410    /// Adds the character to the input buffer of the virtual terminal.
411    ///
412    /// Reference: <https://elixir.bootlin.com/linux/v6.13/source/drivers/tty/vt/keyboard.c#L675>.
413    fn push_input(&self, ch: char, vt: &Tty<VtDriver>, vt_keyboard: &VtKeyboard) {
414        let keyboard_mode = vt_keyboard.mode();
415
416        if keyboard_mode == KeyboardMode::Unicode {
417            let mut buf = [0u8; 4];
418            let s = ch.encode_utf8(&mut buf);
419            let _ = vt.push_input(s.as_bytes());
420        } else if ch.is_ascii() {
421            let _ = vt.push_input(&[ch as u8]);
422        }
423    }
424}
425
426impl InputHandler for VtKeyboardHandler {
427    fn handle_events(&self, events: &[InputEvent]) {
428        for event in events {
429            match event {
430                InputEvent::Key(keycode, key_status) => {
431                    self.handle_key_event(*keycode, *key_status)
432                }
433                InputEvent::Sync(_) => {
434                    // Nothing to do
435                }
436                _ => {
437                    ostd::warn!(
438                        "Virtual terminal keyboard handler received unsupported event: {:?}",
439                        event
440                    );
441                }
442            }
443        }
444    }
445}
