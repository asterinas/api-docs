23/// An opened executable whose format has not yet been detected.
24///
25/// The file has already passed executable-file permission and type checks,
26/// but it is not yet known to be an ELF image or a shebang script.
27/// It also retains the path-access information needed if detection identifies a script.
28pub(crate) struct UndetectedExecutable {
29    file: Arc<dyn FileLike>,
30    script_path: ShebangScriptPath,
31}
32
33impl UndetectedExecutable {
34    /// Opens an executable without detecting its format.
35    ///
36    /// `script_path` describes whether a shebang interpreter can reopen this file after exec.
37    pub(crate) fn open(path: Path, script_path: ShebangScriptPath) -> Result<Self> {
38        let file = open_executable_file(path)?;
39        Ok(Self { file, script_path })
40    }
41
42    /// Returns the path of the opened executable.
43    pub(crate) fn path(&self) -> &Path {
44        self.file.path()
45    }
46
47    /// Consumes this executable and classifies it as an ELF image or a shebang script.
48    fn detect_format(self) -> Result<DetectedExecutable> {
49        let mut first_page = Box::new([0u8; PAGE_SIZE]);
50        let len = self.file.read_bytes_at(0, &mut *first_page)?;
51
52        let Some(interpreter_argv) = parse_shebang_line(&first_page[..len])? else {
53            let headers = ElfHeaders::parse(&first_page[..len])?;
54            return Ok(DetectedExecutable::Elf {
55                file: self.file,
56                headers,
57            });
58        };
59
60        // A shebang interpreter must reopen the original script path after exec. A path
61        // synthesized from a close-on-exec file descriptor will no longer be accessible then,
62        // so fail before changing the process state.
63        // Reference: <https://elixir.bootlin.com/linux/v6.8/source/fs/binfmt_script.c#L93>
64        let script_path = match self.script_path {
65            ShebangScriptPath::Accessible(path) => path,
66            ShebangScriptPath::Inaccessible => {
67                return_errno_with_message!(
68                    Errno::ENOENT,
69                    "the script path is inaccessible after closing close-on-exec file descriptors"
70                );
71            }
72        };
73
74        Ok(DetectedExecutable::Script {
75            interpreter_argv,
76            script_path,
77        })
78    }
79}

81/// An executable classified as an ELF image or a shebang script.
82///
83/// Each variant contains the validated metadata required by the next loading step.
84/// An ELF image retains its opened file and parsed headers,
85/// while a script retains its interpreter arguments and the path passed to that interpreter.
86enum DetectedExecutable {
87    Elf {
88        file: Arc<dyn FileLike>,
89        headers: ElfHeaders,
90    },
91    Script {
92        interpreter_argv: Vec<CString>,
93        script_path: CString,
94    },
95}
96
97/// Opens a path as an executable file.
98fn open_executable_file(path: Path) -> Result<Arc<dyn FileLike>> {
99    check_executable_inode(path.inode().as_ref())?;
100
101    let file: Arc<dyn FileLike> = Arc::new(InodeHandle::new_unchecked_access(
102        path,
103        // Reference: <https://elixir.bootlin.com/linux/v7.0/source/fs/exec.c#L769>.
104        AccessMode::O_RDONLY,
105        StatusFlags::empty(),
106    )?);
107    notify::on_open(file.common());
108
109    Ok(file)
110}

112/// Describes how a shebang interpreter can access the script after exec.
113pub(crate) enum ShebangScriptPath {
114    /// A path accessible to the interpreter after exec.
115    Accessible(CString),
116    /// A path inaccessible to the interpreter after exec.
117    Inaccessible,
118}

120/// Represents an executable file that is ready to be loaded into memory and executed.
121///
122/// This struct encapsulates the ELF file to be executed along with its header data,
123/// the `argv` and the `envp` which is required for the program execution.
124pub(super) struct ProgramToLoad {
125    elf_file: Arc<dyn FileLike>,
126    elf_headers: ElfHeaders,
127    argv: Vec<CString>,
128    envp: Vec<CString>,
129}
130
131impl ProgramToLoad {
132    /// Constructs a new `ProgramToLoad` from an opened executable and handles shebang
133    /// interpretation if necessary.
134    pub(super) fn from_executable(
135        mut executable: UndetectedExecutable,
136        path_resolver: &PathResolver,
137        mut argv: Vec<CString>,
138        envp: Vec<CString>,
139    ) -> Result<Self> {
140        // A limit to the recursion depth of shebang executables.
141        //
142        // If the interpreter is a shebang, then recursion will be triggered. If it loops, we
143        // should fail. We follow the same limit as Linux.
144        let mut recursive_limit = 5;
145
146        let (elf_file, elf_headers) = loop {
147            let (mut new_argv, current_script_path) = match executable.detect_format()? {
148                DetectedExecutable::Elf { file, headers } => break (file, headers),
149                DetectedExecutable::Script {
150                    interpreter_argv,
151                    script_path,
152                } => (interpreter_argv, script_path),
153            };
154
155            if recursive_limit == 0 {
156                return_errno_with_message!(Errno::ELOOP, "the recursive limit is reached");
157            }
158            recursive_limit -= 1;
159
160            let interpreter_filename = new_argv[0].clone();
161            let interpreter = {
162                let filename = interpreter_filename.to_str()?;
163                let fs_path = FsPath::try_from(filename)?;
164                path_resolver.lookup(&fs_path)?
165            };
166
167            // Update the argument list and the executable inode. Then, try again.
168            //
169            // The interpreter receives the script path as its first argument,
170            // regardless of the caller-supplied `argv[0]`. For example,
171            // `exec -a custom ./script arg` with `#!/bin/sh` must execute
172            // `/bin/sh ./script arg`, not `/bin/sh custom arg`.
173            new_argv.push(current_script_path);
174            new_argv.extend(argv.into_iter().skip(1));
175            argv = new_argv;
176            executable = UndetectedExecutable::open(
177                interpreter,
178                ShebangScriptPath::Accessible(interpreter_filename),
179            )?;
180        };
181
182        Ok(Self {
183            elf_file,
184            elf_headers,
185            argv,
186            envp,
187        })
188    }
189
190    /// Returns the `Path` of the ELF file that will be loaded.
191    pub(super) fn elf_path(&self) -> &Path {
192        self.elf_file.path()
193    }
194
195    /// Loads the executable into the specified virtual memory space.
196    ///
197    /// Returns the information about the ELF loading process.
198    pub(super) fn load_to_vmar(
199        self,
200        vmar: &Vmar,
201        path_resolver: &PathResolver,
202    ) -> Result<ElfLoadInfo> {
203        load_elf_to_vmar(
204            vmar,
205            self.elf_file,
206            path_resolver,
207            self.elf_headers,
208            self.argv,
209            self.envp,
210        )
211    }
212}

214fn check_executable_inode(inode: &dyn Inode) -> Result<()> {
215    if inode.type_().is_directory() {
216        return_errno_with_message!(Errno::EISDIR, "the inode is a directory");
217    }
218
219    if inode.type_() == InodeType::SymLink {
220        return_errno_with_message!(Errno::ELOOP, "the inode is a symbolic link");
221    }
222
223    if !inode.type_().is_regular_file() {
224        return_errno_with_message!(Errno::EACCES, "the inode is not a regular file");
225    }
226
227    if inode.check_permission(Permission::MAY_EXEC).is_err() {
228        return_errno_with_message!(Errno::EACCES, "the inode is not executable");
229    }
230
231    Ok(())
