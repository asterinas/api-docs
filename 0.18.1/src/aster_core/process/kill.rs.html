15/// Sends a signal to a process, using the current process as the sender.
16///
17/// The credentials of the current process will be checked to determine
18/// if it is authorized to send the signal to this particular target process.
19///
20/// If `signal` is `None`, this method will only check permission without sending
21/// any signal.
22pub(crate) fn kill(pid: Pid, signal: Option<Box<dyn Signal>>, ctx: &Context) -> Result<()> {
23    // Fast path: If the signal is sent to self, we can skip most checks.
24    if pid == ctx.process.pid() {
25        let Some(signal) = signal else {
26            return Ok(());
27        };
28
29        if !ctx.posix_thread.has_signal_blocked(signal.num()) {
30            // Killing the current thread does not raise any permission issues.
31            ctx.posix_thread.enqueue_signal(signal);
32            return Ok(());
33        }
34
35        return kill_process(ctx.process.as_ref(), Some(signal), ctx);
36    }
37
38    // Slow path
39    let process = pid_table::pid_table_mut()
40        .get_process(pid)
41        .ok_or_else(|| Error::with_message(Errno::ESRCH, "the target process does not exist"))?;
42
43    kill_process(&process, signal, ctx)
44}

46/// Sends a signal to all processes in a group, using the current process
47/// as the sender.
48///
49/// The credentials of the current process will be checked to determine
50/// if it is authorized to send the signal to the target group.
51///
52/// If `signal` is `None`, this method will only check permission without sending
53/// any signal.
54pub(crate) fn kill_group<S: Signal + Clone>(
55    pgid: Pgid,
56    signal: Option<S>,
57    ctx: &Context,
58) -> Result<()> {
59    let process_group = pid_table::pid_table_mut()
60        .get_process_group(&pgid)
61        .ok_or_else(|| Error::with_message(Errno::ESRCH, "the target group does not exist"))?;
62
63    let mut result = Ok(());
64
65    let inner = process_group.lock();
66    for process in inner.iter() {
67        let res = kill_process(
68            &process,
69            signal.clone().map(|s| Box::new(s) as Box<dyn Signal>),
70            ctx,
71        );
72        if res.is_err_and(|err| err.error() != Errno::EPERM) {
73            result = res;
74        }
75    }
76
77    result
78}

80/// Sends a signal to a target thread, using the current process
81/// as the sender.
82///
83/// Checks the target thread belongs to the target process if `tgid` is provided.
84///
85/// If `signal` is `None`, this method will only check permission without sending
86/// any signal.
87pub(crate) fn tgkill(
88    tid: Tid,
89    tgid: Option<Pid>,
90    signal: Option<Box<dyn Signal>>,
91    ctx: &Context,
92) -> Result<()> {
93    let thread = pid_table::pid_table_mut()
94        .get_thread(tid)
95        .ok_or_else(|| Error::with_message(Errno::ESRCH, "the target thread does not exist"))?;
96    let target_posix_thread = thread.as_posix_thread().unwrap();
97
98    if let Some(tgid) = tgid {
99        // Check the TGID
100        let pid = target_posix_thread.process().pid();
101        if pid != tgid {
102            return_errno_with_message!(
103                Errno::ESRCH,
104                "the combination of the TGID and the TID is not valid"
105            );
106        }
107    }
108
109    // Check permission
110    let signum = signal.as_ref().map(|signal| signal.num());
111    check_signal_perm(target_posix_thread, ctx, signum)?;
112
113    if thread.is_exited() {
114        return Ok(());
115    }
116
117    if let Some(signal) = signal {
118        // We've checked the permission issues above.
119        // FIXME: We should take some lock while checking the permission to avoid race conditions.
120        target_posix_thread.enqueue_signal(signal);
121    }
122
123    Ok(())
124}

126/// Sends a signal to all processes except current process and init process, using
127/// the current process as the sender.
128///
129/// The credentials of the current process will be checked to determine
130/// if it is authorized to send the signal to the target group.
131pub(crate) fn kill_all<S: Signal + Clone>(signal: Option<S>, ctx: &Context) -> Result<()> {
132    let mut result = Ok(());
133
134    for process in pid_table::pid_table_mut().iter_processes() {
135        if Arc::ptr_eq(&ctx.process, &process) || process.is_init_process() {
136            continue;
137        }
138
139        let res = kill_process(
140            &process,
141            signal.clone().map(|s| Box::new(s) as Box<dyn Signal>),
142            ctx,
143        );
144        if res.is_err_and(|err| err.error() != Errno::EPERM) {
145            result = res;
146        }
147    }
148
149    result
150}

152fn kill_process(process: &Process, signal: Option<Box<dyn Signal>>, ctx: &Context) -> Result<()> {
153    let signum = signal.as_ref().map(|signal| signal.num());
154    let target_main_thread = process.main_thread();
155    check_signal_perm(target_main_thread.as_posix_thread().unwrap(), ctx, signum)?;
156
157    if let Some(signal) = signal {
158        process.enqueue_signal(signal);
159    }
160
161    Ok(())
162}

164// Reference: <https://elixir.bootlin.com/linux/v6.17/source/kernel/signal.c#L799>.
165fn check_signal_perm(target: &PosixThread, ctx: &Context, signum: Option<SigNum>) -> Result<()> {
166    let target_process = target.process();
167
168    if Arc::ptr_eq(&target_process, &ctx.process) {
169        return Ok(());
170    }
171
172    let current_cred = ctx.posix_thread.credentials();
173    let target_cred = target.credentials();
174    if current_cred.euid() == target_cred.suid()
175        || current_cred.euid() == target_cred.ruid()
176        || current_cred.ruid() == target_cred.suid()
177        || current_cred.ruid() == target_cred.ruid()
178    {
179        return Ok(());
180    }
181
182    if lsm_hooks::on_capable(lsm_hooks::CapableContext::new(
183        target_process.user_ns().lock().as_ref(),
184        ctx.posix_thread,
185        CapSet::KILL,
186    ))
187    .is_ok()
188    {
189        return Ok(());
190    }
191
192    if let Some(signum) = signum
193        && signum == SIGCONT
194    {
195        let target_sid = target_process.sid();
196        let current_sid = ctx.process.sid();
197        if target_sid == current_sid {
198            return Ok(());
199        }
200    }
201
202    return_errno_with_message!(
203        Errno::EPERM,
204        "sending signal to the target process or thread is not allowed"
205    )
