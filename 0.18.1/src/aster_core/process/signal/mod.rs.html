48/// Handles a pending signal for the current process.
49pub(crate) fn handle_pending_signal(user_ctx: &mut UserContext, ctx: &Context) {
50    // FIXME: This function may handle or suppress only one signal per trap, delaying
51    // other pending unmasked signals until the next trap. Consider a looped scan.
52    //
53    // For details, see <https://github.com/asterinas/asterinas/pull/2984/#discussion_r3137275994>.
54    let syscall_restart = if let Some(orig_syscall_ret) = ctx.thread_local.orig_syscall_ret()
55        && user_ctx.syscall_ret() == -(Errno::ERESTARTSYS as i32) as usize
56    {
57        // We should never return `ERESTARTSYS` to the userspace.
58        user_ctx.set_syscall_ret(-(Errno::EINTR as i32) as usize);
59        Some(orig_syscall_ret)
60    } else {
61        None
62    };
63
64    let mut restore_sig_mask = ctx
65        .thread_local
66        .sig_mask_saved()
67        .take()
68        .map(|mask| RestoreSigMaskGuard { ctx, mask });
69
70    let (dequeued, sig_action) = if let Some(dequeued_signal) = dequeue_pending_signal(ctx) {
71        dequeued_signal
72    } else {
73        // Fast path: There is no signal mask to restore.
74        if restore_sig_mask.is_none() {
75            return;
76        }
77        // Restore the signal mask first.
78        let _ = restore_sig_mask.take();
79
80        // Try again with the new signal mask.
81        if let Some(dequeued_signal) = dequeue_pending_signal(ctx) {
82            dequeued_signal
83        } else {
84            return;
85        }
86    };
87
88    let (signal, sig_action) = if dequeued.num() != SIGKILL {
89        match ctx.posix_thread.ptrace_stop(dequeued, ctx, user_ctx) {
90            PtraceStopResult::Continued(Some(dequeued)) => {
91                // Note that this `dequeued` object outputted by `ptrace_stop`
92                // might be different from the input `dequeued` object
93                // because of tracer manipulation.
94                // But we name both `dequeued` anyway.
95
96                let sig_num = dequeued.num();
97                if ctx.posix_thread.sig_mask().contains(sig_num) {
98                    // Requeue the injected signal to the same signal queue as the
99                    // dequeued stop signal (thread queue or process queue).
100                    //
101                    // Reference: <https://elixir.bootlin.com/linux/v6.16.5/source/kernel/signal.c#L2770>
102                    requeue_signal(ctx, dequeued);
103                    return;
104                }
105
106                (dequeued.unwrap(), get_sig_action(ctx, sig_num))
107            }
108            PtraceStopResult::Continued(None) => return,
109            PtraceStopResult::Interrupted => {
110                match dequeue_pending_signal(ctx) {
111                    Some((dequeued, action)) if dequeued.num() == SIGKILL => {
112                        (dequeued.unwrap(), action)
113                    }
114                    // SIGKILL was consumed by a sibling thread; this thread will be
115                    // terminated as part of the process-wide exit.
116                    _ => return,
117                }
118            }
119            PtraceStopResult::NotTraced(dequeued) => (dequeued.unwrap().unwrap(), sig_action),
120        }
121    } else {
122        (dequeued.unwrap(), sig_action)
123    };
124
125    let sig_num = signal.num();
126    match sig_action {
127        SigAction::Ign => {
128            debug!("Ignore signal {:?}", sig_num);
129        }
130        SigAction::User {
131            handler_addr,
132            flags,
133            restorer_addr,
134            mask,
135        } => {
136            if flags.contains(SigActionFlags::SA_RESETHAND) {
137                // In Linux, SA_RESETHAND corresponds to SA_ONESHOT,
138                // which means the user handler will be executed only once and then reset to the default.
139                // Reference: <https://elixir.bootlin.com/linux/v6.0.9/source/kernel/signal.c#L2761>.
140                let sig_dispositions = ctx.process.sig_dispositions().lock();
141                let mut sig_dispositions = sig_dispositions.lock();
142                sig_dispositions.set_default(sig_num);
143            }
144
145            if let Some(orig_syscall_ret) = syscall_restart
146                && flags.contains(SigActionFlags::SA_RESTART)
147            {
148                #[cfg(target_arch = "x86_64")]
149                const SYSCALL_INSTR_LEN: usize = 2; // syscall
150                #[cfg(target_arch = "riscv64")]
151                const SYSCALL_INSTR_LEN: usize = 4; // ecall
152                #[cfg(target_arch = "loongarch64")]
153                const SYSCALL_INSTR_LEN: usize = 4; // syscall
154
155                user_ctx.set_syscall_ret(orig_syscall_ret);
156                user_ctx
157                    .set_instruction_pointer(user_ctx.instruction_pointer() - SYSCALL_INSTR_LEN);
158            }
159
160            if let Err(e) = handle_user_signal(
161                ctx,
162                sig_num,
163                handler_addr,
164                flags,
165                restorer_addr,
166                mask,
167                restore_sig_mask.map(RestoreSigMaskGuard::into_mask),
168                user_ctx,
169                signal.to_info(),
170            ) {
171                debug!("Failed to handle user signal: {:?}", e);
172                // If signal handling fails, the process should be terminated with SIGSEGV.
173                // Reference: <https://elixir.bootlin.com/linux/v6.13/source/kernel/signal.c#L3082>
174                //
175                // FIXME: Linux converts a signal-frame setup failure into a SIGSEGV delivery,
176                // instead of killing the process directly.
177                // This can be observed in LTP test `signal06`.
178                do_exit_group(TermStatus::Killed(SIGSEGV), ctx, user_ctx);
179            }
180        }
181        SigAction::Dfl if ctx.process.is_init_process() => {
182            // From Linux man pages "kill(2)":
183            // "The only signals that can be sent to process ID 1, the init process, are those for
184            // which init has explicitly installed signal handlers."
185        }
186        SigAction::Dfl => {
187            let sig_default_action = SigDefaultAction::from_signum(sig_num);
188            debug!("sig_default_action = {:?}", sig_default_action);
189
190            match sig_default_action {
191                SigDefaultAction::Core | SigDefaultAction::Term => {
192                    warn!(
193                        "PID {}: terminating on signal {}",
194                        ctx.process.pid(),
195                        sig_num.sig_name()
196                    );
197                    // The signal terminates the current process. Therefore, we should exit here.
198                    do_exit_group(TermStatus::Killed(sig_num), ctx, user_ctx);
199                }
200                SigDefaultAction::Ign => {}
201                SigDefaultAction::Stop => ctx.process.stop(sig_num),
202                SigDefaultAction::Cont => ctx.process.resume(),
203            }
204        }
205    }
206}

208/// A guard that restores the signal mask on drop.
209struct RestoreSigMaskGuard<'a> {
210    ctx: &'a Context<'a>,
211    mask: SigMask,
212}
213
214impl RestoreSigMaskGuard<'_> {
215    /// Forgets the guard and returns the signal mask to restore.
216    fn into_mask(self) -> SigMask {
217        let mask = self.mask;
218        // Assert that it's a `Copy` type. So we won't leak resources.
219        let _ = self.mask;
220
221        core::mem::forget(self);
222
223        mask
224    }
225}
226
227impl Drop for RestoreSigMaskGuard<'_> {
228    fn drop(&mut self) {
229        self.ctx.set_sig_mask(self.mask);
230    }
231}

233fn dequeue_pending_signal(ctx: &Context) -> Option<(DequeuedSignal, SigAction)> {
234    let posix_thread = ctx.posix_thread;
235
236    let sig_dispositions = ctx.process.sig_dispositions().lock();
237    let sig_dispositions = sig_dispositions.lock();
238
239    let sig_mask = posix_thread.sig_mask();
240    let (signal, sig_num, sig_action) = loop {
241        let signal = ctx.dequeue_signal(&sig_mask)?;
242        let sig_num = signal.num();
243        let sig_action = sig_dispositions.get(sig_num);
244        if sig_action.will_ignore(sig_num) && !posix_thread.is_traced() {
245            continue;
246        }
247
248        break (signal, sig_num, sig_action);
249    };
250
251    debug!(
252        "sig_num = {:?}, sig_name = {}, sig_action = {:#x?}",
253        sig_num,
254        sig_num.sig_name(),
255        sig_action
256    );
257
258    Some((signal, sig_action))
259}

261/// Re-queue the ptrace-captured signal.
262///
263/// This appends to the tail of the origin queue,
264/// which (a) loses same-number RT-signal send-order FIFO and
265/// (b) allows newer lower-numbered standard signals queued
266/// during the ptrace stop to preempt this signal.
267/// This is intentionally consistent with Linux's behaviors.
268fn requeue_signal(ctx: &Context, signal: DequeuedSignal) {
269    match signal {
270        DequeuedSignal::FromProcess(signal) => ctx.process.enqueue_signal(signal),
271        DequeuedSignal::FromThread(signal) => ctx.posix_thread.enqueue_signal(signal),
272    }
273}

275fn get_sig_action(ctx: &Context, sig_num: SigNum) -> SigAction {
276    let sig_dispositions = ctx.process.sig_dispositions().lock();
277    let sig_dispositions = sig_dispositions.lock();
278    sig_dispositions.get(sig_num)
279}

281#[expect(clippy::too_many_arguments)]
282pub(crate) fn handle_user_signal(
283    ctx: &Context,
284    sig_num: SigNum,
285    handler_addr: Vaddr,
286    flags: SigActionFlags,
287    restorer_addr: Vaddr,
288    mut mask: SigMask,
289    mask_to_restore: Option<SigMask>,
290    user_ctx: &mut UserContext,
291    sig_info: siginfo_t,
292) -> Result<()> {
293    debug!("sig_num = {:?}, signame = {}", sig_num, sig_num.sig_name());
294    debug!("handler_addr = 0x{:x}", handler_addr);
295    debug!("flags = {:?}", flags);
296    debug!("restorer_addr = 0x{:x}", restorer_addr);
297    debug!("mask = {:?}, mask_to_restore = {:?}", mask, mask_to_restore);
298
299    if flags.contains_unsupported_flag() {
300        warn!("Unsupported signal flags: {:?}", flags);
301    }
302
303    if !flags.contains(SigActionFlags::SA_NODEFER) {
304        // Add the current signal to `mask`.
305        mask += sig_num;
306    }
307
308    // Block signals in `mask` while running the signal handler.
309    let old_mask = ctx.posix_thread.sig_mask();
310    let mask_to_restore = mask_to_restore.unwrap_or(old_mask);
311    ctx.set_sig_mask(old_mask + mask);
312
313    // Set up the signal stack.
314    let mut stack_pointer = if let Some(sp) =
315        use_alternate_signal_stack(flags, ctx.thread_local, user_ctx.stack_pointer())
316    {
317        sp as u64
318    } else {
319        // Just use the user stack.
320        let sp = user_ctx.stack_pointer() as u64;
321
322        // Prevent corruption of the current stack. Architectures like as x86-64 have red zones
323        // that can be accessed below the SP. Signal handlers must not write to these zones.
324        // FIXME: This may not be necessary for all architectures.
325        sp.wrapping_sub(128)
326    };
327
328    let user_space = ctx.user_space();
329
330    // 1. Write `siginfo_t`.
331    stack_pointer = alloc_aligned_in_user_stack(
332        stack_pointer,
333        size_of::<siginfo_t>(),
334        align_of::<siginfo_t>(),
335    );
336    user_space.write_val(stack_pointer as _, &sig_info)?;
337    let siginfo_addr = stack_pointer;
338
339    // 2. Write `ucontext_t`.
340
341    // Save the current signal stack information.
342    let uc_stack = {
343        let mut sig_stack = ctx.thread_local.sig_stack().borrow_mut();
344        let stack = stack_t::from(&*sig_stack);
345
346        if sig_stack.flags().contains(SigStackFlags::SS_AUTODISARM) {
347            sig_stack.reset();
348        }
349
350        stack
351    };
352
353    let mut ucontext = ucontext_t {
354        uc_stack,
355        uc_sigmask: mask_to_restore.into(),
356        ..Default::default()
357    };
358
359    // Save general-purpose registers.
360    ucontext.uc_mcontext.copy_user_regs_from(user_ctx);
361
362    // Clone and reset the FPU context.
363    let supp = ctx.thread_local.supp_user_context();
364    let fpu_context = supp.fpu().get();
365    let fpu_context_bytes = fpu_context.as_bytes();
366    supp.fpu().set(FpuContext::new());
367
368    cfg_select! {
369        target_arch = "x86_64" => {
370            // Align the FPU context address to the 64-byte boundary so that the
371            // user program can use the XSAVE/XRSTOR instructions at that address,
372            // if necessary.
373            let fpu_context_addr =
374                alloc_aligned_in_user_stack(stack_pointer, fpu_context_bytes.len(), 64);
375            let ucontext_addr = alloc_aligned_in_user_stack(
376                fpu_context_addr,
377                size_of::<ucontext_t>(),
378                align_of::<ucontext_t>(),
379            );
380            ucontext
381                .uc_mcontext
382                .set_fpu_context_addr(fpu_context_addr as _);
383
384            const UC_FP_XSTATE: u64 = 1 << 0;
385            ucontext.uc_flags = UC_FP_XSTATE;
386        }
387        target_arch = "riscv64" => {
388            // Reference:
389            // <https://elixir.bootlin.com/linux/v6.17.5/source/arch/riscv/include/uapi/asm/ptrace.h#L94-L98>,
390            // <https://elixir.bootlin.com/linux/v6.17.5/source/arch/riscv/include/uapi/asm/ptrace.h#L69-L77>.
391            const FP_STATE_SIZE: usize =
392                size_of::<ostd::arch::cpu::context::QFpuContext>() + 3 * size_of::<u32>();
393
394            let ucontext_addr = alloc_aligned_in_user_stack(
395                stack_pointer,
396                size_of::<ucontext_t>() + FP_STATE_SIZE,
397                align_of::<ucontext_t>(),
398            );
399            let fpu_context_addr = (ucontext_addr as usize) + size_of::<ucontext_t>();
400        }
401        target_arch = "loongarch64" => {
402            // FIXME: It seems that we need to allocate an `sctx_info` structure.
403            // Reference: <https://elixir.bootlin.com/linux/v6.15.7/source/arch/loongarch/kernel/signal.c#L848>
404            let ucontext_addr = alloc_aligned_in_user_stack(
405                stack_pointer,
406                size_of::<ucontext_t>() + fpu_context_bytes.len(),
407                align_of::<ucontext_t>(),
408            );
409            // TODO: Set the flags in the context structure.
410            // Reference: <https://elixir.bootlin.com/linux/v6.15.7/source/arch/loongarch/kernel/signal.c#L805>
411            let fpu_context_addr = (ucontext_addr as usize) + size_of::<ucontext_t>();
412        }
413        _ => {
414            compile_error!("unsupported target");
415        }
416    }
417
418    user_space.write_bytes(fpu_context_addr as _, fpu_context_bytes)?;
419    user_space.write_val(ucontext_addr as _, &ucontext)?;
420
421    // Align the SP to the 16-byte boundary. This is required by the x86-64 ABI before calling any
422    // function.
423    stack_pointer = ucontext_addr.align_down(16);
424
425    // 3. Write the address of the restorer code.
426    let retaddr = if flags.contains(SigActionFlags::SA_RESTORER) {
427        restorer_addr
428    } else {
429        cfg_select! {
430            target_arch = "riscv64" => {
431                ctx.user_space().vmar().process_vm().vdso_base()
432                    + crate::vdso::__VDSO_RT_SIGRETURN_OFFSET
433            }
434            _ => {
435                // Note that this should already be rejected at the `rt_sigaction` system call.
436                return_errno_with_message!(
437                    Errno::EINVAL,
438                    "this architecture currently requires SA_RESTORER"
439                )
440            }
441        }
442    };
443    cfg_select! {
444        target_arch = "x86_64" => {
445            stack_pointer = write_u64_to_user_stack(stack_pointer, retaddr as u64)?;
446        }
447        any(target_arch = "riscv64", target_arch = "loongarch64") => {
448            user_ctx.set_ra(retaddr);
449        }
450        _ => {
451            compile_error!("unsupported target");
452        }
453    }
454
455    debug!(
456        "Before calling to signal handler: stack_pointer = 0x{:x}",
457        stack_pointer
458    );
459
460    // 4. Set correct register values.
461    user_ctx.set_instruction_pointer(handler_addr as _);
462    user_ctx.set_stack_pointer(stack_pointer as usize);
463    // Set parameters of the signal handler.
464    if flags.contains(SigActionFlags::SA_SIGINFO) {
465        user_ctx.set_arguments(sig_num, siginfo_addr as usize, ucontext_addr as usize);
466    } else {
467        user_ctx.set_arguments(sig_num, 0, 0);
468    }
469    // Perform CPU architecture-dependent logic.
470    cfg_select! {
471        target_arch = "x86_64" => {
472            // Clear the DF flag. This is to conform to x86-64 calling conventions.
473            const X86_RFLAGS_DF: usize = 1 << 10; // Bit 10 is the DF flag.
474            user_ctx.general_regs_mut().rflags &= !X86_RFLAGS_DF;
475        }
476        _ => {},
477    }
478
479    Ok(())
480}

482/// Uses the alternate signal stack configured via `sigaltstack`.
483///
484/// If the current stack pointer `sp` is already within the alternate signal stack
485/// or if the stack is disabled, this function returns `None`.
486/// Otherwise, it returns the starting stack pointer of the alternate signal stack.
487fn use_alternate_signal_stack(
488    flags: SigActionFlags,
489    thread_local: &ThreadLocal,
490    sp: usize,
491) -> Option<usize> {
492    if !flags.contains(SigActionFlags::SA_ONSTACK) {
493        return None;
494    }
495
496    let sig_stack = thread_local.sig_stack().borrow();
497
498    if sig_stack.active_status(sp) != SigStackStatus::Inactive {
499        return None;
500    }
501
502    // Overflow checks are done in the `sigaltstack` system call.
503    Some(sig_stack.base() + sig_stack.size())
504}

506/// Writes a `u64` integer to the user's stack.
507#[cfg(target_arch = "x86_64")]
508fn write_u64_to_user_stack(sp: u64, value: u64) -> Result<u64> {
509    use crate::context::current_userspace;
510
511    let sp = sp.wrapping_sub(size_of::<u64>() as u64);
512    current_userspace!().write_val(sp as _, &value)?;
513    Ok(sp)
514}

516/// Allocates `size` bytes on the user's stack.
517///
518/// The allocation will be aligned to `align`, which must be a power of two.
519fn alloc_aligned_in_user_stack(sp: u64, size: usize, align: usize) -> u64 {
520    sp.wrapping_sub(size as u64).align_down(align as u64)
