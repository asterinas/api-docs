36/// Creates a new signalfd or updates an existing one according to the given mask.
37pub(super) fn sys_signalfd(
38    fd: RawFileDesc,
39    mask_ptr: Vaddr,
40    sizemask: usize,
41    ctx: &Context,
42) -> Result<SyscallReturn> {
43    sys_signalfd4(fd, mask_ptr, sizemask, 0, ctx)
44}
45
46/// Creates a new signalfd or updates an existing one according to the given mask and flags.
47pub(super) fn sys_signalfd4(
48    raw_fd: RawFileDesc,
49    mask_ptr: Vaddr,
50    sizemask: usize,
51    flags: i32,
52    ctx: &Context,
53) -> Result<SyscallReturn> {
54    debug!(
55        "fd = {}, mask = {:x}, sizemask = {}, flags = {}",
56        raw_fd, mask_ptr, sizemask, flags
57    );
58
59    if sizemask != size_of::<SigMask>() {
60        return_errno_with_message!(Errno::EINVAL, "invalid mask size");
61    }
62
63    let mut mask = ctx.user_space().read_val::<SigMask>(mask_ptr)?;
64    mask -= SIGKILL;
65    mask -= SIGSTOP;
66
67    let flags = SignalFileFlags::from_bits(flags as u32)
68        .ok_or_else(|| Error::with_message(Errno::EINVAL, "invalid flags"))?;
69
70    let fd_flags = if flags.contains(SignalFileFlags::O_CLOEXEC) {
71        FdFlags::CLOEXEC
72    } else {
73        FdFlags::empty()
74    };
75
76    let non_blocking = flags.contains(SignalFileFlags::O_NONBLOCK);
77
78    let new_fd = if raw_fd == -1 {
79        create_new_signalfd(ctx, mask, non_blocking, fd_flags)?
80    } else {
81        update_existing_signalfd(ctx, raw_fd, mask, non_blocking)?
82    };
83
84    Ok(SyscallReturn::Return(new_fd as _))
85}

87fn create_new_signalfd(
88    ctx: &Context,
89    mask: SigMask,
90    non_blocking: bool,
91    fd_flags: FdFlags,
92) -> Result<RawFileDesc> {
93    let signal_file = {
94        let atomic_mask = AtomicSigMask::new(mask);
95        Arc::new(SignalFile::new(atomic_mask, non_blocking))
96    };
97
98    let file_table = ctx.thread_local.borrow_file_table();
99    let fd = file_table.unwrap().write().insert(signal_file, fd_flags);
100    Ok(fd.into())
101}
102
103fn update_existing_signalfd(
104    ctx: &Context,
105    raw_fd: RawFileDesc,
106    new_mask: SigMask,
107    non_blocking: bool,
108) -> Result<RawFileDesc> {
109    let mut file_table = ctx.thread_local.borrow_file_table_mut();
110    let file = get_file_fast!(&mut file_table, raw_fd.try_into()?);
111    let signal_file = file
112        .downcast_ref::<SignalFile>()
113        .ok_or_else(|| Error::with_message(Errno::EINVAL, "the file is not a signal file"))?;
114
115    signal_file.update_signal_mask(new_mask);
116    signal_file.set_non_blocking(non_blocking);
117    Ok(raw_fd)
118}

124        const O_NONBLOCK = StatusFlags::O_NONBLOCK.bits();

128/// Signal file implementation.
129///
130/// Represents a file that can be used to receive signals
131/// as readable events.
132struct SignalFile {
133    /// An atomic signal mask for filtering signals.
134    signals_mask: AtomicSigMask,
135    /// The common state for this signalfd file.
136    common: FileCommon,
137}
138
139impl SignalFile {
140    /// Creates a new signalfd instance.
141    fn new(mask: AtomicSigMask, non_blocking: bool) -> Self {
142        let pseudo_path = AnonInodeFs::new_path(|_| "anon_inode:[signalfd]".to_string());
143        let status_flags = if non_blocking {
144            StatusFlags::O_NONBLOCK
145        } else {
146            StatusFlags::empty()
147        };
148
149        Self {
150            signals_mask: mask,
151            // Reference: <https://elixir.bootlin.com/linux/v7.0/source/fs/signalfd.c#L276>.
152            common: FileCommon::new(pseudo_path, AccessMode::O_RDWR, status_flags),
153        }
154    }
155
156    fn signal_mask(&self) -> SigMask {
157        self.signals_mask.load(Ordering::Relaxed)
158    }
159
160    fn update_signal_mask(&self, new_mask: SigMask) {
161        self.signals_mask.store(new_mask, Ordering::Relaxed);
162    }
163
164    fn is_non_blocking(&self) -> bool {
165        self.common.is_nonblocking()
166    }
167
168    fn set_non_blocking(&self, non_blocking: bool) {
169        (self as &dyn FileLike).update_status_nonblock(non_blocking);
170    }
171
172    /// Checks current readable I/O events.
173    fn check_io_events(&self, posix_thread: &PosixThread) -> IoEvents {
174        let mask = self.signal_mask();
175        if posix_thread.pending_signals().intersects(mask) {
176            IoEvents::IN
177        } else {
178            IoEvents::empty()
179        }
180    }
181
182    /// Attempts non-blocking read operation.
183    fn try_read(&self, writer: &mut VmWriter, thread: &PosixThread) -> Result<usize> {
184        // We invert `signal_mask` to get the signals that are not blocked.
185        let mask = !self.signal_mask();
186        let max_signals = writer.avail() / size_of::<SignalfdSiginfo>();
187        let mut count = 0;
188
189        for _ in 0..max_signals {
190            match thread.dequeue_signal(&mask) {
191                Some(dequeued) => {
192                    writer.write_val(&dequeued.unwrap().to_signalfd_siginfo())?;
193                    count += 1;
194                }
195                None => break,
196            }
197        }
198
199        if count == 0 {
200            return_errno_with_message!(Errno::EAGAIN, "no signals are pending");
201        }
202        Ok(count * size_of::<SignalfdSiginfo>())
203    }
204}
205
206impl Pollable for SignalFile {
207    fn poll(&self, mask: IoEvents, poller: Option<&mut PollHandle>) -> IoEvents {
208        let thread = current_thread!();
209        let posix_thread = thread.as_posix_thread().unwrap();
210
211        if let Some(poller) = poller {
212            posix_thread.register_signalfd_poller(poller, mask);
213        }
214
215        self.check_io_events(posix_thread) & mask
216    }
217}
218
219impl FileLike for SignalFile {
220    fn read(&self, writer: &mut VmWriter) -> Result<usize> {
221        if writer.avail() < size_of::<SignalfdSiginfo>() {
222            return_errno_with_message!(Errno::EINVAL, "the signal buffer is too small");
223        }
224
225        let thread = current_thread!();
226        let posix_thread = thread.as_posix_thread().unwrap();
227
228        // Fast path: There are already pending signals or the signalfd is non-blocking.
229        // So we don't need to create and register the poller.
230        match self.try_read(writer, posix_thread) {
231            Err(e) if e.error() == Errno::EAGAIN && !self.is_non_blocking() => {}
232            res => return res,
233        }
234
235        // Slow path
236        let mut poller = Poller::new(None);
237        posix_thread.register_signalfd_poller(poller.as_handle_mut(), IoEvents::IN);
238
239        loop {
240            match self.try_read(writer, posix_thread) {
241                Err(e) if e.error() == Errno::EAGAIN => poller.wait()?,
242                res => return res,
243            }
244        }
245    }
246
247    fn common(&self) -> &FileCommon {
248        &self.common
249    }
250
251    fn dump_proc_fdinfo(self: Arc<Self>, fd_flags: FdFlags) -> Box<dyn Display> {
252        struct FdInfo {
253            flags: u32,
254            sigmask: u64,
255        }
256
257        impl Display for FdInfo {
258            fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
259                writeln!(f, "pos:\t{}", 0)?;
260                writeln!(f, "flags:\t0{:o}", self.flags)?;
261                writeln!(f, "mnt_id:\t{}", AnonInodeFs::mount_node().id())?;
262                writeln!(f, "ino:\t{}", AnonInodeFs::shared_inode().ino())?;
263                writeln!(f, "sigmask:\t{:016x}", self.sigmask)
264            }
265        }
266
267        let mut flags = self.common.status_flags().bits() | self.common.access_mode() as u32;
268        if fd_flags.contains(FdFlags::CLOEXEC) {
269            flags |= CreationFlags::O_CLOEXEC.bits();
270        }
271
272        Box::new(FdInfo {
273            flags,
274            sigmask: self.signal_mask().into(),
275        })
276    }
277}
