19pub(super) fn sys_statfs(
20    path_ptr: Vaddr,
21    statfs_buf_ptr: Vaddr,
22    ctx: &Context,
23) -> Result<SyscallReturn> {
24    let user_space = ctx.user_space();
25    let path_name = user_space.read_cstring(path_ptr, PATH_MAX)?;
26    debug!(
27        "path = {:?}, statfs_buf_ptr = 0x{:x}",
28        path_name, statfs_buf_ptr,
29    );
30
31    let statfs = {
32        let path_name = path_name.to_string_lossy();
33        let fs_path = FsPath::try_from(path_name.as_ref())?;
34        let path = ctx
35            .thread_local
36            .borrow_fs()
37            .resolver()
38            .read()
39            .lookup(&fs_path)?;
40        Statfs::new(path.mount_node())
41    };
42    user_space.write_val(statfs_buf_ptr, &statfs)?;
43    Ok(SyscallReturn::Return(0))
44}

46pub(super) fn sys_fstatfs(
47    raw_fd: RawFileDesc,
48    statfs_buf_ptr: Vaddr,
49    ctx: &Context,
50) -> Result<SyscallReturn> {
51    debug!(
52        "raw_fd = {}, statfs_buf_addr = 0x{:x}",
53        raw_fd, statfs_buf_ptr
54    );
55
56    let statfs = {
57        let mut file_table = ctx.thread_local.borrow_file_table_mut();
58        let file = get_file_fast!(&mut file_table, raw_fd.try_into()?);
59        Statfs::new(file.path().mount_node())
60    };
61    ctx.user_space().write_val(statfs_buf_ptr, &statfs)?;
62    Ok(SyscallReturn::Return(0))
63}

65/// FS Stat
66#[repr(C)]
67#[derive(Clone, Copy, Debug, Default, Pod)]
68struct Statfs {
69    /// Type of filesystem
70    f_type: u64,
71    /// Optimal transfer block size
72    f_bsize: usize,
73    /// Total data blocks in filesystem
74    f_blocks: usize,
75    /// Free blocks in filesystem
76    f_bfree: usize,
77    /// Free blocks available to unprivileged user
78    f_bavail: usize,
79    /// Total inodes in filesystem
80    f_files: usize,
81    /// Free inodes in filesystem
82    f_ffree: usize,
83    /// Filesystem ID
84    f_fsid: u64,
85    /// Maximum length of filenames
86    f_namelen: usize,
87    /// Fragment size
88    f_frsize: usize,
89    /// Mount flags of filesystem
90    f_flags: u64,
91    /// Padding bytes reserved for future use
92    f_spare: [u64; 4],
93}
94
95impl Statfs {
96    fn new(mount: &Mount) -> Self {
97        let sb = mount.fs().sb();
98        // TODO: Make `SuperBlock` correctly implement and maintain `FsFlags`,
99        // so they can be retrieved directly here.
100        let statfs_flags =
101            StatfsFlags::new(mount.flags(), FsFlags::from_bits_truncate(sb.flags as u32));
102        Self {
103            f_type: sb.magic,
104            f_bsize: sb.bsize,
105            f_blocks: sb.blocks,
106            f_bfree: sb.bfree,
107            f_bavail: sb.bavail,
108            f_files: sb.files,
109            f_ffree: sb.ffree,
110            f_fsid: sb.fsid,
111            f_namelen: sb.namelen,
112            f_frsize: sb.frsize,
113            f_flags: statfs_flags.bits() as u64,
114            f_spare: [0u64; 4],
115        }
116    }
117}

119bitflags! {
120    /// User-visible flags in [`Statfs`].
121    ///
122    /// Reference: <https://elixir.bootlin.com/linux/v6.16.5/source/include/linux/statfs.h#L31>
123    struct StatfsFlags: u32 {
124        /// Mount read-only.
125        const ST_RDONLY = 1 << 0;
126        /// Ignore suid and sgid bits.
127        const ST_NOSUID = 1 << 1;
128        /// Disallow access to device special files.
129        const ST_NODEV = 1 << 2;
130        /// Disallow program execution.
131        const ST_NOEXEC = 1 << 3;
132        /// Writes are synced at once.
133        const ST_SYNCHRONOUS = 1 << 4;
134        /// Allow mandatory locks on an FS.
135        const ST_MANDLOCK = 1 << 6;
136        /// Do not update access times.
137        const ST_NOATIME = 1 << 10;
138        /// Do not update directory access times.
139        const ST_NODIRATIME = 1 << 11;
140        /// Update atime relative to mtime/ctime.
141        const ST_RELATIME = 1 << 12;
142        /// Do not follow symlinks.
143        const ST_NOSYMFOLLOW = 1 << 13;
144    }
145}
146
147impl StatfsFlags {
148    fn new(per_mount_flags: PerMountFlags, fs_flags: FsFlags) -> Self {
149        let mut statfs_flags = StatfsFlags::from_bits_truncate(per_mount_flags.bits());
150        statfs_flags |= StatfsFlags::from_bits_truncate(fs_flags.bits());
151
152        // Some bits in `StatfsFlags` do not correspond directly to bits in
153        // `PerMountFlags` and `FsFlags`, so we need to populate them manually.
154        // These manually added bits do not exist in `PerMountFlags` or `FsFlags`,
155        // so they will not be populated incorrectly before.
156        if per_mount_flags.contains(PerMountFlags::NOSYMFOLLOW) {
157            statfs_flags |= StatfsFlags::ST_NOSYMFOLLOW;
158        }
159        if per_mount_flags.contains(PerMountFlags::RELATIME) {
160            statfs_flags |= StatfsFlags::ST_RELATIME;
161        }
162
163        statfs_flags
164    }
