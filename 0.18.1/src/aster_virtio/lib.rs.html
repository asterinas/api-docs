41static VIRTIO_BLOCK_MAJOR_ID: Once<MajorIdOwner> = Once::new();
42
43#[init_component]
44fn virtio_component_init() -> Result<(), ComponentInitError> {
45    VIRTIO_BLOCK_MAJOR_ID.call_once(|| aster_block::allocate_major().unwrap());
46
47    // Find all devices and register them to the corresponding crate
48    transport::init();
49
50    device::entropy::init();
51    device::network::init();
52    device::socket::init();
53
54    while let Some(mut transport) = pop_device_transport() {
55        let device_type = transport.device_type();
56
57        // Follow VirtIO 1.3, 3.1.1 "Driver Requirements: Device Initialization".
58        // Reference: <https://docs.oasis-open.org/virtio/virtio/v1.3/virtio-v1.3.html#x1-1230001>
59
60        // Reset the device.
61        transport
62            .write_device_status(DeviceStatus::empty())
63            .unwrap();
64        while transport.read_device_status() != DeviceStatus::empty() {
65            spin_loop();
66        }
67
68        // Set `ACKNOWLEDGE` to report that the guest OS has noticed the device.
69        transport
70            .write_device_status(DeviceStatus::ACKNOWLEDGE)
71            .unwrap();
72
73        // Set `DRIVER` to report that the guest OS knows how to drive the device.
74        transport
75            .write_device_status(DeviceStatus::ACKNOWLEDGE | DeviceStatus::DRIVER)
76            .unwrap();
77
78        // Negotiate the feature subset supported by the driver.
79        negotiate_features(&mut transport);
80
81        if !transport.is_legacy_version() {
82            // Set `FEATURES_OK` to report that feature negotiation is complete.
83            let status =
84                DeviceStatus::ACKNOWLEDGE | DeviceStatus::DRIVER | DeviceStatus::FEATURES_OK;
85            transport.write_device_status(status).unwrap();
86
87            let status = transport.read_device_status();
88            if !status.contains(DeviceStatus::FEATURES_OK) {
89                error!(
90                    "Device rejected negotiated features, device type: {:?}",
91                    device_type
92                );
93                transport
94                    .write_device_status(status | DeviceStatus::FAILED)
95                    .unwrap();
96                continue;
97            }
98        }
99
100        let device_transport = DeviceTransport::new(transport);
101
102        let res = match device_type {
103            VirtioDeviceType::Block => BlockDevice::init(device_transport),
104            VirtioDeviceType::Console => ConsoleDevice::init(device_transport),
105            VirtioDeviceType::Entropy => EntropyDevice::init(device_transport),
106            VirtioDeviceType::Input => InputDevice::init(device_transport),
107            VirtioDeviceType::Network => NetworkDevice::init(device_transport),
108            VirtioDeviceType::Socket => SocketDevice::init(device_transport),
109            VirtioDeviceType::FileSystem => FileSystemDevice::init(device_transport),
110            _ => {
111                warn!("Found unimplemented device: {:?}", device_type);
112                Ok(())
113            }
114        };
115        if res.is_err() {
116            error!(
117                "Device initialization error: {:?}, device type: {:?}",
118                res, device_type
119            );
120        }
121    }
122    Ok(())
123}

125fn pop_device_transport() -> Option<Box<dyn VirtioTransport>> {
126    if let Some(device) = VIRTIO_PCI_DRIVER.get().unwrap().pop_device_transport() {
127        return Some(device);
128    }
129    if let Some(device) = VIRTIO_MMIO_DRIVER.get().unwrap().pop_device_transport() {
130        return Some(Box::new(device));
131    }
132    None
133}

135fn negotiate_features(transport: &mut Box<dyn VirtioTransport>) {
136    let features = transport.read_device_features();
137    let mask = ((1u64 << 24) - 1) | (((1u64 << 24) - 1) << 50);
138    let device_specified_features = features & mask;
139    let device_support_features = match transport.device_type() {
140        VirtioDeviceType::Network => NetworkDevice::negotiate_features(device_specified_features),
141        VirtioDeviceType::Block => BlockDevice::negotiate_features(device_specified_features),
142        VirtioDeviceType::Input => InputDevice::negotiate_features(device_specified_features),
143        VirtioDeviceType::Console => ConsoleDevice::negotiate_features(device_specified_features),
144        VirtioDeviceType::Socket => SocketDevice::negotiate_features(device_specified_features),
145        VirtioDeviceType::FileSystem => {
146            FileSystemDevice::negotiate_features(device_specified_features)
147        }
148        _ => device_specified_features,
149    };
150    let mut support_feature = Feature::from_bits_truncate(features);
151    support_feature.remove(Feature::RING_EVENT_IDX);
152    transport
153        .write_driver_features(features & (support_feature.bits | device_support_features))
154        .unwrap();
155}
156
157bitflags! {
158    /// all device features, bits 0~23 and 50~63 are specified by device.
159    /// if using this struct to translate u64, use from_bits_truncate function instead of from_bits
160    ///
161    struct Feature: u64 {
162
163        // device independent
164        const NOTIFY_ON_EMPTY       = 1 << 24; // legacy
165        const ANY_LAYOUT            = 1 << 27; // legacy
166        const RING_INDIRECT_DESC    = 1 << 28;
167        const RING_EVENT_IDX        = 1 << 29;
168        const UNUSED                = 1 << 30; // legacy
169        const VERSION_1             = 1 << 32; // detect legacy
170
171        // since virtio v1.1
172        const ACCESS_PLATFORM       = 1 << 33;
173        const RING_PACKED           = 1 << 34;
174        const IN_ORDER              = 1 << 35;
175        const ORDER_PLATFORM        = 1 << 36;
176        const SR_IOV                = 1 << 37;
177        const NOTIFICATION_DATA     = 1 << 38;
178        const NOTIF_CONFIG_DATA     = 1 << 39;
179        const RING_RESET            = 1 << 40;
180    }
